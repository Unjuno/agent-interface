N={'s':'http://schemas.openxmlformats.org/spreadsheetml/2006/main'}
def workbook(path):
    with zipfile.ZipFile(path) as z:
        strings=[]
        if 'xl/sharedStrings.xml' in z.namelist():
            for si in E.fromstring(z.read('xl/sharedStrings.xml')).findall('s:si',N):strings.append(''.join(x.text or '' for x in si.findall('.//s:t',N)))
        relations={r.attrib['Id']:r.attrib['Target'] for r in E.fromstring(z.read('xl/_rels/workbook.xml.rels'))}
        sheets={}
        for sheet in E.fromstring(z.read('xl/workbook.xml')).findall('s:sheets/s:sheet',N):
            target=relations[sheet.attrib['{http://schemas.openxmlformats.org/officeDocument/2006/relationships}id']]
            target=target.lstrip('/') if target.startswith('/') else 'xl/'+target
            cells={}
            for c in E.fromstring(z.read(target)).findall('.//s:sheetData/s:row/s:c',N):
                v=c.find('s:v',N);f=c.find('s:f',N);t=c.attrib.get('t')
                value=None if v is None else strings[int(v.text)] if t=='s' else float(v.text) if t not in ['str','e'] else v.text
                cells[c.attrib['r']]={'value':value,'formula':None if f is None else f.text}
            sheets[sheet.attrib['name']]=cells
        return sheets
