def dump(p, obj): p.write_text(json.dumps(obj, indent=2, sort_keys=True) + '\n', encoding='utf-8', newline='\n')
def git(*args, data=None, index=False):
    started = now()
    p = subprocess.run(['git', *args], cwd=REPO, env=dict(os.environ, GIT_TERMINAL_PROMPT='0', **({'GIT_INDEX_FILE': str(INDEX)} if index else {})), input=data, capture_output=True, timeout=30)
    folder = WORK / 'publication-v3-command-streams'
    folder.mkdir(exist_ok=True)
    n = str(len(COMMANDS)).zfill(3)
    for kind in ('stdout', 'stderr'): (folder / (n + '.' + kind)).write_bytes(getattr(p, kind))
    COMMANDS.append({'argv': ['git', *args], 'private_index': index, 'started_utc': started, 'finished_utc': now(), 'exit': p.returncode, 'stdout_sha256': sha(p.stdout), 'stderr_sha256': sha(p.stderr)})
    dump(WORK / 'publication-v3-commands.json', COMMANDS)
    if p.returncode: raise RuntimeError(p.stderr.decode(errors='replace'))
    return p.stdout
