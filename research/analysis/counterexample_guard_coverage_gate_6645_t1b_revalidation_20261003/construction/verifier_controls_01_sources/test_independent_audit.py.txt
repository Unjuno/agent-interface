"""Verifier tests use retained data and copies; no candidate is executed."""
import copy
import importlib.util
import json
import shutil
import tempfile
import unittest
from pathlib import Path

HERE = Path(__file__).resolve().parent
PACKAGE = HERE.parent / "counterexample_guard_coverage_gate_6645_t1b_v1"
SPEC = importlib.util.spec_from_file_location("independent_audit", HERE / "independent_audit.py")
audit = importlib.util.module_from_spec(SPEC)
SPEC.loader.exec_module(audit)


def records():
    return [
        audit.read_json(PACKAGE / "candidate_input/candidate_fixture.json"),
        audit.read_json(PACKAGE / "candidate_input/contract.json"),
        audit.read_json(PACKAGE / "oracle.json"),
        audit.read_json(PACKAGE / "results/formal_01/candidate/raw.json"),
    ]


class IndependentAuditTests(unittest.TestCase):
    def test_retained_package_reconstructs_without_candidate_execution(self):
        result = audit.audit_package(PACKAGE)
        self.assertEqual(result["status"], "PASS_INDEPENDENT_REVALIDATION", result)
        self.assertEqual((result["manifest_files_verified"], result["rows_reconstructed"],
                          result["gate_delta_rows"], result["candidate_invocations"]), (30, 5, 3, 0))
        self.assertTrue(result["candidate_visible_ids_contain_outcome_terms"])
        self.assertFalse(result["candidate_decision_reads_id"])

    def test_twelve_effective_record_corruptions_are_rejected(self):
        original = records()
        controls = {
            "false_hidden_admit": lambda x: x[3]["rows"][2].update(gate_enabled="ADMIT"),
            "omitted_raw_row": lambda x: x[3]["rows"].pop(),
            "duplicate_raw_row": lambda x: x[3]["rows"].append(copy.deepcopy(x[3]["rows"][0])),
            "extra_raw_field": lambda x: x[3]["rows"][0].update(oracle_effect="safe"),
            "bool_scalar_alias": lambda x: x[1].update(contract_complete_for_fixture=1),
            "observation_type": lambda x: x[0]["rows"][0]["observations"].update(app_mode=True),
            "extra_fixture_field": lambda x: x[0]["rows"][2].update(hidden_modal="none"),
            "oracle_mislabel": lambda x: x[2]["labels"][2].update(effect="safe"),
            "duplicate_oracle": lambda x: x[2]["labels"].append(copy.deepcopy(x[2]["labels"][0])),
            "raw_order": lambda x: x[3]["rows"].reverse(),
            "identity": lambda x: x[3].update(contract_id="wrong"),
            "stale_admit": lambda x: x[3]["rows"][1].update(gate_disabled="ADMIT"),
        }
        for name, mutate in controls.items():
            with self.subTest(control=name):
                changed = copy.deepcopy(original)
                mutate(changed)
                self.assertNotEqual(changed, original, name)
                self.assertTrue(audit.audit_records(*changed)[0], name)
        print("RAW_RECORD_CONTROLS_REJECTED=12")

    def test_three_manifest_controls_are_rejected(self):
        for name in ("content", "traversal", "duplicate"):
            with self.subTest(control=name), tempfile.TemporaryDirectory() as temp:
                package = Path(temp) / "parent"
                shutil.copytree(PACKAGE, package)
                manifest = package / "SHA256SUMS"
                if name == "content":
                    (package / "oracle.json").write_bytes(b"{}\n")
                elif name == "traversal":
                    manifest.write_bytes(manifest.read_bytes() + b"0" * 64 + b"  ../outside.json\n")
                else:
                    first = manifest.read_bytes().splitlines(keepends=True)[0]
                    manifest.write_bytes(manifest.read_bytes() + first)
                _, errors = audit.verify_manifest(package)
                self.assertTrue(errors, name)
        print("MANIFEST_CONTROLS_REJECTED=3")

    def test_three_boundary_receipt_controls_are_rejected(self):
        for field, value in (("NetworkMode", "bridge"), ("ReadonlyRootfs", False), ("ExitCode", False)):
            with self.subTest(control=field), tempfile.TemporaryDirectory() as temp:
                package = Path(temp) / "parent"
                shutil.copytree(PACKAGE, package)
                path = package / "results/formal_01/candidate/container_inspect_post.json"
                data = audit.read_json(path)
                if field == "ExitCode":
                    data[0]["State"][field] = value
                else:
                    data[0]["HostConfig"][field] = value
                path.write_text(json.dumps(data), encoding="utf-8")
                errors = audit.inspect_boundary(package, audit.read_json(package / "FREEZE.json"))
                self.assertTrue(errors, field)
        print("BOUNDARY_CONTROLS_REJECTED=3")

    def test_label_checker_control_and_four_effective_corruptions(self):
        fixture, _, _, raw = records()
        neutral, expected = copy.deepcopy(fixture), copy.deepcopy(raw)
        for index, (row, output) in enumerate(zip(neutral["rows"], expected["rows"]), 1):
            row["id"] = output["id"] = f"case_{index:02d}"
        self.assertEqual(audit.audit_label_records(fixture, raw, neutral, expected)["errors"], [])
        for name in ("input_extra", "decision", "id", "extra_output"):
            with self.subTest(control=name):
                changed_input, changed_raw = copy.deepcopy(neutral), copy.deepcopy(expected)
                if name == "input_extra":
                    changed_input["rows"][2]["oracle_label"] = "harmful"
                elif name == "decision":
                    changed_raw["rows"][2]["gate_enabled"] = "ADMIT"
                elif name == "id":
                    changed_raw["rows"][0]["id"] = "case_02"
                else:
                    changed_raw["rows"][0]["extra"] = True
                self.assertNotEqual((changed_input, changed_raw), (neutral, expected))
                self.assertTrue(audit.audit_label_records(fixture, raw, changed_input, changed_raw)["errors"])
        print("LABEL_CHECKER_CONTROLS_REJECTED=4; CANDIDATE_EXECUTIONS=0")

    def test_duplicate_json_key_is_rejected(self):
        with tempfile.TemporaryDirectory() as temp:
            path = Path(temp) / "duplicate.json"
            path.write_text('{"rows":[],"rows":[]}', encoding="utf-8")
            with self.assertRaisesRegex(ValueError, "duplicate_json_key"):
                audit.read_json(path)
        print("DUPLICATE_JSON_CONTROLS_REJECTED=1")


if __name__ == "__main__":
    unittest.main()
