import json
import subprocess
import sys
import unittest
from pathlib import Path


HERE = Path(__file__).resolve().parent
PACKAGE = HERE.parent / "counterexample_guard_coverage_gate_6645_t1b_v1"
AUDITOR = HERE / "independent_audit.py"


class T1bIndependentAuditTests(unittest.TestCase):
    def test_main_package_is_independently_reconstructed_and_hash_valid(self):
        completed = subprocess.run(
            [sys.executable, str(AUDITOR), "--package", str(PACKAGE)],
            capture_output=True,
            text=True,
            check=False,
        )
        self.assertEqual(completed.returncode, 0, completed.stdout + completed.stderr)
        result = json.loads(completed.stdout)
        self.assertEqual(result["status"], "PASS_INDEPENDENT_REVALIDATION")
        self.assertEqual(result["manifest_files_verified"], 30)
        self.assertEqual(result["rows_reconstructed"], 5)
        self.assertEqual(result["gate_delta_rows"], 3)
        self.assertTrue(result["candidate_visible_ids_contain_outcome_terms"])
        self.assertFalse(result["candidate_decision_reads_id"])
        self.assertEqual(result["errors"], [])

    def test_label_bearing_row_ids_do_not_change_candidate_decisions(self):
        completed = subprocess.run(
            [
                sys.executable,
                str(AUDITOR),
                "--package",
                str(PACKAGE),
                "--probe-label-relabel",
            ],
            capture_output=True,
            text=True,
            check=False,
        )
        self.assertEqual(completed.returncode, 0, completed.stdout + completed.stderr)
        result = json.loads(completed.stdout)
        self.assertEqual(result["label_relabel_probe"]["status"], "PASS_LABEL_INVARIANCE")
        self.assertEqual(result["label_relabel_probe"]["candidate_invocations"], 1)
        self.assertEqual(result["label_relabel_probe"]["rows_checked"], 5)
        self.assertEqual(result["label_relabel_probe"]["decisions_matching_frozen_raw"], 5)
        self.assertTrue(result["label_relabel_probe"]["only_row_ids_changed"])
        self.assertEqual(
            result["label_relabel_probe"]["row_ids_replaced_with_neutral_labels"],
            ["case_01", "case_02", "case_03", "case_04", "case_05"],
        )
        self.assertEqual(
            [row["id"] for row in result["label_relabel_probe"]["candidate_raw"]["rows"]],
            ["case_01", "case_02", "case_03", "case_04", "case_05"],
        )
        self.assertEqual(
            result["label_relabel_probe"]["temporary_candidate_payload_files"],
            ["candidate.py", "candidate_fixture_relabelled.json", "contract.json"],
        )
        original = json.loads(
            (PACKAGE / "candidate_input" / "candidate_fixture.json").read_text(encoding="utf-8")
        )
        relabelled = result["label_relabel_probe"]["candidate_fixture"]
        self.assertEqual(
            [{key: value for key, value in row.items() if key != "id"} for row in original["rows"]],
            [{key: value for key, value in row.items() if key != "id"} for row in relabelled["rows"]],
        )


if __name__ == "__main__":
    unittest.main()
