from pathlib import Path
from datetime import datetime,timezone
import subprocess,json,sys
p=Path('/workspace/agent-interface/research/analysis/wslc_receipt_audit_v2_3352_01a10197')
checks={
'unit':[sys.executable,'-B','-m','unittest','discover','-s',str(p),'-p','test_repair.py','-v'],
'unit_optimized':[sys.executable,'-B','-O','-m','unittest','discover','-s',str(p),'-p','test_repair.py','-v'],
'raw_only':[sys.executable,'-B',str(p/'verify_repair.py')],
'raw_only_optimized':[sys.executable,'-B','-O',str(p/'verify_repair.py')],
}
receipts={}
for name,argv in checks.items():
 start=datetime.now(timezone.utc).isoformat();c=subprocess.Popen(argv,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True);out,err=c.communicate(timeout=20)
 receipts[name]={'argv':argv,'pid':c.pid,'started_utc':start,'ended_utc':datetime.now(timezone.utc).isoformat(),'exit_code':c.returncode,'stdout':out,'stderr':err}
 print(name,c.returncode)
(p/'CHECKS.json').write_text(json.dumps(receipts,indent=2)+'\n')
if any(r['exit_code'] for r in receipts.values()):sys.exit(1)
