"""Read-only post-result reconciliation; no producer/helper/SDK execution."""
import hashlib
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parent
E = ROOT / 'v12-owner-record-cleanup'
OUT = E / 'record-check-02-path-separators.json'
if OUT.exists():
    raise SystemExit('exclusive check output exists')

def check(condition, message):
    if not condition:
        raise ValueError(message)

def sha(data):
    return hashlib.sha256(data).hexdigest()

exports = json.loads((E / 'SOURCE-EXPORT.json').read_text(encoding='utf-8'))
for path, pin in exports['files'].items():
    raw = (E / 'original-source' / path).read_bytes()
    check(sha(raw) == pin['sha256'] and len(raw) == pin['bytes'], 'original export custody')
rows = []
for phase_name, source_root, repaired in [
    ('first-original-01', E / 'original-source', False),
    ('repaired-main-01', ROOT / 'v12-cleanup-checkout', True),
    ('repaired-main-02-config-preservation', ROOT / 'v12-cleanup-checkout', True),
]:
    phase = E / phase_name
    before = json.loads((phase / 'BEFORE.json').read_text())
    for index, name in enumerate(['normal', 'owner-file-refusal']):
        prefix = phase / f'{index:02d}-{name}'
        invocation = json.loads(prefix.with_suffix('.invocation.json').read_text())
        for stream in ['stdout', 'stderr']:
            raw = prefix.with_suffix(f'.{stream}.bin').read_bytes()
            check(sha(raw) == invocation[f'{stream}_sha256'], 'raw stream custody')
            check(len(raw) == invocation[f'{stream}_bytes'], 'raw stream byte count')
        check(invocation['exit_code'] == 0, 'case wrapper did not finish')
        result = json.loads((prefix / 'FINAL-RECEIPT.json').read_text())
        at_return = json.loads((prefix / 'AT-MAIN-RETURN.json').read_text())
        check(all(result[key] == value for key, value in at_return.items()), 'return snapshot changed')
        check(result['input_lines'] == 0 and result['actual_executor'] == 'executor_v12', 'boundary changed')
        check(result['backend_close_at_main_return'] == 1, 'backend close count')
        check(result['owned_child_final_exit'] == 0, 'owned child not reaped')
        check((prefix / 'owned-child-stderr.bin').read_bytes() == b'', 'child stderr')
        terminal = (prefix / 'owned-child-terminal.bin').read_bytes()
        check(terminal in [b'PRIVATE-INERT-EOF\n', b'PRIVATE-INERT-EOF\r\n'], 'cap or kill instead of EOF')
        check(result['owned_child_terminal'] == terminal.decode('ascii'), 'terminal receipt custody')
        failed_original = name == 'owner-file-refusal' and not repaired
        expected_close = 0 if failed_original else 1
        check(result['session_close_at_main_return'] == expected_close, 'session close count')
        check(result['game_close_at_main_return'] == expected_close, 'game close count')
        check(result['child_exit_at_main_return'] is None if failed_original
              else result['child_exit_at_main_return'] == 0, 'live/reaped child at main boundary')
        check(result['session_tmp_exists_at_main_return'] == failed_original, 'temporary-directory boundary')
        if name == 'owner-file-refusal':
            check(result['main_error']['is_os_error'] is True, 'owner refusal lost')
            check(result['main_error']['type'] == 'PermissionError', 'actual Windows refusal type changed')
            check((prefix / 'session-out/owner-events.json').is_dir(), 'actual refusal target absent')
            check(b'PermissionError' in prefix.with_suffix('.stderr.bin').read_bytes(), 'error traceback absent')
        else:
            check(result['main_error'] is None, 'normal main error')
            records = json.loads((prefix / 'session-out/owner-events.json').read_text())
            check(records == [{'event': 'explicit_fake_owner', 'input_used': False}], 'fake owner records')
        actions = [x['action'] for x in result['trace_at_main_return']]
        check('outer-wrapper-cleanup-needed' not in actions, 'outer action inserted into main boundary')
        outer = [x['action'] for x in result['trace_after_outer_cleanup']]
        check(('outer-wrapper-cleanup-needed' in outer) == failed_original, 'outer cleanup classification')
        # Source hashes in actual session provenance must match the phase pins.
        provenance = json.loads((prefix / 'session-out/sources.json').read_text())
        check(len(provenance) == 20, 'session literal provenance count')
        normalized = {short.replace('\\', '/'): digest for short, digest in provenance.items()}
        check(len(normalized) == len(provenance), 'duplicate normalized provenance path')
        for short, digest in normalized.items():
            full = 'research/' + short
            expected = before['session_source_sha256'] if short == 'doom/session_map01_v12.py' else exports['files'][full]['sha256']
            check(digest == expected, f'exercised session source hash differs: {short}')
        check(len(result['module_sources']) == 8, 'actual executor closure module count')
        rows.append({'phase': phase_name, 'case': name, 'source_sha256': before['session_source_sha256'],
                     'main_error_type': result['main_error']['type'] if result['main_error'] else None,
                     'child_exit_at_main_return': result['child_exit_at_main_return'],
                     'session_close_at_main_return': expected_close, 'final_child_exit': 0,
                     'stdout_sha256': invocation['stdout_sha256'], 'stderr_sha256': invocation['stderr_sha256']})
unit = []
for phase_name, expected_exit in [('first-unit-red-01', 1), ('first-unit-red-02-fixture-repair', 1),
                                  ('first-unit-green-01', 0), ('unit-green-02-config-preservation', 0)]:
    phase = E / phase_name
    receipt = json.loads((phase / 'RECEIPT.json').read_text())
    for kind in ['stdout', 'stderr']:
        check(sha((phase / f'{kind}.bin').read_bytes()) == receipt[f'{kind}_sha256'], 'unit stream custody')
    check(receipt['exit_code'] == expected_exit, 'unit outcome changed')
    check(sha((phase / 'test-source.py.txt').read_bytes()) == receipt['test_sha256'], 'unit test-source custody')
    check(sha((phase / 'runtime-source.py.txt').read_bytes()) == receipt['source_sha256'], 'unit runtime-source custody')
    check(b'Ran 9 tests' in (phase / 'stderr.bin').read_bytes(), 'unit test count changed')
    unit.append({'phase': phase_name, 'exit_code': expected_exit, 'test_sha256': receipt['test_sha256'],
                 'source_sha256': receipt['source_sha256'], 'stderr_sha256': receipt['stderr_sha256']})
report = {'original_source_exports_verified': len(exports['files']), 'actual_main_cases': rows,
          'owned_private_child_count': len(rows), 'all_owned_child_final_exit_zero': True,
          'unit_phases': unit, 'check_source_sha256': sha(Path(__file__).read_bytes()),
          'kind': 'post-result record reader, not independent physical/scientific validation',
          'limitations': ['VD/X11/backend/screen explicit fakes', 'idle actual Executor',
                          'close methods cooperate; no hang/cancel/physical release/game/task claim']}
OUT.write_text(json.dumps(report, indent=2) + '\n', encoding='utf-8', newline='\n')
print(json.dumps({'source_exports': 21, 'main_cases': len(rows), 'unit_phases': len(unit),
                  'report_sha256': sha(OUT.read_bytes())}))
