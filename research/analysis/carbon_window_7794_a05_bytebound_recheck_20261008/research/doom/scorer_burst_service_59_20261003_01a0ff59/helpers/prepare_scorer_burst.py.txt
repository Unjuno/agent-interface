"""Retain exact source comparators and a one-command service copy."""
import hashlib
import json
import subprocess
from pathlib import Path

ROOT = Path(__file__).resolve().parent
REPO = ROOT / 'agent-interface'
DEST = REPO / 'research/doom/scorer_burst_service_59_20261003_01a0ff59'
BASE = 'a394fcdd4254679df4265622b8ae6d0f8e251849'
PEER = '6a82a3ee5be8b4a0c0929ea26a882b00ac28fd4f'
PREFIX = 'research/doom/scorer_command_fairness_59_20261003_01a0ff52/sources/fair_v2/'
FILES = ('main_thread_scorer_polling_v1.py', 'map01_scorer_stdio_adapter_v1.py', 'independent_progress_clock_v2.py')

def blob(ref, path):
    return subprocess.check_output(['git', '-C', str(REPO), 'show', ref + ':' + path])

DEST.mkdir(parents=True, exist_ok=False)
sources = []
for arm in ('original', 'repair_v2', 'one_command'):
    (DEST / 'sources' / arm).mkdir(parents=True)
    for name in FILES:
        ref = PEER if arm == 'repair_v2' else BASE
        path = PREFIX + name if arm == 'repair_v2' else 'research/doom/' + name
        data = blob(ref, path)
        (DEST / 'sources' / arm / name).write_bytes(data)
        sources.append({'arm': arm, 'file': name, 'original_ref': ref, 'original_path': path,
                        'original_sha256': hashlib.sha256(data).hexdigest()})

poll = DEST / 'sources/one_command/main_thread_scorer_polling_v1.py'
text = poll.read_text()
text = text.replace('                continue\n\n            timeout_s = (next_sample_ns - now) / 1_000_000_000\n',
                    '\n            now = self.clock_ns()\n            timeout_s = max(0, (next_sample_ns - now) / 1_000_000_000)\n', 1)
start = text.index('            if not self.wait_readable(fd, timeout_s):')
end = text.index('\n        ended_ns = self.clock_ns()', start)
text = text[:start] + '''            # At most one complete command before the next sampling opportunity.
            if buffer.find(b"\\n") < 0:
                if not self.wait_readable(fd, timeout_s):
                    continue
                chunk = self.read_fn(fd, read_size)
                if chunk == b"":
                    eof = True
                    if buffer:
                        raise ValueError("unterminated command at EOF")
                    break
                buffer.extend(chunk)
                if len(buffer) > self.max_buffer_bytes:
                    raise ValueError("command buffer exceeded max_buffer_bytes")
            newline = buffer.find(b"\\n")
            if newline < 0:
                continue
            raw = bytes(buffer[:newline])
            del buffer[: newline + 1]
            if not raw:
                continue
            line = raw.decode("utf-8", errors="strict")
            command_count += 1
            if command_handler(line) is False:
                stopped_by_command = True
                break
''' + text[end:]
poll.write_text(text, encoding='utf-8', newline='\n')

stdio = DEST / 'sources/one_command/map01_scorer_stdio_adapter_v1.py'
text = stdio.read_text()
start = text.index('    def __next__(self):')
end = text.index('    def stats(self):', start)
text = text[:start] + '''    def _take_line(self):
        newline=self.buffer.find(b'\\n')
        if newline<0:return None
        raw=bytes(self.buffer[:newline]);del self.buffer[:newline+1]
        if not raw:return ''
        self.commands+=1;return raw.decode('utf-8',errors='strict')
    def __next__(self):
        if threading.get_ident()!=self.owner_thread:raise RuntimeError('scorer stdin left session main thread')
        while True:
            # One due sample, then serve a buffered or newly read complete line.
            self._sample_due()
            line=self._take_line()
            if line is not None:
                if not line:continue
                return line
            now=self.loop.clock_ns();timeout=max(0,(self.next_sample_ns-now)/1e9)
            if not self.loop.wait_readable(self.fd,timeout):continue
            chunk=self.loop.read_fn(self.fd,65536)
            if chunk==b'':
                self.eof=True
                if self.buffer:raise ValueError('unterminated command at EOF')
                raise StopIteration
            self.buffer.extend(chunk)
            if len(self.buffer)>self.loop.max_buffer_bytes:raise ValueError('command buffer exceeded max_buffer_bytes')
            line=self._take_line()
            if line is not None:
                if not line:continue
                return line
''' + text[end:]
stdio.write_text(text, encoding='utf-8', newline='\n')
for item in sources:
    item['retained_sha256'] = hashlib.sha256((DEST / 'sources' / item['arm'] / item['file']).read_bytes()).hexdigest()
(DEST / 'SOURCE_MAP.json').write_text(json.dumps(sources, indent=2) + '\n', encoding='utf-8')
(DEST / 'conftest.py').write_text("collect_ignore_glob = ['sources/**']\n", encoding='utf-8')
print(json.dumps({'directory': str(DEST), 'sources': len(sources)}))
