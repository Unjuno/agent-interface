"""Inert source adapter with real native output-pipe backpressure."""
import dataclasses
import hashlib
import json
import os
from pathlib import Path
import sys
import threading
import time

root = Path(__file__).resolve().parent
case = json.loads(sys.argv[1])
sys.path.insert(0, str(root / 'sources' / case['source']))
from main_thread_scorer_polling_v1 import MainThreadScorerPolling
from map01_scorer_stdio_adapter_v1 import MainThreadScorerStdin
from independent_progress_clock_v2 import ProgressSample

event_index = 0
def event(kind, **fields):
    global event_index
    event_index += 1
    print(json.dumps({'event': kind, 'index': event_index,
                      'native_ns': time.perf_counter_ns(),
                      'thread_id': threading.get_ident(), **fields},
                     sort_keys=True), file=sys.stderr, flush=True)

fd = sys.stdout.fileno()
os.set_blocking(fd, False)
prefill = 0
if case['pipe'] == 'full':
    for width in (4096, 1):
        while True:
            if prefill >= 1048576:
                raise RuntimeError('prefill cap reached without a full pipe')
            try:
                prefill += os.write(fd, b'P' * width)
            except BlockingIOError:
                break
    # Last one-byte write refused: the native pipe is actually full.
event('pipe_ready', prefill_bytes=prefill, nonblocking=not os.get_blocking(fd))
if case['sink'] == 'direct':
    os.set_blocking(fd, True)

pending = bytearray()
frames = []
sample_count = 0
def write_all(data):
    cursor = 0
    while cursor < len(data):
        cursor += os.write(fd, data[cursor:])

def flush_nonblocking():
    while pending:
        try:
            written = os.write(fd, pending)
        except BlockingIOError:
            return
        if written <= 0:
            raise RuntimeError('zero native write')
        del pending[:written]

def sample():
    global sample_count
    sample_count += 1
    event('sample', number=sample_count)
    return ProgressSample(0, 0, 0, False, False, False)

def sink(receipt):
    row = dict(receipt)
    row['payload'] = receipt['payload'].as_dict()
    row['controller_visible'] = False
    data = (json.dumps(row, sort_keys=True, separators=(',', ':')) + '\n').encode('utf-8')
    frames.append(data)
    event('sink_enter', bytes=len(data), sha256=hashlib.sha256(data).hexdigest(), row=row)
    if case['sink'] == 'direct':
        write_all(data)
    else:
        if len(pending) + len(data) > case['capacity_bytes']:
            raise BufferError('bounded spool capacity exceeded')
        pending.extend(data)
        flush_nonblocking()
    event('sink_return', pending_bytes=len(pending))

def wait_readable(_fd, timeout_s):
    event('wait_readable', timeout_s=timeout_s)
    return True

read_count = 0
def read(_fd, _size):
    global read_count
    read_count += 1
    event('read_command', bytes_hex=(b'FINISH\n' if read_count == 1 else b'').hex())
    return b'FINISH\n' if read_count == 1 else b''

def handle(line):
    event('command', line=line)
    return False

loop = MainThreadScorerPolling(sample_hz=1.0, clock_ns=lambda: 0,
                               wait_readable=wait_readable, read_fn=read)
try:
    if case['adapter'] == 'polling':
        stats = dataclasses.asdict(loop.run(123, sample_fn=sample, scorer_sink=sink,
                                           command_handler=handle, max_samples=2))
    else:
        class InertStream:
            def fileno(self):
                return 123
        adapter = MainThreadScorerStdin(InertStream(), sample, sink, loop=loop)
        line = next(adapter)
        handle(line)
        stats = adapter.stats()
    event('source_terminal', stats=stats, pending_bytes=len(pending))
except BufferError as error:
    event('capacity_refused', error_type=type(error).__name__, message=str(error),
          pending_bytes=len(pending))
except Exception as error:
    event('source_exception', error_type=type(error).__name__, message=str(error))

event('waiting_for_drain', pending_bytes=len(pending))
if sys.stdin.buffer.readline() != b'DRAIN\n':
    raise RuntimeError('missing parent drain handshake')
os.set_blocking(fd, True)
if pending:
    write_all(pending)
    pending.clear()
event('sink_reconciled', pending_bytes=len(pending), produced_frames=len(frames),
      produced_sha256=hashlib.sha256(b''.join(frames)).hexdigest())
