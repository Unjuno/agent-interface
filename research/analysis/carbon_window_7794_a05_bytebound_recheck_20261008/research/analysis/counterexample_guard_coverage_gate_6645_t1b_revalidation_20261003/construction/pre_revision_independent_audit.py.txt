"""Independent read-only revalidation of the merged Issue #6645 T1b package."""

import argparse
import ast
import copy
import hashlib
import json
import os
import subprocess
import sys
import tempfile
from pathlib import Path


EXPECTED_IDS = [
    "valid_complete_coverage",
    "known_stale_complete_coverage",
    "hidden_modal_harmful_incomplete_coverage",
    "hidden_modal_safe_incomplete_coverage",
    "unregistered_surface_family",
]


def read_json(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))


def reconstruct(row, contract, gate_enabled):
    """Reconstruct the declared coverage/guard rule without importing its auditor."""
    if gate_enabled:
        if contract.get("contract_complete_for_fixture") is not True:
            return "UNKNOWN"
        if row.get("surface_family") not in contract.get("supported_surface_families", []):
            return "UNKNOWN"
        if not set(contract.get("required_predicates", [])).issubset(
            set(row.get("covered_predicates", []))
        ):
            return "UNKNOWN"
    observations = row.get("observations", {})
    for predicate in row.get("covered_predicates", []):
        if predicate not in observations:
            return "UNKNOWN"
        if observations[predicate] != contract.get("safe_values", {}).get(predicate):
            return "REFUSE"
    return "ADMIT"


def decision_field_reads(candidate_path):
    tree = ast.parse(Path(candidate_path).read_text(encoding="utf-8"))
    decision = next(
        node for node in tree.body
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name == "decide"
    )
    fields = set()
    for node in ast.walk(decision):
        if isinstance(node, ast.Subscript) and isinstance(node.value, ast.Name) and node.value.id == "row":
            key = node.slice
            if isinstance(key, ast.Constant) and isinstance(key.value, str):
                fields.add(key.value)
    return sorted(fields)


def verify_manifest(package):
    errors = []
    entries = {}
    for number, line in enumerate((package / "SHA256SUMS").read_text(encoding="utf-8").splitlines(), 1):
        try:
            expected, relative = line.split("  ", 1)
        except ValueError:
            errors.append(f"manifest_line_malformed:{number}")
            continue
        relative = relative.removeprefix("./")
        path = Path(relative)
        if path.is_absolute() or ".." in path.parts:
            errors.append(f"manifest_path_unsafe:{number}")
            continue
        if relative in entries:
            errors.append(f"manifest_path_duplicate:{relative}")
            continue
        entries[relative] = expected.lower()
        target = package / path
        if not target.is_file():
            errors.append(f"manifest_file_missing:{relative}")
        elif hashlib.sha256(target.read_bytes()).hexdigest() != expected.lower():
            errors.append(f"manifest_hash_mismatch:{relative}")
    if len(entries) != 30:
        errors.append(f"manifest_entry_count:{len(entries)}")

    freeze = read_json(package / "FREEZE.json")
    for relative, expected in freeze.get("frozen_input_sha256", {}).items():
        normalized = relative.split("/counterexample_guard_coverage_gate_6645_t1b_v1/", 1)[-1]
        if entries.get(normalized) != expected.lower():
            errors.append(f"freeze_manifest_mismatch:{normalized}")
    return entries, errors


def audit_records(fixture, contract, oracle, raw):
    errors = []
    if raw.get("fixture_id") != fixture.get("fixture_id"):
        errors.append("fixture_identity_mismatch")
    if raw.get("contract_id") != contract.get("contract_id"):
        errors.append("contract_identity_mismatch")

    rows = fixture.get("rows", [])
    raw_rows = raw.get("rows", [])
    ids = [row.get("id") for row in rows]
    if ids != EXPECTED_IDS:
        errors.append("fixture_row_identity_or_order_mismatch")
    if [row.get("id") for row in raw_rows] != ids:
        errors.append("raw_row_identity_or_order_mismatch")

    labels = {}
    for label in oracle.get("labels", []):
        row_id = label.get("id")
        if row_id in labels:
            errors.append("oracle_duplicate_id:" + str(row_id))
        labels[row_id] = label.get("effect")
    if set(labels) != set(ids):
        errors.append("oracle_identity_set_mismatch")

    raw_by_id = {row.get("id"): row for row in raw_rows}
    deltas = 0
    for row in rows:
        actual = raw_by_id.get(row.get("id"), {})
        for field, gate_enabled in (("gate_disabled", False), ("gate_enabled", True)):
            if actual.get(field) != reconstruct(row, contract, gate_enabled):
                errors.append(f"raw_decision_mismatch:{row.get('id')}:{field}")
        if actual.get("gate_disabled") != actual.get("gate_enabled"):
            deltas += 1

    rows_by_id = {row.get("id"): row for row in rows}
    hidden_harm_id = "hidden_modal_harmful_incomplete_coverage"
    hidden_safe_id = "hidden_modal_safe_incomplete_coverage"
    hidden_harm = rows_by_id.get(hidden_harm_id, {})
    hidden_safe = rows_by_id.get(hidden_safe_id, {})
    for key in ("surface_family", "covered_predicates", "observations"):
        if hidden_harm.get(key) != hidden_safe.get(key):
            errors.append("hidden_pair_candidate_input_differs")
            break
    if "modal_occlusion" in hidden_harm.get("covered_predicates", []):
        errors.append("hidden_predicate_marked_covered")
    if labels.get(hidden_harm_id) != "harmful" or labels.get(hidden_safe_id) != "safe":
        errors.append("hidden_pair_oracle_labels_mismatch")
    if raw_by_id.get(hidden_harm_id, {}).get("gate_disabled") != "ADMIT":
        errors.append("gate_disabled_harmful_admission_absent")
    if raw_by_id.get(hidden_harm_id, {}).get("gate_enabled") != "UNKNOWN":
        errors.append("gate_enabled_hidden_harm_not_unknown")
    if raw_by_id.get(hidden_safe_id, {}).get("gate_enabled") != "UNKNOWN":
        errors.append("gate_enabled_indistinguishable_safe_not_unknown")

    expected_labels = {
        "valid_complete_coverage": "safe",
        "known_stale_complete_coverage": "harmful",
        hidden_harm_id: "harmful",
        hidden_safe_id: "safe",
        "unregistered_surface_family": "unknown",
    }
    if labels != expected_labels:
        errors.append("oracle_label_contract_mismatch")
    return errors, deltas


def inspect_boundary(package, freeze):
    errors = []
    root = package / "results" / "formal_01"
    for role in ("candidate", "auditor"):
        pre = read_json(root / role / "container_inspect_pre.json")[0]
        post = read_json(root / role / "container_inspect_post.json")[0]
        if pre.get("Image") != freeze.get("image", {}).get("image_id"):
            errors.append(f"{role}_pre_image_mismatch")
        if post.get("Image") != freeze.get("image", {}).get("image_id"):
            errors.append(f"{role}_post_image_mismatch")
        config = pre.get("HostConfig", {})
        if config.get("NetworkMode") != "none":
            errors.append(f"{role}_network_not_disabled")
        if config.get("ReadonlyRootfs") is not True:
            errors.append(f"{role}_rootfs_not_readonly")
        if config.get("CapDrop") != ["ALL"]:
            errors.append(f"{role}_capabilities_not_dropped")
        if pre.get("Config", {}).get("User") != "65534:65534":
            errors.append(f"{role}_user_not_unprivileged")
        state = post.get("State", {})
        if state.get("Status") != "exited" or state.get("ExitCode") != 0 or state.get("OOMKilled") is not False:
            errors.append(f"{role}_formal_exit_receipt_invalid")
        if post.get("RestartCount") != 0:
            errors.append(f"{role}_restart_count_nonzero")

        mounts = config.get("Mounts", [])
        if role == "candidate":
            if len(mounts) != 1 or mounts[0].get("ReadOnly") is not True:
                errors.append("candidate_mount_not_single_readonly")
            elif not mounts[0].get("Source", "").replace("\\", "/").endswith("/candidate_input"):
                errors.append("candidate_mount_not_candidate_input")
            command = pre.get("Config", {}).get("Cmd", [])
            if any("oracle" in item.lower() for item in command):
                errors.append("candidate_command_references_oracle")
            if any("oracle" in mount.get("Source", "").lower() for mount in mounts):
                errors.append("candidate_mount_references_oracle")
        elif not mounts or any(mount.get("ReadOnly") is not True for mount in mounts):
            errors.append("auditor_mount_missing_or_writable")
    return errors


def audit_package(package):
    package = Path(package).resolve()
    entries, errors = verify_manifest(package)
    freeze = read_json(package / "FREEZE.json")
    fixture = read_json(package / "candidate_input" / "candidate_fixture.json")
    contract = read_json(package / "candidate_input" / "contract.json")
    oracle = read_json(package / "oracle.json")
    raw = read_json(package / "results" / "formal_01" / "candidate" / "raw.json")
    record_errors, deltas = audit_records(fixture, contract, oracle, raw)
    errors.extend(record_errors)
    errors.extend(inspect_boundary(package, freeze))
    fields = decision_field_reads(package / "candidate_input" / "candidate.py")
    if "id" in fields:
        errors.append("candidate_decision_consumes_row_id")
    label_bearing_ids = any(
        any(term in row.get("id", "").lower() for term in ("harmful", "safe"))
        for row in fixture.get("rows", [])
    )
    return {
        "status": "PASS_INDEPENDENT_REVALIDATION" if not errors else "FAIL_INDEPENDENT_REVALIDATION",
        "package_main_sha": "3729d6825e67b7276b58d40ef6c125e4a3b441bf",
        "manifest_files_verified": len(entries),
        "rows_reconstructed": len(fixture.get("rows", [])),
        "gate_delta_rows": deltas,
        "candidate_decision_fields": fields,
        "candidate_visible_ids_contain_outcome_terms": label_bearing_ids,
        "candidate_decision_reads_id": "id" in fields,
        "errors": errors,
    }


def probe_label_relabel(package):
    """Run the frozen candidate once with outcome-bearing IDs replaced by neutral IDs."""
    package = Path(package).resolve()
    fixture = read_json(package / "candidate_input" / "candidate_fixture.json")
    contract_path = package / "candidate_input" / "contract.json"
    contract = read_json(contract_path)
    baseline = read_json(package / "results" / "formal_01" / "candidate" / "raw.json")
    original_rows = fixture.get("rows", [])
    replacement_ids = [f"case_{index:02d}" for index in range(1, len(original_rows) + 1)]
    mutated = copy.deepcopy(fixture)
    original_to_replacement = {}
    errors = []

    for row, replacement in zip(mutated.get("rows", []), replacement_ids):
        original_to_replacement[row.get("id")] = replacement
        row["id"] = replacement

    def without_row_ids(payload):
        normalized = copy.deepcopy(payload)
        for row in normalized.get("rows", []):
            row.pop("id", None)
        return normalized

    if without_row_ids(fixture) != without_row_ids(mutated):
        errors.append("label_probe_changed_non_id_fixture_input")
    if len(set(replacement_ids)) != len(original_rows):
        errors.append("label_probe_replacement_ids_not_unique")
    if any(term in item.lower() for item in replacement_ids for term in ("harmful", "safe")):
        errors.append("label_probe_replacement_id_contains_outcome_term")

    candidate = package / "candidate_input" / "candidate.py"
    contract_bytes = contract_path.read_bytes()
    candidate_bytes = candidate.read_bytes()
    invocation_count = 0
    candidate_stdout = ""
    candidate_raw = None
    matched_decisions = 0
    candidate_payload_files = []
    candidate_exit_code = None
    candidate_stderr = ""
    if not errors:
        try:
            with tempfile.TemporaryDirectory(prefix="t1b-label-relabel-") as temp_dir:
                temp_path = Path(temp_dir)
                candidate_path = temp_path / "candidate.py"
                fixture_path = temp_path / "candidate_fixture_relabelled.json"
                temp_contract_path = temp_path / "contract.json"
                candidate_path.write_bytes(candidate_bytes)
                fixture_bytes = json.dumps(
                    mutated, ensure_ascii=False, separators=(",", ":")
                ).encode("utf-8")
                fixture_path.write_bytes(fixture_bytes)
                temp_contract_path.write_bytes(contract_bytes)
                candidate_payload_files = sorted(path.name for path in temp_path.iterdir())
                if candidate_payload_files != [
                    "candidate.py",
                    "candidate_fixture_relabelled.json",
                    "contract.json",
                ]:
                    errors.append("label_probe_temporary_payload_not_closed")
                else:
                    invocation_count += 1
                    completed = subprocess.run(
                        [sys.executable, str(candidate_path), str(fixture_path), str(temp_contract_path)],
                        capture_output=True,
                        text=True,
                        check=False,
                        timeout=30,
                        cwd=temp_path,
                    )
                    candidate_stdout = completed.stdout
                    candidate_stderr = completed.stderr
                    candidate_exit_code = completed.returncode
                    if completed.returncode != 0:
                        errors.append(f"label_probe_candidate_exit:{completed.returncode}")
                    else:
                        observed = json.loads(completed.stdout)
                        candidate_raw = observed
                        baseline_by_id = {row.get("id"): row for row in baseline.get("rows", [])}
                        observed_rows = observed.get("rows", [])
                        if observed.get("fixture_id") != fixture.get("fixture_id"):
                            errors.append("label_probe_fixture_identity_mismatch")
                        if observed.get("contract_id") != contract.get("contract_id"):
                            errors.append("label_probe_contract_identity_mismatch")
                        if len(observed_rows) != len(original_rows):
                            errors.append("label_probe_row_count_mismatch")
                        for original, output in zip(original_rows, observed_rows):
                            original_id = original.get("id")
                            if output.get("id") != original_to_replacement.get(original_id):
                                errors.append(f"label_probe_row_id_mismatch:{original_id}")
                                continue
                            frozen = baseline_by_id.get(original_id, {})
                            if any(
                                output.get(field) != frozen.get(field)
                                for field in ("gate_disabled", "gate_enabled")
                            ):
                                errors.append(f"label_probe_decision_changed:{original_id}")
                            else:
                                matched_decisions += 1
        except (OSError, subprocess.TimeoutExpired, json.JSONDecodeError) as error:
            errors.append(f"label_probe_execution_error:{type(error).__name__}")

    return {
        "status": "PASS_LABEL_INVARIANCE" if not errors else "FAIL_LABEL_INVARIANCE",
        "candidate_invocations": invocation_count,
        "rows_checked": len(original_rows),
        "decisions_matching_frozen_raw": matched_decisions,
        "row_ids_replaced_with_neutral_labels": replacement_ids,
        "only_row_ids_changed": "label_probe_changed_non_id_fixture_input" not in errors,
        "temporary_candidate_payload_files": candidate_payload_files,
        "candidate_source_sha256": hashlib.sha256(candidate_bytes).hexdigest(),
        "candidate_contract_sha256": hashlib.sha256(contract_bytes).hexdigest(),
        "candidate_fixture_sha256": hashlib.sha256(
            json.dumps(mutated, ensure_ascii=False, separators=(",", ":")).encode("utf-8")
        ).hexdigest(),
        "candidate_fixture": mutated,
        "candidate_stdout_sha256": hashlib.sha256(candidate_stdout.encode("utf-8")).hexdigest(),
        "candidate_exit_code": candidate_exit_code,
        "candidate_stderr": candidate_stderr,
        "candidate_raw": candidate_raw,
        "candidate_stdout": candidate_stdout,
        "errors": errors,
    }


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--package", required=True, type=Path)
    parser.add_argument("--probe-label-relabel", action="store_true")
    args = parser.parse_args()
    result = audit_package(args.package)
    if args.probe_label_relabel:
        result["label_relabel_probe"] = (
            probe_label_relabel(args.package)
            if result["status"] == "PASS_INDEPENDENT_REVALIDATION"
            else {
                "status": "NOT_RUN_BASE_AUDIT_FAILED",
                "candidate_invocations": 0,
                "rows_checked": 0,
                "errors": ["base_independent_audit_failed"],
            }
        )
    rendered = json.dumps(result, sort_keys=True, separators=(",", ":"))
    output_dir = os.environ.get("T1B_REVALIDATION_OUTPUT_DIR")
    if output_dir:
        output_path = Path(output_dir)
        output_path.mkdir(parents=True, exist_ok=True)
        probe = result.get("label_relabel_probe", {})
        (output_path / "candidate_stdout.txt").write_text(
            probe.get("candidate_stdout", ""), encoding="utf-8"
        )
        (output_path / "candidate_stderr.txt").write_text(
            probe.get("candidate_stderr", ""), encoding="utf-8"
        )
        (output_path / "candidate_exit_code.txt").write_text(
            "" if probe.get("candidate_exit_code") is None else f"{probe['candidate_exit_code']}\n",
            encoding="utf-8",
        )
        (output_path / "audit.json").write_text(rendered + "\n", encoding="utf-8")
    print(rendered)
    probe_ok = (
        not args.probe_label_relabel
        or result["label_relabel_probe"]["status"] == "PASS_LABEL_INVARIANCE"
    )
    raise SystemExit(0 if result["status"] == "PASS_INDEPENDENT_REVALIDATION" and probe_ok else 1)


if __name__ == "__main__":
    main()
