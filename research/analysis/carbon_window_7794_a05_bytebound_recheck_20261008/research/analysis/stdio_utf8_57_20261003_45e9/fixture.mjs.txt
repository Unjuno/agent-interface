/* Inert echo fixture: actual owned child/pipe, no MCP/SDK/native backend. */
import {createInterface} from 'node:readline';
import {writeFileSync} from 'node:fs';
import {join} from 'node:path';
const [dir,nonce]=process.argv.slice(2);let count=0,total=0;
const put=(n,v)=>writeFileSync(join(dir,n),JSON.stringify(v)+'\n',{flag:'wx'});
put('fixture-start.json',{pid:process.pid,ppid:process.ppid,nonce,utc:new Date().toISOString()});
const guard=setTimeout(()=>{process.exitCode=77;process.stdin.destroy();},5000);
process.stdin.on('data',chunk=>{
 total+=chunk.length;
 writeFileSync(join(dir,`fixture-chunk-${++count}.hex.txt`),chunk.toString('hex')+'\n',{flag:'wx'});
});
let requests=0;
const lines=createInterface({input:process.stdin,terminal:false});
lines.on('line',line=>{
 const request=JSON.parse(line);requests++;
 put(`fixture-request-${requests}.json`,request);
 const result={content:[{type:'text',text:JSON.stringify({schema:'own-inert-echo-v1',label:request.arguments.label,actual_native_operations:0})}]};
 process.stdout.write(JSON.stringify({id:request.id,tool:request.tool,next_id:request.id+1,status:'returned',result})+'\n');
});
lines.once('close',()=>clearTimeout(guard));
process.on('exit',code=>put('fixture-exit.json',{pid:process.pid,ppid:process.ppid,nonce,utc:new Date().toISOString(),code,requests,request_bytes:total}));
