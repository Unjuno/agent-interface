"""Research-only cached permit/native boundary; public dispatch stays unchanged."""
import json, os, sys, time
from runtime.backends.x11_v1.backend import X11Backend
from runtime.backends.x11_v1.session import X11RuntimeSession

def emit(kind, **data):
    print(json.dumps(dict(kind=kind, ns=time.monotonic_ns(), pid=os.getpid(), **data)), flush=True)

class Owner(X11Backend):
    def key_state(self, key, down):
        if down:
            emit('positive_check', key=key, permitted=True)
            if key == 'F9' and config['paused']:
                emit('pause_entered', key=key)
                if sys.stdin.buffer.read(1) != b'R':
                    raise RuntimeError('missing bounded resume byte')
                emit('resume_received', key=key)
            emit('native_start', key=key)
            super().key_state(key, down)
            emit('native_end', key=key)
        else:
            super().key_state(key, down)

config=json.loads(sys.argv[1])
backend=Owner(config['display'], {'owned':config['window']})
emit('ready', identity=config['identity'])
if sys.stdin.buffer.read(1) != b'G':
    raise RuntimeError('missing start byte')
program=config['program']
program['authority']['expires_at_ns']=time.monotonic_ns()+4_000_000_000
emit('dispatch_start', program=program)
receipt=X11RuntimeSession(backend).dispatch(program,current_observation_seq=1,current_binding_revision=1)
emit('dispatch_return', receipt=receipt)
backend.close()
emit('owner_done')
