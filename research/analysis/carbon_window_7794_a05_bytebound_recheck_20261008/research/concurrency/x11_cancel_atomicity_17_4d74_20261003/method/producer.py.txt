"""Single first native allocation; all outcomes are raw, no science scorer here."""
from __future__ import annotations
import argparse, hashlib, json, os, pathlib, platform, subprocess, sys, threading, time, traceback
ROOT = pathlib.Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "sources"))
from Xlib import X, XK, display
from runtime.backends.x11_v1.session import X11RuntimeSession
from cancel_gate import CancelRaceBackend

def now():
    return time.monotonic_ns()

def dump(path, value):
    path.write_text(json.dumps(value, sort_keys=True, indent=2) + "\n", encoding="utf-8")

def pin_sources(plan):
    return {p: hashlib.sha256((ROOT/p).read_bytes()).hexdigest()
            for p in plan["source_hashes"]}

def physical(d):
    start = now()
    bits = d.query_keymap()
    keys = [code for code in range(256) if bits[code//8] & (1 << (code%8))]
    mask = d.screen().root.query_pointer().mask & (X.Button1Mask | X.Button2Mask | X.Button3Mask)
    return {"started_ns": start, "ended_ns": now(),
            "keys_down": keys, "buttons_mask": int(mask)}

def wait_until(deadline):
    while now() < deadline:
        time.sleep(min(.005, max(0, (deadline-now())/1e9)))

def run_cell(spec, plan, output, index):
    row = dict(spec)
    row.update(samples=[], events=[], errors=[])
    started = now()
    xvfb_log = output / (spec["id"] + ".xvfb.log")
    log = xvfb_log.open("wb")
    display_name = ":" + str(91 + index)
    xvfb = subprocess.Popen(["Xvfb", display_name, "-screen", "0", "320x240x24",
                             "-nolisten", "tcp", "-ac"], stdout=log, stderr=subprocess.STDOUT)
    row["Xvfb_pid"] = xvfb.pid
    row["Xvfb_argv"] = xvfb.args
    d = backend = None
    program_thread = cancel_thread = monitor_thread = None
    stop_monitor = threading.Event()
    ready = threading.Event()
    receipt_box, thread_errors = [], []
    def collect_error():
        thread_errors.append(traceback.format_exc())
    try:
        deadline = time.monotonic() + plan["constants"]["readiness_timeout_s"]
        while True:
            try:
                d = display.Display(display_name)
                break
            except Exception:
                if xvfb.poll() is not None or time.monotonic() >= deadline:
                    raise
                time.sleep(.01)
        root = d.screen().root
        window = root.create_window(10, 10, 120, 80, 0, d.screen().root_depth,
                                    X.InputOutput, X.CopyFromParent,
                                    background_pixel=d.screen().white_pixel,
                                    event_mask=X.KeyPressMask | X.KeyReleaseMask)
        window.map()
        d.change_keyboard_control(auto_repeat_mode=X.AutoRepeatModeOff)
        d.sync()
        row["window_id"] = window.id
        row["keycodes"] = {key: int(d.keysym_to_keycode(XK.string_to_keysym(key)))
                          for key in ("F8", "F9")}
        def monitor():
            try:
                ready.set()
                while not stop_monitor.is_set():
                    row["samples"].append(physical(d))
                    while d.pending_events():
                        event = d.next_event()
                        if event.type in (X.KeyPress, X.KeyRelease):
                            row["events"].append({
                                "type": "press" if event.type == X.KeyPress else "release",
                                "keycode": int(event.detail), "received_ns": now(),
                                "server_time_ms": int(event.time)})
                    time.sleep(plan["constants"]["sample_period_s"])
                d.sync()
                while d.pending_events():
                    event = d.next_event()
                    if event.type in (X.KeyPress, X.KeyRelease):
                        row["events"].append({
                            "type": "press" if event.type == X.KeyPress else "release",
                            "keycode": int(event.detail), "received_ns": now(),
                            "server_time_ms": int(event.time)})
            except Exception:
                collect_error()
        monitor_thread = threading.Thread(target=monitor, name="independent-X11-observer")
        monitor_thread.start()
        if not ready.wait(1):
            raise RuntimeError("observer never started")
        backend = CancelRaceBackend(display_name, {"owned": window.id},
                                    arm=spec["arm"], paused=spec["paused"])
        program = {"schema": "agent-interface/program-v1",
                   "program_id": "program-" + spec["id"],
                   "source": {"observation_seq": 1, "binding_revision": 1},
                   "authority": {"lease_id": "owned-" + spec["id"],
                                 "expires_at_ns": now() + 4_000_000_000},
                   "terminal": {"release_all_required": True},
                   "ops": plan["program_ops"]}
        row["program"] = program
        session = X11RuntimeSession(backend)
        def dispatch():
            try:
                row["dispatch_started_ns"] = now()
                receipt_box.append(session.dispatch(
                    program, current_observation_seq=1, current_binding_revision=1))
                row["dispatch_ended_ns"] = now()
            except Exception:
                collect_error()
        program_thread = threading.Thread(target=dispatch, name="owned-program")
        program_thread.start()
        if spec["paused"]:
            if not backend.pause_entered.wait(2):
                raise RuntimeError("positive F9 check/pause not reached")
            # Independent OS evidence must expose F8 down before cancellation.
            deadline = time.monotonic() + 1
            while not any(s["keys_down"] == [row["keycodes"]["F8"]] for s in row["samples"]):
                if time.monotonic() >= deadline:
                    raise RuntimeError("independent held-F8 exposure missing")
                time.sleep(.002)
            def cancel():
                try:
                    backend.cancel_program(program["program_id"])
                except Exception:
                    collect_error()
            cancel_thread = threading.Thread(target=cancel, name="same-owner-cancellation")
            cancel_thread.start()
            deadline = time.monotonic() + 1
            while backend.cancel_row is None:
                if time.monotonic() >= deadline:
                    raise RuntimeError("cancel request entry missing")
                time.sleep(.001)
            request = backend.cancel_row["request_ns"]
            wait_until(request + plan["constants"]["checkpoint_delay_ns"])
            # The observer exclusively owns its connection while running.
            query_deadline = time.monotonic() + .1
            while not any(s["started_ns"] >= request + plan["constants"]["checkpoint_delay_ns"]
                          for s in row["samples"]):
                if time.monotonic() >= query_deadline:
                    raise RuntimeError("fresh checkpoint query missing")
                time.sleep(.001)
            snapshot = dict(next(s for s in row["samples"]
                                 if s["started_ns"] >= request + plan["constants"]["checkpoint_delay_ns"]))
            row["checkpoint"] = {
                "at_ns": now(), "program_alive": program_thread.is_alive(),
                "canceller_alive": cancel_thread.is_alive(),
                "cancel_accepted": backend.cancelled.is_set(), "query": snapshot}
            wait_until(request + plan["constants"]["resume_delay_ns"])
            backend.resume.set()
        else:
            row["checkpoint"] = None
        program_thread.join(2)
        if cancel_thread is not None:
            cancel_thread.join(2)
        if program_thread.is_alive() or (cancel_thread is not None and cancel_thread.is_alive()):
            raise RuntimeError("owned thread failed retirement")
        if thread_errors:
            raise RuntimeError("owned thread failure")
        if (len(receipt_box) != 1 or not isinstance(receipt_box[0], dict)
                or receipt_box[0].get("status") != "completed"
                or receipt_box[0].get("admission") != "accepted"):
            raise RuntimeError("first public dispatch source/exposure outcome not completed")
        time.sleep(plan["constants"]["drain_after_dispatch_s"])
        stop_monitor.set()
        monitor_thread.join(1)
        if monitor_thread.is_alive():
            raise RuntimeError("observer failed retirement")
        row.update(program_receipt=receipt_box[0], checks=backend.checks,
                   native_calls=backend.native_calls, pause=backend.pause_row,
                   cancel=backend.cancel_row)
        final = physical(d)
        row["terminal"] = dict(keys_down=final["keys_down"],
                               buttons_mask=final["buttons_mask"], query=final,
                               threads_alive=[], Xvfb_exit=None)
    except Exception:
        row["errors"].append(traceback.format_exc())
    finally:
        if backend is not None:
            row.update(checks=backend.checks, native_calls=backend.native_calls,
                       pause=backend.pause_row, cancel=backend.cancel_row)
            if receipt_box:
                row["program_receipt"] = receipt_box[0]
            backend.resume.set()
        for t in (program_thread, cancel_thread):
            if t is not None:
                t.join(2)
        if backend is not None:
            row.update(checks=backend.checks, native_calls=backend.native_calls,
                       pause=backend.pause_row, cancel=backend.cancel_row)
            if receipt_box:
                row["program_receipt"] = receipt_box[0]
        if backend is not None and not any(t is not None and t.is_alive()
                                          for t in (program_thread, cancel_thread)):
            try:
                row["cleanup_receipt"] = backend.release_all()
                backend.close()
            except Exception:
                row["errors"].append(traceback.format_exc())
        stop_monitor.set()
        if monitor_thread is not None:
            monitor_thread.join(1)
        if d is not None and (monitor_thread is None or not monitor_thread.is_alive()):
            try:
                d.close()
            except Exception:
                row["errors"].append(traceback.format_exc())
        xvfb.terminate()
        try:
            row.setdefault("terminal", {})["Xvfb_exit"] = xvfb.wait(2)
        except subprocess.TimeoutExpired:
            xvfb.kill()
            row.setdefault("terminal", {})["Xvfb_exit"] = xvfb.wait(2)
            row["errors"].append("Xvfb required SIGKILL")
        row.setdefault("terminal", {})["threads_alive"] = [
            t.name for t in (program_thread, cancel_thread, monitor_thread)
            if t is not None and t.is_alive()]
        log.close()
        row["thread_errors"] = thread_errors
        row["cell_elapsed_ns"] = now() - started
        row["xvfb_log_sha256"] = hashlib.sha256(xvfb_log.read_bytes()).hexdigest()
    return row

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", required=True)
    args = parser.parse_args()
    output = pathlib.Path(args.out)
    output.mkdir(parents=True, exist_ok=True)
    raw_path = output / "raw.json"
    if raw_path.exists():
        raise RuntimeError("first outcome path already exists; refusing replay")
    plan = json.loads((ROOT/"PLAN.json").read_text(encoding="utf-8"))
    pins = pin_sources(plan)
    if pins != plan["source_hashes"]:
        raise RuntimeError("source custody mismatch")
    data = {"schema": plan["raw_schema"], "allocation": plan["allocation"],
            "source_hashes_before": pins, "source_hashes_after": None, "cells": [],
            "plan_sha256": hashlib.sha256((ROOT/"PLAN.json").read_bytes()).hexdigest(),
            "environment": {"python": sys.version, "platform": platform.platform(),
                            "pid": os.getpid(), "clock": str(time.get_clock_info("monotonic"))}}
    for cgroup in ("/sys/fs/cgroup/cpu.max", "/sys/fs/cgroup/memory.max",
                   "/sys/fs/cgroup/pids.max"):
        path = pathlib.Path(cgroup)
        data["environment"][cgroup] = path.read_text() if path.exists() else "absent"
    dump(raw_path, data)
    for index, spec in enumerate(plan["cells"]):
        cell = run_cell(spec, plan, output, index)
        data["cells"].append(cell)
        dump(raw_path, data)
        if cell["errors"] or cell["thread_errors"] or cell["terminal"]["threads_alive"]:
            data["first_disposition"] = "STOP_NATIVE_CELL"
            data["source_hashes_after"] = pin_sources(plan)
            dump(raw_path, data)
            print(json.dumps({"disposition": data["first_disposition"], "cells": len(data["cells"])}))
            return 2
    data["source_hashes_after"] = pin_sources(plan)
    data["first_disposition"] = "COLLECTED_NOT_SCORED"
    dump(raw_path, data)
    print(json.dumps({"disposition": data["first_disposition"], "cells": len(data["cells"])}))
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
