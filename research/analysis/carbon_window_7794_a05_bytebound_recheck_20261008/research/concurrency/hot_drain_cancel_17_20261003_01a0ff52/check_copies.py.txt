"""Ordinary synthetic/raw-only negative controls; never replay the producer."""
import copy
import json
import pathlib
import runpy
import sys

root = pathlib.Path(__file__).resolve().parent
audit = runpy.run_path(str(root/'audit.py.txt'),run_name='raw_only_import')
inspect = audit['inspect']


def synthetic():
    spec = dict(id=6,profile='co_ready',budget=0,cancel_after=0)
    events = []
    def event(tag, **fields):
        events.append(dict(seq=len(events),t=len(events),tag=tag,**fields))
    event('open',fds={r:dict(fd=n,inode=100 if r.startswith('D') else 101) for r,n in [('DR',3),('DW',4),('CR',5),('CW',6)]},selector_fd=7)
    event('write',role='D',data='44',wrote=1,cause_read=0)
    event('write',role='C',data='43',wrote=1,cause_read=0)
    event('poll',cycle=1,ready=[dict(role='D',fd=3,mask=1,inode=100),dict(role='C',fd=5,mask=1,inode=101)])
    event('read',role='C',data='43',cycle=1,count=0)
    event('terminal',kind='CANCEL_ACK',read_count=0,cycles=1)
    event('cleanup_empty',role='C')
    event('cleanup_read',role='D',data='44')
    event('cleanup_empty',role='D')
    for role,fd in [('SELECTOR',7),('DR',3),('DW',4),('CR',5),('CW',6)]:
        event('closed',role=role,fd=fd,err=9)
    return dict(spec=spec,events=events), spec


def altered(record, kind):
    result = copy.deepcopy(record)
    events = result['events']
    if kind == 'float_spec':result['spec']['id'] = float(result['spec']['id'])
    elif kind == 'boolean_seq':events[0]['seq'] = False
    elif kind == 'false_ack_byte':next(e for e in events if e['tag']=='read' and e['role']=='C')['data']='44'
    elif kind == 'wrong_poll_fd':next(e for e in events if e['tag']=='poll')['ready'][0]['fd']=999
    elif kind == 'wrong_poll_inode':next(e for e in events if e['tag']=='poll')['ready'][0]['inode']=999
    elif kind == 'wrong_cancel_cause':next(e for e in events if e['tag']=='write' and e['role']=='C')['cause_read']+=1
    elif kind == 'missing_close':events.pop()
    elif kind == 'float_closed_error':events[-1]['err']=9.0
    elif kind == 'false_cleanup_byte':next(e for e in events if e['tag']=='cleanup_read')['data']='43'
    elif kind == 'terminal_before_ack':
        a=next(i for i,e in enumerate(events) if e['tag']=='read' and e['role']=='C')
        b=next(i for i,e in enumerate(events) if e['tag']=='terminal')
        events[a],events[b]=events[b],events[a]
        for i,e in enumerate(events):e['seq']=i;e['t']=i
    else:raise ValueError(kind)
    return result


if __name__ == '__main__':
    if len(sys.argv) == 1:
        record,spec=synthetic()
        classification='ordinary synthetic construction; no native experiment'
    else:
        records=[json.loads(line) for line in pathlib.Path(sys.argv[1]).read_text().splitlines()]
        record=next(row for row in records if row['spec']['profile']=='co_ready' and row['spec']['budget']==0)
        spec=json.loads((root/'input.json').read_text())[record['spec']['id']]
        classification='copied retained records; no producer replay'
    baseline=inspect(record,spec)
    controls=[]
    for kind in ['float_spec','boolean_seq','false_ack_byte','wrong_poll_fd','wrong_poll_inode','wrong_cancel_cause','missing_close','float_closed_error','false_cleanup_byte','terminal_before_ack']:
        changed=altered(record,kind)
        try:inspect(changed,spec)
        except (ValueError,KeyError,TypeError) as exc:controls.append(dict(kind=kind,rejected=True,reason=str(exc)))
        else:raise RuntimeError('control admitted: '+kind)
    print(json.dumps(dict(classification=classification,baseline=baseline,controls=controls),indent=2))
