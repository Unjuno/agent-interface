def require(v,label):

def load_deck():
    deck={}
    for arm in ('red-first','green-first'):
        deck[arm]=[]

def verify(deck):
    coverage={}
    for arm in ('red-first','green-first'):
        require(len(deck[arm])==5,'five distinct endpoint directories')
        require(len({r['directory'] for r in deck[arm]})==5,'unique directories')

        for row in deck[arm]:

            require(type(request['id']) is int and request['id']==1,'typed request identity')
            require(request['arguments']=={} and type(request['arguments']) is dict,'unchanged empty arguments')
            require(len([k for k in files if k.startswith('request-')])==1,'no retry attempt')
            terminal=files.get('exit.json')
            tool=request['tool']
            if tool=='invalid-json':
                require(reply=='{broken\n','original invalid JSON bytes');label='invalid-json'
            elif tool=='valid' and reply==sentinel:label='occupied-reply'
            else:
                parsed=json.loads(reply)
                require(type(parsed['id']) is int and type(parsed['next_id']) is int,'typed reply counters')
                require(parsed['tool']==tool and parsed['status']=='returned' and parsed['next_id']==2
                    and parsed['result']=={'content':[]},'complete retained reply identity/result')
                if tool=='wrong-id':
                    require(parsed['id']==2,'original wrong ID preserved')
                    label='occupied-exit' if terminal==sentinel else 'wrong-id'
                else:
                    require(tool=='valid' and parsed['id']==1,'valid positive reply')
                    require(terminal==sentinel,'healthy occupied exit remains original bytes');label='healthy-occupied-exit'
            labels.append(label)
            if label in ('occupied-exit','healthy-occupied-exit'):
                require(terminal==sentinel,'existing terminal bytes unchanged')
            elif arm=='red-first':require(terminal is None,'first missing terminal outcome retained')
            else:
                t=json.loads(terminal)
                require(set(t)=={'code','signal','journal_error'},'complete new terminal shape')
                require(type(t['code']) is int and t['code']==0 and t['signal'] is None,'typed actual terminal')
                require(type(t['journal_error']) is str and bool(t['journal_error']),'first fault retained')
                expected={'wrong-id':'identity/status mismatch','invalid-json':'SyntaxError','occupied-reply':'EEXIST'}[label]
                require(expected in t['journal_error'],'correct original fault class')
            pre=json.loads(files['storage-preflight.json'])
            require(pre['child_started'] is False and pre['capacity_reserved'] is False and pre['future_writes_guaranteed'] is False,'capacity remains scoped')
        require(sorted(labels)==sorted(['wrong-id','invalid-json','occupied-reply','occupied-exit','healthy-occupied-exit']),'complete five-case coverage')
        coverage[arm]=labels
