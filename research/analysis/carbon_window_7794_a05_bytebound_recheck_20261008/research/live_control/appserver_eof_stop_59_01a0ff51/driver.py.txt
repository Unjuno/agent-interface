"""One fixed ordinary native comparison arm, preserving all eight case rows."""
import hashlib
import json
import locale
import os
from pathlib import Path
import platform
import sys
import threading
import time
import traceback
import types

HERE = Path(__file__).resolve().parent
arm, output_text = sys.argv[1:]
out = Path(output_text)
out.mkdir(parents=True, exist_ok=False)
source = HERE / (arm + '_client.py.txt')
client_module = types.ModuleType('fixed_' + arm)
exec(compile(source.read_bytes(), str(source), 'exec'), client_module.__dict__)
plan = json.loads((HERE / 'PLAN.json').read_text())
if hashlib.sha256(source.read_bytes()).hexdigest() != plan['source_sha256'][arm]:
    raise RuntimeError('source pin mismatch before native child')
rows = []


def wait_until(predicate, seconds=2):
    end = time.monotonic() + seconds
    while not predicate():
        if time.monotonic() >= end:
            raise RuntimeError('construction handshake timed out')
        time.sleep(0.001)


def frame_snapshot(thread):
    frame = sys._current_frames().get(thread.ident)
    stack = []
    while frame is not None:
        row = {'function': frame.f_code.co_name, 'line': frame.f_lineno,
               'file': Path(frame.f_code.co_filename).name}
        if frame.f_code.co_filename == str(source):
            row['deadline'] = frame.f_locals.get('deadline')
            row['remaining'] = frame.f_locals.get('remaining')
            row['snapshot_monotonic'] = time.monotonic()
        stack.append(row)
        frame = frame.f_back
    return stack


for api in plan['apis']:
    for scenario in plan['scenarios']:
        identity = api + '-' + scenario
        case = out / identity
        case.mkdir()
        journal = case / 'journal.jsonl'
        command = [sys.executable, '-B', str(HERE / 'peer.py.txt'), api, scenario, str(case)]
        client = client_module.CodexAppServerClient(command, journal_path=str(journal))
        row = {'identity': identity, 'api': api, 'scenario': scenario,
               'pid': client.process.pid, 'command': command, 'events': [], 'returned': None}

        def event(kind):
            row['events'].append({'kind': kind, 'ns': time.monotonic_ns()})

        def invoke():
            event('call_started')
            try:
                value = (client.request('fixture/request', {'owned': True}, timeout=plan['timeout_s'])
                         if api == 'request' else client.wait_notification(
                             lambda message: message.get('method') == 'fixture/ready',
                             timeout=plan['timeout_s']))
                row['returned'] = {'type': 'result', 'value': value}
            except BaseException as error:
                row['returned'] = {'type': type(error).__name__, 'message': str(error)}
                row['traceback'] = traceback.format_exc()
            finally:
                event('call_ended')

        try:
            wait_until(lambda: (case / 'ready.json').exists())
            row['ready'] = json.loads((case / 'ready.json').read_text())
            event('child_ready')
            caller = threading.Thread(target=invoke, daemon=True)
            caller.start()
            if api == 'notification':
                (case / 'go').write_text('owned fixture start\n')
            wait_until(lambda: client._closed)
            event('stdout_eof_observed')
            checkpoint = time.monotonic() + plan['checkpoint_after_eof_s']
            while time.monotonic() < checkpoint:
                time.sleep(max(0, min(0.002, checkpoint - time.monotonic())))
            event('checkpoint')
            row['checkpoint'] = {'caller_alive': caller.is_alive(),
                                 'process_exit': client.process.poll(),
                                 'reader_alive': client._reader.is_alive(),
                                 'stdout_closed': client._closed,
                                 'frames': frame_snapshot(caller),
                                 'returned': row['returned']}
        finally:
            if scenario == 'held':
                (case / 'release').write_text('owned fixture recovery only\n')
                event('owned_release_sent')
            try:
                row['native_exit'] = client.process.wait(timeout=2)
            except Exception:
                client.process.kill()
                row['native_exit'] = client.process.wait(timeout=2)
                row['forced_kill'] = True
            event('child_reaped')
            if 'caller' in locals():
                caller.join(timeout=1)
                row['caller_alive_final'] = caller.is_alive()
            client.close(timeout=0.2)
            event('client_close_returned')
            row['client_close_pipes'] = {name: getattr(client.process, name).closed
                                         for name in ('stdin', 'stdout', 'stderr')}
            row['stderr_remaining_hex'] = client.process.stderr.buffer.read().hex()
            for name in ('stdin', 'stdout', 'stderr'):
                getattr(client.process, name).close()
            event('driver_pipes_closed')
            row['final'] = {'caller_alive': caller.is_alive(), 'reader_alive': client._reader.is_alive(),
                            'journal_closed': client._journal.closed,
                            'pipes_closed': {name: getattr(client.process, name).closed
                                             for name in ('stdin', 'stdout', 'stderr')}}
            for name in ('wire', 'finished'):
                row[name] = json.loads((case / (name + '.json')).read_text())
            row['journal_sha256'] = hashlib.sha256(journal.read_bytes()).hexdigest()
            row['journal'] = [json.loads(line) for line in journal.read_text().splitlines()]
            (case / 'row.json').write_text(json.dumps(row, indent=2, sort_keys=True) + '\n')
            rows.append(row)

result = {'arm': arm, 'source_sha256': hashlib.sha256(source.read_bytes()).hexdigest(),
          'plan_sha256': hashlib.sha256((HERE / 'PLAN.json').read_bytes()).hexdigest(),
          'python': sys.version, 'platform': platform.platform(),
          'preferred_encoding': locale.getpreferredencoding(False), 'pid': os.getpid(), 'rows': rows}
(out / 'raw.json').write_text(json.dumps(result, indent=2, sort_keys=True) + '\n')
print(json.dumps({'arm': arm, 'rows': len(rows), 'returns': [row['returned']['type'] for row in rows]}))
