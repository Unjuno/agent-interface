import datetime
import hashlib
import json
from pathlib import Path
import subprocess

ROOT = Path(__file__).resolve().parent
LEAF = ROOT.parent / 'agent-interface-eof-59/research/live_control/appserver_eof_stop_59_01a0ff51'
VM = 'research-clipboard-36-01a0ff51-docker'
IMAGE = 'sha256:65ae56ebdddc2742bc1441ee9eca0fcca7e232cb611a09ea8b177b36fc23c291'


def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


plan = json.loads((LEAF / 'PLAN.json').read_text())
for name, expected in plan['file_sha256'].items():
    if hashlib.sha256((LEAF / name).read_bytes()).hexdigest() != expected:
        raise RuntimeError('frozen source changed before first transmission: ' + name)
for arm in ('baseline', 'candidate'):
    run = ROOT / ('capture-' + arm)
    run.mkdir(exist_ok=False)
    if (ROOT / ('native-' + arm)).exists():
        raise RuntimeError('no replay of existing native output')
    name = 'eof-stop-59-01a0ff51-' + arm + '-01'
    command = ['orb', '-m', VM, '-u', 'root', 'docker', 'run', '--name', name,
               '--pull', 'never', '--network', 'none', '--read-only', '--memory', '512m',
               '--cpus', '0.5', '--pids-limit', '64', '--tmpfs', '/tmp:rw,size=16m',
               '--mount', 'type=bind,src=' + str(LEAF) + ',dst=/src,readonly',
               '--mount', 'type=bind,src=' + str(ROOT) + ',dst=/out',
               '--workdir', '/tmp', IMAGE, 'timeout', '25s', 'python3', '-B',
               '/src/driver.py.txt', arm, '/out/native-' + arm]
    with (run / 'PRE_SEND.json').open('x') as handle:
        handle.write(json.dumps({'utc': utc(), 'command': command,
                  'plan_sha256': hashlib.sha256((LEAF / 'PLAN.json').read_bytes()).hexdigest()}, indent=2) + '\n')
    started = utc()
    process = subprocess.Popen(command, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    with (run / 'CALLER.json').open('x') as handle:
        handle.write(json.dumps({'pid': process.pid, 'started_utc': started}) + '\n')
    stdout, stderr = process.communicate(timeout=40)
    (run / 'stdout.txt').write_bytes(stdout)
    (run / 'stderr.txt').write_bytes(stderr)
    (run / 'RECEIPT.json').write_text(json.dumps({'started_utc': started, 'ended_utc': utc(),
             'native_wrapper_exit': process.returncode, 'stdout_sha256': hashlib.sha256(stdout).hexdigest(),
             'stderr_sha256': hashlib.sha256(stderr).hexdigest()}, indent=2) + '\n')
    inspected = subprocess.run(['orb', '-m', VM, '-u', 'root', 'docker', 'inspect', name],
                               capture_output=True, check=True)
    (run / 'container-inspect.json').write_bytes(inspected.stdout)
    details = json.loads(inspected.stdout)[0]
    print(json.dumps({'arm': arm, 'wrapper_exit': process.returncode,
                      'container_exit': details['State']['ExitCode'], 'running': details['State']['Running'],
                      'stdout': stdout.decode(), 'stderr': stderr.decode()}), flush=True)
    if process.returncode != 0:
        raise SystemExit(process.returncode)
