from pathlib import Path
import hashlib,json,subprocess,sys
from datetime import datetime,timezone
ROOT=Path(__file__).resolve().parent
def sha(data):return hashlib.sha256(data).hexdigest()
def now():return datetime.now(timezone.utc).isoformat()
if (ROOT/'EXECUTION.json').exists() or (ROOT/'cells').exists():raise RuntimeError('never replay first native cells')
argv=[sys.executable,'-I','-B',str(ROOT/'driver.py')];started=now()
result=subprocess.run(argv,cwd=ROOT,capture_output=True)
(ROOT/'producer.stdout').write_bytes(result.stdout);(ROOT/'producer.stderr').write_bytes(result.stderr)
receipt={'argv':argv,'started_utc':started,'ended_utc':now(),'exit_code':result.returncode,'stdout_bytes':len(result.stdout),'stderr_bytes':len(result.stderr),'stdout_sha256':sha(result.stdout),'stderr_sha256':sha(result.stderr),'source_runtime_modified':False,'formal_allocation':False,'main_request_sent':False}
(ROOT/'EXECUTION.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8');print(json.dumps(receipt));raise SystemExit(result.returncode)
