"""Single ordinary construction against the unchanged client snapshot."""
from pathlib import Path
import importlib.util, json, os, sys, threading, time, traceback

ROOT = Path(__file__).resolve().parent
cell_id, api, mode, out_path = sys.argv[1:]
OUT = Path(out_path)
OUT.mkdir()
spec = importlib.util.spec_from_file_location('exact_client', ROOT / 'codex_app_server_client_v2.py')
module = importlib.util.module_from_spec(spec)
spec.loader.exec_module(module)
plan = json.loads((ROOT / 'PLAN.json').read_text())
limits = plan['limits']
events = OUT / 'child-events.jsonl'
client = module.CodexAppServerClient([sys.executable, '-B', str(ROOT / 'inert_child.py'),
               mode, api, str(events)], journal_path=OUT / 'journal.jsonl')
result = {'id': cell_id, 'api': api, 'mode': mode, 'probe_pid': os.getpid(),
          'child_pid': client.process.pid, 'source_sha256': plan['source']['sha256'],
          'public_timeout_s': limits['public_timeout_s'], 'checkpoint_s': limits['checkpoint_s']}
call = {}
caller = None

def child_rows():
    if not events.exists():
        return []
    data = events.read_text()
    # An incomplete final line is still a pending write, not a completed record.
    return [json.loads(line) for line in data.splitlines(keepends=True) if line.endswith('\n')]

def invoke():
    call['started_ns'] = time.monotonic_ns()
    try:
        if api == 'request':
            call['value'] = client.request('owned/probe', timeout=limits['public_timeout_s'])
        else:
            call['value'] = client.wait_notification(lambda r: r.get('method') == 'owned/probe',
                                                   timeout=limits['public_timeout_s'])
        call['outcome'] = 'RETURN'
    except BaseException as error:
        call['outcome'] = 'RAISE'
        call['exception'] = {'type': type(error).__name__, 'message': str(error)}
    finally:
        call['ended_ns'] = time.monotonic_ns()

try:
    ready_end = time.monotonic() + limits['setup_ready_s']
    while not any(r['event'] == 'ready' for r in child_rows()):
        if time.monotonic() >= ready_end:
            raise TimeoutError('owned inert child ready deadline')
        time.sleep(.002)
    if mode != 'healthy':
        client._reader.join(limits['setup_ready_s'])
        if client._reader.is_alive() or client._closed is not True:
            raise RuntimeError('expected completed stdout reader before invocation')
    result['before_call'] = {'at_ns': time.monotonic_ns(), 'transport_closed': client._closed,
                            'reader_alive': client._reader.is_alive(), 'child_exit': client.process.poll()}
    caller = threading.Thread(target=invoke, name='owned-probe-caller', daemon=True)
    caller.start()
    caller.join(limits['checkpoint_s'])
    local_deadline = None
    frame = sys._current_frames().get(caller.ident)
    stack = []
    while frame is not None:
        if Path(frame.f_code.co_filename).name == 'codex_app_server_client_v2.py' and frame.f_code.co_name in ('request', 'wait_notification'):
            local_deadline = {'deadline_s': frame.f_locals.get('deadline'),
                              'remaining_s_at_read_entry': frame.f_locals.get('remaining'),
                              'observed_monotonic_s': time.monotonic()}
        stack.append({'file': Path(frame.f_code.co_filename).name,
                      'function': frame.f_code.co_name, 'line': frame.f_lineno})
        frame = frame.f_back
    result['checkpoint'] = {'at_ns': time.monotonic_ns(), 'caller_alive': caller.is_alive(),
                            'call_record': dict(call), 'stack': stack, 'local_deadline': local_deadline,
                            'transport_closed': client._closed,
                            'reader_alive': client._reader.is_alive(), 'child_exit': client.process.poll(),
                            'child_events': child_rows()}
except BaseException as error:
    result['construction_error'] = {'type': type(error).__name__, 'message': str(error),
                                    'traceback': traceback.format_exc()}
finally:
    result['cleanup_started_ns'] = time.monotonic_ns()
    try:
        client.close(timeout=limits['owned_client_close_timeout_s'])
        result['client_close_returned_ns'] = time.monotonic_ns()
    except BaseException as error:
        result['cleanup_error'] = {'type': type(error).__name__, 'message': str(error)}
    if caller is not None:
        caller.join(limits['caller_final_join_s'])
    result['final'] = {'at_ns': time.monotonic_ns(), 'child_exit': client.process.poll(),
                       'reader_alive': client._reader.is_alive(), 'caller_alive': caller is not None and caller.is_alive(),
                       'journal_closed': client._journal.closed, 'call_record': dict(call),
                       'child_events': child_rows()}
    # Pipes are closed only after their actual reader/caller has stopped.
    if not result['final']['reader_alive'] and not result['final']['caller_alive']:
        for stream in [client.process.stdin, client.process.stdout, client.process.stderr]:
            stream.close()
        result['final']['pipes_closed'] = True
    else:
        result['final']['pipes_closed'] = False
    (OUT / 'result.json').write_text(json.dumps(result, indent=2) + '\n')
print(json.dumps(result))
raise SystemExit(bool(result.get('construction_error') or result.get('cleanup_error') or
                      result['final']['reader_alive'] or result['final']['caller_alive'] or
                      result['final']['child_exit'] is None))
