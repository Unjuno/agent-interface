import json,sys,time
from pathlib import Path
mode,events=sys.argv[1:]; events=Path(events)
def event(kind,**kw):
 with events.open('a') as f: f.write(json.dumps({'kind':kind,'peer_ns':time.perf_counter_ns(),**kw},sort_keys=True)+'\n')
event('ready',mode=mode)
print(json.dumps({'method':'fixture/ready','params':{'mode':mode}}),flush=True)
if mode=='healthy':
 line=sys.stdin.buffer.readline();event('read',bytes=len(line),wire_hex=line.hex())
 request=json.loads(line);reply={'id':request['id'],'result':{'label':'inert','padding':request['params']['padding']}}
 print(json.dumps(reply,separators=(',',':')),flush=True);event('reply',message=reply)
while True: time.sleep(1)
