import hashlib, json, os, subprocess, sys
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parent.parent
private = root / 'work/repairs/appserver-journal-close-01'
out = root / 'outputs/appserver-journal-close-01'
stage = sys.argv[1]
assert stage in ('red', 'red-corrected', 'green-normal', 'green-optimized', 'compatibility-red', 'final-normal', 'final-optimized')
target = out / stage
target.mkdir(exist_ok=False)
cases = target / 'cases'
cases.mkdir()
source = (private / 'codex_app_server_client_v2.py').read_bytes()
tests = (private / 'test_appserver_journal_close_01a0ff2d.py').read_bytes()
freeze_name = ('FREEZE_BEFORE_RED_V3.json' if stage in ('compatibility-red', 'final-normal', 'final-optimized') else
               'FREEZE_BEFORE_RED.json' if stage == 'red' else 'FREEZE_BEFORE_RED_V2.json')
freeze = json.loads((out / freeze_name).read_bytes())
assert hashlib.sha256(tests).hexdigest() == freeze['test_sha256']
(target / 'executed-client.py.txt').write_bytes(source)
(target / 'executed-test.py.txt').write_bytes(tests)
selected = ([
    'test_appserver_journal_close_01a0ff2d.JournalCloseTests.test_explicit_none_timeout_preserves_unlimited_journal_mode',
    'test_appserver_journal_close_01a0ff2d.JournalCloseTests.test_explicit_none_timeout_without_journal_preserves_reader_join'
] if stage == 'compatibility-red' else ['test_appserver_journal_close_01a0ff2d'])
command = [sys.executable, '-B'] + (['-O'] if stage in ('green-optimized', 'final-optimized') else []) + ['-m', 'unittest', '-v', *selected]
before = {'stage': stage, 'argv': command, 'cwd': str(private), 'source_sha256': hashlib.sha256(source).hexdigest(),
          'test_sha256': hashlib.sha256(tests).hexdigest(), 'source_bytes': len(source),
          'created_before_command_utc': datetime.now(timezone.utc).isoformat(), 'native_child_count_expected': 0 if stage == 'compatibility-red' else 3}
with (target / 'INTENT.json').open('x', encoding='utf-8') as f:
    json.dump(before, f, indent=2)
    f.write('\n')
env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1', JOURNAL_CLOSE_EVIDENCE_DIR=str(cases))
start = datetime.now(timezone.utc).isoformat()
p = subprocess.Popen(command, cwd=private, env=env, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
stdout, stderr = p.communicate(timeout=20)
end = datetime.now(timezone.utc).isoformat()
(target / 'stdout.log').write_bytes(stdout)
(target / 'stderr.log').write_bytes(stderr)
receipt = dict(before, pid=p.pid, started_utc=start, ended_utc=end, exit_code=p.returncode,
               stdout_sha256=hashlib.sha256(stdout).hexdigest(), stderr_sha256=hashlib.sha256(stderr).hexdigest(),
               ordinary_regression=True, producer_or_formal_replay=False)
with (target / 'receipt.json').open('x', encoding='utf-8') as f:
    json.dump(receipt, f, indent=2)
    f.write('\n')
print(json.dumps(receipt, indent=2))
print(stderr.decode(errors='replace'))
sys.exit(p.returncode)
