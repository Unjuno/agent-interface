import ast, hashlib, json
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parent.parent
private = root / 'work/repairs/appserver-journal-close-01'
out = root / 'outputs/appserver-journal-close-01'
path = private / 'codex_app_server_client_v2.py'
before = path.read_bytes()
assert hashlib.sha256(before).hexdigest() == '8ed4e3496ef8154d3c4573fb1386efe6e036292a3809e43aa37fc22aecc5b42b'
newline = b'\r\n' if b'\r\n' in before else b'\n'
old = newline.join([b'            with self._journal_lock:', b'                self._journal.close()'])
new = newline.join([b'            if not self._journal_lock.acquire(timeout=timeout):',
                    b'                raise TimeoutError("app-server journal close timed out")',
                    b'            try:', b'                self._journal.close()', b'            finally:',
                    b'                self._journal_lock.release()'])
assert before.count(old) == 1
after = before.replace(old, new, 1)
def methods(source):
    return {node.name: ast.dump(node, include_attributes=False) for node in ast.walk(ast.parse(source))
            if isinstance(node, ast.FunctionDef)}
old_methods, new_methods = methods(before), methods(after)
assert old_methods.keys() == new_methods.keys()
assert [key for key in old_methods if old_methods[key] != new_methods[key]] == ['close']
tests = (private / 'test_appserver_journal_close_01a0ff2d.py').read_bytes()
freeze = json.loads((out / 'FREEZE_BEFORE_RED_V2.json').read_bytes())
assert hashlib.sha256(tests).hexdigest() == freeze['test_sha256']
record = {'created_before_green_utc': datetime.now(timezone.utc).isoformat(),
          'original_source_sha256': hashlib.sha256(before).hexdigest(), 'repaired_source_sha256': hashlib.sha256(after).hexdigest(),
          'test_sha256': hashlib.sha256(tests).hexdigest(), 'repaired_source_bytes': len(after),
          'only_changed_method': 'close', 'exact_old_block': old.decode(), 'exact_new_block': new.decode(),
          'scope': 'Bound journal mutex acquisition by existing per-stage timeout; typed incomplete cleanup; release only successfully acquired lock even on journal exception. No overall close/OS I/O/pipe closure guarantee.'}
with (out / 'FIX_BEFORE_GREEN.json').open('x', encoding='utf-8') as f:
    json.dump(record, f, indent=2)
    f.write('\n')
path.write_bytes(after)
print(json.dumps(record, indent=2))
