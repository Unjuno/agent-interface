import hashlib,json,subprocess,sys
from datetime import datetime,timezone
from pathlib import Path
root=Path(__file__).resolve().parent.parent;source=root/'work/repairs/6920-audit-v3'
stage=sys.argv[1];assert stage in ('red','final-normal','final-optimized')
out=root/'outputs/6920-audit-v3'/stage;out.mkdir(exist_ok=False)
for name,path in [('executed-audit.py.txt',source/'audit_v3.py.txt'),('executed-regressions.py.txt',source/'regressions.py.txt')]:
    (out/name).write_bytes(path.read_bytes())
command=[sys.executable,'-B']+(['-O'] if stage=='final-optimized' else [])+[str(source/'regressions.py.txt'),str(out/'results')]
start=datetime.now(timezone.utc).isoformat()
(out/'INTENT.json').write_text(json.dumps({'argv':command,'cwd':str(source),'start_utc':start,'formal_allocation':None,'producer_invocations':0})+'\n',encoding='utf-8')
with (out/'stdout.log').open('xb') as stdout,(out/'stderr.log').open('xb') as stderr:
    p=subprocess.Popen(command,cwd=source,stdout=stdout,stderr=stderr)
    (out/'HANDLE.json').write_text(json.dumps({'pid':p.pid,'start_utc':start})+'\n',encoding='utf-8')
    code=p.wait(timeout=30)
receipt={'stage':stage,'argv':command,'cwd':str(source),'pid':p.pid,'start_utc':start,'end_utc':datetime.now(timezone.utc).isoformat(),'exit_code':code,
         'audit_sha256':hashlib.sha256((out/'executed-audit.py.txt').read_bytes()).hexdigest(),
         'test_sha256':hashlib.sha256((out/'executed-regressions.py.txt').read_bytes()).hexdigest(),
         'streams':{n:{'bytes':(out/n).stat().st_size,'sha256':hashlib.sha256((out/n).read_bytes()).hexdigest()} for n in ('stdout.log','stderr.log')},
         'browser_candidate_fixture_server_model_original_A01_invocations':0}
(out/'RECEIPT.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps(receipt,indent=2));sys.exit(code)
