import copy
import hashlib
import importlib.util
import json
from datetime import datetime,timezone
from pathlib import Path

root=Path(__file__).parent; source=root.parent/'temporal-goal-C02/audit-v2.py'
spec=importlib.util.spec_from_file_location('audit_v2',source); module=importlib.util.module_from_spec(spec); spec.loader.exec_module(module)
data=module.read(root.parent/'temporal-goal-C02/primary-C02',json.loads((root.parent/'temporal-goal-C02/cases.json').read_bytes()))
results=[]
for label,all_same in [('one_duplicate_one_omitted',False),('sixteen_duplicates_one_actual_event',True)]:
    copied=copy.deepcopy(data)
    if all_same: copied['fixtures']=[copy.deepcopy(copied['fixtures'][0]) for _ in range(16)]
    else: copied['fixtures'][1]=copy.deepcopy(copied['fixtures'][0])
    ids=[json.loads(f['response']['body'])['event_id'] for f in copied['fixtures']]
    result=module.audit(copied)
    record=dict(case=label,fixture_records=len(ids),distinct_fixture_event_ids=len(set(ids)),unexpected_status=result['status'],
        intended='reject duplicate/missing external provenance',actual_primary_changes=0,producer_replays=0)
    (root/(label+'.copied.json')).write_text(json.dumps(copied,ensure_ascii=False,indent=2),encoding='utf-8')
    results.append(record)
result=dict(utc=datetime.now(timezone.utc).isoformat(),source_sha256=hashlib.sha256(source.read_bytes()).hexdigest(),
    finding='4171886014',actual_copied_reproductions=results,original_fixture_event_ids=[json.loads(f['response']['body'])['event_id'] for f in data['fixtures']],
    original_fixture_distinct=16,actual_original_fields_changed=0,scope='v2 defect real; saved original fixture set is complete, not a new primary execution')
(root/'V2_FIRST_REPRODUCTION.json').write_text(json.dumps(result,indent=2),encoding='utf-8');print(json.dumps(result))
