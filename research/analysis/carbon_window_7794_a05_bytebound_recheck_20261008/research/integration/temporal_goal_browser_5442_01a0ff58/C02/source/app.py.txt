"""Disposable loopback editor; standard version-conditional SQLite writes."""
import argparse
from contextlib import closing
import html
import json
import os
import sqlite3
import threading
import time
from datetime import datetime, timezone
from http.server import BaseHTTPRequestHandler, HTTPServer
from pathlib import Path
from urllib.parse import parse_qs, urlsplit


def stamp():
    return datetime.now(timezone.utc).isoformat(), time.monotonic_ns()


class Store:
    def __init__(self, path, namespace):
        self.path = Path(path)
        self.namespace = namespace

    def connection(self):
        connection = sqlite3.connect(self.path, timeout=5, isolation_level=None)
        connection.row_factory = sqlite3.Row
        return connection

    def create(self, case_ids):
        if self.path.exists():
            raise ValueError('new instance database already exists')
        with closing(self.connection()) as db:
            db.executescript('''
            PRAGMA journal_mode=DELETE;
            PRAGMA synchronous=FULL;
            CREATE TABLE meta(key TEXT PRIMARY KEY,value TEXT NOT NULL);
            CREATE TABLE docs(case_id TEXT,target TEXT,value TEXT NOT NULL,version INTEGER NOT NULL,
                              PRIMARY KEY(case_id,target));
            CREATE TABLE events(event_id INTEGER PRIMARY KEY AUTOINCREMENT,role TEXT NOT NULL,
                opid TEXT UNIQUE NOT NULL,nonce TEXT UNIQUE NOT NULL,case_id TEXT NOT NULL,target TEXT NOT NULL,
                expected_version INTEGER,value TEXT NOT NULL,goal_hash TEXT NOT NULL,outcome TEXT NOT NULL,
                before_value TEXT NOT NULL,before_version INTEGER NOT NULL,
                after_value TEXT NOT NULL,after_version INTEGER NOT NULL,utc TEXT NOT NULL,mono_ns INTEGER NOT NULL);
            CREATE TABLE requests(request_id INTEGER PRIMARY KEY AUTOINCREMENT,method TEXT NOT NULL,
                path TEXT NOT NULL,body TEXT NOT NULL,status INTEGER NOT NULL,event_id INTEGER,
                utc TEXT NOT NULL,mono_ns INTEGER NOT NULL);
            ''')
            db.executemany('INSERT INTO meta VALUES (?,?)', [('schema','temporal-goal-editor-v1'),('namespace',self.namespace)])
            for case_id in case_ids:
                db.execute('INSERT INTO docs VALUES (?,?,?,?)',(case_id,'doc:A','initial '+case_id,0))
                db.execute('INSERT INTO docs VALUES (?,?,?,?)',(case_id,'doc:B','untouched '+case_id,0))

    def state(self, case_id):
        with closing(self.connection()) as db:
            rows = {row['target']:dict(row) for row in db.execute('SELECT * FROM docs WHERE case_id=? ORDER BY target',(case_id,))}
        if set(rows) != {'doc:A','doc:B'}:
            raise ValueError('unknown case')
        return dict(namespace=self.namespace,case_id=case_id,target=rows['doc:A'],sibling=rows['doc:B'])

    def mutate(self, role, item):
        required = {'namespace','case_id','target','opid','nonce','value','goal_hash'}
        if role == 'task':
            required.add('expected_version')
        if set(item) != required or role not in {'task','disturbance'}:
            raise ValueError('mutation shape')
        if any(type(item[key]) is not str for key in required-{'expected_version'}):
            raise ValueError('text field type')
        if item['namespace'] != self.namespace or item['target'] not in {'doc:A','doc:B'}:
            raise ValueError('target binding')
        if not item['opid'] or not item['nonce'] or len(item['value'].encode('utf-8')) > 1024:
            raise ValueError('bounded fields')
        if role == 'task' and (type(item['expected_version']) is not int or item['expected_version'] < 0):
            raise ValueError('version must be exact nonnegative integer')
        db = self.connection()
        try:
            db.execute('BEGIN IMMEDIATE')
            if db.execute('SELECT 1 FROM events WHERE opid=? OR nonce=?',(item['opid'],item['nonce'])).fetchone():
                raise ValueError('duplicate opid/nonce; no implicit replay')
            counts = dict(db.execute('SELECT role,COUNT(*) FROM events GROUP BY role').fetchall())
            if counts.get(role,0) >= (20 if role == 'task' else 20):
                raise ValueError('instance mutation budget exhausted')
            row = db.execute('SELECT value,version FROM docs WHERE case_id=? AND target=?',
                             (item['case_id'],item['target'])).fetchone()
            if row is None:
                raise ValueError('unknown document')
            before_value, before_version = row
            expected = item.get('expected_version')
            if role == 'task':
                changed = db.execute('UPDATE docs SET value=?,version=version+1 WHERE case_id=? AND target=? AND version=?',
                    (item['value'],item['case_id'],item['target'],expected)).rowcount
                outcome = 'APPLIED' if changed == 1 else 'CONFLICT'
            else:
                db.execute('UPDATE docs SET value=?,version=version+1 WHERE case_id=? AND target=?',
                           (item['value'],item['case_id'],item['target']))
                outcome = 'DISTURBED'
            after = db.execute('SELECT value,version FROM docs WHERE case_id=? AND target=?',
                               (item['case_id'],item['target'])).fetchone()
            utc, mono = stamp()
            cursor = db.execute('''INSERT INTO events(role,opid,nonce,case_id,target,expected_version,value,goal_hash,outcome,
                before_value,before_version,after_value,after_version,utc,mono_ns) VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)''',
                (role,item['opid'],item['nonce'],item['case_id'],item['target'],expected,item['value'],item['goal_hash'],outcome,
                 before_value,before_version,after[0],after[1],utc,mono))
            event = dict(db.execute('SELECT * FROM events WHERE event_id=?',(cursor.lastrowid,)).fetchone())
            db.execute('COMMIT')
            return event
        except BaseException:
            if db.in_transaction:
                db.execute('ROLLBACK')
            raise
        finally:
            db.close()

    def log_request(self, method, path, body, status, event_id=None):
        utc, mono = stamp()
        with closing(self.connection()) as db:
            if db.execute('SELECT COUNT(*) FROM requests').fetchone()[0] >= 200:
                raise ValueError('bounded request journal exhausted')
            db.execute('INSERT INTO requests(method,path,body,status,event_id,utc,mono_ns) VALUES (?,?,?,?,?,?,?)',
                       (method,path,body,status,event_id,utc,mono))


def json_node(identifier, value):
    return '<script type="application/json" id="'+identifier+'">'+json.dumps(value,ensure_ascii=True).replace('<','\\u003c')+'</script>'


def page(title, body):
    return ('<!doctype html><html lang="en"><meta charset="utf-8"><title>'+html.escape(title)+
        '</title><style>body{font:15px system-ui;margin:24px;color:#132638}input{font:inherit;min-width:520px;padding:7px}'
        'label{display:block;margin:12px 0}table{border-collapse:collapse;font-size:13px}td,th{border:1px solid #ccd;padding:6px}'
        'button{font:inherit;padding:8px 16px}pre{white-space:pre-wrap}</style><body><h1>'+html.escape(title)+'</h1>'+body+'</body></html>').encode()


def state_html(state):
    a,b = state['target'],state['sibling']
    return ('<p>Namespace: '+html.escape(state['namespace'])+'</p><p>Case: '+html.escape(state['case_id'])+
        '</p><p>Target doc:A — current value: '+html.escape(a['value'])+' — version '+str(a['version'])+
        '</p><p>Sibling doc:B — '+html.escape(b['value'])+' — version '+str(b['version'])+'</p>'+json_node('state-json',state))


class Handler(BaseHTTPRequestHandler):
    def log_message(self,*args):
        pass

    def finish_response(self, status, data, body='', event_id=None, json_response=False):
        self.server.store.log_request(self.command,self.path,body,status,event_id)
        self.send_response(status)
        self.send_header('Content-Type','application/json; charset=utf-8' if json_response else 'text/html; charset=utf-8')
        self.send_header('Cache-Control','no-store')
        self.send_header('Content-Length',str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        parts = urlsplit(self.path)
        try:
            if parts.path == '/':
                with closing(self.server.store.connection()) as db:
                    docs = [dict(row) for row in db.execute('SELECT * FROM docs ORDER BY case_id,target')]
                    counts = [dict(row) for row in db.execute('SELECT role,outcome,COUNT(*) AS count FROM events GROUP BY role,outcome')]
                rows = ''.join('<tr>'+''.join('<td>'+html.escape(str(row[key]))+'</td>' for key in ('case_id','target','value','version'))+'</tr>' for row in docs)
                data = page('Owned temporal goal editor','<p>'+html.escape(self.server.store.namespace)+'</p><table><tr><th>Case</th><th>Document</th><th>Saved value</th><th>Version</th></tr>'+rows+'</table><pre>'+html.escape(json.dumps(counts,indent=2))+'</pre>')
            elif parts.path.startswith('/view/'):
                state = self.server.store.state(parts.path.removeprefix('/view/'))
                data = page('Current saved document',state_html(state))
            elif parts.path.startswith('/edit/'):
                case_id = parts.path.removeprefix('/edit/')
                state = self.server.store.state(case_id)
                query = parse_qs(parts.query,keep_blank_values=True)
                if set(query) != {'opid','nonce','goal_hash'} or any(len(v)!=1 for v in query.values()):
                    raise ValueError('explicit edit identity required')
                values = dict(namespace=state['namespace'],case_id=case_id,target='doc:A',expected_version=str(state['target']['version']),
                              opid=query['opid'][0],nonce=query['nonce'][0],goal_hash=query['goal_hash'][0])
                fields = ''.join('<label>Binding '+html.escape(key)+' <input type="text" readonly name="'+key+'" value="'+html.escape(value,quote=True)+'"></label>' for key,value in values.items())
                data = page('Version guarded document edit',state_html(state)+'<form method="post" action="/apply">'+fields+
                    '<label>Desired value <input name="value" aria-label="Desired value" value="'+html.escape(state['target']['value'],quote=True)+
                    '"></label><button type="submit">Apply guarded edit</button></form>')
            else:
                self.finish_response(404,page('No route','<p>No document action.</p>'))
                return
            self.finish_response(200,data)
        except (ValueError,KeyError) as error:
            self.finish_response(400,page('Rejected request','<p>'+html.escape(str(error))+'</p>'))

    def do_POST(self):
        size = int(self.headers.get('Content-Length','0'))
        if size > 4096:
            self.finish_response(413,page('Body too large',''))
            return
        raw = self.rfile.read(size).decode('utf-8')
        try:
            if self.path == '/apply':
                fields = parse_qs(raw,keep_blank_values=True)
                if any(len(value)!=1 for value in fields.values()):
                    raise ValueError('duplicate form field')
                item = {key:value[0] for key,value in fields.items()}
                version = item.get('expected_version','')
                if not version.isascii() or not version.isdecimal():
                    raise ValueError('decimal expected version required')
                item['expected_version'] = int(version)
                event = self.server.store.mutate('task',item)
                state = self.server.store.state(event['case_id'])
                data = page('Delivery accepted','<h2>Delivery accepted</h2><p>Guard outcome: '+event['outcome']+'</p>'+
                            state_html(state)+json_node('receipt-json',event))
                self.finish_response(200,data,raw,event['event_id'])
            elif self.path == '/disturb':
                event = self.server.store.mutate('disturbance',json.loads(raw))
                self.finish_response(200,json.dumps(event).encode(),raw,event['event_id'],True)
            elif self.path == '/shutdown':
                self.finish_response(200,b'{"shutdown":"requested"}',raw,json_response=True)
                threading.Thread(target=self.server.shutdown,daemon=True).start()
            else:
                self.finish_response(404,page('No route',''),raw)
        except (ValueError,KeyError) as error:
            self.finish_response(400,page('Rejected mutation','<p>'+html.escape(str(error))+'</p>'),raw)


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--output',required=True)
    parser.add_argument('--cases',required=True)
    args = parser.parse_args()
    out = Path(args.output).resolve()
    if not out.is_dir() or (out/'READY.json').exists():
        raise ValueError('unique pre-created output required')
    cases = json.loads(Path(args.cases).read_bytes())
    store = Store(out/'live.sqlite',cases['namespace'])
    store.create([row['case_id'] for row in cases['cases']])
    server = HTTPServer(('127.0.0.1',0),Handler)
    server.store = store
    utc,mono = stamp()
    ready = dict(pid=os.getpid(),port=server.server_port,host='127.0.0.1',namespace=store.namespace,
                 database=str(store.path),utc=utc,mono_ns=mono,sqlite_version=sqlite3.sqlite_version)
    (out/'READY.json').write_bytes(json.dumps(ready,indent=2).encode())
    print(json.dumps(ready),flush=True)
    try:
        server.serve_forever(poll_interval=0.1)
    finally:
        server.server_close()
        utc,mono = stamp()
        (out/'SERVER_EXIT.json').write_bytes(json.dumps(dict(utc=utc,mono_ns=mono,pid=os.getpid(),status='serve_forever_returned')).encode())
        print('owned server closed',flush=True)


if __name__ == '__main__':
    main()
