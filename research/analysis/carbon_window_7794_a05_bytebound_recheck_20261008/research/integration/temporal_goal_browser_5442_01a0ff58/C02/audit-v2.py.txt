"""Separate raw-only implementation; never imports app, actor or observer."""
import argparse
import copy
import hashlib
import json
import sqlite3
from datetime import datetime, timezone
from pathlib import Path
from urllib.parse import parse_qs, urlsplit

def typed(value):
    if type(value) is dict: return ('object',tuple((k,typed(v)) for k,v in sorted(value.items())))
    if type(value) is list: return ('array',tuple(map(typed,value)))
    return (type(value).__name__,value)

def need(condition,label):
    if not condition: raise ValueError(label)

def read(run,cases):
    phases={p:json.loads((run/('observer-'+p)/'raw.json').read_bytes()) for p in ['empty','initial','post','final']}
    for p,raw in phases.items():
        snap=run/('observer-'+p)/'snapshot.sqlite'
        need(hashlib.sha256(snap.read_bytes()).hexdigest()==raw['snapshot_sha256'],'snapshot byte hash '+p)
        db=sqlite3.connect(snap.resolve().as_uri()+'?mode=ro',uri=True); db.row_factory=sqlite3.Row
        try:
            sql={name:[dict(r) for r in db.execute('SELECT * FROM '+name+' ORDER BY '+key)] for name,key in
                 [('meta','key'),('docs','case_id,target'),('events','event_id'),('requests','request_id')]}
        finally: db.close()
        need(typed(sql)==typed(raw['tables']),'whole SQL tables match raw '+p)
    ui=[json.loads(line) for line in (run/'ui.jsonl').read_text(encoding='utf-8').splitlines()]
    fixtures=[]
    for path in sorted(run.glob('fixture-*/[0-9]*-intent.json')):
        response=path.with_name(path.name.replace('-intent','-response'))
        intent=json.loads(path.read_bytes()); result=json.loads(response.read_bytes())
        if intent['route']=='/shutdown': continue
        fixtures.append(dict(intent=intent,response=result))
    return dict(cases=cases,phases=phases,ui=ui,fixtures=fixtures)

def audit(data):
    cases=data['cases']; namespace=cases['namespace']; cs={c['case_id']:c for c in cases['cases']}
    phases=data['phases']; ui=data['ui']; final=phases['final']['tables']
    need(len(cs)==10 and cases['max_repair_attempts']==2,'frozen cohort and budget')
    state={}
    for c in cs.values():
        h=hashlib.sha256(json.dumps(dict(namespace=namespace,case_id=c['case_id'],target='doc:A',goal=c['goal']),
            sort_keys=True,separators=(',',':'),ensure_ascii=True).encode()).hexdigest()
        need(h==c['goal_hash'],'semantic binding hash')
        state[(c['case_id'],'doc:A')]=dict(case_id=c['case_id'],target='doc:A',value=c['initial_value'],version=0)
        state[(c['case_id'],'doc:B')]=dict(case_id=c['case_id'],target='doc:B',value=c['sibling_initial'],version=0)
    events=final['events']; requests=final['requests']
    need(len(events)==36 and len(requests)<=200,'exact mutation and request bound')
    need([e['event_id'] for e in events]==list(range(1,37)),'ordered event identity')
    need(len({e['opid'] for e in events})==36 and len({e['nonce'] for e in events})==36,'unique intent identities')
    snapshots=[copy.deepcopy(state)]
    for e in events:
        need(type(e['before_version']) is int and type(e['after_version']) is int and
             type(e['mono_ns']) is int and type(e['event_id']) is int,'exact event integer types')
        need(e['role'] in {'task','disturbance'} and e['case_id'] in cs and e['target']=='doc:A','event target and role')
        c=cs[e['case_id']]; doc=state[(e['case_id'],e['target'])]
        need(e['goal_hash']==c['goal_hash'] and (e['before_value'],e['before_version'])==(doc['value'],doc['version']),'event predecessor')
        if e['role']=='task':
            need(type(e['expected_version']) is int and e['expected_version']>=0 and e['value']==c['goal'],'task version and rich goal')
            outcome='APPLIED' if e['expected_version']==doc['version'] else 'CONFLICT'
            if outcome=='APPLIED': doc.update(value=e['value'],version=doc['version']+1)
        else:
            need(e['expected_version'] is None,'actor has no task guard')
            outcome='DISTURBED'; doc.update(value=e['value'],version=doc['version']+1)
        need(e['outcome']==outcome and (e['after_value'],e['after_version'])==(doc['value'],doc['version']),'independent transition oracle')
        snapshots.append(copy.deepcopy(state))
    need([r['request_id'] for r in requests]==list(range(1,len(requests)+1)),'request identity')
    apply=[r for r in requests if r['method']=='POST' and r['path']=='/apply']
    disturb=[r for r in requests if r['method']=='POST' and r['path']=='/disturb']
    need(len(apply)==20 and len(disturb)==16,'actual task20 external16 sends')
    by_event={}
    for r in apply+disturb:
        need(type(r['status']) is int and r['status']==200 and type(r['event_id']) is int,'prepared HTTP acceptance type')
        need(r['event_id'] not in by_event,'one prepared response per event')
        by_event[r['event_id']]=r; e=events[r['event_id']-1]
        if r['path']=='/apply':
            fields=parse_qs(r['body'],keep_blank_values=True)
            need(set(fields)=={'namespace','case_id','target','opid','nonce','value','goal_hash','expected_version'} and
                 all(len(v)==1 for v in fields.values()),'actual form shape')
            item={k:v[0] for k,v in fields.items()}
            need(item['expected_version'].isascii() and item['expected_version'].isdecimal(),'wire decimal version')
            item['expected_version']=int(item['expected_version'])
        else: item=json.loads(r['body'])
        need(item.pop('namespace')==namespace,'request namespace')
        need(typed(item)==typed({k:e[k] for k in item}),'actual request-event typed join')
        need(r['mono_ns']>=e['mono_ns'],'commit before prepared response')
    for p,count in [('empty',0),('initial',10),('post',20),('final',36)]:
        raw=phases[p]; tables=raw['tables']
        need(raw['phase']==p,'snapshot phase label')
        need(typed(tables['events'])==typed(events[:count]),'historical receipt prefix immutable '+p)
        need(typed(tables['docs'])==typed([snapshots[count][k] for k in sorted(state)]),'snapshot replay '+p)
        need(typed(tables['requests'])==typed(requests[:len(tables['requests'])]),'request prefix immutable '+p)
        need(typed(tables['meta'])==typed([dict(key='namespace',value=namespace),dict(key='schema',value='temporal-goal-editor-v1')]),'meta binding '+p)
        predicted=[dict(case_id=c['case_id'],goal_current=snapshots[count][(c['case_id'],'doc:A')]['value']==c['goal'],
            sibling_preserved=snapshots[count][(c['case_id'],'doc:B')]['value']==c['sibling_initial'] and
            snapshots[count][(c['case_id'],'doc:B')]['version']==0) for c in cases['cases']]
        need(typed(raw['verdicts'])==typed(predicted),'observer typed labels '+p)
    need([row['index'] for row in ui]==list(range(len(ui))),'UI trace prefix ordering')
    intents={}; responses={}; observations={}; repairs={c:0 for c in cs}; latest={}; decisions={}; last_request=0
    for row in ui:
        kind=row['kind']; cid=row['case_id']; c=cs[cid]; phase=row['phase']; need(kind in
            {'form','filled','intent','send_started','response','decision','observe','goal_observation'},'unexpected UI stage')
        if kind in {'form','filled','response','observe'}:
            dom=row['dom']; s=row['state']; need(type(s) is dict,'public state present')
            need(s['namespace']==namespace and s['case_id']==cid and type(s['target']['version']) is int,'public typed binding')
            need(typed(json.loads(dom['state_text']))==typed(s),'DOM state extraction equality')
            if kind=='filled':
                need(typed(s)==typed(latest[(cid,phase)]['state']),'fill does not change committed state')
                fields={f['name']:f['value'] for f in dom['fields']}; need(fields['value']==c['goal'],'actual filled semantic goal')
                continue
            if kind=='response':
                e=row['receipt']; need(type(e['event_id']) is int and e['event_id'] in by_event,'actual receipt identity')
                req=by_event[e['event_id']]; need(typed(e)==typed(events[e['event_id']-1]),'public receipt immutable raw join')
                need(typed(json.loads(dom['receipt_text']))==typed(e) and dom['title']=='Delivery accepted','actual accepted DOM')
                responses[e['opid']]=row
            else:
                path=urlsplit(dom['url']).path; query=urlsplit(dom['url']).query; route=path+('?' + query if query else '')
                matches=[r for r in requests if r['request_id']>last_request and r['method']=='GET' and r['path']==route]
                need(bool(matches),'actual GET for public observation'); req=matches[0]
            need(req['request_id']>last_request,'UI request order'); last_request=req['request_id']
            count=max((r['event_id'] for r in requests if r['event_id'] is not None and r['request_id']<=req['request_id']),default=0)
            expected=dict(namespace=namespace,case_id=cid,target=snapshots[count][(cid,'doc:A')],sibling=snapshots[count][(cid,'doc:B')])
            need(typed(s)==typed(expected),'public current view matches server event history')
            need(s['target']['value'] in dom['body'] and cid in row['ax'],'visible public state represented')
            latest[(cid,phase)]=row
            if kind=='form':
                fields={f['name']:f['value'] for f in dom['fields']}; op=c['operations'][phase]
                expected_fields=dict(namespace=namespace,case_id=cid,target='doc:A',expected_version=str(s['target']['version']),
                    opid=op['opid'],nonce=op['nonce'],goal_hash=c['goal_hash'],value=s['target']['value'])
                need(typed(fields)==typed(expected_fields),'public form exact bound fields')
        elif kind=='intent':
            fields=row['fields']; op=c['operations'][phase]; need(fields['opid'] not in intents,'one UI intent')
            need(fields['opid']==op['opid'] and fields['nonce']==op['nonce'] and fields['value']==c['goal'],'intent semantic binding')
            intents[fields['opid']]=row
        elif kind=='send_started':
            need(row['fields']['opid'] in intents,'intent before send')
        elif kind=='goal_observation':
            current=latest[(cid,phase)]['state']['target']; need(type(row['goal_current']) is bool and
                row['goal_current']==(current['value']==c['goal']) and type(row['version']) is int and row['version']==current['version'],'fresh point observation')
            observations[(cid,phase)]=row['goal_current']
        elif kind=='decision':
            current=latest[(cid,phase)]['state']['target']; goal=current['value']==c['goal']
            if phase!='initial' and 'outcome' in row: repairs[cid]+=1
            need(type(row['attempts']) is int and row['attempts']==repairs[cid] and repairs[cid]<=2,'finite repair count')
            need(type(row['goal_current']) is bool and row['goal_current']==goal,'decision fresh goal predicate')
            expected='CONFIRMED' if goal else 'YIELD_BUDGET' if repairs[cid]>=2 else 'NEED_REPAIR'
            need(row['status']==expected,'no false completion; finite yield')
            if phase!='initial': decisions[cid]=row['status']
    need(len(intents)==20 and set(intents)==set(responses),'all and only20 actual UI submissions')
    for r in apply:
        e=events[r['event_id']-1]; intent=intents[e['opid']]; fields=intent['fields'].copy()
        fields['expected_version']=int(fields['expected_version']); fields.pop('namespace')
        need(typed(fields)==typed({k:e[k] for k in fields}),'intent to actual applied request join')
    need(len(data['fixtures'])==16,'sixteen actual external response records')
    for f in data['fixtures']:
        item=json.loads(f['intent']['body']); response=f['response']; e=json.loads(response['body'])
        need(f['intent']['attempts']==1 and type(response['status']) is int and response['status']==200,'fixture one send')
        need(typed(e)==typed(events[e['event_id']-1]) and e['role']=='disturbance','fixture actual receipt join')
        req=by_event[e['event_id']]; need(typed(item)==typed(json.loads(req['body'])),'fixture intent actual body')
    golden={'stable':(1,0,True),'overwrite':(3,1,True),'one_race':(4,2,True),'persistent_race':(4,2,False),'aba_restore':(3,0,True)}
    rows=[]
    for c in cases['cases']:
        cid=c['case_id']; version,k,goal=golden[c['history']]; doc=state[(cid,'doc:A')]
        need((doc['version'],repairs[cid],doc['value']==c['goal'])==(version,k,goal),'history discrimination '+cid)
        need(observations[(cid,'post')]==(c['history'] in {'stable','aba_restore'}),'historical applied versus post goal')
        need(decisions[cid]==('CONFIRMED' if goal else 'YIELD_BUDGET'),'final truthful status '+cid)
        rows.append(dict(case_id=cid,history=c['history'],historical_initial='APPLIED',post_goal=observations[(cid,'post')],
            final_goal=goal,repair_attempts=k,final_version=version,status=decisions[cid]))
    need(sum(e['role']=='task' and e['outcome']=='APPLIED' for e in events)==14 and
         sum(e['role']=='task' and e['outcome']=='CONFLICT' for e in events)==6,'guard discrimination14/6')
    return dict(status='PASS_TEMPORAL_GOAL_SCOPED',cases=rows,task_sends=20,external_sends=16,
        initial_goal=10,post_goal=4,final_goal=8,yield_budget=2,sibling_preserved=10,raw_rows=len(events),ui_stages=len(ui))

def controls(data):
    tests=[]
    def attempt(name,change):
        mutated=copy.deepcopy(data); change(mutated)
        try: audit(mutated)
        except (ValueError,KeyError,TypeError,IndexError) as error: tests.append(dict(name=name,rejected=True,reason=str(error)))
        else: raise ValueError('negative control escaped: '+name)
    attempt('observer bool alias',lambda d:d['phases']['final']['verdicts'][0].update(goal_current=1))
    attempt('event version bool alias',lambda d:d['phases']['final']['tables']['events'][0].update(expected_version=False))
    attempt('event target swap',lambda d:d['phases']['final']['tables']['events'][0].update(target='doc:B'))
    attempt('conflict relabel applied',lambda d:next(e for e in d['phases']['final']['tables']['events'] if e['outcome']=='CONFLICT').update(outcome='APPLIED'))
    attempt('collateral sibling',lambda d:d['phases']['final']['tables']['docs'][1].update(value='changed'))
    attempt('historical prefix rewritten',lambda d:d['phases']['post']['tables']['events'][0].update(value='rewritten'))
    attempt('false completion',lambda d:next(r for r in reversed(d['ui']) if r['kind']=='decision' and r['status']=='YIELD_BUDGET').update(status='CONFIRMED'))
    attempt('third repair counter',lambda d:next(r for r in reversed(d['ui']) if r['kind']=='decision' and r['status']=='YIELD_BUDGET').update(attempts=3))
    attempt('fixture namespace swap',lambda d:d['fixtures'][0]['intent'].update(body=d['fixtures'][0]['intent']['body'].replace(d['cases']['namespace'],'foreign')))
    attempt('extra task send',lambda d:d['phases']['final']['tables']['requests'].append(copy.deepcopy(next(r for r in d['phases']['final']['tables']['requests'] if r['path']=='/apply'))))
    return tests

def main():
    p=argparse.ArgumentParser(); p.add_argument('--run',required=True); p.add_argument('--cases',required=True); p.add_argument('--output',required=True)
    a=p.parse_args(); out=Path(a.output); out.mkdir(exist_ok=False); start=datetime.now(timezone.utc).isoformat()
    data=read(Path(a.run),json.loads(Path(a.cases).read_bytes()))
    result=audit(data); result['negative_controls']=controls(data); result['started_utc']=start; result['ended_utc']=datetime.now(timezone.utc).isoformat()
    (out/'RESULT.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print(json.dumps({k:result[k] for k in ['status','task_sends','external_sends','initial_goal','post_goal','final_goal','yield_budget','ui_stages']}))

if __name__=='__main__': main()
