import hashlib
import json
from pathlib import Path

root=Path(__file__).parent/'temporal-goal-C02'
path=root/'cases.json'
original=path.read_bytes()
(root/'construction-preflight'/'cases-before-hash-repair.json').write_bytes(original)
cases=json.loads(original)
for c in cases['cases']:
    item=dict(namespace=cases['namespace'],case_id=c['case_id'],target=c['target'],goal=c['goal'])
    c['goal_hash']=hashlib.sha256(json.dumps(item,sort_keys=True,separators=(',',':'),ensure_ascii=True).encode()).hexdigest()
path.write_text(json.dumps(cases,ensure_ascii=False,indent=2),encoding='utf-8',newline='\n')
print(json.dumps(dict(reason='C02 namespace requires new semantic bindings before freeze',cases=10,
                     before_sha256=hashlib.sha256(original).hexdigest(),after_sha256=hashlib.sha256(path.read_bytes()).hexdigest())))
