import datetime,hashlib,json,re,urllib.request,zipfile
from pathlib import Path

p=Path(__file__).resolve().parent
runtime=p/'node22';runtime.mkdir(exist_ok=False)
start=datetime.datetime.now(datetime.timezone.utc).isoformat()
index_url='https://nodejs.org/download/release/index.json'
index=urllib.request.urlopen(index_url,timeout=30).read()
(runtime/'index.json').write_bytes(index)
entry=next(x for x in json.loads(index) if x['version'].startswith('v22.') and 'win-x64-zip' in x['files'])
version=entry['version'];assert re.fullmatch(r'v22\.\d+\.\d+',version)
name='node-'+version+'-win-x64.zip'
prefix='https://nodejs.org/download/release/'+version+'/'
sums=urllib.request.urlopen(prefix+'SHASUMS256.txt',timeout=30).read()
(runtime/'SHASUMS256.txt').write_bytes(sums)
expected=next(line.split()[0] for line in sums.decode().splitlines() if line.split()[-1]==name)
archive=runtime/name
with urllib.request.urlopen(prefix+name,timeout=60) as source,archive.open('xb') as output:
    while data:=source.read(1024*1024):output.write(data)
actual=hashlib.sha256(archive.read_bytes()).hexdigest();assert actual==expected
with zipfile.ZipFile(archive) as z:
    for basename in ['node.exe','LICENSE']:
        with (runtime/basename).open('xb') as output:
            output.write(z.read('node-'+version+'-win-x64/'+basename))
r={'started_utc':start,'ended_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'version':version,'release_date':entry['date'],'index_url':index_url,'zip_url':prefix+name,
    'archive_sha256':actual,'archive_bytes':archive.stat().st_size,
    'node_exe_sha256':hashlib.sha256((runtime/'node.exe').read_bytes()).hexdigest(),
    'node_exe_bytes':(runtime/'node.exe').stat().st_size,'license_retained':True,
    'scope':'Private Windows test runtime only; no global install/PATH/configuration changes, not redistributed.'}
(runtime/'download-receipt.json').write_text(json.dumps(r,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps(r))
