import cp from 'node:child_process';
import { syncBuiltinESMExports } from 'node:module';
import { readdir } from 'node:fs/promises';
import { once } from 'node:events';

const [moduleUrl, scenario, evidenceDirectory] = process.argv.slice(1);
const originalSpawn = cp.spawn;
let child;
cp.spawn = (...args) => {
  child = originalSpawn(...args);
  return child;
};
syncBuiltinESMExports();
const { createRelayClient } = await import(moduleUrl);
const fixture = `
const readline=require('node:readline');
const timer=setTimeout(()=>process.exit(31),2000);
const input=readline.createInterface({input:process.stdin});
input.on('close',()=>{clearTimeout(timer);process.exit(0);});
input.on('line',async line=>{
  const r=JSON.parse(line);
  process.stderr.write('COMMITTED '+r.id+'\\n');
  if(r.tool==='pending')return;
  const row={id:r.id,tool:r.tool,status:'returned',next_id:r.id+1,
    result:{content:[{type:'text',text:r.arguments.literal}]}};
  const bytes=Buffer.from(JSON.stringify(row)+'\\n','utf8');
  for(const byte of bytes){process.stdout.write(Buffer.from([byte]));
    await new Promise(resolve=>setImmediate(resolve));}
});`;
const client = await createRelayClient({ command: process.execPath,
  args: ['-e', fixture], evidenceDirectory });
console.log(JSON.stringify({ stage: 'created', child_pid: child.pid, scenario }));
const fault = () => child.stdout.destroy(new Error('injected relay reply read fault'));
let pending, outcome, samePromise = null, committed = false;
const literal = '\u685c\u306e\u8a18\u9332\ud83d\ude00';
if (scenario === 'idle') {
  const closed = once(child.stdout, 'close').catch(() => {});
  fault();
  await closed;
} else {
  let stderr = '';
  const commitment = new Promise(resolve => child.stderr.on('data', chunk => {
    stderr += chunk.toString('utf8');
    if (stderr.includes('COMMITTED 1\n')) resolve();
  }));
  pending = client.send(scenario === 'pending' ? 'pending' : 'unicode', { literal });
  await commitment;
  committed = true;
  console.log(JSON.stringify({ stage: 'committed', scenario, state: client.state() }));
  samePromise = client.wait() === pending;
  if (scenario === 'pending') {
    fault();
    try { await pending; outcome = 'unexpected-return'; }
    catch (error) { outcome = String(error); }
  } else {
    outcome = await pending;
    if (scenario === 'settled') {
      const closed = once(child.stdout, 'close').catch(() => {});
      fault();
      await closed;
      samePromise = samePromise && client.wait() === pending && await client.wait() === outcome;
    }
  }
}
let newSendBlocked = null;
if (scenario !== 'unicode') {
  try { client.send('unicode', { literal }); newSendBlocked = false; }
  catch (error) { newSendBlocked = String(error).includes('injected relay reply read fault'); }
}
const beforeClose = client.state();
const exit = await client.close();
const entries = await readdir(evidenceDirectory);
console.log(JSON.stringify({ stage: 'result', scenario, committed, outcome,
  samePromise, newSendBlocked, beforeClose, exit,
  requestFiles: entries.filter(name => name.startsWith('request-')).sort(),
  replyFiles: entries.filter(name => name.startsWith('reply-')).sort(),
  literal, fixture: 'real inert Node child; injected real stdout stream destroy; responsive private files; child self-exit cap2000ms' }));
