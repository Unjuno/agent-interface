import hashlib,json
from pathlib import Path
root=Path(__file__).resolve().parent;repo=root/'candidate'
source=json.loads((root/'SOURCE_READBACK.json').read_text(encoding='utf-8'))
for remote,text in source['sources'].items():
    p=repo/remote;p.parent.mkdir(parents=True,exist_ok=True);p.write_bytes(text.encode())
target=repo/'research/live_control/test_adaptive_acquisition_caller_refusal_v3.py'
target.write_bytes((root/target.name).read_bytes())
pins=[]
for p in [root/'PLAN.md',root/'SOURCE_READBACK.json',root/'prepare.py',target]:
    d=p.read_bytes();pins.append({'path':str(p.relative_to(root)),'bytes':len(d),'sha256':hashlib.sha256(d).hexdigest()})
(root/'FREEZE.json').write_text(json.dumps({'source_commit':source['commit'],'kind':'ordinary-repair-not-formal','pins':pins},indent=2)+'\n',encoding='utf-8')
print(json.dumps({'source_commit':source['commit'],'baseline_files':len(source['sources']),'pins':len(pins)}))
