"""Capture first ordinary subprocess output, return code and UTC boundaries."""
import argparse
import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

parser = argparse.ArgumentParser()
parser.add_argument('--record-dir', type=Path, required=True)
parser.add_argument('command', nargs=argparse.REMAINDER)
args = parser.parse_args()
args.record_dir.mkdir()
command = args.command[1:] if args.command[:1] == ['--'] else args.command
started = datetime.now(timezone.utc).isoformat()
result = subprocess.run(command, capture_output=True)
(args.record_dir / 'stdout.txt').write_bytes(result.stdout)
(args.record_dir / 'stderr.txt').write_bytes(result.stderr)
(args.record_dir / 'RESULT.json').write_text(json.dumps({'command': command, 'started_utc': started,
    'ended_utc': datetime.now(timezone.utc).isoformat(), 'returncode': result.returncode}, indent=2), encoding='utf-8')
print(result.stdout.decode('utf-8', errors='replace'))
if result.stderr:
    print(result.stderr.decode('utf-8', errors='replace'), file=sys.stderr)
sys.exit(result.returncode)
