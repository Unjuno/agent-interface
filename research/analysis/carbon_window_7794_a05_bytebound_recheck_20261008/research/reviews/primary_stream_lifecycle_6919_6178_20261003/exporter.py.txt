from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import subprocess

ROOT = Path(__file__).resolve().parents[1]
REPO = ROOT / 'work/agent-interface'
OUT = ROOT / 'outputs/review6919'
WORK = ROOT / 'work/review6919'
EXPORT = WORK / 'export'
P = json.loads((OUT / 'PROPOSAL.json').read_bytes())
HEAD, BASE = P['head_sha'], P['review_base_sha']
OLD = 'dbe05f5ce33b6827dbd3d09e8a85da1e2d20f51a'
PREFIX1 = 'research/integration/primary_input_error_57_20261003_01a0ff53/'
PREFIX2 = 'research/integration/primary_stream_lifecycle_57_20261003_01a0ff53/'
record = {'started_utc': datetime.now(timezone.utc).isoformat(), 'commands': [], 'images': [], 'producer_auditor_test_runtime_calls': 0}
WORK.mkdir(exist_ok=False)
EXPORT.mkdir()
streams = WORK / 'command-streams'
streams.mkdir()
def sha(b): return hashlib.sha256(b).hexdigest()
def save(): (OUT / 'EXPORTS.json').write_text(json.dumps(record, indent=2, sort_keys=True) + '\n', encoding='utf-8', newline='\n')
def git(*args, data=None, timeout=45, no_lazy=False):
    start = datetime.now(timezone.utc).isoformat()
    env = dict(os.environ, GIT_TERMINAL_PROMPT='0')
    if no_lazy: env['GIT_NO_LAZY_FETCH'] = '1'
    p = subprocess.run(['git', *args], cwd=REPO, env=env, input=data, capture_output=True, timeout=timeout)
    n = str(len(record['commands'])).zfill(3)
    for kind in ('stdout', 'stderr'): (streams / (n + '.' + kind)).write_bytes(getattr(p, kind))
    record['commands'].append({'argv': ['git', *args], 'stdin_sha256': None if data is None else sha(data), 'started_utc': start, 'finished_utc': datetime.now(timezone.utc).isoformat(), 'exit': p.returncode, 'stdout_sha256': sha(p.stdout), 'stderr_sha256': sha(p.stderr)})
    save()
    if p.returncode: raise RuntimeError(p.stderr.decode(errors='replace'))
    return p.stdout
def tree(ref, paths):
    entries = git('ls-tree', '-r', '-z', ref, '--', *paths).split(b'\0')
    out = {}
    for entry in entries:
        if entry:
            fields, name = entry.split(b'\t', 1)
            mode, kind, oid = fields.decode().split()
            if kind != 'blob' or mode != '100644': raise RuntimeError('nonregular image')
            out[name.decode()] = {'mode': mode, 'blob_oid': oid}
    return out

try:
    canon = {k:v for k,v in P.items() if k != 'proposal_digest'}
    if sha(json.dumps(canon, sort_keys=True, separators=(',', ':'), ensure_ascii=True).encode()) != P['proposal_digest']: raise RuntimeError('proposal digest')
    if P['committee'] != ['01a0ff59-2f23-7ff0-ba99-cc5cb8d494fc','01a0ff2e-17a3-70a1-908e-0d6dd78c45e9','01a0ff58-6178-7ea1-a6d5-cf09260b91e3']: raise RuntimeError('assigned committee')
    paths = git('diff', '--name-only', '--no-renames', BASE, HEAD).decode().splitlines()
    if len(paths) != 259: raise RuntimeError('scope size')
    before, after = tree(BASE, P['scope_paths']), tree(HEAD, P['scope_paths'])
    if any(path not in after for path in paths): raise RuntimeError('deleted image')
    ids = sorted({o['blob_oid'] for path in paths for o in (before.get(path), after.get(path)) if o})
    checked = git('cat-file', '--batch-check=%(objectname) %(objecttype) %(objectsize)', data=('\n'.join(ids) + '\n').encode(), no_lazy=True).decode().splitlines()
    missing = [line.split()[0] for line in checked if line.endswith(' missing')]
    if missing:
        git('-c', 'fetch.negotiationAlgorithm=noop', 'fetch', '--no-tags', '--no-write-fetch-head', '--recurse-submodules=no', '--filter=blob:none', '--stdin', 'origin', data=('\n'.join(missing) + '\n').encode(), timeout=60)
    batch = git('cat-file', '--batch', data=('\n'.join(ids) + '\n').encode(), no_lazy=True)
    blobs = {}; cursor = 0
    for oid in ids:
        eol = batch.index(b'\n', cursor)
        fields = batch[cursor:eol].split()
        if fields[0].decode() != oid or fields[1] != b'blob': raise RuntimeError('batch image')
        size = int(fields[2]); cursor = eol + 1
        blobs[oid] = batch[cursor:cursor + size]; cursor += size + 1
    if cursor != len(batch): raise RuntimeError('batch trailing bytes')
    for path in paths:
        image = {'path': path, 'before': None, 'after': None}
        for stage, mapping in [('before', before), ('after', after)]:
            if path in mapping:
                meta = mapping[path]
                blob = blobs[meta['blob_oid']]
                image[stage] = dict(meta, bytes=len(blob), sha256=sha(blob))
                if stage == 'after':
                    target = EXPORT / path
                    if not target.resolve().is_relative_to(EXPORT.resolve()): raise RuntimeError('unsafe path')
                    target.parent.mkdir(parents=True, exist_ok=True)
                    target.write_bytes(blob)
        record['images'].append(image)
    map_digest = sha(json.dumps(record['images'], sort_keys=True, separators=(',', ':'), ensure_ascii=True).encode())
    if map_digest != P['changed_blobs_sha256']: raise RuntimeError('canonical259 image map')
    old = tree(OLD, [PREFIX1])
    current1 = {k:v for k,v in after.items() if k.startswith(PREFIX1)}
    if current1 != old or len(old) != 51: raise RuntimeError('original v1 custody')
    manifests = []
    for prefix in (PREFIX1, PREFIX2):
        pkg = EXPORT / prefix
        meta = P['evidence_manifests'][prefix.rstrip('/')]
        b = (pkg / 'SHA256SUMS').read_bytes()
        if sha(b) != meta['sha256']: raise RuntimeError('manifest original')
        targets = []
        for line in b.decode().splitlines():
            digest, name = line.split('  ', 1)
            target = (pkg / name).resolve()
            if name in targets or not target.is_relative_to(pkg.resolve()) or sha(target.read_bytes()) != digest: raise RuntimeError('manifest member')
            targets.append(name)
        actual = sorted(p.relative_to(pkg).as_posix() for p in pkg.rglob('*') if p.is_file() and p != pkg / 'SHA256SUMS')
        if sorted(targets) != actual or len(targets) != meta['members']: raise RuntimeError('manifest full coverage')
        manifests.append({'prefix': prefix, 'targets': len(targets), 'delivered': len(actual) + 1})
    patch = git('--attr-source=' + HEAD, 'diff', '--binary', '--full-index', '--no-ext-diff', '--no-renames', BASE, HEAD)
    (OUT / 'proposal.diff').write_bytes(patch)
    git('--attr-source=' + HEAD, 'diff', '--check', BASE, HEAD)
    record.update({'proposal_digest': P['proposal_digest'], 'map_digest': map_digest, 'all259_images_match': True, 'original_v1_51_unchanged': True, 'manifests': manifests, 'patch_sha256': sha(patch), 'patch_bytes': len(patch), 'finished_utc': datetime.now(timezone.utc).isoformat()})
    save()
    print(json.dumps({k:record[k] for k in ('all259_images_match','map_digest','original_v1_51_unchanged','manifests','patch_bytes','producer_auditor_test_runtime_calls')}, sort_keys=True))
except BaseException as exc:
    record['first_failure'] = repr(exc)
    save()
    raise
finally:
    (OUT / 'exporter.py.txt').write_bytes(Path(__file__).read_bytes())
