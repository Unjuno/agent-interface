import datetime
import hashlib
import json
from pathlib import Path
import subprocess
import sys

repo = Path(__file__).parent / 'agent-interface'
label = sys.argv[1]
argv = sys.argv[2:]
out = repo / 'runtime/results/kernel-begin-causality-01a0ff58/evidence'
private = Path(__file__).parent / 'private-begin-causality'
out.mkdir(parents=True, exist_ok=True)
private.mkdir(exist_ok=True)
start = datetime.datetime.now(datetime.timezone.utc).isoformat()
run = subprocess.run(argv, cwd=repo, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=60)
end = datetime.datetime.now(datetime.timezone.utc).isoformat()
with (private / (label + '.log')).open('xb') as f:
    f.write(run.stdout)
public = run.stdout.decode('utf-8', errors='replace').replace(str(repo), '<repo>').replace(repo.as_posix(), '<repo>')
with (out / (label + '.log')).open('x', encoding='utf-8', newline='\n') as f:
    f.write(public)
record = {'label': label, 'argv': argv, 'cwd': 'repository', 'started_utc': start, 'ended_utc': end, 'exit_code': run.returncode, 'original_log_sha256': hashlib.sha256(run.stdout).hexdigest(), 'public_log_sha256': hashlib.sha256(public.encode()).hexdigest(), 'redacted_repository_prefix': public.encode() != run.stdout}
with (out / (label + '.command.json')).open('x', encoding='utf-8', newline='\n') as f:
    json.dump(record, f, indent=2)
    f.write('\n')
print(json.dumps(record))
sys.exit(run.returncode)
