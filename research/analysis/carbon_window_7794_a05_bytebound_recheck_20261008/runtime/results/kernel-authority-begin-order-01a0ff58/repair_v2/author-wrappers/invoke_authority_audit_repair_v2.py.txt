from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

root = Path(__file__).resolve().parent
package = root/'agent-interface/runtime/results/kernel-authority-begin-order-01a0ff58'
out = package/'repair_v2'
private = root/'authority-audit-repair-v2-private'
def redact_text(value):
    return value.replace(str(package),'<PACKAGE>').replace(str(root),'<WORKSPACE>').replace(str(Path.home()),'<HOME>')
def redact_bytes(value):
    return value.replace(str(package).encode(),b'<PACKAGE>').replace(str(root).encode(),b'<WORKSPACE>').replace(str(Path.home()).encode(),b'<HOME>')
profile = sys.argv[1]
commands = {
    'RED': [sys.executable,str(out/'check_audit_v2.py'),str(package/'audit_retained.py')],
    'OLD_ROW1': [sys.executable,str(package/'audit_retained.py'),str(out/'row1_false_expiration.json')],
    'OLD_ROW100': [sys.executable,str(package/'audit_retained.py'),str(out/'row100_false_mismatch.json')],
    'GREEN': [sys.executable,str(out/'check_audit_v2.py')],
    'OPTIMIZED': [sys.executable,'-O',str(out/'check_audit_v2.py')],
    'AUDIT_V2': [sys.executable,str(out/'audit_reasons_v2.py'),str(package/'matrix.raw.json')],
    'V2_ROW1': [sys.executable,str(out/'audit_reasons_v2.py'),str(out/'row1_false_expiration.json')],
    'V2_ROW100': [sys.executable,str(out/'audit_reasons_v2.py'),str(out/'row100_false_mismatch.json')],
}
command = commands[profile]
owned = private/profile
owned.mkdir()
(out/'logs').mkdir(exist_ok=True)
start = datetime.now(timezone.utc).isoformat()
env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1')
proc = subprocess.Popen(command,cwd=package,stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=env)
stdout,stderr = proc.communicate(timeout=30)
end = datetime.now(timezone.utc).isoformat()
for name,payload in [('stdout',stdout),('stderr',stderr)]:
    (owned/(name+'.txt')).write_bytes(payload)
    public = redact_bytes(payload)
    (out/'logs'/(profile+'.'+name+'.txt')).write_bytes(public)
receipt = {'label':profile,'actual_argv':command,'cwd':str(package),'pid':proc.pid,'started_utc':start,'ended_utc':end,'actual_child_exit':proc.returncode,'stdout_sha256':hashlib.sha256(stdout).hexdigest(),'stderr_sha256':hashlib.sha256(stderr).hexdigest(),'stdout_bytes':len(stdout),'stderr_bytes':len(stderr),'scope':'Raw-only ordinary retained-data check, no producer/candidate/kernel/backend/formal replay'}
(owned/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8')
public = dict(receipt)
public['actual_argv']=[redact_text(s) for s in command]
public['cwd']='<PACKAGE>'
public['redaction']='Only owned PACKAGE/WORKSPACE/HOME path prefixes in published argv/cwd/traceback; originals retained privately'
public['published_stdout_sha256']=hashlib.sha256(redact_bytes(stdout)).hexdigest()
public['published_stderr_sha256']=hashlib.sha256(redact_bytes(stderr)).hexdigest()
(out/'logs'/(profile+'.receipt.json')).write_text(json.dumps(public,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps({'label':profile,'actual_child_exit':proc.returncode,'stdout_sha256':receipt['stdout_sha256'],'stderr_sha256':receipt['stderr_sha256'],'started_utc':start,'ended_utc':end}))
raise SystemExit(proc.returncode)
