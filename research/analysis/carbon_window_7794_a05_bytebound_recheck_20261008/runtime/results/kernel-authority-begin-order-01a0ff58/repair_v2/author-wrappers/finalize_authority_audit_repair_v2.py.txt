join={}
for arm,name in source_names.items():
    payload=(package/name).read_bytes()
    module=ast.parse(payload)
    cls=next(n for n in module.body if isinstance(n,ast.ClassDef) and n.name=='RequestLifecycle')
    begin=next(n for n in cls.body if isinstance(n,ast.FunctionDef) and n.name=='begin_execution')
    guards=[]
    for node in begin.body:
        if isinstance(node,ast.If):
            raise_node=node.body[0]
            assert isinstance(raise_node,ast.Raise) and isinstance(raise_node.exc,ast.Call)
            assert isinstance(raise_node.exc.func,ast.Name) and raise_node.exc.func.id=='ContractError'
            reason=raise_node.exc.args[0].value
            guards.append({'source_line':node.lineno,'condition':ast.unparse(node.test),'message':reason})
    expect=['execution requires active authority','execution has already begun','now_ns must be nonnegative integer']
    if arm=='observation_floor':expect+=['execution precedes observation capture']
    if arm=='authorization_floor':expect+=['execution precedes accepted authorization']
    expect+=['authority expired before execution','execution request changed bound target or authority']
    assert [g['message'] for g in guards]==expect
    counts=Counter(r['error_message'] for r in raw['rows'] if r['arm']==arm and not r['accepted'])
    assert set(counts).issubset(set(expect))
    join[arm]={'frozen_source':name,'sha256':hashlib.sha256(payload).hexdigest(),'begin_guards_in_order':guards,'actual_original_refusal_cause_counts':dict(counts)}
