"""Continue archive checks after preserving reader01 line-ending mismatch."""
import datetime as dt
import hashlib
import json
import pathlib
import re
import subprocess
import sys

ROOT = pathlib.Path(__file__).resolve().parent.parent
REPO = ROOT / 'work/agent-interface'
PREFIX = 'research/analysis/checkpoint_delivery_6089_20261003_01a0ff59/'
DEST = REPO / PREFIX

def git(*args):
    return subprocess.run(['git', *args], cwd=REPO, capture_output=True, check=True).stdout

def sha(data):
    return hashlib.sha256(data).hexdigest()

def put(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as stream:
        stream.write(data)

def jb(value):
    return (json.dumps(value, indent=2, sort_keys=True) + '\n').encode()

freeze = json.loads((DEST / 'FREEZE.json').read_bytes())
base = freeze['prospective_base_main']
if git('rev-parse', 'HEAD').decode().strip() != base:
    raise ValueError('unexpected head')
for name, pin in freeze['source_files'].items():
    data = (DEST / name).read_bytes()
    if sha(data) != pin['sha256'] or len(data) != pin['bytes']:
        raise ValueError('frozen source changed: ' + name)
check_path = REPO / 'research/analysis/check_index.py'
before = check_path.read_bytes()
exact = git('show', f'{base}:research/analysis/check_index.py')
if before.replace(b'\r\n', b'\n') != exact or before == exact:
    raise ValueError('reader01 cause is not solely CRLF checkout conversion')
put(DEST / 'execution/reader01-index-checker-checkout.py.txt', before)
put(DEST / 'execution/reader01-failure.tool-record.json', jb({'kind': 'ordinary delivery reader preparation failure; not scientific assay failure', 'command': ['python', '-B', 'work/deliver_checkpoint_delivery.py'], 'exit_code': 1, 'tool_wall_time_seconds': 0.3049996, 'exact_start_end_utc': None, 'error': 'ValueError: unexpected existing index checker', 'cause': '155 CRLF checkout line endings; normalized content is identical to prospective Git blob', 'local_bytes': len(before), 'local_sha256': sha(before), 'git_bytes': len(exact), 'git_sha256': sha(exact), 'assay_rerun': False, 'frozen_files_changed': False, 'reader01_source': 'helpers/deliver_checkpoint_delivery.py.txt'}))
check_path.write_bytes(exact)
put(DEST / 'helpers/deliver_checkpoint_delivery_reader02.py.txt', pathlib.Path(__file__).read_bytes())
report = DEST / 'REPORT.md'
report.write_bytes(report.read_bytes() + b'\nDelivery reader01 stopped before running checks because the existing checkout of the index checker had155 CRLF endings while its Git blob had LF. The normalized contents were identical. The original reader and checkout witness are retained, and reader02 materialized the exact pinned checker before continuing. No scientific source, raw, threshold or matrix was changed or rerun.\n')
syntax = []
for name in ['assay.py', 'audit.py', 'test_reference.py', 'conftest.py', 'sources/candidate.py.txt', 'sources/audit.py.txt']:
    compile((DEST / name).read_bytes(), name, 'exec')
    syntax.append(name)
put(DEST / 'execution/delivery-syntax.json', jb({'kind': 'compile-only, no execution or imports', 'checked': syntax, 'python': sys.version, 'passed': True}))

def check(label, executable, args, expected):
    started = dt.datetime.now(dt.timezone.utc).isoformat()
    child = subprocess.run([str(executable), *args], cwd=REPO, capture_output=True, timeout=30)
    ended = dt.datetime.now(dt.timezone.utc).isoformat()
    put(DEST / f'execution/{label}.stdout.txt', child.stdout)
    put(DEST / f'execution/{label}.stderr.txt', child.stderr)
    put(DEST / f'execution/{label}.receipt.json', jb({'command': ['python', *args], 'cwd_alias': 'repository root', 'started_utc': started, 'ended_utc': ended, 'exit_code': child.returncode, 'expected_exit_code': expected, 'stdout_sha256': sha(child.stdout), 'stderr_sha256': sha(child.stderr), 'producer_invocations': 0, 'formal_invocations': 0}))
    if child.returncode != expected:
        raise ValueError('delivery check unexpected exit: ' + label)
    return child.stdout.decode(errors='replace')

collection = check('pytest-noncollection', ROOT / 'work/venv/Scripts/python.exe', ['-B', '-m', 'pytest', '--collect-only', '-q', '-p', 'no:cacheprovider', PREFIX], 5)
if 'no tests collected' not in collection:
    raise ValueError('unexpected collection transcript')
index = check('analysis-index', pathlib.Path(sys.executable), ['-B', 'research/analysis/check_index.py'], 0)
workflow_names = git('ls-tree', '-r', '--name-only', base, '.github/workflows').decode().splitlines()
workflow_checks = []
for path in workflow_names:
    if not path.endswith(('.yml', '.yaml')):
        continue
    data = git('show', f'{base}:{path}')
    if PREFIX.encode() in data or DEST.name.encode() in data:
        raise ValueError('existing workflow references this archive')
    if re.search(rb'(?m)^\s*run:.*(?:pytest|unittest discover).*research/analysis(?:\s|/)', data):
        workflow_checks.append(path)
put(DEST / 'execution/delivery-scope.json', jb({'base_sha': base, 'frozen_files_unchanged': len(freeze['source_files']), 'pytest_collected_tests': 0, 'pytest_version': '9.1.1, dedicated host venv', 'pytest_exit_5_is_expected_noncollection_not_test_pass': True, 'analysis_index_checker_sha256': sha(exact), 'index_change': 'one sorted directory entry; preexisting bytes retained', 'workflow_files_inspected': len([x for x in workflow_names if x.endswith(('.yml', '.yaml'))]), 'existing_workflows_reference_new_archive': False, 'analysis_discovery_workflows': workflow_checks, 'runtime_or_workflow_files_changed': False, 'scope': 'new archive plus one directory index entry; no prior formal run or full runtime suite'}))
for name, pin in freeze['source_files'].items():
    if sha((DEST / name).read_bytes()) != pin['sha256']:
        raise ValueError('post-delivery frozen source change: ' + name)
paths = sorted(p for p in DEST.rglob('*') if p.is_file())
manifest = ''.join(f'{sha(path.read_bytes())}  {path.relative_to(DEST).as_posix()}\n' for path in paths)
put(DEST / 'SHA256SUMS', manifest.encode())
print(json.dumps({'files': len(paths) + 1, 'manifest_entries': len(paths), 'collection': collection.strip(), 'index': index.strip(), 'raw_sha256': sha((DEST / 'raw.json').read_bytes()), 'manifest_sha256': sha(manifest.encode())}, indent=2))
