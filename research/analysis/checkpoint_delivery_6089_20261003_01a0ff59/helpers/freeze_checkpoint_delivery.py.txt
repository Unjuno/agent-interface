import datetime as dt
import hashlib
import json
import pathlib
import subprocess
import sys

ROOT = pathlib.Path(__file__).resolve().parent.parent
REPO = ROOT / 'work/agent-interface'
PREFIX = 'research/analysis/checkpoint_delivery_6089_20261003_01a0ff59/'
DEST = REPO / PREFIX

def git(*args):
    return subprocess.run(['git', *args], cwd=REPO, capture_output=True, check=True).stdout

def sha(data):
    return hashlib.sha256(data).hexdigest()

def put(path, data):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as stream:
        stream.write(data)

def json_bytes(value):
    return (json.dumps(value, indent=2, sort_keys=True) + '\n').encode()

def now():
    return dt.datetime.now(dt.timezone.utc).isoformat()

base = git('rev-parse', 'HEAD').decode().strip()
if git('rev-parse', 'origin/main').decode().strip() != base:
    raise ValueError('prospective branch/main mismatch')
source_map = json.loads((DEST / 'SOURCE_MAP.json').read_bytes())
for path, pin in source_map['sources'].items():
    current = git('show', f'{base}:{path}')
    if sha(current) != pin['sha256'] or current != (DEST / pin['snapshot']).read_bytes():
        raise ValueError('legacy source changed before freeze: ' + path)
for path in ['FREEZE.json', 'raw.json', 'audit.json', 'execution/assay.stdout.txt', 'execution/audit.stdout.txt']:
    if (DEST / path).exists():
        raise ValueError('output already exists: ' + path)
transcript = '''test_empty_is_not_delivered_fresh (__main__.ReferenceTests.test_empty_is_not_delivered_fresh) ... ok
test_explicit_missing_has_identical_full_output (__main__.ReferenceTests.test_explicit_missing_has_identical_full_output) ... ok
test_hypothesized_fallback_hides_planted_optimism (__main__.ReferenceTests.test_hypothesized_fallback_hides_planted_optimism) ... ok
test_path_prefix_and_release_boundary (__main__.ReferenceTests.test_path_prefix_and_release_boundary) ... ok
test_valid_receipt_and_invalidation_are_distinct (__main__.ReferenceTests.test_valid_receipt_and_invalidation_are_distinct) ... ok

----------------------------------------------------------------------
Ran 5 tests in 0.024s

OK
'''
put(DEST / 'execution/pre-freeze-reference.tool-transcript.txt', transcript.encode())
put(DEST / 'execution/pre-freeze-reference.tool-record.json', json_bytes({'source': 'retained exec_command tool result before freeze, combined output transcription', 'command': ['python', '-B', 'test_reference.py', '-v'], 'cwd_alias': 'study package', 'exit_code': 0, 'tests': 5, 'tool_wall_time_seconds': 0.2729121, 'exact_start_end_utc': None, 'timing_limit': 'Exact UTC was not independently recorded for this construction test; not reconstructed from later timestamps.', 'transcript_sha256': sha(transcript.encode())}))
files = {}
for path in sorted(DEST.rglob('*')):
    if path.is_file():
        data = path.read_bytes()
        files[path.relative_to(DEST).as_posix()] = {'bytes': len(data), 'sha256': sha(data)}
freeze = {'schema': 'checkpoint-delivery-freeze-v1', 'worker_id': '01a0ff59-0820-7c81-8ef1-f7c3e48ab67d', 'kind': 'ordinary new-input finite contract assay, not a historical/formal allocation', 'branch': git('branch', '--show-current').decode().strip(), 'prospective_base_main': base, 'legacy_snapshot_base': source_map['base_main'], 'legacy_sources_byte_equal_at_prospective_base': True, 'frozen_utc': now(), 'matrix': {'contexts': 3, 'encodings': 4, 'arms': 2, 'rows': 24}, 'raw_byte_cap': 1 << 20, 'max_path_steps': 8, 'producer_invocations_before_freeze': 0, 'raw_auditor_invocations_before_freeze': 0, 'ordinary_reference_tests_before_freeze': 5, 'source_files': files, 'first_outcome_rule': 'one producer then one separate raw auditor; source/output/process failure is retained STOP, no replacement matrix or outcome-dependent source change', 'historical_formal_invocations': 0, 'shared_resources': [], 'live_or_main_authority': False}
put(DEST / 'FREEZE.json', json_bytes(freeze))

private_receipts = []
def invoke(label, args):
    started = now()
    try:
        child = subprocess.run([sys.executable, '-B', *args], cwd=DEST, capture_output=True, timeout=20)
    except subprocess.TimeoutExpired as exc:
        stdout, stderr, exit_code = exc.stdout or b'', exc.stderr or b'', None
        process_result = 'TIMEOUT_STOP'
    else:
        stdout, stderr, exit_code = child.stdout, child.stderr, child.returncode
        process_result = 'EXITED'
    ended = now()
    put(DEST / f'execution/{label}.stdout.txt', stdout)
    put(DEST / f'execution/{label}.stderr.txt', stderr)
    public = {'command': ['python', '-B', *args], 'cwd_alias': 'study package', 'python': sys.version, 'started_utc': started, 'ended_utc': ended, 'exit_code': exit_code, 'process_result': process_result, 'stdout_sha256': sha(stdout), 'stdout_bytes': len(stdout), 'stderr_sha256': sha(stderr), 'stderr_bytes': len(stderr)}
    put(DEST / f'execution/{label}.receipt.json', json_bytes(public))
    private_receipts.append({**public, 'actual_executable': sys.executable, 'actual_cwd': str(DEST)})
    if exit_code != 0:
        put(DEST / 'STOP.json', json_bytes({'stage': label, 'process_result': process_result, 'exit_code': exit_code, 'candidate_not_replaced': True}))
        return None
    if len(stdout) > 1 << 20:
        put(DEST / 'STOP.json', json_bytes({'stage': label, 'reason': 'RAW_BYTE_CAP', 'candidate_not_replaced': True}))
        return None
    json.loads(stdout)
    return stdout

raw = invoke('assay', ['assay.py'])
if raw is not None:
    put(DEST / 'raw.json', raw)
    result = invoke('audit', ['audit.py', 'raw.json'])
    if result is not None:
        put(DEST / 'audit.json', result)
        for name, pin in files.items():
            if sha((DEST / name).read_bytes()) != pin['sha256']:
                raise ValueError('frozen input changed after execution: ' + name)
        record = {'finished_utc': now(), 'frozen_files_read_back_unchanged': len(files), 'raw_sha256': sha(raw), 'raw_bytes': len(raw), 'audit_sha256': sha(result), 'producer_invocations': 1, 'raw_auditor_invocations': 1, 'historical_formal_invocations': 0, 'status': json.loads(result)['status']}
        put(DEST / 'ENDPOINTS.json', json_bytes(record))
        print(json.dumps({'endpoints': record, 'audit': json.loads(result)}, indent=2))
put(ROOT / 'work/checkpoint-delivery-private-execution.json', json_bytes(private_receipts))
