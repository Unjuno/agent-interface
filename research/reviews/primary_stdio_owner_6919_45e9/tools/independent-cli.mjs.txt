/* Frozen public --config entry witnesses with actual owned OS stdout pipes. */
import assert from 'node:assert/strict';
import {spawn} from 'node:child_process';
import {readFile,writeFile} from 'node:fs/promises';
import {join} from 'node:path';
import {fileURLToPath} from 'node:url';
const root=fileURLToPath(new URL('.',import.meta.url));
const out=join(root,'independent-cli-first');const source=join(root,'source/runtime/host_v1/primary_stdio.mjs');
const results=[];
function gone(pid){try{process.kill(pid,0);return false;}catch(error){assert.equal(error.code,'ESRCH');return true;}}
for(const mode of ['healthy-eof','ready-pipe-closed','terminal-pipe-closed']){
 const dir=join(out,mode);const started=new Date().toISOString();
 const child=spawn(process.execPath,[source,'--config',join(dir,'config.json')],{stdio:['pipe','pipe','pipe'],windowsHide:true});
 const pid=child.pid;assert.equal(typeof pid,'number');let stdout='',stderr='',seenReady=false,timedOut=false,spawnError=null;
 const guard=setTimeout(()=>{timedOut=true;child.kill();},5500);
 child.on('error',error=>{spawnError=String(error);});
 child.stderr.on('data',chunk=>{stderr+=chunk.toString('utf8');});
 child.stdout.on('data',chunk=>{
  stdout+=chunk.toString('utf8');
  if(!seenReady&&stdout.includes('\n')){
   const first=JSON.parse(stdout.slice(0,stdout.indexOf('\n')));assert.equal(first.status,'ready');seenReady=true;
   if(mode==='terminal-pipe-closed')child.stdout.destroy();
   child.stdin.end();
  }
 });
 if(mode==='ready-pipe-closed')child.stdout.destroy();
 const endpoint=await new Promise(resolve=>child.once('close',(code,signal)=>resolve({code,signal})));clearTimeout(guard);
 await writeFile(join(dir,'cli.stdout.txt'),stdout,{flag:'wx'});await writeFile(join(dir,'cli.stderr.txt'),stderr,{flag:'wx'});
 const receipt={mode,pid,command:[process.execPath,source,'--config',join(dir,'config.json')],started_utc:started,
  ended_utc:new Date().toISOString(),...endpoint,timed_out:timedOut,spawn_error:spawnError,observed_ready:seenReady,
  stdout_bytes:Buffer.byteLength(stdout),stderr_bytes:Buffer.byteLength(stderr)};
 await writeFile(join(dir,'cli.receipt.json'),JSON.stringify(receipt,null,2)+'\n',{flag:'wx'});
 assert.equal(timedOut,false);assert.equal(spawnError,null);assert.equal(endpoint.signal,null);
 const fixtureStart=JSON.parse(await readFile(join(dir,'fixture-start.json'),'utf8'));
 const fixtureExit=JSON.parse(await readFile(join(dir,'fixture-exit.json'),'utf8'));
 const hostExit=JSON.parse(await readFile(join(dir,'host/exit.json'),'utf8'));
 assert.equal(fixtureStart.pid,fixtureExit.pid);assert.equal(fixtureStart.ppid,pid);
 assert.equal(fixtureExit.code,0);assert.equal(fixtureExit.request_bytes,0);
 assert.deepEqual(hostExit,{code:0,signal:null});assert.equal(gone(pid),true);assert.equal(gone(fixtureStart.pid),true);
 if(mode==='healthy-eof'){
  assert.equal(endpoint.code,0);assert.equal(stderr,'');const rows=stdout.trim().split('\n').map(JSON.parse);
  assert.deepEqual(rows.map(x=>x.status),['ready','terminal']);assert.equal(rows[0].state.next_id,1);assert.equal(rows[1].state.next_id,1);
  assert.deepEqual(rows[1].exit,hostExit);
 }else{
  assert.equal(endpoint.code,2);assert.match(stderr,/EPIPE|broken pipe/i);
  if(mode==='ready-pipe-closed')assert.equal(stdout,'');
  else assert.deepEqual(stdout.trim().split('\n').map(JSON.parse).map(x=>x.status),['ready']);
 }
 const proof={...receipt,fixture_pid:fixtureStart.pid,fixture_code:fixtureExit.code,fixture_request_bytes:fixtureExit.request_bytes,
  host_exit:hostExit,parent_and_owned_fixture_absent:true,no_native_backend_calls:true};
 await writeFile(join(dir,'RECONCILIATION.json'),JSON.stringify(proof,null,2)+'\n',{flag:'wx'});
 results.push(proof);
}
assert.equal(results.length,3);
console.log(JSON.stringify({status:'PASS_SCOPED_PUBLIC_CLI_OS_OUTPUT_PIPE',actual_cases:3,cases:results,
 original_author_producer_or_formal_replays:0,actual_backend_or_input_operations:0}));
