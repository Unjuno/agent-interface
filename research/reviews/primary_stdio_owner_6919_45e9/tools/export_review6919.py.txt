 result={}

  metadata,path=row.split('\t');mode,kind,oid=metadata.split();assert kind=='blob';result[path]={'mode':mode,'blob_oid':oid}
 return result

 'bytes':sum(x['bytes'] for x in records),'primary_index_unchanged':True}))
