"""Read-only ordinary evidence verification; no author producer or old formal replay."""
from datetime import datetime, timezone
import ast
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys

ROOT = Path(__file__).resolve().parent
REPO = ROOT.parent / 'agent-interface'
PREFIX = 'research/analysis/checkpoint_delivery_6089_20261003_01a0ff59'
PACK = ROOT / 'source' / PREFIX
OUT = ROOT / 'proof-v2'
OUT.mkdir(exist_ok=False)
START = datetime.now(timezone.utc).isoformat()
INDEX = (REPO / '.git/index').read_bytes()
ENV = dict(os.environ, GIT_NO_LAZY_FETCH='1')
def sha(b): return hashlib.sha256(b).hexdigest()
def load(p): return json.loads(p.read_bytes())
def put(name, value):
    (OUT / name).write_text(json.dumps(value, sort_keys=True, indent=2) + '\n', encoding='utf-8')
def git(*args):
    proc = subprocess.run(['git', *args], cwd=REPO, env=ENV, capture_output=True, timeout=20)
    assert proc.returncode == 0, (args, proc.stderr.decode(errors='replace'))
    return proc.stdout
def canonical(x): return json.dumps(x, sort_keys=True, separators=(',', ':'), ensure_ascii=True).encode()

# First unwrapped verifier observation is retained, not replaced by this repeat.
put('FIRST_VERIFIER_OBSERVATION.json', {'command': ['python', '-B', '.\\work\\review6918\\independent.py'],
    'tool_chunk_id': 'b49df1', 'exit_code': 0, 'tool_wall_time_seconds': 0.4514723,
    'exact_start_end_utc': None, 'full_stream_file_custody': False,
    'qualification': 'First result/full output retained in tool conversation; no independent UTC or original stream files. This later read-only repeat does not recover missing first bytes/timing.',
    'observed_result': {'rows':24,'reconstruction_errors':0,'legacy_empty_mismatches':3,
    'normalized_mismatches':0,'nonempty_pairs':9,'normalized_missing_equivalences':3,'new_controls_rejected':12}})
export = load(ROOT / 'EXPORT.json')
for item in export['blobs']:
    b = (ROOT / item['side'] / item['path']).read_bytes()
    assert len(b) == item['bytes'] and sha(b) == item['sha256']
    assert hashlib.sha1(('blob %d\0' % len(b)).encode()+b).hexdigest() == item['blob_oid']
freeze = load(PACK / 'FREEZE.json')
assert len(freeze['source_files']) == 16
for path, item in freeze['source_files'].items():
    b = (PACK/path).read_bytes()
    assert len(b) == item['bytes'] and sha(b) == item['sha256']
source = load(PACK / 'SOURCE_MAP.json')
pins=[]
for path, item in source['sources'].items():
    b = (PACK/item['snapshot']).read_bytes()
    assert len(b) == item['bytes'] and sha(b) == item['sha256']
    for ref in (source['base_main'],freeze['prospective_base_main']):
        got = git('ls-tree',ref,'--',path).decode().strip().split('\t')[0].split()[2]
        assert got == item['blob']
    assert git('cat-file','blob',item['blob']) == b
    pins.append({'path':path,**item,'both_historical_refs_exact':True})
put('LEGACY_PINS.json',pins)
receipts={}
for label, output in [('assay','raw.json'),('audit','audit.json')]:
    r=load(PACK/f'execution/{label}.receipt.json')
    assert r['exit_code']==0 and r['process_result']=='EXITED'
    for stream in ('stdout','stderr'):
        b=(PACK/f'execution/{label}.{stream}.txt').read_bytes()
        assert sha(b)==r[stream+'_sha256'] and len(b)==r[stream+'_bytes']
    assert (PACK/output).read_bytes()==(PACK/f'execution/{label}.stdout.txt').read_bytes()
    assert datetime.fromisoformat(r['started_utc']) <= datetime.fromisoformat(r['ended_utc'])
    receipts[label]=r
end=load(PACK/'ENDPOINTS.json')
assert datetime.fromisoformat(freeze['frozen_utc']) <= datetime.fromisoformat(receipts['assay']['started_utc'])
assert datetime.fromisoformat(receipts['assay']['ended_utc']) <= datetime.fromisoformat(receipts['audit']['started_utc'])
assert datetime.fromisoformat(receipts['audit']['ended_utc']) <= datetime.fromisoformat(end['finished_utc'])
assert end['raw_sha256']==sha((PACK/'raw.json').read_bytes()) and end['audit_sha256']==sha((PACK/'audit.json').read_bytes())
assert end['producer_invocations']==1 and end['raw_auditor_invocations']==1 and end['historical_formal_invocations']==0
put('RETAINED_PROCESS_JOINS.json',{'receipts':receipts,'endpoints':end,'exact_stream_joins':True,
    'qualification':'Byte and timestamp consistency, not independent observation of historical execution; equal freeze/start timestamp allowed by clock resolution.'})
proposal=load(ROOT/'proposal-v2.json')
digest=sha(canonical(proposal))
assert digest=='281cd872799ab2d7c7a23d669611d74c56bdd5779c1a362a7b3038bb51da7be6'
head=proposal['head_sha'];base=proposal['review_base_sha']
diff=git(*proposal['diff_command'][1:])
(OUT/'PROPOSAL.diff').write_bytes(diff)
assert sha(diff)==proposal['diff_sha256'], ('diff mismatch',sha(diff))
paths=git('diff','--name-only','--no-renames',base,head).decode().splitlines()
assert paths==proposal['changed_paths'] and len(paths)==47
all_dirs=[]
for row in git('ls-tree',head+':research/analysis').decode().splitlines():
    metadata,path=row.split('\t')
    if metadata.split()[1]=='tree':all_dirs.append(path)
dirs=sorted({path.split('/')[2] for path in git('ls-tree','-r','--name-only',head,'--','research/analysis').decode().splitlines()
    if len(path.split('/'))==4 and path.split('/')[3] in {'REPORT.md','FORMAL_FAILURE.md','STOP.md'}
    and not path.split('/')[2].startswith('.')})
readme=(ROOT/'source/research/analysis/README.md').read_text(encoding='utf-8')
block=readme.split('<!-- BEGIN GENERATED ANALYSIS RESULT INDEX -->',1)[1].split('<!-- END GENERATED ANALYSIS RESULT INDEX -->',1)[0]
listed=[]
for label,target in re.findall(r'\[`([^`]+?)/`\]\(([^)]+)/\)',block):
    assert label==target
    listed.append(target)
# Index qualifies REPORT.md / FORMAL_FAILURE.md / STOP.md directories only.
assert len(dirs)==605, len(dirs)
assert len(listed)==605 and sorted(listed)==sorted(dirs) and listed==sorted(listed), (len(listed),len(dirs))
attrs=git('--attr-source='+head,'check-attr','--source='+head,'text','whitespace','eol','--',PREFIX+'/audit.json').decode()
assert ': text: unset' in attrs and ': whitespace: unset' in attrs and ': eol: unspecified' in attrs
(OUT/'ATTRIBUTES.txt').write_text(attrs,encoding='utf-8')
assert ast.literal_eval(ast.parse((PACK/'conftest.py').read_text()).body[1].value)==['test_reference.py']
assert not list(PACK.rglob('__init__.py'))
col=load(PACK/'execution/pytest-noncollection.receipt.json')
assert col['exit_code']==col['expected_exit_code']==5
for stream in ('stdout','stderr'):
    assert sha((PACK/f'execution/pytest-noncollection.{stream}.txt').read_bytes())==col[stream+'_sha256']
assert 'no tests collected' in (PACK/'execution/pytest-noncollection.stdout.txt').read_text()
put('IDENTITY_AND_SCOPE.json',{'source_export_hashes':len(export['blobs']), 'frozen_pins':16,'legacy_pins':6,
    'manifest_entries':44,'unchanged_archive_v1_blobs':45,'proposal_digest':digest,'proposal_diff_sha256':sha(diff),
    'exact_changed_paths':47,'all_analysis_child_directories':len(all_dirs),'head_retained_result_directories':605,'full_sorted_index_entries':len(listed),
    'attribute_scope':PREFIX+'/audit.json','retained_pytest_collection':col,
    'collection_qualification':'Archived command/exit5/zero collection joins and hook checked; no fresh pytest command here.'})

# separate process timestamps/streams. No author producer or retained legacy execution.

(OUT/'independent-repeat.stderr.txt').write_bytes(p.stderr)

    'python':sys.version,'source_sha256':sha((ROOT/'independent.py').read_bytes()),
    'stdout_bytes':len(p.stdout),'stdout_sha256':sha(p.stdout),'stderr_bytes':len(p.stderr),'stderr_sha256':sha(p.stderr),
    'kind':'ordinary raw-only reviewer repeat; first observation qualification retained separately',

result=json.loads(p.stdout)
put('INDEPENDENT_RESULT.json',result)

    assert sha((ROOT/item['side']/item['path']).read_bytes())==item['sha256']

put('COMPLETION.json',{'started_utc':START,'ended_utc':datetime.now(timezone.utc).isoformat(),
    'status':'PASS_SCOPED_TECHNICAL_REVIEW','head':head,'proposal_digest':digest,'primary_index_unchanged':True,
    'source_and_raw_unchanged':True,'author_producer_invocations':0,'historical_formal_invocations':0,

print(json.dumps({'status':'PASS_SCOPED_TECHNICAL_REVIEW','identity_paths':47,'archive_unchanged':45,
    'freeze_pins':16,'source_pins':6,'index_entries':605,'row_reconstruction_errors':0,
