"""Prepare the existing scalar PR on observed main without changing any ref."""
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

workspace = Path(__file__).resolve().parent.parent
repo = workspace / "work/agent-interface"
root = workspace / "work/compose6866-403a21a"
assert not root.exists(), "preserve first preparation; use a separate diagnostic repair"
root.mkdir()
source = root / "source"
base = "403a21a6951662401925835fbc70670bab84b457"
head = "abd318643112150d073a242926494145c6461e33"
p = json.loads((workspace / "outputs/proposal-6866-v2.json").read_bytes())
primary = (repo / ".git/index").read_bytes()
started = datetime.now(timezone.utc).isoformat()


def git(*args, data=None, env=None):
    return subprocess.check_output(["git", *args], cwd=repo, input=data, env=env)


def entries(ref, *paths):
    result = {}
    for line in git("ls-tree", "-r", ref, "--", *paths).decode().splitlines():
        metadata, path = line.split("\t")
        mode, kind, oid = metadata.split()
        assert kind == "blob"
        result[path] = {"mode": mode, "blob_oid": oid}
    return result


assert git("rev-parse", "origin/main").decode().strip() == base
assert git("rev-parse", "HEAD").decode().strip() == head
closure = entries(base, "runtime/core_v1", "runtime/backends/win32_v1/session.py",
                  "runtime/backends/x11_v1/session.py", "runtime/backends/quartz_v1/session.py",
                  ".github/workflows/runtime-core-portability.yml", ".gitattributes")
contract_path = "runtime/core_v1/contract.py"
ancestor_oid = entries(p["review_base_sha"], contract_path)[contract_path]["blob_oid"]
oids = {v["blob_oid"] for v in closure.values()} | {ancestor_oid}
oids.update(v["after"]["blob_oid"] for v in p["changed_blobs"])
fetch = subprocess.run(["git", "-c", "fetch.negotiationAlgorithm=noop", "fetch", "origin",
                        "--no-tags", "--no-write-fetch-head", "--recurse-submodules=no",
                        "--filter=blob:none", "--stdin"], cwd=repo,
                       input=("\n".join(sorted(oids)) + "\n").encode(), capture_output=True)
(root / "fetch.stdout").write_bytes(fetch.stdout)
(root / "fetch.stderr").write_bytes(fetch.stderr)
assert fetch.returncode == 0, fetch.stderr.decode(errors="replace")


def blob(oid):
    data = git("cat-file", "blob", oid)
    assert hashlib.sha1(f"blob {len(data)}\0".encode() + data).hexdigest() == oid
    return data


contract_record = next(v for v in p["changed_blobs"] if v["path"] == contract_path)
for label, oid in (("main", closure[contract_path]["blob_oid"]),
                   ("ancestor", ancestor_oid), ("head", contract_record["after"]["blob_oid"])):
    (root / (label + ".contract.py.txt")).write_bytes(blob(oid))
merge = subprocess.run(["git", "merge-file", "-p", str(root / "main.contract.py.txt"),
                        str(root / "ancestor.contract.py.txt"), str(root / "head.contract.py.txt")],
                       cwd=repo, capture_output=True)
(root / "merge.stdout").write_bytes(merge.stdout)
(root / "merge.stderr").write_bytes(merge.stderr)
(root / "merge.receipt.json").write_text(json.dumps({"exit_code": merge.returncode,
    "kind": "ordinary source integration; no ref update", "base_sha": base,
    "head_sha": head}, indent=2) + "\n")
assert merge.returncode == 0, "retain any actual conflict and diagnose separately"
combined = merge.stdout
assert combined.count(b"_bounded_int(now_ns,") == 1
assert combined.count(b"_bounded_int(current_observation_seq,") == 1
assert combined.count(b"_bounded_int(current_binding_revision,") == 1
normalized = hashlib.sha256(combined.replace(b"\r\n", b"\n")).hexdigest()
assert normalized == "1709078c2a144d78ab61a9bc59fbb9973c120b9a58101c079d69db65d43a66b1"
combined_oid = git("hash-object", "-w", "--stdin", data=combined).decode().strip()
closure[contract_path] = {"mode": "100644", "blob_oid": combined_oid}
test_path = "runtime/core_v1/test_admission_current_evidence.py"
test_record = next(v for v in p["changed_blobs"] if v["path"] == test_path)
closure[test_path] = {key: test_record["after"][key] for key in ("mode", "blob_oid")}
freeze = []
for path, item in sorted(closure.items()):
    data = blob(item["blob_oid"])
    target = source / path
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(data)
    freeze.append(dict(path=path, **item, bytes=len(data), sha256=hashlib.sha256(data).hexdigest()))
probe_path = "research/integration/current_evidence_scalars_57_20261003_45e9/session_probe.py"
probe_record = next(v for v in p["changed_blobs"] if v["path"] == probe_path)
probe_bytes = blob(probe_record["after"]["blob_oid"])
probe = source / probe_path
probe.parent.mkdir(parents=True, exist_ok=True)
probe.write_bytes(probe_bytes)
freeze.append(dict(path=probe_path, **probe_record["after"]))
(root / "FREEZE.json").write_text(json.dumps({"observed_base_sha": base, "head_sha": head,
    "scope": "separate ordinary current-main combination; no historical producer/formal replay",
    "blobs": freeze}, indent=2) + "\n", encoding="utf-8", newline="\n")

index_env = dict(os.environ, GIT_INDEX_FILE=str(root / "candidate.index"))
git("read-tree", base, env=index_env)
for item in p["changed_blobs"]:
    path = item["path"]
    actual = item["after"]
    data = blob(actual["blob_oid"])
    assert len(data) == actual["bytes"] and hashlib.sha256(data).hexdigest() == actual["sha256"]
    oid = combined_oid if path == contract_path else actual["blob_oid"]
    git("update-index", "--add", "--cacheinfo", actual["mode"], oid, path, env=index_env)
delta = git("diff", "--cached", "--name-status", base, env=index_env).decode().splitlines()
assert len(delta) == 42
assert {line.split("\t")[1] for line in delta} == set(p["changed_paths"])
assert sum(line.startswith("M\t") for line in delta) == 1
tree = git("write-tree", "--missing-ok", env=index_env).decode().strip()
candidate = git("commit-tree", tree, "-p", base, "-p", head,
                data=b"Merge PR #6866 bounded current evidence (local preparation only)\n").decode().strip()


def check(label, command):
    begin = datetime.now(timezone.utc).isoformat()
    run = subprocess.run(command, cwd=source, capture_output=True)
    end = datetime.now(timezone.utc).isoformat()
    (root / (label + ".stdout.txt")).write_bytes(run.stdout)
    (root / (label + ".stderr.txt")).write_bytes(run.stderr)
    (root / (label + ".receipt.json")).write_text(json.dumps({"command": command,
        "started_utc": begin, "ended_utc": end, "exit_code": run.returncode,
        "source_freeze_sha256": hashlib.sha256((root / "FREEZE.json").read_bytes()).hexdigest(),
        "kind": "ordinary integration regression, no formal allocation", "python": sys.version},
        indent=2) + "\n", encoding="utf-8", newline="\n")
    assert run.returncode == 0, (label, run.stderr.decode(errors="replace")[-2000:])
    return run


core = check("core-current", [sys.executable, "-m", "unittest", "discover", "-s",
                              "runtime/core_v1", "-p", "test_*.py", "-v"])
assert b"Ran 84 tests" in core.stderr and core.stderr.rstrip().endswith(b"OK")
spy = check("sessions-current", [sys.executable, str(probe)])
raw = json.loads(spy.stdout)
assert len(raw["rows"]) == 21
for row in raw["rows"]:
    valid = row["case"] == "valid"
    assert row["spy_execution_count"] == (1 if valid else 0), row
    assert row["status"] == ("ok" if valid else "refused"), row
    assert row["calls"] == (["clock", "preflight", "execute", "release_all"] if valid else ["clock"]), row
for item in freeze:
    assert hashlib.sha256((source / item["path"]).read_bytes()).hexdigest() == item["sha256"]
assert (repo / ".git/index").read_bytes() == primary
result = {"worker_id": p["author_worker"], "policy_version": "FINAL-v5",
    "base_sha": base, "head_sha": head, "planned_tree_sha": tree,
    "local_candidate_commit": candidate, "parents": [base, head], "changed_paths": 42,
    "contract_blob_oid": combined_oid, "contract_sha256": hashlib.sha256(combined).hexdigest(),
    "contract_normalized_lf_sha256": normalized, "frozen_source_files": len(freeze),
    "core_tests": 84, "session_spy_cases": 21, "valid_inert_executions": 3,
    "refused_cases_without_preflight_or_execute": 18, "source_after_checks_unchanged": True,
    "primary_index_unchanged": True, "started_utc": started,
    "ended_utc": datetime.now(timezone.utc).isoformat(), "main_send_attempted": False,
    "qualification": "Author preparation only; independent current-tuple check and content quorum remain required. No native backend/input or historical producer/formal replay."}
(root / "RESULT.json").write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8", newline="\n")
(workspace / "outputs/6866-current-composition.json").write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8", newline="\n")
print(json.dumps(result, sort_keys=True))
