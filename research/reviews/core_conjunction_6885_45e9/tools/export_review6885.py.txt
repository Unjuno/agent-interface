"""Exact, bounded Git export for an assigned nonauthor evidence review."""
from datetime import datetime, timezone
import hashlib
import io
import json
from pathlib import Path
import subprocess
import tarfile

workspace = Path(__file__).resolve().parent.parent
repo = workspace / "work/agent-interface"
root = workspace / "work/review6885"
root.mkdir(exist_ok=False)
proposal = json.loads((workspace / "outputs/author-proposal-6885-v2.json").read_text(encoding="utf-8"))
digest = proposal.pop("proposal_digest")
canonical = lambda value: json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode()
assert hashlib.sha256(canonical(proposal)).hexdigest() == digest
head, base, old = proposal["head_sha"], proposal["review_base_sha"], proposal["original_head_sha"]
prefix = "research/integration/core_boundary_composition_57_20261003_01a0ff58"


def git(*args):
    return subprocess.check_output(["git", *args], cwd=repo)


assert git("rev-parse", "origin/review-6885").decode().strip() == head
delta = git("diff", "--name-status", base, head).decode().splitlines()
paths = []
for line in delta:
    status, path = line.split("\t")
    assert status == "A" and path.startswith(prefix + "/")
    paths.append(path)
assert len(paths) == 136
tree_entries = git("ls-tree", "-r", head, "--", prefix).decode().splitlines()
objects = {}
for line in tree_entries:
    metadata, path = line.split("\t")
    mode, kind, oid = metadata.split()
    assert kind == "blob" and mode == "100644"
    objects[path] = (mode, oid)
assert set(objects) == set(paths)
started = datetime.now(timezone.utc).isoformat()
archive = git("archive", "--format=tar", head, prefix)
ended = datetime.now(timezone.utc).isoformat()
assert len(archive) < 41943040
source = root / "source"
source.mkdir()
records = []
with tarfile.open(fileobj=io.BytesIO(archive)) as bundle:
    members = [member for member in bundle.getmembers() if not member.isdir()]
    assert set(member.name for member in members) == set(paths)
    for member in members:
        assert member.isfile() and member.name in objects
        target = (source / member.name).resolve()
        assert target.is_relative_to(source.resolve())
        data = bundle.extractfile(member).read()
        assert len(data) == member.size
        target.parent.mkdir(parents=True, exist_ok=True)
        with target.open("xb") as stream:
            stream.write(data)
        mode, oid = objects[member.name]
        assert hashlib.sha1(f"blob {len(data)}\0".encode() + data).hexdigest() == oid
        records.append({"path": member.name, "mode": mode, "blob_oid": oid,
                        "bytes": len(data), "sha256": hashlib.sha256(data).hexdigest()})
records.sort(key=lambda item: item["path"])
assert hashlib.sha256(canonical(records)).hexdigest() == proposal["content_blobs_sha256"]
old_entries = git("ls-tree", "-r", old, "--", prefix).decode().splitlines()
original_count = 0
for line in old_entries:
    metadata, path = line.split("\t")
    mode, kind, oid = metadata.split()
    assert kind == "blob" and objects[path] == (mode, oid)
    original_count += 1
assert original_count == 116
packet = source / prefix
manifest_counts = {}
for relative, expected in (("SHA256SUMS", proposal["original_manifest_sha256"]),
                           ("audit_v2/SHA256SUMS", proposal["audit_v2_manifest_sha256"])):
    manifest = packet / relative
    assert hashlib.sha256(manifest.read_bytes()).hexdigest() == expected
    lines = manifest.read_text(encoding="utf-8").splitlines()
    for line in lines:
        expected_hash, path = line.split("  ", 1)
        target = (manifest.parent / path).resolve()
        assert target.is_relative_to(packet.resolve())
        assert hashlib.sha256(target.read_bytes()).hexdigest() == expected_hash
    manifest_counts[relative] = len(lines)
result = {"proposal_digest": digest, "head_sha": head, "review_base_sha": base,
          "source_files": len(records), "bytes": sum(item["bytes"] for item in records),
          "unchanged_original_files": original_count, "manifest_counts": manifest_counts,
          "portable_blob_digest": proposal["content_blobs_sha256"], "archive_started_utc": started,
          "archive_ended_utc": ended, "command": ["git", "archive", "--format=tar", head, prefix],
          "exit_code": 0, "blobs": records}
(root / "EXPORT.json").write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8", newline="\n")
print(json.dumps({key: value for key, value in result.items() if key != "blobs"}))
