"""Freeze actual inputs and capture the first two bounded reviewer processes."""
from datetime import datetime, timezone
import hashlib,json,subprocess,sys
from pathlib import Path
root=Path(__file__).resolve().parent;out=root/'checks';out.mkdir(exist_ok=False)
def sha(b):return hashlib.sha256(b).hexdigest()
def put(n,v):(out/n).write_text(json.dumps(v,indent=2,sort_keys=True)+'\n',encoding='utf-8')
source=json.loads((root/'EXPORT.json').read_bytes())
for r in source['blobs']:assert sha((root/r['side']/r['path']).read_bytes())==r['sha256']
python=Path(sys.executable)
jobs=[('nominal-seven',[str(python),'-B','-m','unittest','runtime.kernel.test_nominal_types','-v'],root/'source'),
 ('independent-fourteen',[str(python),'-B',str(root/'independent.py')],root)]
put('FREEZE.json',{'frozen_utc':datetime.now(timezone.utc).isoformat(),'head':source['head'],
 'proposal_digest':source['proposal_digest'],'python_version':sys.version,'python_binary_sha256':sha(python.read_bytes()),
 'source_manifest_sha256':sha((root/'EXPORT.json').read_bytes()),'independent_source_sha256':sha((root/'independent.py').read_bytes()),
 'capture_source_sha256':sha(Path(__file__).read_bytes()),'planned_jobs':[{'label':l,'command':c,'cwd':str(d)} for l,c,d in jobs],
 'original_producer_or_formal_replays':0})
for label,cmd,cwd in jobs:
 started=datetime.now(timezone.utc).isoformat();timed_out=False
 try:
  p=subprocess.run(cmd,cwd=cwd,capture_output=True,timeout=15);stdout,stderr,code=p.stdout,p.stderr,p.returncode
 except subprocess.TimeoutExpired as e:stdout,stderr,code=e.stdout or b'',e.stderr or b'',None;timed_out=True
 (out/(label+'.stdout.txt')).write_bytes(stdout);(out/(label+'.stderr.txt')).write_bytes(stderr)
 put(label+'.receipt.json',{'command':cmd,'cwd':str(cwd),'started_utc':started,'ended_utc':datetime.now(timezone.utc).isoformat(),
  'exit_code':code,'timed_out':timed_out,'stdout_sha256':sha(stdout),'stderr_sha256':sha(stderr),
  'stdout_bytes':len(stdout),'stderr_bytes':len(stderr),'scope':'ordinary pure sequential lifecycle record check'})
 assert code==0 and not timed_out,(label,code,stdout.decode(errors='replace'),stderr.decode(errors='replace'))
 if label=='nominal-seven':assert b'Ran 7 tests' in stderr and stderr.rstrip().endswith(b'OK')
 else:assert json.loads(stdout)['actual_cases']==14
 print(json.dumps({'check':label,'exit_code':code,'stdout_sha256':sha(stdout),'stderr_sha256':sha(stderr)}),flush=True)
for r in source['blobs']:assert sha((root/r['side']/r['path']).read_bytes())==r['sha256']
put('COMPLETION.json',{'ended_utc':datetime.now(timezone.utc).isoformat(),'status':'PASS_SCOPED',
 'actual_nominal_methods':7,'independent_cases':14,'source_and_raw_unchanged':True,'vote_yet':False})
