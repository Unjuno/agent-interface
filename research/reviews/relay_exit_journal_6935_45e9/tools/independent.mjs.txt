// Reviewer-owned ordinary real-child boundary checks. No author fixture/producer import.
import assert from 'node:assert/strict';
import { mkdir,writeFile,readFile,readdir,stat } from 'node:fs/promises';
import { join,resolve } from 'node:path';
import { createRelayClient } from './source/runtime/host_v1/relay_client.mjs';

const ROOT=resolve(process.argv[2]);
await mkdir(ROOT,{recursive:false});
const sentinel='reviewer first exclusive bytes\n';
const fixture=String.raw`
const fs=require('node:fs'),rl=require('node:readline'),path=require('node:path');
const marker=process.argv[1],mode=process.argv[2];let count=0;
const guard=setTimeout(()=>{fs.writeFileSync(path.join(marker,'guard.json'),JSON.stringify({pid:process.pid,count}),{flag:'wx'});process.exit(75);},2500);
fs.writeFileSync(path.join(marker,'spawn.json'),JSON.stringify({pid:process.pid,mode}),{flag:'wx'});
if(mode==='stderr-directory')setTimeout(()=>process.stderr.write('one inert diagnostic\n'),20);
const input=rl.createInterface({input:process.stdin});
input.on('line',line=>{const request=JSON.parse(line);count++;
 const row={id:request.id,tool:mode==='wrong-tool'||mode==='exit-directory'?'other-tool':request.tool,
  status:'returned',next_id:request.id+1,result:{content:[]}};
 process.stdout.write(JSON.stringify(row)+'\n');
});
input.on('close',()=>{clearTimeout(guard);const code=mode==='wrong-tool'?23:0;
 fs.writeFileSync(path.join(marker,'terminal.json'),JSON.stringify({pid:process.pid,count,code}),{flag:'wx'});
 process.exitCode=code;
});`;
async function exists(path){try{await stat(path);return true;}catch(e){if(e.code==='ENOENT')return false;throw e;}}
async function read(path){return JSON.parse(await readFile(path,'utf8'));}
async function waitBlocked(client){
 const deadline=Date.now()+1200;
 while(!client.state().blocked){assert.ok(Date.now()<deadline,'idle fault becomes blocked');await new Promise(r=>setTimeout(r,5));}
}
const plan=[
 {name:'wrong-tool-nonzero-terminal',mode:'wrong-tool',code:23,requests:1,fault:/identity\/status mismatch/},
 {name:'occupied-request-no-delivery',mode:'request-occupied',code:0,requests:0,fault:/EEXIST/},
 {name:'idle-stderr-directory',mode:'stderr-directory',code:0,requests:0,fault:/EISDIR|EACCES|EPERM/},
 {name:'failed-reply-exit-directory',mode:'exit-directory',code:0,requests:1,fault:/identity\/status mismatch/,second:true},
 {name:'healthy-terminal-control',mode:'healthy',code:0,requests:1,healthy:true},
];
const results=[];
for(const item of plan){
 const caseRoot=join(ROOT,item.name);await mkdir(caseRoot);
 const evidence=join(caseRoot,'evidence');
 const client=await createRelayClient({command:process.execPath,args:['-e',fixture,caseRoot,item.mode],evidenceDirectory:evidence});
 let closed=false,pending;
 try{
  if(item.mode==='request-occupied')await writeFile(join(evidence,'request-1.json'),sentinel,{flag:'wx'});
  if(item.mode==='stderr-directory'){
   await mkdir(join(evidence,'stderr.log'));await waitBlocked(client);
   assert.equal(client.state().attempts,0);assert.throws(()=>client.send('synthetic',{}));
  }else{
   if(item.second)await mkdir(join(evidence,'exit.json'));
   pending=client.send('synthetic',{unchanged:true});assert.equal(client.wait(),pending);
   if(item.healthy){const reply=await pending;assert.equal(reply.status,'returned');assert.equal(reply.id,1);}
   else{await assert.rejects(pending,/delivery is uncertain/);assert.equal(client.wait(),pending);assert.throws(()=>client.send('synthetic',{}));}
  }
  const beforeClose=client.state();const outcome=await client.close();closed=true;
  assert.equal(outcome.code,item.code);assert.equal(outcome.signal,null);
  assert.deepEqual(await client.close(),outcome,'repeat close reconciles same terminal without rewriting');
  assert.throws(()=>client.send('synthetic',{}));
  const marker=await read(join(caseRoot,'terminal.json'));const spawned=await read(join(caseRoot,'spawn.json'));
  assert.equal(marker.pid,spawned.pid);assert.equal(marker.count,item.requests);assert.equal(marker.code,outcome.code);
  assert.equal(await exists(join(caseRoot,'guard.json')),false,'self-guard not reached');
  let absent=false;try{process.kill(marker.pid,0);}catch(e){assert.equal(e.code,'ESRCH');absent=true;}
  assert.equal(absent,true,'this owned fixture child is absent after close');
  if(item.healthy){assert.equal(Object.hasOwn(outcome,'journal_error'),false);assert.deepEqual(await read(join(evidence,'exit.json')),outcome);}
  else{
   assert.match(outcome.journal_error,item.fault);
   if(item.second){assert.match(outcome.exit_journal_error,/EEXIST|EISDIR|EACCES|EPERM/);assert.equal((await stat(join(evidence,'exit.json'))).isDirectory(),true);}
   else{assert.equal(Object.hasOwn(outcome,'exit_journal_error'),false);assert.deepEqual(await read(join(evidence,'exit.json')),outcome);}
  }
  if(item.mode==='request-occupied'){
   assert.equal(await readFile(join(evidence,'request-1.json'),'utf8'),sentinel);
   assert.equal(await exists(join(evidence,'reply-1.json')),false);
  }
  const files=await readdir(evidence);assert.equal(files.filter(n=>n.startsWith('request-')).length,item.mode==='stderr-directory'?0:1);
  const result={name:item.name,outcome,beforeClose,requests_received:marker.count,owned_pid:marker.pid,
   owned_child_absent:absent,guard_not_reached:true,repeat_close_equal:true,source_scope:'transport/exclusive files only, no task or input effect'};
  await writeFile(join(caseRoot,'RESULT.json'),JSON.stringify(result,null,2)+'\n',{flag:'wx'});
  results.push(result);
 }finally{
  if(!closed){if(pending)await pending.catch(()=>{});await client.close();}
 }
}
await writeFile(join(ROOT,'SUMMARY.json'),JSON.stringify({status:'PASS_SCOPED',node:process.version,cases:results},null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({status:'PASS_SCOPED',cases:results.length,actual_child_exits:results.map(r=>r.outcome.code),requests:results.map(r=>r.requests_received),all_owned_children_absent:true}));
