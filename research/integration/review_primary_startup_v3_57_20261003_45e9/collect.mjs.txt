import {spawnSync} from 'node:child_process';
import {readFileSync,writeFileSync} from 'node:fs';
import {join,dirname} from 'node:path';
import {fileURLToPath} from 'node:url';
const root=dirname(fileURLToPath(import.meta.url)),plan=JSON.parse(readFileSync(join(root,'MATRIX_PLAN.json')));
const captured=[];
for(const variant of plan.variants)for(const cell of plan.cells){
 const directory=join(root,'matrix-first',variant.name,cell.name);
 const nonce=plan.configs.find(x=>x.variant===variant.name&&x.cell===cell.name).nonce;
 const argv=[process.execPath,join(root,'worker.mjs'),variant.name,cell.name,directory,nonce];
 const started=new Date().toISOString(),r=spawnSync(argv[0],argv.slice(1),{cwd:root,encoding:null,timeout:7000,windowsHide:true});
 const stdout=r.stdout||Buffer.alloc(0),stderr=r.stderr||Buffer.alloc(0);
 writeFileSync(join(directory,'stdout.txt'),stdout,{flag:'wx'});writeFileSync(join(directory,'stderr.txt'),stderr,{flag:'wx'});
 let workerAbsent=false;const pid=r.pid;
 try{process.kill(pid,0);}catch(e){if(e.code==='ESRCH')workerAbsent=true;else throw e;}
 const receipt={argv,pid,started_utc:started,ended_utc:new Date().toISOString(),exit_code:r.status,signal:r.signal,
 error:r.error?String(r.error):null,timed_out:r.error?.code==='ETIMEDOUT',
 stdout_bytes:stdout.length,stderr_bytes:stderr.length,original_worker_pid_absent:workerAbsent};
 writeFileSync(join(directory,'RECEIPT.json'),JSON.stringify(receipt,null,2)+'\n',{flag:'wx'});
 captured.push({variant:variant.name,cell:cell.name,exit_code:r.status,signal:r.signal,timed_out:receipt.timed_out});
}
writeFileSync(join(root,'matrix-first','COLLECTOR.json'),JSON.stringify(captured,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({status:'CAPTURED_NOT_ADJUDICATED',cells:captured.length,exits:captured.map(x=>x.exit_code)}));
