env=dict(os.environ,GIT_NO_LAZY_FETCH='1')
def sha(b):return hashlib.sha256(b).hexdigest()
def put(p,v):p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(v,indent=2,sort_keys=True)+'\n',encoding='utf-8',newline='\n')
def command(label,argv,stdin=None,timeout=45):
 out=root/'command-first'/label;out.mkdir(parents=True,exist_ok=False)
 start=datetime.now(timezone.utc).isoformat()
 r=subprocess.run(argv,cwd=repo,env=env,input=stdin,capture_output=True,timeout=timeout)
 (out/'stdout.txt').write_bytes(r.stdout);(out/'stderr.txt').write_bytes(r.stderr)
 put(out/'RECEIPT.json',{'argv':argv,'started_utc':start,'ended_utc':datetime.now(timezone.utc).isoformat(),'exit_code':r.returncode,'stdout_sha256':sha(r.stdout),'stderr_sha256':sha(r.stderr)})
 assert r.returncode==0,(label,r.returncode,r.stderr[:800])
 return r.stdout
def git(*a,**kw):return subprocess.check_output(['git',*a],cwd=repo,env=env,**kw)
def image(ref,path):
 b=git('ls-tree',ref,'--',path)
 if not b:return None
 meta,name=b.decode().strip().split('\t');mode,kind,oid=meta.split()
 assert name==path and kind=='blob'
 return {'mode':mode,'blob_oid':oid}
