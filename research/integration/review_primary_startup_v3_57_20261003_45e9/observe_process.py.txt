"""Read only the actual exporter in this owned directory and its descendant handles."""
from pathlib import Path
from datetime import datetime,timezone
import json
import psutil
root=Path(__file__).resolve().parent
rows=[]
for process in psutil.process_iter(['pid','ppid','name','cmdline','create_time']):
 try:
  cmd=process.info['cmdline'] or []
  if any(str(x).endswith('export.py') for x in cmd) and Path(process.cwd()).resolve()==root:
   chain=[process]+process.children(recursive=True)
   for p in chain:
    rows.append({'pid':p.pid,'ppid':p.ppid(),'name':p.name(),'cwd':p.cwd(),'argv':p.cmdline(),'status':p.status(),
      'created_utc':datetime.fromtimestamp(p.create_time(),timezone.utc).isoformat(),'cpu':list(p.cpu_times()[:2]),'io':list(p.io_counters()[:4])})
 except (psutil.NoSuchProcess,psutil.AccessDenied):continue
out={'observed_utc':datetime.now(timezone.utc).isoformat(),'owned_exporter_chain':rows}
(root/'PROCESS_OBSERVATION.json').write_text(json.dumps(out,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps(out))
