def save(name,value):

        json.dump(value,f,indent=2,sort_keys=True,allow_nan=False);f.write('\n');f.flush();os.fsync(f.fileno())

def custody():
    ns=_win32_logon_namespace_digest()
    path=Path(os.environ['LOCALAPPDATA'])/'Unjuno/AgentInterface/custody-v1'/(ns+'.json')
    with path.open('rb') as f: raw=f.read(65537);st=os.fstat(f.fileno())
    if len(raw)>65536:raise RuntimeError('custody exceeds bound')
    return {'namespace':ns,'logical_path':str(path),'physical_path':str(path.resolve()),
      'inode':st.st_ino,'device':st.st_dev,'sha256':hashlib.sha256(raw).hexdigest(),
      'raw_utf8':raw.decode('utf-8'),'state':json.loads(raw)}

def process_state(pid):
    handle=K.OpenProcess(0x1000|0x100000,False,pid)
    if not handle:
        error=ctypes.get_last_error()
        if error!=87:raise OSError(error,'cannot verify process')
        return {'pid':pid,'absent':True,'winerror':error,'observed_ns':time.monotonic_ns()}
    try:
        times=[W.FILETIME() for _ in range(4)]
        if not K.GetProcessTimes(handle,*[ctypes.byref(t) for t in times]):raise ctypes.WinError(ctypes.get_last_error())
        waited=K.WaitForSingleObject(handle,0)
        if waited not in (0,258):raise RuntimeError('process wait failed')
        return {'pid':pid,'absent':False,'exited':waited==0,
          'creation_time_100ns':(times[0].dwHighDateTime<<32)|times[0].dwLowDateTime,
          'observed_ns':time.monotonic_ns()}
    finally:K.CloseHandle(handle)

def source_absent():
    row=process_state(SOURCE_ID['pid'])
    absent=row.get('absent') or row.get('exited') or row.get('creation_time_100ns')!=SOURCE_ID['creation_time_100ns']
    if not absent:raise RuntimeError('exact source producer remains live')
    return row

def target_gate(meta):
    pid=W.DWORD();hwnd=meta['hwnd']
    if not U.IsWindow(hwnd) or not U.GetWindowThreadProcessId(hwnd,ctypes.byref(pid)) or pid.value!=meta['pid']:
        raise RuntimeError('owned window identity failed')
    identity=process_state(pid.value)
    if identity.get('creation_time_100ns')!=meta['created_filetime_100ns'] or identity.get('exited'):
        raise RuntimeError('owned process identity failed')
    if int(U.GetForegroundWindow() or 0)!=hwnd:raise RuntimeError('owned fixture is not foreground')
    if any(U.GetAsyncKeyState(k)&0x8000 for k in (0x10,0x11,0x12,0x01)):
        raise RuntimeError('modifier/mouse neutral gate failed')
    return identity

def report_in(body):
    rows=[r for r in walk(body) if r.get('source_operation_id')==SOURCE_OP
          and 'recovery_status' in r and 'source_action_status' in r]
    if not rows:return None
    if any(row!=rows[0] for row in rows):raise RuntimeError('conflicting public recovery reports')
    return rows[0]
