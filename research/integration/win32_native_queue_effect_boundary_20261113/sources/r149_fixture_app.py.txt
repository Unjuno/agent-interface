@WNDPROC
def window_proc(hwnd,message,wparam,lparam):
 try:
  if message==0x10:
   event('close_requested');U.DestroyWindow(hwnd);return 0
  if message==0x02:
   state['alive']=False;event('destroyed');return 0
  if message==0x100 or message==0x101 or message==0x102:
   row=event('wndproc_keyboard',message=message,wparam=int(wparam),lparam=int(lparam),
             phase=state['phase'],a_async_down=bool(U.GetAsyncKeyState(0x41)&0x8000),
             foreground=int(U.GetForegroundWindow() or 0))
   if message==0x100 and wparam==0x41:
    state['effects']+=1
    effect=event('task_effect',counter=state['effects'],phase=state['phase'],
                 first_transition=not bool(lparam&(1<<30)),key_event_ns=row['monotonic_ns'])
    with (P/'effects.jsonl').open('a',encoding='utf-8') as f:f.write(json.dumps(effect,sort_keys=True)+'\n')
    U.InvalidateRect(hwnd,None,True)
   return 0
  if message==0x0F:
   dc=U.GetDC(hwnd)
   if dc:paint(dc);U.ReleaseDC(hwnd,dc)
   U.ValidateRect(hwnd,None);return 0
  if message==0x318:paint(W.HDC(wparam));return 0
  return U.DefWindowProcW(hwnd,message,wparam,lparam)
 except BaseException as error:
  state['callback_error']=repr(error)
  event('callback_error',error=repr(error));return 0

def remove_and_dispatch(low,high):
 # Each message is used directly from the OS-returned MSG, immediately.
 # No deferred message list, synthetic MSG, PostMessage or SendInput exists.
 for _ in range(64):
  msg=W.MSG()
  if not U.PeekMessageW(ctypes.byref(msg),None,low,high,1):return
  if msg.message==0x12:state['alive']=False;return
  if 0x100<=msg.message<=0x109:
   if state['phase']=='holding':raise RuntimeError('keyboard removed during hold')
   event('native_keyboard_removed',message=int(msg.message),wparam=int(msg.wParam),
         lparam=int(msg.lParam),os_time_ms=int(msg.time),hwnd=int(msg.hWnd or 0),phase=state['phase'])
  U.TranslateMessage(ctypes.byref(msg));U.DispatchMessageW(ctypes.byref(msg))

def service_nonkeyboard():
 remove_and_dispatch(0,0xFF)
 remove_and_dispatch(0x10A,0xFFFF)
