/** Independently written NEW busy/fault interaction checks; no host operation. */
import {PassThrough,Writable} from 'node:stream';
import {writeFile} from 'node:fs/promises';
import assert from 'node:assert/strict';
import {randomUUID} from 'node:crypto';
const turn=()=>new Promise(r=>setImmediate(r)),rows=[];
for(const variant of ['peer','combined']) {
 const {servePrimaryLines}=await import(new URL(`./${variant}/runtime/host_v1/primary_stdio.mjs`,import.meta.url));
 for(const name of ['backlog-first','input-first','malformed-first']) {
  const nonce=randomUUID(),first=Error('owned input first '+nonce),later=Error('later output fault '+nonce);
  const input=new PassThrough();let gate,blockedDone,open=false,calls=0,completed=0,settled=false;const values=[];
  const foreignIn=()=>{},foreignOut=()=>{};input.on('error',foreignIn);
  const output=new Writable({highWaterMark:1,write(chunk,_enc,done){values.push(JSON.parse(chunk));if(open)done();else blockedDone=done;}});
  output.on('error',foreignOut);
  const exchange={state:()=>({next_id:calls+1}),execute:async value=>{calls++;assert.equal(value.id,1);await new Promise(r=>{gate=r;});completed++;return {id:1,nonce};}};
  const started=new Date().toISOString();
  const owner=servePrimaryLines({exchange,input,output}).then(()=>({status:'resolved'}),error=>({status:'rejected',error})).finally(()=>{settled=true;});
  input.write('{"id":1}\n');await turn();input.write('{"id":2}\n');await turn();
  assert.equal(calls,1);assert.equal(completed,0);assert.equal(settled,false);assert.deepEqual(values.map(x=>x.status),['busy']);
  if(name==='backlog-first')input.write(Array.from({length:64},(_,i)=>JSON.stringify({id:i+3})+'\n').join(''));
  else if(name==='input-first')input.emit('error',first);
  else input.write(Buffer.concat([Buffer.from('{"id":3,"note":"'),Buffer.from([255]),Buffer.from('"}\n')]));
  await turn();output.emit('error',later);await turn();
  const beforeRelease={calls,completed,settled,writableLength:output.writableLength,rows:values.map(x=>x.status)};
  console.log('PROBE_BEFORE '+JSON.stringify({variant,name,nonce,started_utc:started,beforeRelease}));
  assert.equal(calls,1);assert.equal(completed,0);assert.equal(settled,false);assert.equal(input.isPaused(),true);assert.ok(output.writableLength<1024);assert.deepEqual(values.map(x=>x.status),['busy']);
  gate();input.end();open=true;const done=blockedDone;blockedDone=null;done();
  const outcome=await owner;await turn();
  console.log('PROBE_AFTER '+JSON.stringify({variant,name,nonce,calls,completed,outcome:{status:outcome.status,error:outcome.error?.message,code:outcome.error?.code},rows:values,original_input_error_identity:outcome.error===first,later_error_replaced_first:outcome.error===later,listeners:{input:input.listenerCount('error'),output:output.listenerCount('error')}}));
  assert.equal(outcome.status,'rejected');assert.notEqual(outcome.error,later);assert.equal(calls,1);assert.equal(completed,1);
  if(name==='input-first')assert.equal(outcome.error,first);
  else if(name==='malformed-first'&&variant==='combined')assert.match(outcome.error.message,/UTF-8/i);
  else assert.match(outcome.error.message,/busy response backlog/);
  assert.deepEqual(values.map(x=>x.status),['busy','returned']);assert.deepEqual(values[1].result,{id:1,nonce});
  assert.equal(input.listenerCount('error'),1);assert.equal(output.listenerCount('error'),1);assert.equal(input.listeners('error')[0],foreignIn);assert.equal(output.listeners('error')[0],foreignOut);
  rows.push({variant,name,nonce,started_utc:started,ended_utc:new Date().toISOString(),beforeRelease,calls,completed,status:outcome.status,error:outcome.error.message,original_input_error_identity:outcome.error===first,later_error_replaced_first:outcome.error===later,rows:values,foreign_listeners_retained:true,owner_pid:process.pid,owner_ppid:process.ppid});
 }
}
await writeFile(new URL('./PROBE_ROWS.json',import.meta.url),JSON.stringify(rows,null,2)+'\n',{flag:'wx'});
console.log(JSON.stringify({new_busy_fault_rows:rows.length,all_pass:true}));
