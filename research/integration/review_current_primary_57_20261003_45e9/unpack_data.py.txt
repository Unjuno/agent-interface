"""Validate all public packet bytes and materialize only inert known data paths."""
from pathlib import Path,PurePosixPath
import base64,hashlib,json
r=Path(__file__).resolve().parent;d=json.loads((r/'DECODED_CUSTODY.json').read_bytes())
allowed={'PROPOSAL.json','CHANGED_IMAGES.json','SOURCE_DELIVERY.json','FINAL_SOURCE_CHECK.json','PLAN.json','check-first/RECEIPT.json','source-push-first/RECEIPT.json','source-push-first/stderr.txt','source-push-reconciled-fullpack/RECEIPT.json','source-push-reconciled-fullpack/stderr.txt'}
assert len(d['files'])==10 and {x['path'] for x in d['files']}==allowed
for x in d['files']:
 p=PurePosixPath(x['path']);assert not p.is_absolute() and '..' not in p.parts
 b=base64.b64decode(x['data_base64'],validate=True)
 assert len(b)==x['bytes'] and hashlib.sha256(b).hexdigest()==x['sha256']
 if x['projection']=='exact':assert x['bytes']==x['original_bytes'] and x['sha256']==x['original_sha256']
 t=r/'custody'/p;t.parent.mkdir(parents=True,exist_ok=True);assert not t.exists();t.write_bytes(b)
for n in ['PLAN.json','SOURCE_DELIVERY.json','FINAL_SOURCE_CHECK.json','check-first/RECEIPT.json']:
 print(n);print((r/'custody'/n).read_text(encoding='utf-8'))
