from pathlib import Path
from datetime import datetime,timezone
import subprocess,json,hashlib,os,re,sys
W=Path('<OWN_WORKSPACE>/work/current-primary-successor');plan=json.loads((W/'PLAN.json').read_bytes());D=W/'check-first';D.mkdir(exist_ok=False);(D/'cases').mkdir()
def now():return datetime.now(timezone.utc).isoformat()
def sha(b):return hashlib.sha256(b).hexdigest()
assert sha(Path(plan['node_binary']).read_bytes())==plan['node_binary_sha256']
for x in plan['source_pins']:assert sha((Path(plan['cwd'])/x['path']).read_bytes())==x['sha256']
start=now();r=subprocess.run(plan['argv'],cwd=plan['cwd'],env={**os.environ,**plan['environment_delta']},capture_output=True);end=now()
for name,data in [('stdout',r.stdout),('stderr',r.stderr)]: (D/(name+'.txt')).write_bytes(data)
counts={k:int(v) for k,v in re.findall(r'^# (tests|pass|fail|cancelled|skipped|todo) (\d+)',r.stdout.decode(),re.M)}
rc={'started_utc':start,'ended_utc':end,'actual_exit_code':r.returncode,'counts':counts,'argv':plan['argv'],'cwd':plan['cwd'],'stdout_bytes':len(r.stdout),'stdout_sha256':sha(r.stdout),'stderr_bytes':len(r.stderr),'stderr_sha256':sha(r.stderr),'source_pins_unchanged_after':all(sha((Path(plan['cwd'])/x['path']).read_bytes())==x['sha256'] for x in plan['source_pins']),'main_requests':0}
(D/'RECEIPT.json').write_text(json.dumps(rc,indent=2)+'\n');print(json.dumps(rc));print(r.stderr.decode())
assert r.returncode==0 and counts=={'tests':39,'pass':39,'fail':0,'cancelled':0,'skipped':0,'todo':0} and rc['source_pins_unchanged_after']
rows=[json.loads(line.split('STARTUP_EXCHANGE_WITNESS ',1)[1]) for line in r.stdout.decode().splitlines() if 'STARTUP_EXCHANGE_WITNESS ' in line];assert len(rows)==2
(W/'STARTUP_ROWS.json').write_text(json.dumps(rows,indent=2)+'\n')
print(json.dumps({'startup_witness_rows':2,'row_keys':list(rows[0])}))
