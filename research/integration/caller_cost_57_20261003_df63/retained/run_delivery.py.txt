import datetime,hashlib,json,os,subprocess,sys
from pathlib import Path
root=Path(__file__).resolve().parent;folder=root/'candidate'
pin=lambda p:{'bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()}
utc=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
for phase in ['delivery-normal','delivery-optimized']:
    out=root/phase;out.mkdir(exist_ok=False)
    cmd=[sys.executable]+(['-O'] if phase.endswith('optimized') else [])+['-m','unittest','-v','test_adaptive_acquisition_cost_coverage','test_adaptive_acquisition_caller_v3']
    env=os.environ.copy();env.pop('TRACE_OUTPUT',None);env['PYTHONDONTWRITEBYTECODE']='1'
    freeze={p.name:pin(p) for p in folder.glob('*.py')}
    (out/'PLAN.json').write_text(json.dumps({'utc_before_spawn':utc(),'command':cmd,'cwd':str(folder),'TRACE_OUTPUT':'absent; portable os.devnull sink','freeze':freeze,'tests':21,'guard_seconds':10,'supervisor_pid':os.getpid()},indent=2)+'\n',encoding='utf-8')
    start=utc();p=subprocess.Popen(cmd,cwd=folder,env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    timeout=False
    try:stdout,stderr=p.communicate(timeout=10)
    except subprocess.TimeoutExpired:timeout=True;p.kill();stdout,stderr=p.communicate(timeout=3)
    (out/'stdout.bin').write_bytes(stdout);(out/'stderr.bin').write_bytes(stderr)
    r={'pid':p.pid,'supervisor_pid':os.getpid(),'start_utc':start,'end_utc':utc(),'exit_code':p.returncode,'guard_expired':timeout,'stdout':pin(out/'stdout.bin'),'stderr':pin(out/'stderr.bin'),'freeze_unchanged':all(pin(folder/k)==v for k,v in freeze.items()),'model_native_gui_calls':0}
    (out/'RESULT.json').write_text(json.dumps(r,indent=2)+'\n',encoding='utf-8');print(json.dumps(r))
    if p.returncode!=0:print(stderr.decode('utf-8'));break
