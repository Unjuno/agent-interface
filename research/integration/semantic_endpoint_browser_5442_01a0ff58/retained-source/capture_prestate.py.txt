import argparse
import hashlib
import json
import sqlite3
from contextlib import closing
from datetime import datetime, timezone
from pathlib import Path

parser = argparse.ArgumentParser()
parser.add_argument('--out', required=True, type=Path)
args = parser.parse_args()
out = args.out.resolve()
with closing(sqlite3.connect((out / 'endpoint.sqlite').as_uri() + '?mode=ro', uri=True)) as db:
    db.row_factory = sqlite3.Row
    documents = [dict(r) for r in db.execute('SELECT * FROM documents ORDER BY room,target')]
    events = [dict(r) for r in db.execute('SELECT * FROM events')]
    requests = [dict(r) for r in db.execute('SELECT * FROM requests')]
if len(documents) != 16 or events or requests or any(r['version'] != 0 or r['value'] != '' for r in documents):
    raise ValueError('unexpected initial state; no GUI allowed')
result = {'utc': datetime.now(timezone.utc).isoformat(), 'documents': documents, 'events': events, 'requests': requests,
          'database_sha256': hashlib.sha256((out / 'endpoint.sqlite').read_bytes()).hexdigest()}
with (out / 'PRESTATE.json').open('x', encoding='utf-8') as handle:
    json.dump(result, handle, indent=2)
print(json.dumps({'rows': len(documents), 'events': 0, 'requests': 0}))
