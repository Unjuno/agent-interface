"""Separate-process endpoint observation, using fresh read-only SQLite connections."""
import argparse
import hashlib
import json
import sqlite3
from contextlib import closing
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parent


def exact_integer(value, wanted):
    return type(value) is int and value == wanted


def decide(row):
    intent, event, state = row['intent'], row['event'], row['state']
    if not event:
        return 'UNKNOWN'
    initial = row['phase'] == 'initial'
    pre = 0 if initial else intent['expected_versions']['A']
    before_b = {'version': 0, 'value': ''} if initial else intent['previous_state']['B']
    expected_b = {'version': before_b['version'] + (0 if initial else int(bool(before_b['value']))), 'value': ''}
    conditions = (
        intent['namespace'] == 'owned-endpoint-5442-01a0ff58',
        intent['target'] == 'A', event['requested_target'] == intent['target'],
        event['actual_target'] == intent['target'], event['room'] == row['trial_id'],
        event['phase'] == row['phase'], event['nonce'] == intent['nonce'],
        exact_integer(event['applied_pre'], pre), exact_integer(event['applied_post'], pre + 1),
        exact_integer(event['written'], 1), event['text'] == intent['desired_text'],
        set(state) == {'A', 'B'}, exact_integer(state['A']['version'], pre + 1),
        state['A']['value'] == intent['desired_text'],
        exact_integer(state['B']['version'], expected_b['version']), state['B']['value'] == '',
    )
    return 'CONFIRMED' if all(conditions) else 'UNKNOWN'


def observe(out, phase):
    started = datetime.now(timezone.utc).isoformat()
    freeze = json.loads((ROOT / 'FREEZE.json').read_bytes())
    for name, digest in freeze['files'].items():
        assert hashlib.sha256((ROOT / name).read_bytes()).hexdigest() == digest, name
    scenarios = json.loads((ROOT / 'scenarios.json').read_bytes())
    initial = json.loads((out / 'initial-observation.json').read_bytes()) if phase == 'repair' else None
    previous = {r['trial_id']: r for r in initial['rows']} if initial else {}
    chosen = [s for s in scenarios if phase == 'initial' or previous[s['trial_id']]['semantic_status'] == 'UNKNOWN']
    rows = []
    with closing(sqlite3.connect((out / 'endpoint.sqlite').as_uri() + '?mode=ro', uri=True)) as db:
        db.row_factory = sqlite3.Row
        db.execute('BEGIN')
        documents = [dict(r) for r in db.execute('SELECT * FROM documents ORDER BY room,target')]
        events = [dict(r) for r in db.execute('SELECT * FROM events ORDER BY room,phase')]
        requests = [dict(r) for r in db.execute('SELECT * FROM requests ORDER BY sequence')]
        for item in chosen:
            room = item['trial_id']
            state = {r['target']: {'version': r['version'], 'value': r['value']} for r in documents if r['room'] == room}
            found = [r for r in events if r['room'] == room and r['phase'] == phase]
            assert len(found) <= 1
            intent = {k: item[k] for k in ('namespace', 'target', 'desired_text')}
            intent['nonce'] = item['operation_id'] + ':' + phase
            intent['expected_pre_version'] = 0
            if phase == 'repair':
                intent['previous_state'] = previous[room]['state']
                intent['expected_versions'] = {t: r['version'] for t, r in previous[room]['state'].items()}
                intent['expected_pre_version'] = intent['expected_versions']['A']
            row = {'trial_id': room, 'phase': phase, 'intent': intent, 'event': found[0] if found else None,
                   'state': state, 'dispatch_status': 'accepted' if found else 'missing'}
            row['semantic_status'] = decide(row)
            rows.append(row)
        db.rollback()
    result = {'study': freeze['study'], 'phase': phase, 'started_utc': started,
              'ended_utc': datetime.now(timezone.utc).isoformat(),
              'freeze_sha256': hashlib.sha256((ROOT / 'FREEZE.json').read_bytes()).hexdigest(),
              'observer_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
              'rows': rows, 'documents': documents, 'events': events, 'requests': requests,
              'authority_scope': 'owned SQLite rows; same authored system, distinct process/read-only connection'}
    with (out / (phase + '-observation.json')).open('x', encoding='utf-8') as handle:
        json.dump(result, handle, indent=2, ensure_ascii=False)
    with closing(sqlite3.connect((out / 'endpoint.sqlite').as_uri() + '?mode=ro', uri=True)) as source:
        destination = out / ('endpoint-after-' + phase + '.sqlite')
        assert not destination.exists()
        with closing(sqlite3.connect(destination)) as copy:
            source.backup(copy)
    print(json.dumps({'phase': phase, 'rows': len(rows), 'confirmed': sum(r['semantic_status'] == 'CONFIRMED' for r in rows)}, ensure_ascii=False))


if __name__ == '__main__':
    parser = argparse.ArgumentParser()
    parser.add_argument('--out', required=True, type=Path)
    parser.add_argument('--phase', required=True, choices=('initial', 'repair'))
    args = parser.parse_args()
    observe(args.out.resolve(), args.phase)
