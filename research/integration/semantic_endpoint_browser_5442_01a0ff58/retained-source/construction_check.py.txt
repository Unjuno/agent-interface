"""Ordinary SQLite unit construction check, not a primary browser run."""
import hashlib
import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parent
out = ROOT / 'construction-02'
out.mkdir()
started = datetime.now(timezone.utc).isoformat()
source = (ROOT / 'app.py').read_bytes()
(out / 'app.source.py.txt').write_bytes(source)
check = subprocess.run([sys.executable, '-m', 'py_compile', str(ROOT / 'app.py')], capture_output=True)
(out / 'compile.stdout.txt').write_bytes(check.stdout)
(out / 'compile.stderr.txt').write_bytes(check.stderr)
assert check.returncode == 0
import app
items = json.loads((ROOT / 'scenarios.json').read_bytes())
db = out / 'construction.sqlite'
app.initialize(db, items)
checks = []
for item in items:
    room = item['trial_id']
    assert app.snapshot(db, room) == {'A': {'version': 0, 'value': ''}, 'B': {'version': 0, 'value': ''}}
    app.apply(db, item, 'initial', item['desired_text'])
    state = app.snapshot(db, room)
    mode = item['mode']
    target = 'B' if mode == 'wrong_target' else 'A'
    assert state[target]['value'] == ('' if mode == 'noop' else item['desired_text'])
    assert state['A']['version'] == {'valid': 1, 'wrong_target': 0, 'noop': 0, 'stale_precondition': 2}[mode]
    try:
        app.apply(db, item, 'initial', item['desired_text'])
    except ValueError:
        pass
    else:
        raise AssertionError('duplicate accepted')
    assert app.snapshot(db, room) == state
    if mode != 'valid':
        versions = {t: r['version'] for t, r in state.items()}
        try:
            app.apply(db, item, 'repair', item['desired_text'], {'A': -1, 'B': -1})
        except ValueError:
            pass
        else:
            raise AssertionError('stale repair accepted')
        assert app.snapshot(db, room) == state
        app.apply(db, item, 'repair', item['desired_text'], versions)
        after = app.snapshot(db, room)
        assert after['A'] == {'version': versions['A'] + 1, 'value': item['desired_text']}
        assert after['B']['value'] == ''
        assert after['B']['version'] == versions['B'] + int(bool(state['B']['value']))
    checks.append({'trial_id': room, 'mode': mode, 'passed': True})
result = {'started_utc': started, 'ended_utc': datetime.now(timezone.utc).isoformat(),
          'app_sha256': hashlib.sha256(source).hexdigest(), 'compile_exit': check.returncode,
          'unit_scenarios': checks, 'duplicate_guard_checks': 8, 'stale_guard_checks': 6,
          'fresh_repair_checks': 6, 'GUI_actions': 0, 'primary_server_invocations': 0}
(out / 'RESULT.json').write_text(json.dumps(result, indent=2, ensure_ascii=False), encoding='utf-8')
print(json.dumps(result, ensure_ascii=False))
