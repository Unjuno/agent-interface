from datetime import datetime, timezone
from pathlib import Path
import hashlib, json, subprocess, sys
ROOT=Path(__file__).resolve().parents[1]/'work/iab-golden-port-C01'
names=['audit-v2.py','PLAN.md','inputs.json','FREEZE.json','READY.json','FINAL.json','ui-raw.json','UI_EXPORT.json','TAB_CLOSED.json','TRANSPORT_RECEIPT.json','executed-function.js.txt','executed-goals.json','application/submission-history.jsonl','fixture.py.txt','fixture_server.py','run-ui.js.txt','negative.jpg','audit.py','AUDIT.json','AUDIT_FREEZE.json','AUDIT_RECEIPT.json','audit.stdout.txt','audit.stderr.txt']
pins={n:{'bytes':len((ROOT/n).read_bytes()),'sha256':hashlib.sha256((ROOT/n).read_bytes()).hexdigest()} for n in names}
pre={'utc':datetime.now(timezone.utc).isoformat(),'argv':[sys.executable,str(ROOT/'audit-v2.py')],'inputs':pins,'read_only_originals':True,'producer_reexecution':False}
with (ROOT/'AUDIT_V2_FREEZE.json').open('x',encoding='utf8',newline='\n') as f:json.dump(pre,f,indent=2);f.write('\n')
start=datetime.now(timezone.utc).isoformat()
run=subprocess.run(pre['argv'],capture_output=True,timeout=15)
for name,raw in [('audit-v2.stdout.txt',run.stdout),('audit-v2.stderr.txt',run.stderr)]:
    with (ROOT/name).open('xb') as f:f.write(raw)
for name,pin in pins.items():
    if hashlib.sha256((ROOT/name).read_bytes()).hexdigest()!=pin['sha256']:raise ValueError('original changed during read-only check')
receipt={'utc_start':start,'utc_end':datetime.now(timezone.utc).isoformat(),'exit_code':run.returncode,'stdout_bytes':len(run.stdout),'stderr_bytes':len(run.stderr),'originals_unchanged':True}
with (ROOT/'AUDIT_V2_RECEIPT.json').open('x',encoding='utf8',newline='\n') as f:json.dump(receipt,f,indent=2);f.write('\n')
print(json.dumps(receipt));print(run.stdout.decode());print(run.stderr.decode())
raise SystemExit(run.returncode)
