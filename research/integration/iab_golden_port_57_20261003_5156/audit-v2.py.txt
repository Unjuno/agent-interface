"""Separate retained-data reconstruction, never starts the fixture or browser."""
from datetime import datetime, timezone
from pathlib import Path
from collections import Counter
import copy
import hashlib
import json

ROOT = Path(__file__).resolve().parent
sha = lambda raw: hashlib.sha256(raw).hexdigest()

def object_pairs(pairs):
    out = {}
    for key, value in pairs:
        if key in out:
            raise ValueError('duplicate JSON key')
        out[key] = value
    return out

def parse(raw):
    return json.loads(raw, object_pairs_hook=object_pairs,
                      parse_constant=lambda token: (_ for _ in ()).throw(ValueError(token)))

def need(value, message):
    if not value:
        raise ValueError(message)

def utc(text):
    need(type(text) is str, 'timestamp type')
    result = datetime.fromisoformat(text.replace('Z', '+00:00'))
    need(result.tzinfo is not None, 'timestamp timezone')
    return result

def load_data():
    return {'config':parse((ROOT/'inputs.json').read_bytes()),
            'freeze':parse((ROOT/'FREEZE.json').read_bytes()),
            'ready':parse((ROOT/'READY.json').read_bytes()),
            'final':parse((ROOT/'FINAL.json').read_bytes()),
            'ui':parse((ROOT/'ui-raw.json').read_bytes()),
            'export':parse((ROOT/'UI_EXPORT.json').read_bytes()),
            'closed':parse((ROOT/'TAB_CLOSED.json').read_bytes()),
            'receipt':parse((ROOT/'TRANSPORT_RECEIPT.json').read_bytes()),
            'goals':parse((ROOT/'executed-goals.json').read_bytes()),
            'records':[parse(line) for line in (ROOT/'application/submission-history.jsonl').read_bytes().splitlines()],
            'inputs':{name:(ROOT/name).read_bytes() for name in parse((ROOT/'FREEZE.json').read_bytes())['inputs']},
            'executed_function':(ROOT/'executed-function.js.txt').read_bytes()}

def validate(data):
    config, freeze, ready, final, ui = [data[n] for n in ('config','freeze','ready','final','ui')]
    for name, pin in freeze['inputs'].items():
        raw = data['inputs'][name]
        need(type(pin['bytes']) is int and pin['bytes'] == len(raw), 'pinned byte size')
        need(pin['sha256'] == sha(raw), 'pinned hash')
    need(sha(data['inputs']['fixture.py.txt']) == '5dd92db6aeddc180fccec563baed0440d9faa1fa6ac136aee45e479549d48976','literal unchanged fixture')
    function = data['inputs']['run-ui.js.txt'].decode().split('\n',1)[1]
    need(data['executed_function'].decode() == function,'executed function differs from prospective source')
    need(config['task_order'] == [f'task-{i}' for i in range(1,7)],'literal task order')
    need(config['expected_tokens'] == [f't991032-{i}' for i in range(1,7)],'literal tokens')
    for record in (freeze, ready, final, ui):
        need(record['id'] == config['id'],'execution identity')
    need(ready['fixture_sha256'] == freeze['inputs']['fixture.py.txt']['sha256'],'ready source join')
    need(type(ready['pid']) is int and ready['pid'] == final['pid'] == data['receipt']['owned_pid'] == 29576,'actual process join')
    need(len(ready['goals']) == len(data['goals']) == 6,'goals census')
    base = ready['goals'][0]['url'].rsplit('/task/',1)[0]
    need(base == 'http://127.0.0.1:55240','actual own loopback origin')
    for i,(ready_goal, sent_goal) in enumerate(zip(ready['goals'],data['goals']),1):
        expected = {'task_id':f'task-{i}','token':f't991032-{i}','url':f'{base}/task/{i}'}
        need(sent_goal == expected and all(ready_goal[k] == v for k,v in expected.items()),'intent-source join')
        need(ready_goal['layout'] == ('A' if i<=3 else 'B'),'literal layout split')
    records = data['records']
    need(len(records) == 6 and final['records'] == records,'literal final/raw census')
    need(Counter(r['task_id'] for r in records) == Counter(config['task_order']),'one effect per own task')
    last_ns = 0
    for i, record in enumerate(records,1):
        need(record['schema'] == 'integrated_efficiency_submission_v1','record schema')
        need(record['task_id'] == f'task-{i}' and record['layout'] == ('A' if i<=3 else 'B'),'actual target/order/layout')
        token = config['expected_tokens'][i-1]
        need(record['submitted_values'] == [token] and record['expected_token'] == token,'actual application token, not exact flag alone')
        need(record['exact'] is True,'typed application exact flag')
        need(type(record['received_ns']) is int and record['received_ns'] > last_ns,'typed ordered server clock')
        last_ns = record['received_ns']
    need(len(ui['rows']) == 6 and ui['tab_id'] == data['closed']['tab_id'] == '1','actual UI/tab census')
    previous = utc(ui['utc_start'])
    for i, row in enumerate(ui['rows'],1):
        need(row['task_id'] == f'task-{i}' and row['token'] == config['expected_tokens'][i-1],'UI intent join')
        need(row['url'] == f'{base}/task/{i}' and row['final_url'] == f'{base}/submit/{i}','UI task origin/route')
        need(row['field_count'] == row['save_count'] == row['saved_heading_count'] == 1 and all(type(row[k]) is int for k in ('field_count','save_count','saved_heading_count')),'typed unique controls')
        need(row['status'] == 'UI_SAVED' and row['error'] is None,'UI terminal')
        need(f'AI INTEGRATED task-{i} READY' in row['before'] and 'text field (settable) Value' in row['before'] and 'button Save' in row['before'],'actual pre-control observation')
        need('heading AI INTEGRATED SAVED' in row['after'] and f'/submit/{i}' in row['after'],'actual post-effect observation')
        start,end = utc(row['utc_start']),utc(row['utc_end'])
        need(previous <= start <= end,'UI local timestamp order')
        previous = end
    negative = ui['negative']
    need(negative['url'] == base+'/task/7' and negative['page_title'] == 'NOT FOUND','actual changed route')
    need(negative['field_count'] == negative['save_count'] == 0 and type(negative['save_count']) is int and type(negative['field_count']) is int,'typed missing controls')
    need('NOT FOUND' in negative['before'] and 'NOT FOUND' in negative['after'],'negative observed state')
    need(negative['status'] == 'MISSING_CONTROL_ERROR_RETURN' and negative['error']['name'] == 'Error','actual missing-control return')
    need('selector deadline exceeded' in negative['error']['message'] and '"matchCount":0' in negative['error']['message'],'actual no-match diagnostic')
    times = [utc(freeze['frozen_utc']),utc(ready['utc_start']),utc(ready['utc_ready']),utc(ui['utc_start']),previous,
             utc(negative['utc_start']),utc(negative['click_start']),utc(negative['click_end']),utc(negative['utc_end']),utc(ui['utc_end']),
             utc(data['closed']['utc']),utc(final['utc_end']),utc(data['receipt']['process_absence_observed_utc'])]
    need(times == sorted(times),'actual phase UTC order (not server perf clock)')
    need(data['export']['rows'] == 6 and data['export']['original_tab_closed']['utc'] == data['closed']['utc'],'original in-memory export join')
    need(data['closed']['close_returned'] is True and data['closed']['remaining_task_local_iab_tabs'] == [],'own tab release observation')
    need(final['stop_reason'] == 'own_stop_flag' and final['error'] is None,'own natural supervisor finish')
    need(final['cleanup'] == 'own_fixture_close_returned' and final['server_thread_alive'] is False and type(final['server_socket_fileno']) is int and final['server_socket_fileno'] == -1,'actual owned server cleanup')
    need(type(data['receipt']['process_exit_code']) is int and data['receipt']['process_exit_code'] == 0 and data['receipt']['pid_still_present'] is False,'actual process exit/absence')
    oracle = final['oracle']
    need(oracle['success'] is True and oracle['record_count'] == 6 and oracle['exact_counts'] == {f'task-{i}':1 for i in range(1,7)} and oracle['unexpected'] == [] and oracle['duplicates'] == {} and oracle['missing'] == [],'producer oracle consistency, separately reconstructed above')
    return {'decision':'PORT_PASS_SCOPED','actual_effects':6,'layouts':{'A':3,'B':3},
            'healthy_ui_seconds':(previous-utc(ui['utc_start'])).total_seconds(),
            'whole_ui_seconds':(utc(ui['utc_end'])-utc(ui['utc_start'])).total_seconds(),
            'missing_click_observed_seconds':(utc(negative['click_end'])-utc(negative['click_start'])).total_seconds(),
            'requested_missing_seconds':config['missing_timeout_ms']/1000,
            'model_tokens':'unavailable','physical_input_release':'not observed'}

def main():
    data = load_data()
    result = validate(data)
    controls = []
    def change_both(d, index, **fields):
        d['records'][index].update(fields)
        d['final']['records'][index].update(fields)
    def duplicate_both(d):
        d['records'].append(copy.deepcopy(d['records'][0]))
        d['final']['records'].append(copy.deepcopy(d['final']['records'][0]))
        d['final']['oracle'].update(record_count=7)
        d['final']['oracle']['exact_counts']['task-1']=2
    mutations = {
        'extra_duplicate_effect':duplicate_both,
        'wrong_submitted_value_despite_true_flag':lambda d:change_both(d,0,submitted_values=['wrong']),
        'wrong_task':lambda d:change_both(d,3,task_id='task-1'),
        'boolean_server_clock':lambda d:change_both(d,0,received_ns=True),
        'false_ui_heading':lambda d:d['ui']['rows'][0].update(after='AI INTEGRATED REJECTED'),
        'missing_click_claimed_success':lambda d:d['ui']['negative'].update(status='UNEXPECTED_CLICK_RETURN'),
        'open_final_socket':lambda d:d['final'].update(server_socket_fileno=0),
        'source_byte_changed':lambda d:d['inputs'].update({'fixture.py.txt':d['inputs']['fixture.py.txt'][:-1]+b' '})}
    for name, mutate in mutations.items():
        changed = copy.deepcopy(data)
        mutate(changed)
        try:
            validate(changed)
        except (ValueError,KeyError,TypeError) as error:
            controls.append({'name':name,'refused':True,'reason':str(error)})
        else:
            raise ValueError('copied control accepted: '+name)
    result.update(controls=controls,source_arm_replays=0,original_child_replays=0,
                  checker_recorded_utc=datetime.now(timezone.utc).isoformat())
    with (ROOT/'AUDIT_V2.json').open('x',encoding='utf8',newline='\n') as f:
        json.dump(result,f,indent=2);f.write('\n')
    print(json.dumps(result))

if __name__ == '__main__':
    main()
