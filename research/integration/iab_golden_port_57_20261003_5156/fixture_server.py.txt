"""Own bounded supervisor; exact original Fixture executes in this one process."""
from datetime import datetime, timezone
from pathlib import Path
import hashlib
import importlib.util
import json
import os
import sys
import time
import traceback

ROOT = Path(__file__).resolve().parent

def stamp():
    return datetime.now(timezone.utc).isoformat()

def dump(name, value):
    with (ROOT / name).open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

def main():
    start = stamp()
    freeze = json.loads((ROOT / 'FREEZE.json').read_bytes())
    for name, pin in freeze['inputs'].items():
        raw = (ROOT / name).read_bytes()
        if len(raw) != pin['bytes'] or hashlib.sha256(raw).hexdigest() != pin['sha256']:
            raise ValueError('prospective input changed: ' + name)
    if (ROOT / 'READY.json').exists() or (ROOT / 'FINAL.json').exists():
        raise ValueError('this construction process has already started')
    spec = importlib.util.spec_from_file_location('owned_golden_fixture', ROOT / 'fixture.py.txt')
    from importlib.machinery import SourceFileLoader
    spec = importlib.util.spec_from_loader('owned_golden_fixture', SourceFileLoader('owned_golden_fixture', str(ROOT / 'fixture.py.txt')))
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    config = json.loads((ROOT / 'inputs.json').read_bytes())
    fixture = module.Fixture(ROOT / 'application', config['seed'])
    record = {'id': freeze['id'], 'pid': os.getpid(), 'utc_start': start,
              'utc_ready': stamp(), 'python': sys.version, 'goals': fixture.goals(),
              'fixture_sha256': freeze['inputs']['fixture.py.txt']['sha256']}
    dump('READY.json', record)
    print(json.dumps({'ready': True, 'pid': os.getpid(), 'base_url': record['goals'][0]['url'].rsplit('/task/', 1)[0]}), flush=True)
    deadline = time.monotonic() + config['server_max_seconds']
    reason = 'own_stop_flag'
    error = None
    try:
        while not (ROOT / 'STOP.flag').exists():
            if time.monotonic() >= deadline:
                reason = 'server_lifetime_expired'
                break
            if fixture.history.exists() and fixture.history.stat().st_size > config['output_cap_bytes']:
                reason = 'owned_output_cap'
                break
            time.sleep(0.05)
    except BaseException:
        reason = 'wrapper_exception'
        error = traceback.format_exc()
    finally:
        records = fixture.records()
        oracle = fixture.evaluate()
        fixture.close()
        final = {'id': freeze['id'], 'pid': os.getpid(), 'utc_end': stamp(),
                 'stop_reason': reason, 'error': error, 'records': records, 'oracle': oracle,
                 'server_thread_alive': fixture.thread.is_alive(),
                 'server_socket_fileno': fixture.server.socket.fileno(),
                 'cleanup': 'own_fixture_close_returned'}
        dump('FINAL.json', final)
        print(json.dumps({'final': True, 'reason': reason, 'records': len(records), 'oracle_success': oracle['success'], 'server_thread_alive': final['server_thread_alive'], 'socket_fileno': final['server_socket_fileno']}), flush=True)
    return 0 if reason == 'own_stop_flag' and error is None and not fixture.thread.is_alive() and fixture.server.socket.fileno() == -1 else 1

if __name__ == '__main__':
    raise SystemExit(main())
