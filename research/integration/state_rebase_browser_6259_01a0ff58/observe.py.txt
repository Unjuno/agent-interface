"""One fresh process/one read-only SQLite transaction, no app mutation."""
import json
import os
from pathlib import Path
import sqlite3
import sys
from datetime import datetime, timezone


def observe(path, stage):
    db = sqlite3.connect(Path(path).resolve().as_uri()+"?mode=ro", uri=True)
    db.row_factory = sqlite3.Row
    db.execute("PRAGMA query_only=ON")
    db.execute("BEGIN")
    m = dict(db.execute("SELECT * FROM metadata").fetchone())
    rows = [dict(row) for row in db.execute("SELECT * FROM cases ORDER BY case_id")]
    effects = [dict(row) for row in db.execute("SELECT * FROM effects ORDER BY id")]
    public = []
    for row in rows:
        covered = ["target_value", "protected_sibling", "notifications", "release_status"]
        p = dict(m, case_id=row["case_id"], read_complete=True, covered_fields=covered,
                 **{k: row[k] for k in covered})
        if row["mask_notifications"]:
            p.pop("notifications")
            p["covered_fields"] = [k for k in covered if k != "notifications"]
            p["read_complete"] = False
        public.append(p)
    db.rollback()
    db.close()
    return dict(stage=stage, observed_utc=datetime.now(timezone.utc).isoformat(),
                observer_pid=os.getpid(), metadata=m, cases=rows, effects=effects,
                public_projection=public)


if __name__ == "__main__":
    print(json.dumps(observe(sys.argv[1], sys.argv[2]), indent=2))
