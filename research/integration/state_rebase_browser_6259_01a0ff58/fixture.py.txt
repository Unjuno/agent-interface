"""Owned loopback HTML/SQLite fixture. Private ledger is never served to UI."""
import argparse
import hashlib
import html
import json
import os
from pathlib import Path
import sqlite3
import threading
from http.server import BaseHTTPRequestHandler, HTTPServer
from urllib.parse import parse_qs, urlparse
from datetime import datetime, timezone


def utc():
    return datetime.now(timezone.utc).isoformat()


def connect():
    db = sqlite3.connect(DB)
    db.row_factory = sqlite3.Row
    return db


def public_row(row):
    covered = ["target_value", "protected_sibling", "notifications", "release_status"]
    p = dict(namespace=NAMESPACE, session=SESSION, case_id=row["case_id"],
             target_value=row["target_value"], protected_sibling=row["protected_sibling"],
             notifications=row["notifications"], release_status=row["release_status"],
             covered_fields=covered, read_complete=True)
    if row["mask_notifications"]:
        p.pop("notifications")
        p["covered_fields"] = [k for k in covered if k != "notifications"]
        p["read_complete"] = False
    return p


def effect(db, case_id, actor, mode):
    row = db.execute("SELECT * FROM cases WHERE case_id=?", (case_id,)).fetchone()
    after = row["target_value"] if mode in {"NOOP", "UNSATISFIED"} else GOALS[case_id]["desired"]
    delta = 1 if mode == "NOTIFY" else 0
    db.execute("UPDATE cases SET target_value=?,notifications=notifications+?,"
               "event_count=event_count+? WHERE case_id=?",
               (after, delta, int(mode == "EVENT"), case_id))
    db.execute("INSERT INTO effects(case_id,actor,mode,before_value,after_value,notification_delta,utc)"
               " VALUES(?,?,?,?,?,?,?)", (case_id, actor, mode, row["target_value"], after, delta, utc()))


def page(row):
    c = GOALS[row["case_id"]]
    p = public_row(row)
    fields = {"Namespace": p["namespace"], "Study session": p["session"],
              "Case": p["case_id"], "Contract kind": c["kind"],
              "Desired value": c["desired"], "Current value": p["target_value"],
              "Protected sibling": p["protected_sibling"],
              "Release evidence": p["release_status"],
              "Complete observation": str(p["read_complete"]).lower()}
    if "notifications" in p:
        fields["Notification count"] = str(p["notifications"])
    controls = "".join(f'<label>{html.escape(k)}<input type="text" readonly value="{html.escape(v)}"></label>'
                       for k, v in fields.items())
    ack = "Dispatch accepted; effect attribution unavailable." if row["root_count"] else "No dispatch yet."
    button = '<button type="submit">Submit once</button>' if not row["root_count"] else '<p>One study dispatch recorded. Further submission is disabled.</p>'
    return ('<!doctype html><html><head><meta charset="utf-8"><title>State rebase study</title>'
            '<style>body{font:16px system-ui;background:#eef2f6;color:#142130;margin:24px auto;max-width:900px}'
            'main{background:white;padding:24px;border-radius:12px}section{display:grid;grid-template-columns:1fr 1fr;gap:12px}'
            'label{display:block;font-weight:600}input{display:block;width:95%;padding:6px;margin-top:4px;color:#142130}'
            'button{padding:10px 20px;background:#145fa8;color:white;border:0;border-radius:6px;margin-top:20px}'
            '.note{padding:12px;background:#eef5fc}</style></head><body><main>'
            f'<h1>State rebase: {row["case_id"]}</h1>'
            '<p>Controlled research fixture · invented records · read-only evidence fields</p>'
            f'<section>{controls}</section><p class="note" role="status">{ack}</p>'
            f'<form method="post" action="/dispatch/{row["case_id"]}">{button}</form>'
            '<p>Current state and request origin are separate observations. The acknowledgement contains no bound event receipt.</p>'
            '</main></body></html>')


class Handler(BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        print(json.dumps(dict(utc=utc(), method=self.command, path=self.path,
                              detail=fmt % args)), flush=True)

    def response(self, code, body, kind="text/plain; charset=utf-8"):
        data = body.encode("utf-8")
        self.send_response(code)
        self.send_header("Content-Type", kind)
        self.send_header("Cache-Control", "no-store")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        path = urlparse(self.path).path
        if path.startswith("/case/") and path[6:] in GOALS:
            with connect() as db:
                row = db.execute("SELECT * FROM cases WHERE case_id=?", (path[6:],)).fetchone()
            self.response(200, page(row), "text/html; charset=utf-8")
        else:
            self.response(404, "Unknown owned route")

    def do_POST(self):
        path = urlparse(self.path).path
        body = self.rfile.read(int(self.headers.get("Content-Length", "0")))
        if path == "/stop":
            self.response(200, "Owned server stopping")
            threading.Thread(target=self.server.shutdown, daemon=True).start()
            return
        case_id = path.rsplit("/", 1)[-1]
        if case_id not in GOALS or path not in {"/dispatch/"+case_id, "/actor/"+case_id}:
            self.response(404, "Unknown owned route")
            return
        with connect() as db:
            db.execute("BEGIN IMMEDIATE")
            row = db.execute("SELECT * FROM cases WHERE case_id=?", (case_id,)).fetchone()
            actor = path.startswith("/actor/")
            deck = DECK[case_id]
            if actor:
                permitted = deck["actor_order"]
                order_ok = (permitted == "BEFORE_ROOT" and row["root_count"] == 0 or
                            permitted == "AFTER_ROOT" and row["root_count"] == 1)
                if row["actor_count"] or not order_ok:
                    self.response(409, "Actor order or multiplicity rejected")
                    return
                db.execute("UPDATE cases SET actor_count=actor_count+1 WHERE case_id=?", (case_id,))
                effect(db, case_id, "EXPERIMENT_ACTOR", deck["actor_mode"])
            else:
                before_ok = deck["actor_order"] != "BEFORE_ROOT" or row["actor_count"] == 1
                if row["root_count"] or not before_ok:
                    self.response(409, "Root order or multiplicity rejected")
                    return
                db.execute("UPDATE cases SET root_count=root_count+1 WHERE case_id=?", (case_id,))
                effect(db, case_id, "ROOT_DOM", deck["root_mode"])
            db.commit()
        if actor:
            self.response(200, "Serialized experimental actor accepted")
        else:
            self.send_response(303)
            self.send_header("Location", "/case/"+case_id)
            self.send_header("Content-Length", "0")
            self.end_headers()


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("output")
    ap.add_argument("session")
    args = ap.parse_args()
    OUT = Path(args.output)
    OUT.mkdir(exist_ok=False)
    SOURCE = Path(__file__).resolve().parent
    NAMESPACE = "state-rebase-browser-6259-01a0ff58"
    SESSION = args.session
    DB = OUT / "app.sqlite"
    GOALS = {g["case_id"]: dict(g, session=SESSION) for g in json.loads((SOURCE / "GOALS.json").read_text())}
    DECK = {d["case_id"]: d for d in json.loads((SOURCE / "SCENARIOS.json").read_text())}
    with connect() as db:
        db.executescript('CREATE TABLE metadata(namespace TEXT, session TEXT);'
                        'CREATE TABLE cases(case_id TEXT PRIMARY KEY,target_value TEXT,protected_sibling TEXT,'
                        'notifications INTEGER,event_count INTEGER,root_count INTEGER,actor_count INTEGER,'
                        'mask_notifications INTEGER,release_status TEXT);'
                        'CREATE TABLE effects(id INTEGER PRIMARY KEY,case_id TEXT,actor TEXT,mode TEXT,'
                        'before_value TEXT,after_value TEXT,notification_delta INTEGER,utc TEXT);')
        db.execute("INSERT INTO metadata VALUES(?,?)", (NAMESPACE, SESSION))
        for case_id, d in DECK.items():
            db.execute("INSERT INTO cases VALUES(?,?,?,?,?,?,?,?,?)",
                       (case_id, "waiting", "keep", 0, 0, 0, 0,
                        int(d["mask_notifications"]), d["release_status"]))
    server = HTTPServer(("127.0.0.1", 0), Handler)
    start = dict(utc=utc(), pid=os.getpid(), session=SESSION,
                 url=f"http://127.0.0.1:{server.server_port}",
                 source_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest())
    (OUT / "SERVER_STARTED.json").write_text(json.dumps(start, indent=2))
    print(json.dumps(start), flush=True)
    try:
        server.serve_forever(poll_interval=0.1)
    finally:
        server.server_close()
        (OUT / "SERVER_ENDED.json").write_text(json.dumps(dict(utc=utc(), pid=os.getpid(),
            session=SESSION, normal_serve_return=True), indent=2))
