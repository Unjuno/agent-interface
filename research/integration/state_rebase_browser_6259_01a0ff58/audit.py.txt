"""Independent saved-data evaluator; does not import/run fixture, observer or gate.

Reconstruct complete serial app histories from the prospective actor deck and join
each public/UI view. Literal task results are separately fixed in the plan.
"""
import copy
from datetime import datetime
import hashlib
import json
from pathlib import Path
import sys


def need(condition, message):
    if not condition:
        raise ValueError(message)


def load(path):
    return json.loads(path.read_text(encoding="utf-8"))


def independent_decision(c, p):
    if tuple(p[k] for k in ("namespace", "session", "case_id")) != tuple(c[k] for k in ("namespace", "session", "case_id")):
        task, cause = "UNKNOWN", "binding"
    elif c["kind"] == "EVENT":
        task, cause = "NOT_ELIGIBLE", "event_specific"
    elif p["read_complete"] is False:
        task, cause = "UNKNOWN", "coverage"
    elif c["release_required"] and p["release_status"] == "UNVERIFIED":
        task, cause = "UNKNOWN", "release_unverified"
    elif all((p["target_value"] == c["desired"], p["protected_sibling"] == c["protected_sibling"],
              p["notifications"] <= c["max_notifications"])):
        task, cause = "SATISFIED", "state_predicate"
    else:
        task, cause = "NOT_SATISFIED", "state_mismatch"
    return dict(task_state=task, cause=cause, request_origin="UNKNOWN", next_action="NONE")


def check(root, decisions=None, snapshots=None, inputs=None, ui=None):
    source = root.parent
    goals = load(source/"GOALS.json")
    deck = load(source/"SCENARIOS.json")
    ids = [f"C{i:02}" for i in range(1, 9)]
    need([r["case_id"] for r in goals] == ids == [r["case_id"] for r in deck], "complete prospective cases")
    snapshots = snapshots or {name: load(root/"receipts"/(name+".stdout.log")) for name in ["initial", *ids, "after_classifier"]}
    decisions = decisions if decisions is not None else load(root/"receipts"/"classifier.stdout.log")
    inputs = inputs if inputs is not None else load(root/"candidate_input.json")
    ui = ui if ui is not None else {cid: load(root/"ui"/(cid+".json")) for cid in ids}
    need(len(decisions) == len(inputs) == 8, "decision/input denominator")
    states = [dict(case_id=d["case_id"], target_value="waiting", protected_sibling="keep",
                   notifications=0, event_count=0, root_count=0, actor_count=0,
                   mask_notifications=int(d["mask_notifications"]), release_status=d["release_status"]) for d in deck]
    effects = []
    metadata = {k: goals[0][k] for k in ("namespace", "session")}
    need(snapshots["initial"]["cases"] == states and snapshots["initial"]["effects"] == [], "initial state")
    previous_utc = None
    for n, (c, d) in enumerate(zip(goals, deck)):
        cid = c["case_id"]
        actual = snapshots[cid]
        row = states[n]
        schedule = [("ROOT_DOM", d["root_mode"])]
        if d["actor_order"] == "BEFORE_ROOT":
            schedule.insert(0, ("EXPERIMENT_ACTOR", d["actor_mode"]))
        elif d["actor_order"] == "AFTER_ROOT":
            schedule.append(("EXPERIMENT_ACTOR", d["actor_mode"]))
        for actor, mode in schedule:
            before = row["target_value"]
            after = before if mode in ("NOOP", "UNSATISFIED") else c["desired"]
            delta = int(mode == "NOTIFY")
            row["target_value"] = after
            row["notifications"] += delta
            row["event_count"] += int(mode == "EVENT")
            row["root_count" if actor == "ROOT_DOM" else "actor_count"] += 1
            index = len(effects)
            need(len(actual["effects"]) > index, "missing actual ledger effect")
            effect = actual["effects"][index]
            need(set(effect) == {"id", "case_id", "actor", "mode", "before_value", "after_value", "notification_delta", "utc"}, "typed effect schema")
            need(type(effect["id"]) is int and type(effect["notification_delta"]) is int, "typed effect counts")
            expected = dict(id=index+1, case_id=cid, actor=actor, mode=mode, before_value=before,
                            after_value=after, notification_delta=delta, utc=effect["utc"])
            now = datetime.fromisoformat(effect["utc"])
            need(now.tzinfo is not None and (previous_utc is None or now >= previous_utc), "serial ledger UTC")
            previous_utc = now
            need(effect == expected, "exact ledger reconstruction")
            effects.append(expected)
        need(actual["stage"] == cid and actual["metadata"] == metadata, "snapshot binding")
        need(actual["cases"] == states and actual["effects"] == effects, "complete snapshot reconstruction")
        for row_check in actual["cases"]:
            for k in ("notifications", "event_count", "root_count", "actor_count", "mask_notifications"):
                need(type(row_check[k]) is int, "typed full state")
        covered = ["target_value", "protected_sibling", "notifications", "release_status"]
        p = dict(metadata, case_id=cid, **{k: row[k] for k in covered}, read_complete=True, covered_fields=covered)
        if d["mask_notifications"]:
            p.pop("notifications")
            p["covered_fields"] = [k for k in covered if k != "notifications"]
            p["read_complete"] = False
        actual_public = actual["public_projection"][n]
        need(actual_public == p, "public/private projection join")
        need(inputs[n] == {"contract": c, "public": p}, "candidate input excludes private truth")
        labels = {"Namespace": p["namespace"], "Study session": p["session"], "Case": cid,
                  "Contract kind": c["kind"], "Desired value": c["desired"],
                  "Current value": p["target_value"], "Protected sibling": p["protected_sibling"],
                  "Release evidence": p["release_status"], "Complete observation": str(p["read_complete"]).lower()}
        if "notifications" in p:
            labels["Notification count"] = str(p["notifications"])
        need(ui[cid]["labels"] == labels and ui[cid]["case_id"] == cid, "visible UI/public field join")
        need("Dispatch accepted; effect attribution unavailable." in ui[cid]["ax"], "untagged acknowledgement")
        need(ui[cid]["action"] == "ONE_DOM_SUBMIT" and ui[cid]["tab_id"] == ui[ids[0]]["tab_id"], "owned UI action binding")
        need(datetime.fromisoformat(ui[cid]["start_utc"]) <= datetime.fromisoformat(ui[cid]["end_utc"]), "actual UI receipt UTC")
        wanted = dict(case_id=cid, decision=independent_decision(c, p),
                      naive_value_equal=p["target_value"] == c["desired"], strict_global_unknown="UNKNOWN")
        need(decisions[n] == wanted, "complete decision/causes/actions/reference comparison")
    terminal = snapshots["after_classifier"]
    need(terminal["cases"] == states and terminal["effects"] == effects and terminal["metadata"] == metadata,
         "no post-classifier state/effect change")
    need([r["decision"]["task_state"] for r in decisions] ==
         ["SATISFIED", "SATISFIED", "NOT_ELIGIBLE", "NOT_ELIGIBLE", "NOT_SATISFIED", "UNKNOWN", "UNKNOWN", "NOT_SATISFIED"],
         "prospective literal outcomes")
    return dict(cases=8, root_dispatches=8, experimental_actor_writes=2, private_ledger_rows=10,
                event_specific_actual_event_counts=[states[2]["event_count"], states[3]["event_count"]],
                task_counts={x: sum(r["decision"]["task_state"] == x for r in decisions)
                             for x in ("SATISFIED", "NOT_ELIGIBLE", "NOT_SATISFIED", "UNKNOWN")},
                naive_value_equal=7, naive_unsupported_completions=5, strict_unknown_withheld_eligible=2,
                post_rebase_actions=0, request_origin_claims=0,
                scientific_result="PASS_CONTROLLED_APP_TRANSFER_SCOPED")


if __name__ == "__main__":
    root = Path(sys.argv[1]).resolve()
    result = check(root)
    original = load(root/"receipts"/"classifier.stdout.log")
    controls = []
    edits = [("case_binding", "case_id", "CXX"), ("false_origin", "request_origin", "ROOT_DOM"),
             ("extra_action", "next_action", "RETRY"), ("false_state", "task_state", "NOT_SATISFIED"),
             ("wrong_cause", "cause", "event_specific")]
    for name, field, value in edits:
        rows = copy.deepcopy(original)
        target = rows[0] if field == "case_id" else rows[0]["decision"]
        target[field] = value
        try:
            check(root, decisions=rows)
        except ValueError as err:
            controls.append(dict(id=name, detected=True, reason=str(err)))
        else:
            raise ValueError("ineffective copied control: "+name)
    for name, position, value in (("event_naive_accept", 3, "SATISFIED"), ("forbidden_collateral_accept", 4, "SATISFIED"),
                                 ("incomplete_accept", 5, "SATISFIED"), ("release_accept", 6, "SATISFIED")):
        rows = copy.deepcopy(original)
        rows[position]["decision"]["task_state"] = value
        try:
            check(root, decisions=rows)
        except ValueError as err:
            controls.append(dict(id=name, detected=True, reason=str(err)))
        else:
            raise ValueError("ineffective copied control: "+name)
    snapshots = {name: load(root/"receipts"/(name+".stdout.log")) for name in ["initial", *[f"C{i:02}" for i in range(1, 9)], "after_classifier"]}
    for name in ("hidden_extra_effect", "wrong_actor", "bool_as_count", "protected_clobber"):
        modified = copy.deepcopy(snapshots)
        if name == "hidden_extra_effect":
            modified["C04"]["effects"].pop()
        elif name == "wrong_actor":
            modified["C02"]["effects"][1]["actor"] = "ROOT_DOM"
        elif name == "bool_as_count":
            modified["C01"]["cases"][0]["notifications"] = False
        else:
            modified["after_classifier"]["cases"][0]["protected_sibling"] = "clobbered"
        try:
            check(root, snapshots=modified)
        except ValueError as err:
            controls.append(dict(id=name, detected=True, reason=str(err)))
        else:
            raise ValueError("ineffective copied control: "+name)
    rows = copy.deepcopy(original)
    rows.pop()
    try:
        check(root, decisions=rows)
    except ValueError as err:
        controls.append(dict(id="missing_case", detected=True, reason=str(err)))
    else:
        raise ValueError("ineffective missing-row control")
    print(json.dumps(dict(result=result, effective_copied_controls=controls,
                         audit_source_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest()), indent=2))
