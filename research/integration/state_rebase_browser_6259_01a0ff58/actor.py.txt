"""Explicit experimental actor API; distinct from ROOT DOM operation."""
import json
import sys
from urllib.request import Request, urlopen


if __name__ == "__main__":
    req = Request(sys.argv[1]+"/actor/"+sys.argv[2], data=b"", method="POST")
    with urlopen(req, timeout=5) as r:
        print(json.dumps(dict(status=r.status, body=r.read().decode(), case_id=sys.argv[2])))
