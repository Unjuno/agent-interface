"""Pure scoped predicate: no app, SQLite, network or input capability."""
import json
import sys


def classify(contract, public):
    # The contract is prospective semantic input. Event requests require receipts.
    result = {"request_origin": "UNKNOWN", "next_action": "NONE"}
    expected = {"namespace", "session", "case_id", "desired", "kind",
                "protected_sibling", "max_notifications", "release_required"}
    if set(contract) != expected or contract["kind"] not in {"STATE", "EVENT"}:
        raise ValueError("invalid contract")
    for key in ("namespace", "session", "case_id", "desired", "kind", "protected_sibling"):
        if type(contract[key]) is not str or not contract[key]:
            raise ValueError("invalid contract string")
    if type(contract["max_notifications"]) is not int or contract["max_notifications"] < 0:
        raise ValueError("invalid notification bound")
    if type(contract["release_required"]) is not bool:
        raise ValueError("invalid release obligation")
    identity = ("namespace", "session", "case_id")
    if any(type(public.get(k)) is not str or public[k] != contract[k] for k in identity):
        return dict(result, task_state="UNKNOWN", cause="binding")
    if contract["kind"] == "EVENT":
        return dict(result, task_state="NOT_ELIGIBLE", cause="event_specific")
    fields = {"target_value", "protected_sibling", "notifications", "release_status"}
    if (public.get("read_complete") is not True or
            type(public.get("covered_fields")) is not list or
            any(type(k) is not str for k in public["covered_fields"]) or
            set(public["covered_fields"]) != fields or len(public["covered_fields"]) != 4):
        return dict(result, task_state="UNKNOWN", cause="coverage")
    if (type(public.get("target_value")) is not str or
            type(public.get("protected_sibling")) is not str or
            type(public.get("notifications")) is not int or public["notifications"] < 0 or
            type(public.get("release_status")) is not str or
            public["release_status"] not in {"NO_INPUT_OBLIGATION", "VERIFIED", "UNVERIFIED"}):
        return dict(result, task_state="UNKNOWN", cause="typed_state")
    if contract["release_required"] and public["release_status"] != "VERIFIED":
        return dict(result, task_state="UNKNOWN", cause="release_unverified")
    ok = (public["target_value"] == contract["desired"] and
          public["protected_sibling"] == contract["protected_sibling"] and
          public["notifications"] <= contract["max_notifications"])
    return dict(result, task_state="SATISFIED" if ok else "NOT_SATISFIED",
                cause="state_predicate" if ok else "state_mismatch")


if __name__ == "__main__":
    bundle = json.load(open(sys.argv[1], encoding="utf-8"))
    rows = []
    for record in bundle:
        c, p = record["contract"], record["public"]
        rows.append({"case_id": c["case_id"], "decision": classify(c, p),
                     "naive_value_equal": p.get("target_value") == c["desired"],
                     "strict_global_unknown": "UNKNOWN"})
    print(json.dumps(rows, indent=2))
