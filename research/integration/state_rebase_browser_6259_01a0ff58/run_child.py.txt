"""Actual ordinary child receipts; stdout and failures remain immutable."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
from datetime import datetime, timezone


def utc():
    return datetime.now(timezone.utc).isoformat()


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("receipt_directory")
    ap.add_argument("label")
    ap.add_argument("command", nargs=argparse.REMAINDER)
    a = ap.parse_args()
    out = Path(a.receipt_directory).resolve()
    out.mkdir(parents=True, exist_ok=True)
    receipt_path = out / (a.label+".receipt.json")
    if receipt_path.exists() or (out / (a.label+".started.json")).exists():
        raise RuntimeError("receipt ID already consumed")
    cmd = [sys.executable if s == "@python" else s for s in a.command]
    source_pins = {str(Path(x).resolve()): hashlib.sha256(Path(x).read_bytes()).hexdigest()
                   for x in cmd if x.endswith(".py") and Path(x).is_file()}
    start = dict(utc=utc(), argv=cmd, cwd=str(Path.cwd()), parent_pid=os.getpid(), source_pins=source_pins)
    (out / (a.label+".started.json")).write_text(json.dumps(start, indent=2))
    with (out / (a.label+".stdout.log")).open("wb") as stdout, (out / (a.label+".stderr.log")).open("wb") as stderr:
        proc = subprocess.Popen(cmd, stdout=stdout, stderr=stderr)
        (out / (a.label+".pid.json")).write_text(json.dumps(dict(pid=proc.pid, utc=utc())))
        code = proc.wait()
    receipt = dict(start, child_pid=proc.pid, end_utc=utc(), exit_code=code,
                   stdout_sha256=hashlib.sha256((out/(a.label+".stdout.log")).read_bytes()).hexdigest(),
                   stderr_sha256=hashlib.sha256((out/(a.label+".stderr.log")).read_bytes()).hexdigest())
    receipt_path.write_text(json.dumps(receipt, indent=2))
    print(json.dumps(receipt))
    sys.exit(code)
