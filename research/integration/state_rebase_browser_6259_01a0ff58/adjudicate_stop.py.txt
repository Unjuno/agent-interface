"""Read-only independent reconciliation of a stopped partial experiment."""
import copy
from datetime import datetime
import hashlib
import json
from pathlib import Path
import sys


def need(ok, text):
    if not ok:
        raise ValueError(text)


def read(p):
    return json.loads(p.read_text(encoding="utf-8"))


def verify(root, override=None):
    override = override or {}
    def data(rel):
        return copy.deepcopy(override[rel]) if rel in override else read(root/rel)
    freeze = data("FREEZE.json")
    for name, expected in freeze["source_sha256"].items():
        need(hashlib.sha256((root/name).read_bytes()).hexdigest() == expected, "frozen source identity "+name)
    goals, deck = data("GOALS.json"), data("SCENARIOS.json")
    ids = [f"C{i:02}" for i in range(1, 9)]
    need([r["case_id"] for r in goals] == ids == [r["case_id"] for r in deck], "prospective complete denominator")
    s = data("primary/receipts/STOP_READBACK.stdout.log")
    metadata = {k: goals[0][k] for k in ("namespace", "session")}
    need(s["metadata"] == metadata and s["stage"] == "STOP_READBACK", "STOP snapshot identity")
    states = [dict(case_id=d["case_id"], target_value="waiting", protected_sibling="keep", notifications=0,
                   event_count=0, root_count=0, actor_count=0, mask_notifications=int(d["mask_notifications"]),
                   release_status=d["release_status"]) for d in deck]
    initial = data("primary/receipts/initial.stdout.log")
    need(initial["cases"] == states and initial["effects"] == [], "initial saved state")
    sequence = [(0, "ROOT_DOM", "SET"), (1, "EXPERIMENT_ACTOR", "SET"),
                (1, "ROOT_DOM", "NOOP"), (2, "ROOT_DOM", "EVENT"), (3, "ROOT_DOM", "EVENT")]
    need(len(s["cases"]) == 8 and len(s["effects"]) == 5, "actual prefix denominator")
    effects = []
    for index, (position, actor, mode) in enumerate(sequence):
        c, row = goals[position], states[position]
        before = row["target_value"]
        after = before if mode == "NOOP" else c["desired"]
        row["target_value"] = after
        row["root_count" if actor == "ROOT_DOM" else "actor_count"] += 1
        row["event_count"] += int(mode == "EVENT")
        actual = s["effects"][index]
        need(type(actual["id"]) is int and type(actual["notification_delta"]) is int, "typed ledger counts")
        expected = dict(id=index+1, case_id=c["case_id"], actor=actor, mode=mode,
                        before_value=before, after_value=after, notification_delta=0, utc=actual["utc"])
        need(actual == expected, "independent exact effect prefix")
        event_utc = datetime.fromisoformat(actual["utc"])
        need(event_utc.tzinfo is not None, "actual effect time")
        effects.append(expected)
        if actor == "ROOT_DOM":
            ui_rel = "primary/ui/"+c["case_id"]+(".post-root" if position == 3 else "")+".json"
            ui = data(ui_rel)
            intent = data("primary/ui/"+c["case_id"]+".intent.json")
            labels = {"Namespace": metadata["namespace"], "Study session": metadata["session"],
                      "Case": c["case_id"], "Contract kind": c["kind"], "Desired value": c["desired"],
                      "Current value": after, "Protected sibling": "keep", "Notification count": "0",
                      "Release evidence": "NO_INPUT_OBLIGATION", "Complete observation": "true"}
            need(ui["labels"] == labels and ui["case_id"] == c["case_id"] and ui["tab_id"] == "2", "typed visible UI join")
            need(intent["case_id"] == ui["case_id"] and intent["start_utc"] == ui["start_utc"] and
                 ui["action"] == intent["action"] == "ONE_DOM_SUBMIT", "actual pre-send intent join")
            need(datetime.fromisoformat(ui["start_utc"]) <= event_utc <= datetime.fromisoformat(ui["end_utc"]), "actual UI/effect time interval")
            need("Dispatch accepted; effect attribution unavailable." in ui["ax"], "actual ambiguous public acknowledgement")
            if position < 3:
                old = data("primary/receipts/"+c["case_id"]+".stdout.log")
                need(old["cases"] == states and old["effects"] == effects and old["metadata"] == metadata, "complete historical SQL prefix")
    need(s["cases"] == states and s["effects"] == effects, "complete stopped state")
    for row in s["cases"]:
        for field in ("root_count", "actor_count", "event_count", "notifications", "mask_notifications"):
            need(type(row[field]) is int, "integer state types")
    public = []
    for row in states:
        covered = ["target_value", "protected_sibling", "notifications", "release_status"]
        p = dict(metadata, case_id=row["case_id"], read_complete=True, covered_fields=covered,
                 **{k: row[k] for k in covered})
        if row["mask_notifications"]:
            p.pop("notifications")
            p["covered_fields"] = [k for k in covered if k != "notifications"]
            p["read_complete"] = False
        public.append(p)
    need(s["public_projection"] == public, "complete public projection")
    # Actual child receipts bind source, output and time, including the failed actor.
    labels = ["initial", "C01", "C02", "C03", "actor_C02", "actor_C04", "STOP_READBACK"]
    for label in labels:
        r = data("primary/receipts/"+label+".receipt.json")
        for stream in ("stdout", "stderr"):
            need(r[stream+"_sha256"] == hashlib.sha256((root/"primary"/"receipts"/(label+"."+stream+".log")).read_bytes()).hexdigest(), "actual receipt stream binding")
        need(type(r["child_pid"]) is int and type(r["exit_code"]) is int, "actual child receipt types")
        need(r["exit_code"] == (1 if label == "actor_C04" else 0), "actual exit semantics")
        need(datetime.fromisoformat(r["utc"]) <= datetime.fromisoformat(r["end_utc"]), "actual child UTC")
    failed = (root/"primary"/"receipts"/"actor_C04.stderr.log").read_text(errors="replace")
    need("ConnectionRefusedError" in failed and "10061" in failed, "pre-transmission connection refusal")
    logs = [json.loads(line) for line in (root/"primary-receipts"/"server.stdout.log").read_text().splitlines() if line]
    writes = [r["path"] for r in logs if r.get("method") == "POST"]
    need(writes == ["/dispatch/C01", "/actor/C02", "/dispatch/C02", "/dispatch/C03", "/dispatch/C04"], "HTTP accepted-write prefix")
    need(not (root/"primary-receipts"/"server.receipt.json").exists() and not (root/"primary"/"SERVER_ENDED.json").exists(), "missing server completion remains missing")
    stop = data("STOP.json")
    need(stop["server_exit_code"] is None and stop["server_end_utc"] is None, "do not fabricate termination")
    need(stop["status"] == "STOP_PRIMARY_INFRASTRUCTURE_LOSS" and stop["eight_case_scientific_PASS"] is False, "retain scientific STOP")
    for rel in ("primary/receipts/classifier.receipt.json", "primary/receipts/audit_v1.receipt.json",
                "primary/receipts/after_classifier.stdout.log", "primary/candidate_input.json"):
        need(not (root/rel).exists(), "unexecuted primary stage remains absent")
    return dict(evidence_check="VERIFIED_SAVED_PREFIX_ONLY", scientific_status=stop["status"],
                planned_case_denominator=8, root_dom_actions=4, actor_requests_attempted=2,
                actor_effects_applied=1, actual_ledger_rows=5, c04_second_actor_applied=False,
                unexecuted_cases=ids[4:], original_candidate_invocations=0, original_full_auditor_invocations=0,
                server_end_utc=None, server_exit_code=None, termination_cause="UNKNOWN",
                missing_fields_not_repaired=True, primary_restarts_retries=0)


if __name__ == "__main__":
    root = Path(sys.argv[1]).resolve()
    report = verify(root)
    checks = []
    for cid in ("wrong_actor", "missing_effect", "bool_count", "protected_clobber", "wrong_UI", "fake_exit", "fake_PASS", "missing_case"):
        overrides = {}
        if cid in {"wrong_actor", "missing_effect", "bool_count", "protected_clobber", "missing_case"}:
            rel = "primary/receipts/STOP_READBACK.stdout.log"
            obj = read(root/rel)
            if cid == "wrong_actor":
                obj["effects"][1]["actor"] = "ROOT_DOM"
            elif cid == "missing_effect":
                obj["effects"].pop()
            elif cid == "bool_count":
                obj["cases"][0]["notifications"] = False
            elif cid == "protected_clobber":
                obj["cases"][0]["protected_sibling"] = "clobbered"
            else:
                obj["cases"].pop()
        elif cid == "wrong_UI":
            rel = "primary/ui/C01.json"
            obj = read(root/rel)
            obj["labels"]["Study session"] = "wrong"
        else:
            rel = "STOP.json"
            obj = read(root/rel)
            if cid == "fake_exit":
                obj["server_exit_code"] = 0
            else:
                obj["eight_case_scientific_PASS"] = True
        overrides[rel] = obj
        try:
            verify(root, overrides)
        except ValueError as err:
            checks.append(dict(id=cid, detected=True, reason=str(err)))
        else:
            raise ValueError("ineffective copied control: "+cid)
    print(json.dumps(dict(report=report, copied_controls=checks,
                         source_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest()), indent=2))
