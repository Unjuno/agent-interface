from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import shutil
import subprocess

WS = Path(__file__).resolve().parent.parent
ROOT = WS/"work"/"state-rebase-browser-6259"
DEST = WS/"work"/"agent-interface"/"research"/"integration"/"state_rebase_browser_6259_01a0ff58"
PUBLIC = WS/"outputs"/"state-rebase-6259"


def sha(data):
    return hashlib.sha256(data).hexdigest()


if __name__ == "__main__":
    if DEST.exists():
        raise RuntimeError("archive namespace already populated; preserve and inspect")
    DEST.mkdir(parents=True)
    prefixes = [(str(WS), "<WORKSPACE>"), (str(Path.home()), "<PROFILE>")]
    custody = []
    original_inventory = []
    for source in sorted(ROOT.rglob("*")):
        if not source.is_file() or "__pycache__" in source.parts:
            continue
        rel = source.relative_to(ROOT).as_posix()
        original = source.read_bytes()
        original_inventory.append(dict(path=rel, bytes=len(original), sha256=sha(original)))
        target_rel = rel+(".txt" if source.suffix in (".py", ".log") else "")
        data = original
        if source.suffix != ".sqlite":
            text = data.decode("utf-8")
            for prefix, replacement in prefixes:
                for spelling in (json.dumps(prefix)[1:-1], prefix, prefix.replace("\\", "/")):
                    text = text.replace(spelling, replacement)
            data = text.encode("utf-8")
        target = DEST/target_rel
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(data)
        custody.append(dict(original_relative_path=rel, private_original_sha256=sha(original),
                            public_relative_path=target_rel, public_sha256=sha(data),
                            byte_identity=original == data,
                            transformation="none" if original == data else "workspace/profile prefixes only; original retained"))
    (ROOT.parent/"rebase-stop-original-inventory.json").write_text(json.dumps(original_inventory, indent=2))
    (DEST/"CUSTODY.json").write_text(json.dumps(dict(utc=datetime.now(timezone.utc).isoformat(),
        private_root="<PRIVATE_STUDY_ROOT>", path_maps=custody,
        source_extensions=".py.txt inert; original .py names retained in freeze/custody",
        receipt_hash_domain="original private streams; public derivatives have their own SHA256SUMS"), indent=2))
    (DEST/".gitattributes").write_text("* -text\n")
    (DEST/"README.md").write_text((ROOT/"REPORT.md").read_text())
    files = sorted(p for p in DEST.rglob("*") if p.is_file())
    total = sum(p.stat().st_size for p in files)
    if total > 16000000:
        raise RuntimeError("archive output bound exceeded")
    lines = [sha(p.read_bytes())+"  "+p.relative_to(DEST).as_posix() for p in files]
    (DEST/"SHA256SUMS").write_text("\n".join(lines)+"\n")
    for p in files:
        if p.suffix == ".py" or p.name in ("AGENTS.md", "__init__.py"):
            raise RuntimeError("unsafe source/discovery path")
    for line in (DEST/"SHA256SUMS").read_text().splitlines():
        digest, rel = line.split("  ", 1)
        if sha((DEST/rel).read_bytes()) != digest:
            raise RuntimeError("archive readback failed: "+rel)
    subprocess.run(["git", "-C", str(WS/"work"/"agent-interface"), "add", "--force", "--",
                    "research/integration/state_rebase_browser_6259_01a0ff58"], check=True)
    PUBLIC.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(DEST/"README.md", PUBLIC/"REPORT.md")
    shutil.copyfile(DEST/"reconciliation"/"partial_audit_v1.stdout.log.txt", PUBLIC/"SAVED_PREFIX_AUDIT.json")
    print(json.dumps(dict(archive_files=len(files)+1, manifest_entries=len(files), bytes=total,
                         custody_source_files=len(custody), byte_identical=sum(c["byte_identity"] for c in custody),
                         prefix_derivatives=sum(not c["byte_identity"] for c in custody),
                         manifest_sha256=sha((DEST/"SHA256SUMS").read_bytes()),
                         scientific_status="STOP_PRIMARY_INFRASTRUCTURE_LOSS", main_sent=False)))
