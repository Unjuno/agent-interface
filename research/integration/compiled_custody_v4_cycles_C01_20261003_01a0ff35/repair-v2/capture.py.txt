from pathlib import Path
import datetime,hashlib,json,os,subprocess,sys
HERE=Path(__file__).resolve().parent
stage,helper,mode=sys.argv[1:]
if helper not in {'test_schedule.py','auditor_v2.py'} or mode not in {'test','audit','optimized-test'}:raise ValueError('saved-only helpers')
out=HERE/stage;capture=HERE/'receipts'/stage
if out.exists():raise ValueError('fresh output required')
capture.mkdir(parents=True,exist_ok=False)
sha=lambda b:hashlib.sha256(b).hexdigest()
utc=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
def write(name,obj):(capture/name).write_text(json.dumps(obj,sort_keys=True,indent=2)+'\n',encoding='utf8')
argv=[sys.executable,'-B']+(['-O'] if mode=='optimized-test' else [])+[str(HERE/helper)]+(['audit'] if mode=='audit' else [])+[str(out)]
start=dict(argv=argv,cwd=str(HERE),started_utc=utc(),parent_pid=os.getpid(),helper_sha256=sha((HERE/helper).read_bytes()),runner_sha256=sha(Path(__file__).read_bytes()),
    plan_sha256=sha((HERE/'PLAN.md').read_bytes()),interpreter_sha256=sha(Path(sys.executable).read_bytes()),
    auditor_v2_sha256=sha((HERE/'auditor_v2.py').read_bytes()),frozen_original_auditor_sha256=sha((HERE.parent/'auditor.py').read_bytes()),
    raw_sha256=sha((HERE.parent/'measure-01/raw.jsonl').read_bytes()),measurement_replays=0,formal_producer_replays=0)
write('start.json',start)
p=subprocess.Popen(argv,cwd=HERE,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
write('child.json',dict(child_pid=p.pid,observed_utc=utc()))
stdout,stderr=p.communicate(timeout=45)
for name,data in (('stdout',stdout),('stderr',stderr)):(capture/name).write_bytes(data)
result=dict(child_pid=p.pid,ended_utc=utc(),exit_code=p.returncode,terminal=True,stdout_bytes=len(stdout),stderr_bytes=len(stderr),stdout_sha256=sha(stdout),stderr_sha256=sha(stderr),start_sha256=sha((capture/'start.json').read_bytes()))
write('terminal.json',result);print(json.dumps(result,sort_keys=True));print(stdout.decode('utf8','replace'))
raise SystemExit(p.returncode)
