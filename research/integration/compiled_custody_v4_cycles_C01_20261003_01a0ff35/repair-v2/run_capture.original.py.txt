from pathlib import Path
import datetime
import hashlib
import json
import os
import subprocess
import sys

HERE=Path(__file__).resolve().parent
stage,helper,mode=sys.argv[1:]
if not stage.replace('-','').isalnum() or helper not in {'collector.py','auditor.py'} or mode not in {'measure','audit'}:
    raise RuntimeError('fixed own unit stages only')
output=HERE/stage
if output.exists():raise RuntimeError('previous output; do not resend')
capture=HERE/'receipts'/stage;capture.mkdir(parents=True,exist_ok=False)
sha=lambda raw:hashlib.sha256(raw).hexdigest()
utc=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
def write(path,obj):
    with path.open('x',encoding='utf-8',newline='\n') as f:json.dump(obj,f,sort_keys=True,indent=2);f.write('\n')
argv=[sys.executable,'-B',str(HERE/helper),mode,str(output)]
start=dict(argv=argv,cwd=str(HERE),parent_pid=os.getpid(),started_utc=utc(),helper_sha256=sha((HERE/helper).read_bytes()),
           runner_sha256=sha(Path(__file__).read_bytes()),freeze_sha256=sha((HERE/'FREEZE.json').read_bytes()),
           interpreter_sha256=sha(Path(sys.executable).read_bytes()),mode=mode,stage=stage)
write(capture/'start.json',start)
if mode=='measure':write(HERE/'MEASUREMENT_SEND.json',dict(unit_id='compiled-custody-v4-cycles-C01-2ef3-20261003',**start))
p=subprocess.Popen(argv,cwd=HERE,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
write(capture/'child.json',dict(child_pid=p.pid,observed_utc=utc(),argv=argv))
timed_out=False
try:stdout,stderr=p.communicate(timeout=45)
except subprocess.TimeoutExpired:
    timed_out=True;p.kill();stdout,stderr=p.communicate(timeout=5)
ended=utc()
for name,raw in (('stdout',stdout),('stderr',stderr)):
    with (capture/name).open('xb') as f:f.write(raw)
result=dict(stage=stage,ended_utc=ended,child_pid=p.pid,exit_code=p.returncode,terminal=True,timed_out=timed_out,
            stdout_bytes=len(stdout),stderr_bytes=len(stderr),stdout_sha256=sha(stdout),stderr_sha256=sha(stderr),
            helper_sha256=sha((HERE/helper).read_bytes()),start_sha256=sha((capture/'start.json').read_bytes()))
write(capture/'terminal.json',result)
print(json.dumps(result,sort_keys=True));print(stdout.decode('utf-8','replace'));print(stderr.decode('utf-8','replace'))
raise SystemExit(p.returncode if not timed_out else 124)
