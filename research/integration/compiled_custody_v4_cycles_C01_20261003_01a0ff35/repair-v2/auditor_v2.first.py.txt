"""Versioned saved-data gate; original C01 files and measurements stay intact."""
import ast
import hashlib
import json
from pathlib import Path
import sys

ROOT=Path(__file__).resolve().parent.parent
canon=lambda v:json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=True).encode()+b'\n'
sha=lambda b:hashlib.sha256(b).hexdigest()

def original_path():
    p=ROOT/'auditor.py.txt'
    return p if p.exists() else ROOT/'auditor.py'

def load_original():
    raw=original_path().read_bytes()
    if sha(raw)!='a12e582286cb470ecdc9c93a00b13f7ef420e2df4c2620679cf8cc1bd4e3f942':
        raise ValueError('immutable original auditor bytes')
    wanted={'require','equal','verify_method'}
    nodes=[n for n in ast.parse(raw).body if isinstance(n,ast.FunctionDef) and n.name in wanted]
    if {n.name for n in nodes}!=wanted:raise ValueError('literal selected original definitions')
    ns=dict(canon=canon,sha=sha)
    exec(compile(ast.Module(body=nodes,type_ignores=[]),'quoted-original-auditor-functions','exec'),ns)
    return ns['verify_method']

ORIGINAL_VERIFY=load_original()

def verify_method(method,spec):
    return ORIGINAL_VERIFY(method,spec)

def run_saved(mode,out):
    # Execute only the unchanged saved-data auditor, with its byte-exact file pin.
    # No collector/runtime/native cycle API is imported or invoked.
    path=original_path();nodes=ast.parse(path.read_bytes()).body
    index=next(i for i,n in enumerate(nodes) if isinstance(n,ast.If) and isinstance(n.test,ast.Compare)
               and isinstance(n.test.left,ast.Name) and n.test.left.id=='MODE')
    ns={'__file__':str(path),'__name__':'saved_C01_auditor_v1_with_v2_gate'}
    exec(compile(ast.Module(body=nodes[:index],type_ignores=[]),str(path),'exec'),ns)
    ns['MODE']=mode;ns['OUT']=Path(out);ns['verify_method']=verify_method
    exec(compile(ast.Module(body=nodes[index:],type_ignores=[]),str(path),'exec'),ns)

if __name__=='__main__':
    if len(sys.argv)!=3 or sys.argv[1]!='audit':raise ValueError('saved audit mode and fresh output only')
    run_saved(sys.argv[1],sys.argv[2])
