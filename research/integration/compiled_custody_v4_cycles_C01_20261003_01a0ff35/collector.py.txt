"""One bounded inert profile; complete per-method callback objects are retained."""
import ast
import ctypes
from ctypes import wintypes
import hashlib
import json
from pathlib import Path
import platform
import sys
import time

HERE = Path(__file__).resolve().parent
MODE, OUT = sys.argv[1], Path(sys.argv[2])
PINS = json.loads((HERE/'sources.json').read_bytes())
MAX_RAW = 96*1024*1024
canon = lambda obj: json.dumps(obj,sort_keys=True,separators=(',',':'),ensure_ascii=True).encode()+b'\n'
sha = lambda data: hashlib.sha256(data).hexdigest()
def require(value,message):
    if not value:
        raise RuntimeError(message)

def graph(predicates, actions):
    names = ['phase']+['p'+str(i) for i in range(1,predicates)]
    value = lambda phase: dict(phase=phase,**{name:0 for name in names[1:]})
    states = {}
    for i in range(actions+1):
        states['s'+str(i)] = {'branches':[dict(when=value(i),outcome='action' if i<actions else 'complete',
            action='a'+str(i) if i<actions else None,next_state='s'+str(i+1) if i<actions else None,reason=None)]}
    return {'format':'compiled-gui-interface-v1','interface_id':'custody-cycle-C01','session_scope':'private-inert',
            'surface':'form','predicates':names,'symbols':{'field':{'kind':'target_reference','target_reference':'field',
            'identity_predicate':'phase','dependencies':names.copy()}},
            'actions':{'a'+str(i):{'target_symbol':'field','operation':'step'+str(i),'expected_effect':value(i+1)} for i in range(actions)},
            'method':{'name':'healthy-'+str(predicates)+'-'+str(actions),'version':'1','initial_state':'s0',
            'max_transitions':actions,'max_runtime_ms':10,'states':states}}

INPUTS = {'p1-a2':graph(1,2),'p32-a2':graph(32,2),'p32-a8':graph(32,8)}
functions = {}
for arm in ('baseline','candidate'):
    raw = (HERE/PINS[arm]['file']).read_bytes()
    require(sha(raw) == PINS[arm]['sha256'], 'source pin '+arm)
    module = ast.parse(raw)
    require(all(isinstance(n,(ast.Expr,ast.Import,ast.FunctionDef,ast.ClassDef,ast.Assign)) for n in module.body),'top-level source scope')
    require([n.names[0].name for n in module.body if isinstance(n,ast.Import)] == ['copy','time'],'pure module imports')
    scope = {'__name__':'inert_profile_'+arm}
    exec(compile(raw,'pinned-'+arm,'exec'),scope)
    functions[arm] = scope['run']

class Driver:
    def __init__(self, spec):
        self.spec=spec;self.phase=0;self.sequence=0;self.trace=[]
    def record(self,kind,request,returned):
        self.trace.append({'kind':kind,'request':request,'returned':returned})
        return returned
    def clock(self):
        return self.record('clock',None,0)
    def cancelled(self):
        return self.record('cancelled',None,False)
    def observe(self,payload):
        self.sequence+=1
        predicates={name:self.phase if name=='phase' else 0 for name in self.spec['predicates']}
        return self.record('observe',payload,dict(sequence=self.sequence,captured_ns=0,surface='form',predicates=predicates,
                           evidence_ref='frame-'+str(self.sequence),evidence_digest='digest-'+str(self.sequence)))
    def admit(self,payload):
        return self.record('admit',payload,dict(eligible=True,status='revalidated',authorization='auth-'+str(self.sequence),
                           expected_sequence=self.sequence,valid_until_ns=100_000_000))
    def execute(self,payload):
        self.phase+=1
        return self.record('execute',payload,dict(status='completed',action_id='action-'+str(self.phase),effect_ref='effect-'+str(self.phase),
                           release=dict(verified=True,keys_down=[],buttons_down=[])))
    def verify(self,payload):
        return self.record('verify_effect',payload,dict(status='succeeded',evidence_ref=payload['observation']['evidence_ref']))
    def journal(self,payload):
        return self.record('journal',payload,None)

def invoke(arm,case):
    spec=INPUTS[case];driver=Driver(spec)
    result=functions[arm](spec,dict(observe=driver.observe,admit=driver.admit,execute=driver.execute,
                            verify_effect=driver.verify,cancelled=driver.cancelled,journal=driver.journal),clock=driver.clock)
    return dict(input=spec,input_sha256=sha(canon(spec)),receipt=result,trace=driver.trace)

kernel = ctypes.WinDLL('kernel32',use_last_error=True)
kernel.GetCurrentThread.argtypes=[];kernel.GetCurrentThread.restype=wintypes.HANDLE
kernel.QueryThreadCycleTime.argtypes=[wintypes.HANDLE,ctypes.POINTER(ctypes.c_ulonglong)]
kernel.QueryThreadCycleTime.restype=wintypes.BOOL
kernel.GetCurrentProcessorNumber.argtypes=[];kernel.GetCurrentProcessorNumber.restype=wintypes.DWORD
handle=kernel.GetCurrentThread()
def cycles():
    value=ctypes.c_ulonglong()
    if not kernel.QueryThreadCycleTime(handle,ctypes.byref(value)):
        raise ctypes.WinError(ctypes.get_last_error())
    return value.value

if MODE=='preflight':
    OUT.mkdir(exist_ok=False)
    rows=[dict(arm=arm,case=case,method=invoke(arm,case)) for case in INPUTS for arm in functions]
    counter_before=cycles();counter_after=cycles()
    with (OUT/'preflight.json').open('xb') as f:
        f.write(canon(dict(rows=rows,counter_before=counter_before,counter_after=counter_after,inputs=INPUTS,
                          source_pins=PINS,measurement_started=False)))
    for row in rows:
        expected=row['method']['input']['method']['max_transitions']
        require(row['method']['receipt']['outcome']=='TASK_SUCCEEDED' and row['method']['receipt']['completed_transitions']==expected,'healthy construction')
    print(json.dumps(dict(preflight_methods=len(rows),measurement_started=False,counter_api_succeeded=True)))
elif MODE=='measure':
    freeze=json.loads((HERE/'FREEZE.json').read_bytes())
    require(freeze['collector_sha256']==sha(Path(__file__).read_bytes()),'frozen collector')
    require(freeze['source_pins']==PINS and freeze['inputs']==INPUTS,'frozen sources/inputs')
    require(freeze['interpreter_sha256']==sha(Path(sys.executable).read_bytes()),'frozen interpreter')
    OUT.mkdir(exist_ok=False)
    with (OUT/'raw.jsonl').open('xb') as sink:
        def write(row):
            raw=canon(row)
            if sink.tell()+len(raw)>MAX_RAW:
                raise RuntimeError('prospective raw cap exceeded; no retry')
            sink.write(raw)
        write(dict(kind='header',freeze=freeze,freeze_sha256=sha((HERE/'FREEZE.json').read_bytes()),
                   python=sys.version,platform=platform.platform(),counter_units='CPU cycle counter, not seconds',
                   frequency_power_thermal_during_run_workload='unmeasured',clock_info=str(time.get_clock_info('perf_counter'))))
        for i in range(64):
            c0=cycles();t0=time.perf_counter_ns();t1=time.perf_counter_ns();c1=cycles()
            write(dict(kind='calibration',index=i,cycle_start=c0,cycle_end=c1,wall_start_ns=t0,wall_end_ns=t1))
        for case in INPUTS:
            for arm in functions:
                write(dict(kind='warmup',case=case,arm=arm,methods=[invoke(arm,case) for _ in range(8)]))
        cases=list(INPUTS)
        for pair in range(31):
            case_order=cases[pair%3:]+cases[:pair%3]
            for case in case_order:
                arms=['baseline','candidate'] if pair%2==0 else ['candidate','baseline']
                for arm in arms:
                    processor_before=kernel.GetCurrentProcessorNumber()
                    c0=cycles();t0=time.perf_counter_ns()
                    methods=[invoke(arm,case) for _ in range(8)]
                    t1=time.perf_counter_ns();c1=cycles()
                    processor_after=kernel.GetCurrentProcessorNumber()
                    write(dict(kind='batch',pair=pair,case=case,arm=arm,methods=methods,
                               cycle_start=c0,cycle_end=c1,wall_start_ns=t0,wall_end_ns=t1,
                               processor_before=processor_before,processor_after=processor_after))
        write(dict(kind='terminal',measured_methods=1488,warmup_methods=48,paired_case_rounds=93,status='collection_complete'))
    print(json.dumps(dict(raw_bytes=(OUT/'raw.jsonl').stat().st_size,raw_sha256=sha((OUT/'raw.jsonl').read_bytes()),
                          measured_methods=1488,warmup_methods=48,measurement_retries=0)))
else:
    raise RuntimeError('unsupported collector mode')
