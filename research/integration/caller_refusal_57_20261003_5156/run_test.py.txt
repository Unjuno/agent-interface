import datetime,json,subprocess,sys
from pathlib import Path
root=Path(__file__).resolve().parent;label=sys.argv[1]
if (root/(label+'-PROCESS.json')).exists():raise RuntimeError('same first label already exists')
stamp=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
args=[sys.executable,'-B']+(['-O'] if label=='green-optimized' else [])+[str(root/'suite.py'),label]
proc=subprocess.Popen(args,stdout=subprocess.PIPE,stderr=subprocess.PIPE);record={'label':label,'pid':proc.pid,'started_utc':stamp(),'args':args}
(root/(label+'-PROCESS.json')).write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8')
timeout=False
try:out,err=proc.communicate(timeout=20)
except subprocess.TimeoutExpired:timeout=True;proc.kill();out,err=proc.communicate(timeout=5)
(root/(label+'-stdout.bin')).write_bytes(out);(root/(label+'-stderr.bin')).write_bytes(err)
record.update(ended_utc=stamp(),exit_code=proc.returncode,timeout=timeout,stdout_bytes=len(out),stderr_bytes=len(err),stream_caps_pass=len(out)<=65536 and len(err)<=65536)
(root/(label+'-PROCESS.json')).write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8')
print(json.dumps(record));print(out.decode('utf-8','replace'));print(err.decode('utf-8','replace'))
