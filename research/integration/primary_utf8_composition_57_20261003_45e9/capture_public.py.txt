"""Freeze three new public CLI controls and exact maintained regression selection."""
from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,platform,shutil,subprocess,sys,uuid
root=Path(__file__).resolve().parent;out=root/'public-first';out.mkdir(exist_ok=False)
node=Path(shutil.which('node'))
def sha(b):return hashlib.sha256(b).hexdigest()
def put(p,v):p.write_text(json.dumps(v,indent=2,sort_keys=True,ensure_ascii=False)+'\n',encoding='utf-8',newline='\n')
def command(i,label):return b'{"id":'+str(i).encode()+b',"method":"call","args":["interface_clock",{"label":"'+label+b'"}]}\n'
valid=command(1,'あ😀'.encode());cut=valid.index(b'\xe3')+1
specs=[('split-valid-japanese-emoji',[valid[:cut],valid[cut:]],[],0,1),
 ('malformed-before-admission',[command(1,b'\xff')],[],2,0),
 ('accepted-before-byte-stop',[command(1,b'original-held')],[command(2,b'\xff')],2,1)]
cells=[]
for name,initial,later,code,count in specs:
 d=out/name;d.mkdir(exist_ok=False);nonce=str(uuid.uuid4())
 cfg={'host':{'command':str(node),'args':[str(root/'fixture.mjs'),str(d),nonce,'yes' if later else 'no'],'evidenceDirectory':str(d/'host')},
  'route':'guarded-local','exchangeDirectory':str(d/'exchange')}
 put(d/'config.json',cfg)
 for stem,chunks in [('initial',initial),('later',later)]:
  (d/(stem+'.original.bin')).write_bytes(b''.join(chunks));(d/(stem+'.hex.txt')).write_text(b''.join(chunks).hex(),encoding='ascii')
 cells.append({'name':name,'nonce':nonce,'initial_hex':[x.hex() for x in initial],'later_hex':[x.hex() for x in later],
  'expected_cli_code':code,'expected_fixture_requests':count,'config_sha256':sha((d/'config.json').read_bytes())})
put(root/'PUBLIC_PLAN.json',{'H':'strict byte admission must prevent a malformed raw command while preserving valid multibyte and the same already accepted original request',
 'T':'three new ordinary public --config controls;split Japanese+emoji,malformed first request,one fixture-held accepted request then malformed next bytes;all initial/later bytes and config frozen',
 'D':'CLI0/2/2;fixture counts1/0/1 and all original host/fixture exits0;no repeated request/guard/owned PID;original-held success retained once before byte fault;first outcomes retained',
 'C':'writer chunks need not map to OS read chunks;explicit fixture-request endpoint precedes malformed writer and owned reply release, but exact internal validation/reply scheduling is not directly timestamped',
 'U':'own inert echo children/controlled streams,zero SDK/backend/native/input/model/task/performance/formal producer invocation;no hard deadline/line cap added',
 'cells':cells})
jobs=[('maintained-stdio',[str(node),'--test','--test-concurrency=1','--test-reporter=tap','runtime/host_v1/test_primary_stdio.mjs','runtime/host_v1/test_primary_stdio_utf8.mjs','runtime/host_v1/test_relay_client.mjs'],root/'source'),
 ('public-cli',[str(node),str(root/'collect_public.mjs')],root)]
source_paths=sorted((x for x in (root/'source').rglob('*') if x.is_file()),key=lambda p:p.relative_to(root/'source').as_posix())
put(root/'PUBLIC_FREEZE.json',{'frozen_utc':datetime.now(timezone.utc).isoformat(),'source':[{'path':p.relative_to(root/'source').as_posix(),'sha256':sha(p.read_bytes()),'bytes':p.stat().st_size} for p in source_paths],
 'public_plan_sha256':sha((root/'PUBLIC_PLAN.json').read_bytes()),'fixture_sha256':sha((root/'fixture.mjs').read_bytes()),
 'collector_sha256':sha((root/'collect_public.mjs').read_bytes()),'capture_sha256':sha(Path(__file__).read_bytes()),
 'auditor_sha256':sha((root/'audit_public.py').read_bytes()),'node_version':subprocess.check_output([str(node),'--version']).decode().strip(),
 'node_binary_sha256':sha(node.read_bytes()),'python_version':sys.version,'platform':platform.platform(),
 'jobs':[{'name':n,'argv':a,'cwd':str(c)} for n,a,c in jobs]})
for name,argv,cwd in jobs:
 start=datetime.now(timezone.utc).isoformat();r=subprocess.run(argv,cwd=cwd,capture_output=True,timeout=30)
 (out/(name+'.stdout.txt')).write_bytes(r.stdout);(out/(name+'.stderr.txt')).write_bytes(r.stderr)
 put(out/(name+'.receipt.json'),{'argv':argv,'cwd':str(cwd),'started_utc':start,'ended_utc':datetime.now(timezone.utc).isoformat(),'exit_code':r.returncode,
  'stdout_bytes':len(r.stdout),'stderr_bytes':len(r.stderr),'stdout_sha256':sha(r.stdout),'stderr_sha256':sha(r.stderr)})
 assert r.returncode==0,(name,r.returncode,r.stderr)
 print(json.dumps({'name':name,'exit_code':r.returncode,'summary':[x for x in r.stdout.decode('utf-8',errors='replace').splitlines() if x.startswith('# tests') or x.startswith('# pass') or x.startswith('# fail') or x.startswith('{')][-3:]}))
for p in source_paths:assert sha(p.read_bytes())==next(x['sha256'] for x in json.loads((root/'PUBLIC_FREEZE.json').read_bytes())['source'] if x['path']==p.relative_to(root/'source').as_posix())
