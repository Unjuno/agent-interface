"""Keep the first composition command's complete streams, UTC, argv and typed endpoint."""
from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,subprocess,sys
root=Path(__file__).resolve().parent
name=sys.argv[1]
assert name in ['capture','audit']
folder=root/(name+'-outer-first');folder.mkdir(exist_ok=False)
command=[sys.executable,'-X','utf8','-B',str(root/('capture_public.py' if name=='capture' else 'audit_composition.py'))]
if name=='capture':
 source={'utc':datetime.now(timezone.utc).isoformat(),'auditor_sha256':hashlib.sha256((root/'audit_composition.py').read_bytes()).hexdigest(),
 'wrapper_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
 'context_sha256':hashlib.sha256((root/'COMPOSITION_CONTEXT.json').read_bytes()).hexdigest()}
 (root/'COMPOSITION_AUDITOR_FREEZE.json').write_text(json.dumps(source,indent=2)+'\n',encoding='utf-8',newline='\n')
started=datetime.now(timezone.utc).isoformat()
r=subprocess.run(command,cwd=root,capture_output=True,timeout=50)
(folder/'stdout.txt').write_bytes(r.stdout);(folder/'stderr.txt').write_bytes(r.stderr)
receipt={'argv':command,'cwd':str(root),'started_utc':started,'ended_utc':datetime.now(timezone.utc).isoformat(),'exit_code':r.returncode,
 'stdout_sha256':hashlib.sha256(r.stdout).hexdigest(),'stderr_sha256':hashlib.sha256(r.stderr).hexdigest(),
 'stdout_bytes':len(r.stdout),'stderr_bytes':len(r.stderr)}
(folder/'RECEIPT.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8',newline='\n')
print(json.dumps(receipt));print(r.stdout.decode('utf-8'),end='');print(r.stderr.decode('utf-8'),end='')
sys.exit(r.returncode)
