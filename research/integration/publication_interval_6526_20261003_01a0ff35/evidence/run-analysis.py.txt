"""Private bounded command receipt and preservation for the frozen analysis."""
import datetime
import hashlib
import json
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent
PKG = ROOT / "agent-interface/research/integration/publication_interval_6526_20261003_01a0ff35"
WORK = ROOT / "interval-6526/export-v3"
EVIDENCE = PKG / "evidence"

def sha(data):
    return hashlib.sha256(data).hexdigest()

def main():
    freeze = json.loads((PKG / "FREEZE.json").read_bytes())
    for name, digest in freeze["pre_result_files"].items():
        if sha((PKG / name).read_bytes()) != digest:
            raise RuntimeError("changed-pre-result-code:" + name)
    manifest = PKG / "INPUTS.json"
    inputs = WORK / "inputs"
    commands = [
        ("retained-analysis", [sys.executable, "-B", str(PKG / "intervals.py"), str(inputs), str(manifest),
                               str(PKG / "FREEZE.json"), str(EVIDENCE / "interval-result.json")]),
        ("retained-controls", [sys.executable, "-B", str(PKG / "controls.py"), "--inputs", str(inputs),
                               "--manifest", str(manifest), "--freeze", str(PKG / "FREEZE.json")]),
        ("optimized-controls", [sys.executable, "-O", "-B", str(PKG / "controls.py"), "--inputs", str(inputs),
                                "--manifest", str(manifest), "--freeze", str(PKG / "FREEZE.json")]),
        ("independent-check", [sys.executable, "-B", str(PKG / "independent_check.py"), str(inputs), str(manifest),
                               str(PKG / "FREEZE.json"), str(EVIDENCE / "interval-result.json")]),
    ]
    receipts = []
    receipt_path = EVIDENCE / "commands.json"
    if receipt_path.exists():
        raise RuntimeError("command-sequence-already-started")
    receipt_path.write_text("[]\n", encoding="utf-8")
    for name, argv in commands:
        start = datetime.datetime.now(datetime.timezone.utc).isoformat()
        proc = subprocess.run(argv, cwd=PKG, capture_output=True, timeout=60)
        end = datetime.datetime.now(datetime.timezone.utc).isoformat()
        stdout_name, stderr_name = name + ".stdout.txt", name + ".stderr.txt"
        (EVIDENCE / stdout_name).write_bytes(proc.stdout)
        (EVIDENCE / stderr_name).write_bytes(proc.stderr)
        receipts.append({"name": name, "argv": argv, "cwd": str(PKG), "started_utc": start,
                         "ended_utc": end, "exit_code": proc.returncode,
                         "stdout": stdout_name, "stdout_sha256": sha(proc.stdout),
                         "stderr": stderr_name, "stderr_sha256": sha(proc.stderr),
                         "freeze_sha256": sha((PKG / "FREEZE.json").read_bytes())})
        receipt_path.write_text(json.dumps(receipts, sort_keys=True, indent=2) + "\n", encoding="utf-8")
        print(json.dumps({"command": name, "exit": proc.returncode, "stdout": proc.stdout.decode().strip()[:2000]}), flush=True)
        if proc.returncode != 0:
            raise RuntimeError("construction-command-failed:" + name)

if __name__ == "__main__":
    main()
