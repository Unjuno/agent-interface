import datetime
import hashlib
import json
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent
PKG = ROOT / "agent-interface/research/integration/publication_interval_6526_20261003_01a0ff35"
INPUTS = ROOT / "interval-6526/export-v3/inputs"
E = PKG / "evidence"
def sha(data):
    return hashlib.sha256(data).hexdigest()

def main():
    freeze = json.loads((PKG / "FREEZE.json").read_bytes())
    if any(sha((PKG / n).read_bytes()) != h for n, h in freeze["pre_result_files"].items()):
        raise RuntimeError("changed-original-plan-or-code")
    result_digest = sha((E / "interval-result.json").read_bytes())
    commands = []
    for flags, suffix in ((["-B"], "normal"), (["-O", "-B"], "optimized")):
        commands.append(("independent-v2-" + suffix, [sys.executable, *flags, str(PKG / "independent_check_v2.py"),
             str(INPUTS), str(PKG / "INPUTS.json"), str(PKG / "FREEZE.json"), str(E / "interval-result.json")]))
        commands.append(("custody-v2-" + suffix, [sys.executable, *flags, str(PKG / "custody_controls.py"),
             str(INPUTS), str(PKG / "INPUTS.json"), str(PKG / "FREEZE.json"), str(E / "interval-result.json"),
             str(PKG / "independent_check_v2.py")]))
    commands.append(("materialize-roundtrip", [sys.executable, "-B", str(PKG / "materialize_inputs.py"),
                                             str(ROOT / "agent-interface"), str(ROOT / "interval-6526/roundtrip-inputs")]))
    receipt_path = E / "v2-commands.json"
    if receipt_path.exists():
        raise RuntimeError("v2-commands-already-started")
    receipts = []
    receipt_path.write_text("[]\n", encoding="utf-8")
    for name, argv in commands:
        start = datetime.datetime.now(datetime.timezone.utc).isoformat()
        run = subprocess.run(argv, cwd=PKG, capture_output=True, timeout=60)
        (E / (name + ".stdout.txt")).write_bytes(run.stdout)
        (E / (name + ".stderr.txt")).write_bytes(run.stderr)
        receipts.append({"name": name, "argv": argv, "started_utc": start,
                         "ended_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
                         "exit_code": run.returncode, "stdout_sha256": sha(run.stdout),
                         "stderr_sha256": sha(run.stderr), "result_sha256": result_digest,
                         "checker_v2_sha256": sha((PKG / "independent_check_v2.py").read_bytes())})
        receipt_path.write_text(json.dumps(receipts, sort_keys=True, indent=2) + "\n", encoding="utf-8")
        if run.returncode != 0:
            raise RuntimeError(name + ":" + run.stderr.decode())
        if name.startswith("custody") and json.loads(run.stdout)["accepted_result_corruption_count"] != 0:
            raise RuntimeError("v2-accepted-corruption")
        if name == "materialize-roundtrip":
            for item in json.loads((PKG / "INPUTS.json").read_bytes())["files"]:
                roundtrip = ROOT / "interval-6526/roundtrip-inputs" / item["path"]
                if sha(roundtrip.read_bytes()) != item["sha256"]:
                    raise RuntimeError("roundtrip-bytes")
        print(json.dumps({"name": name, "exit": run.returncode, "stdout": run.stdout.decode().strip()}), flush=True)
    if sha((E / "interval-result.json").read_bytes()) != result_digest:
        raise RuntimeError("original-result-changed")

if __name__ == "__main__":
    main()
