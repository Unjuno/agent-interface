const fs=require('node:fs'),fsp=require('node:fs/promises'),path=require('node:path');
const {PassThrough}=require('node:stream'),{TextDecoder}=require('node:util');

 const owner=runPrimaryStdio(config,{input,output}).then(()=>{settled=true;event('owner_resolved');},error=>{settled=true;rejection=error;event('owner_rejected',{name:error.name,code:error.code??null,message:error.message,same_decoder:error===decoderError,same_stream_fault:error===streamFault,aggregate:error instanceof AggregateError});});

 }else input.write(wire);

 await waitFor('fixture-entered.json');event('fixture_entered');
 const malformed=Buffer.concat([Buffer.from('{"id":2,"method":"call","args":["interface_clock",{"label":"'),Buffer.from([255]),Buffer.from('"}]}\n')]);
 fs.writeFileSync(path.join(root,'second-input.raw'),which==='healthy_split'?Buffer.alloc(0):malformed,{flag:'wx'});
 if(which==='stream_then_malformed'){event('emit_input_error');input.emit('error',streamFault);input.write(malformed);event('malformed_written_after_stream_error');}
 else if(which!=='healthy_split'){
  input.write(malformed);event('malformed_write_returned',{decoder_observed:decoderError!==null});
  if(which==='malformed_then_stream'){event('emit_input_error');input.emit('error',streamFault);}
 }
 await new Promise(r=>setImmediate(r));const settledBeforeRelease=settled;event('before_release',{settled});
 fs.writeFileSync(path.join(root,'release.signal'),'release original fixture once\n',{flag:'wx'});event('release_original');
 input.end();await owner;TextDecoder.prototype.decode=originalDecode;
 await new Promise(r=>setImmediate(r));
 const fixtureStart=JSON.parse(fs.readFileSync(path.join(root,'fixture-start.json')));
 let fixtureAbsent=false;try{process.kill(fixtureStart.pid,0);}catch(error){if(error.code==='ESRCH')fixtureAbsent=true;else throw error;}
 const errorJSON=error=>error?{name:error.name,code:error.code??null,message:error.message,cause:error.cause?errorJSON(error.cause):null,errors:error instanceof AggregateError?Array.from(error.errors,errorJSON):null}:null;
 const requestFiles=fs.readdirSync(path.join(root,'exchange')).filter(x=>/^request-\d+\.json$/.test(x)).sort();
 const result={which,collector_pid:process.pid,fixture_start:fixtureStart,fixture_exit:JSON.parse(fs.readFileSync(path.join(root,'fixture-exit.json'))),fixture_absent:fixtureAbsent,host_exit:JSON.parse(fs.readFileSync(path.join(root,'host/exit.json'))),rows,events,settledBeforeRelease,requestFiles,decoderEvents,decoderError:errorJSON(decoderError),rejection:errorJSON(rejection),sameDecoder:rejection===decoderError&&decoderError!==null,sameStreamFault:rejection===streamFault,listeners:{input_errors:input.listenerCount('error'),output_errors:output.listenerCount('error'),input_data:input.listenerCount('data'),input_end:input.listenerCount('end')},foreign_input_listener_retained:input.listeners('error').includes(foreignInputError),guard:fs.existsSync(path.join(root,'fixture-guard.json'))};
 save('RESULT.json',result);save('collector-end.json',{pid:process.pid,utc:new Date().toISOString(),exit_intended:0});console.log(JSON.stringify(result));
})().catch(error=>{console.error(error.stack);process.exitCode=3;});
