import pathlib,hashlib,json,subprocess,datetime,os,sys
ROOT=pathlib.Path(__file__).resolve().parents[2];OUT=pathlib.Path(__file__).resolve().parent
WORK=ROOT/'work/primary-startup-utf8-union-01a0ff32'
F=json.loads((OUT/'FREEZE.json').read_text());NODE=F['runtime']['node']
def utc():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(b):return hashlib.sha256(b).hexdigest()
for name,row in F['pins'].items():
 data=(ROOT/name).read_bytes();assert len(data)==row['bytes'] and sha(data)==row['sha256'],name
assert sha(pathlib.Path(NODE).read_bytes())==F['runtime']['node_sha256']
CASE=OUT/'cases';CASE.mkdir()
records=[]
def capture(label,argv,cwd,timeout):
 directory=OUT/label;directory.mkdir()
 env=os.environ.copy()
 if label=='affected':env['PRIMARY_STDIO_STARTUP_EVIDENCE']=str(OUT/'affected-startup-endpoints');pathlib.Path(env['PRIMARY_STDIO_STARTUP_EVIDENCE']).mkdir()
 start=utc();p=subprocess.Popen(argv,cwd=cwd,env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
 timed_out=False
 try:so,se=p.communicate(timeout=timeout)
 except subprocess.TimeoutExpired:
  timed_out=True
  # Bound termination to this exact collector and its own descendants only.
  killer=subprocess.run(['taskkill','/PID',str(p.pid),'/T','/F'],stdout=subprocess.PIPE,stderr=subprocess.PIPE)
  (directory/'owned-timeout-kill.stdout').write_bytes(killer.stdout);(directory/'owned-timeout-kill.stderr').write_bytes(killer.stderr)
  so,se=p.communicate(timeout=5)
 end=utc()
 (directory/'stdout.raw').write_bytes(so);(directory/'stderr.raw').write_bytes(se)
 receipt={'label':label,'argv':argv,'cwd':str(cwd),'pid':p.pid,'started_utc':start,'ended_utc':end,'exit':p.returncode,'timeout':timed_out,'stdout_bytes':len(so),'stdout_sha256':sha(so),'stderr_bytes':len(se),'stderr_sha256':sha(se),'freeze_sha256':sha((OUT/'FREEZE.json').read_bytes())}
 (directory/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf8')
 records.append(receipt);(OUT/'COMMANDS.json').write_text(json.dumps(records,indent=2)+'\n',encoding='utf8')
 print(json.dumps({k:v for k,v in receipt.items() if k not in ('argv','cwd') }),flush=True)
 if len(so)+len(se)>F['allocation']['output_cap_B']:raise RuntimeError('output cap exceeded; no more cases')
capture('affected',[NODE,'--test','--test-concurrency=1',*F['T']['affected_modules']],WORK,30)
for case in F['T']['cases']:
 capture('case-'+case,[NODE,str(OUT/'probe.cjs'),str(CASE/case),case,str(WORK),str(OUT/'fixture.cjs')],WORK,12)
for name,row in F['pins'].items(): assert sha((ROOT/name).read_bytes())==row['sha256'],name
assert sha(pathlib.Path(NODE).read_bytes())==F['runtime']['node_sha256']
(OUT/'POST_RUN_PINS.json').write_text(json.dumps({'utc':utc(),'pins_checked':len(F['pins']),'all_unchanged':True,'node_unchanged':True,'commands':len(records)},indent=2)+'\n',encoding='utf8')
print(json.dumps({'state':'ORIGINAL_COMMANDS_CAPTURED','commands':len(records),'exits':[r['exit'] for r in records],'all_pins_unchanged':True}))
