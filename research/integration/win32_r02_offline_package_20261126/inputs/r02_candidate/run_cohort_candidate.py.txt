def fence_is_clear(state):
    return isinstance(state, dict) and state.get("status") == "clear" and "pending" in state and state["pending"] is None


def fence_matches_action(state, operation_id, generation):
    if not fence_is_clear(state) or not operation_id or not generation:
        return False
    last = state.get("last_resolved")
    return (
        isinstance(last, dict)
        and last.get("operation_id") == operation_id
        and last.get("target_generation") == generation
    )


def exact_action(result, call_id, generation):
    action = result.get("action")
    if (
        result.get("call_id") == call_id
        and isinstance(action, dict)
        and action.get("target_generation") == generation
        and isinstance(action.get("operation_id"), str) and action["operation_id"]
        and action.get("replay_allowed") is False
    ):
        return action
    return None


def verified_task_success(result, call_id, action):
    effect = result.get("task_effect")
    if not isinstance(effect, dict):
        return None
    receipt = effect.get("receipt")
    success = result.get("task_success")
    if (
        isinstance(receipt, dict) and type(success) is bool
        and result.get("call_id") == call_id and result.get("status") == "verified"
        and result.get("replay_allowed") is False and effect.get("status") == "verified"
        and effect.get("task_success") is success and receipt.get("call_id") == call_id
        and receipt.get("operation_id") == action["operation_id"]
        and receipt.get("target_generation") == action["target_generation"]
        and receipt.get("outcome") == ("success" if success else "failure")
    ):
        return success
    return None


def recovery_matches_action(result, action, generation, horizon_ns):
    receipt = result.get("recovery_receipt", {})
    return (
        result.get("status") == "recovered"
        and result.get("source_action_status") == "unknown"
        and result.get("source_operation_id") == action["operation_id"]
        and result.get("source_target_generation") == action["target_generation"]
        and result.get("recovery_target_generation") == generation
        and result.get("release_confirmed") is True
        and "task_success" in result and result["task_success"] is None
        and result.get("replay_allowed") is False
        and receipt.get("readback", {}).get("neutral") is True
        and receipt.get("late_input_horizon_ns") == horizon_ns
        and receipt.get("late_input_horizon_verified") is True
    )


async def poll_exact_action(call, call_id, *, timeout_s=25, interval_s=.25):
    deadline = time.monotonic() + timeout_s
    while True:
        result = await call("interface_results", {"call_id": call_id, "include_image": False})
        action = result.get("action")
        pending = result.get("status") == "pending" or (
            isinstance(action, dict) and action.get("status") in
            ("pending", "admitted", "starting", "running", "cancel_pending", "recovery_pending")
        )
        if not pending or time.monotonic() >= deadline:
            return result
        await asyncio.sleep(interval_s)


async def poll_recovery(call, result):
    call_id = result.get("call_id")
    deadline = time.monotonic() + 35
    while result.get("status") == "pending" and isinstance(call_id, str) and time.monotonic() < deadline:
        await asyncio.sleep(.2)
        result = await call("interface_results", {
            "call_id": call_id, "include_image": False, "compact": True, "detail": "brief",
        })
        result_call_id = result.get("call_id")
        pending_call = result.get("call")
        if result.get("status") == "pending" and isinstance(pending_call, dict):
            result_call_id = pending_call.get("call_id")
        if result_call_id != call_id:
            return {"status": "unexpected_recovery_call_id", "result": result}
    return result


async def recover_successor_action(call, result, *, target_name, hwnd, pid, title, folder, cfg, record=None):
    record = dump if record is None else record
    action = result["action"]
    cancel = result.get("cancellation", {}).get("result", {})
    identity = cancel.get("producer_identity")
    absent, identity_check = producer_absent(identity)
    record(folder / "followup-producer-absence.json", identity_check)
    if not absent:
        return None
    obligations = cancel.get("obligations")
    if not isinstance(obligations, list) or not obligations or cancel.get("late_input_horizon_ns") not in (None, 0, 2_000_000_000):
        return None
    custody = source_custody(action, obligations)
    dump(folder / "followup-source-custody-record.json", custody)
    observation = await call("interface_observe", {
        "target": target_name, "frame": cfg["capture_frame"], "region": cfg["capture_region"],
    })
    recovery_generation = observation_generation(observation)
    if recovery_generation is None:
        return None
    fg = subprocess.run(
        [sys.executable, str(FG_PATH), str(hwnd), str(pid),
         str(folder / "foreground-before-followup-recovery.json"), title],
        cwd=str(PACKET), capture_output=True, text=True,
    )
    record(folder / "foreground-before-followup-recovery-process.json", {
        "returncode": fg.returncode, "stdout": fg.stdout, "stderr": fg.stderr,
    })
    if fg.returncode != 0:
        return None
    recovery = await call("interface_recover_orphaned_input", {
        "source_operation_id": action["operation_id"], "source_target_generation": action["target_generation"],
        "recovery_target_generation": recovery_generation, "obligations": obligations,
        "source_custody_record": custody,
    })
    recovery = await poll_recovery(call, recovery)
    record(folder / "followup-recovery-final.json", recovery)
    if recovery_matches_action(recovery, action, recovery_generation, 2_000_000_000):
        return recovery
    return None


def source_custody(action, obligations):
    return {
        "classification": "UNKNOWN_RELEASE_UNCONFIRMED_STOPPED",
        "operation_id": action["operation_id"],
        "target_generation": action["target_generation"],
        "obligations": obligations,
        "do_not_replay": True,
        "observed_producer_live": False,
        "late_server_request_possible": True,
        "dispatch": {
            "input_dispatched": action.get("input_dispatched"), "release_confirmed": False,
            "task_success": None, "replay_allowed": False,
        },
    }


def observation_generation(observation):
    report = report_from_review(observation)
    generation = report.get("target_generation")
    if (
        report.get("status") == "returned" and report.get("input_dispatched") is False
        and report.get("side_effect_authority") is False
        and isinstance(generation, str) and len(generation) == 64
    ):
        return generation
    return None


def exception_record(exc):
    return {"type": type(exc).__name__, "message": str(exc)}


def launch_fixture(fixture, position, title, delay_ms):
    folder = fixture.folder
    args = [
        sys.executable, str(FIXTURE_PATH), "--meta", str(fixture.meta),
        "--effect", str(folder / "effect.json"), "--events", str(folder / "events.jsonl"),
        "--effect-delay-ms", str(delay_ms), "--state", str(folder / "fixture-state.json"),
        "--x", str(position[0]), "--y", str(position[1]), "--title", title,
        "--shutdown-ack", str(folder / ACK_NAME),
    ]
    with (folder / "fixture.stdout.txt").open("wb") as stdout, (folder / "fixture.stderr.txt").open("wb") as stderr:
        fixture.process = subprocess.Popen(
            args, cwd=str(PACKET), stdin=subprocess.PIPE, stdout=stdout, stderr=stderr,
            bufsize=0, close_fds=True,
        )


async def wait_until(predicate, timeout_s):
    deadline = time.monotonic() + timeout_s
    while not predicate():
        if time.monotonic() >= deadline:
            return False
        await asyncio.sleep(.05)
    return True


async def fixture_ready(fixture):
    await wait_until(lambda: fixture.meta.exists() or fixture.process.poll() is not None, 10)
    if not fixture.meta.exists():
        return False
    fixture.hwnd = int(json.loads(fixture.meta.read_text(encoding="utf-8"))["hwnd"])
    return fixture.hwnd > 0 and fixture.process.poll() is None

    return subprocess.run(

        cwd=str(PACKET), capture_output=True, text=True,
    )


def score_source(cfg, folder, record):
    score_summary = {"task_success": None, "state_sha256": None, "score_error": None}
    scorer_sha256 = None
    score = None
    try:
        spec = spec_from_file_location(f'r34_scorer_{cfg["ordinal"]}', SCORER_PATH)
        scorer = module_from_spec(spec)
        spec.loader.exec_module(scorer)
        scorer_sha256 = hashlib.sha256(SCORER_PATH.read_bytes()).hexdigest()
        success, state_bytes, score = scorer.score(folder / "events.jsonl", folder / "effect.json")
        score_summary["state_sha256"] = hashlib.sha256(state_bytes).hexdigest()
        score_summary["task_success"] = success
    except Exception as exc:
        score_summary["score_error"] = exception_record(exc)
    record(folder / "independent-score.json", {
        "scorer_id": "fixture-delayed-a-independent-v1", "scorer_sha256": scorer_sha256,
        **score_summary, "effect": score,
    })
    return score_summary


def finalize_fixtures(trial, source, fresh, initial_fence, record):
    latest_dispatch = next((fixture for fixture in (fresh, source) if fixture.dispatch_started), None)

    def matches_trial(fence):
        if latest_dispatch is None:
            return fence_is_clear(fence) and fence == initial_fence
        return latest_dispatch.settlement is not None and fence_matches_action(fence, *latest_dispatch.settlement)

    for label, fixture in (("followup", fresh), ("source", source)):
        if fixture.process is None:
            continue
        cleanup = {"dispatch_started": fixture.dispatch_started, "settlement": fixture.settlement, "closed": False}
        fence = None
        try:
            if fixture.dispatch_started and fixture.settlement is None:
                cleanup["reason"] = "fixture_custody_unresolved"
            elif latest_dispatch is not None and latest_dispatch.settlement is None:
                cleanup["reason"] = "cohort_custody_unresolved"
            else:
                fence = json.loads(FENCE.read_text(encoding="utf-8"))
                cleanup["fence_verified"] = matches_trial(fence)
                if cleanup["fence_verified"]:
                    cleanup.update(close_fixture(fixture))
                else:
                    cleanup["reason"] = "durable_fence_unverified"
        except Exception as exc:
            cleanup["reason"] = "cleanup_exception"
            cleanup["error"] = exception_record(exc)
        trial["cleanup"][label] = cleanup
        if fence is not None:
            record(fixture.folder / "fence-before-cleanup.json", fence, required=True)
        record(fixture.folder / "fixture-cleanup.json", cleanup, required=True)

    trial["fence_verified_after"] = False
    trial["fence_status_after"] = None
    trial["fence_last_resolved"] = None
    try:
        final_fence = json.loads(FENCE.read_text(encoding="utf-8"))
        trial["fence_verified_after"] = matches_trial(final_fence)
        if isinstance(final_fence, dict):
            trial["fence_status_after"] = final_fence.get("status")
            trial["fence_last_resolved"] = final_fence.get("last_resolved")
        record(source.folder / "fence-after.json", final_fence, required=True)
    except Exception as exc:
        trial["fence_after_error"] = exception_record(exc)
        record(source.folder / "fence-after.json", {"error": trial["fence_after_error"]}, required=True)

    trial["fixture_closed"] = trial["cleanup"].get("source", {}).get("closed") is True
    trial["fresh_fixture_closed"] = trial["cleanup"].get("followup", {}).get("closed") is True
    if trial["outcome"] == "RECOVERY_AND_NEW_TASK_VERIFIED":
        if not trial["fence_verified_after"] or any(
            row.get("fence_verified") is not True for row in trial["cleanup"].values()
        ):
            trial["outcome"] = "STOP_DURABLE_FENCE_NOT_CLEAR"
        elif not trial["fixture_closed"]:
            trial["outcome"] = "STOP_FIXTURE_CLOSE_UNVERIFIED"
        elif not trial["fresh_fixture_closed"]:
            trial["outcome"] = "STOP_FRESH_FIXTURE_CLOSE_UNVERIFIED"
        elif any(error.get("required") is True for error in trial["evidence_errors"]):
            trial["outcome"] = "STOP_REQUIRED_EVIDENCE_UNRECORDED"


async def run_trial(cfg):
    folder = RAW / f'trial-{cfg["ordinal"]:02d}'
    source = TrialFixture(folder)
    fresh = TrialFixture(folder / "fresh-task-target")
    initial_fence = None
    timeline = []
    trial = {
        "ordinal": cfg["ordinal"], "trial_id": cfg["trial_id"], "outcome": "STOP_TRIAL_INCOMPLETE",
        "source_calls": timeline, "recovery_attempted": False,
        "recovery_release_confirmed": None, "recovery_horizon_verified": None,
        "cleanup": {}, "evidence_errors": [], "session_close": {},
    }

    def stop(reason):
        trial["outcome"] = reason
        return trial

    def record(path, value, *, required=False):
        try:
            dump(path, value)
        except Exception as exc:
            trial["evidence_errors"].append({"file": str(path), "required": required, **exception_record(exc)})

    async def call(session, label, name, args):
        st = time.monotonic_ns()
        response = await session.call_tool(name, args)
        en = time.monotonic_ns()
        timeline.append({"session": label, "tool": name, "started_ns": st, "ended_ns": en})
        try:
            record(folder / f"{label}-{name}-{uuid.uuid4().hex}.json", raw_response(response))
        except Exception as exc:
            trial["evidence_errors"].append({"tool": name, **exception_record(exc)})
        return body(response), response

    async def close_session(label, invoke):
        try:
            result = await invoke("interface_close", {})
        except Exception as exc:
            result = {"status": "close_exception", **exception_record(exc)}
        trial["session_close"][label] = result
        record(folder / f"{label}-session-close.json", result)

    def save_image(path, raw, *, required):
        images = [block for block in raw.get("content", []) if block.get("type") == "image" and block.get("mimeType") == "image/png"]
        if len(images) == 1:
            path.write_bytes(base64.b64decode(images[0]["data"], validate=True))
        elif required:
            raise RuntimeError("public observation PNG missing")

    try:
        folder.mkdir()
    except Exception as exc:
        trial["exception"] = exception_record(exc)
        return stop("STOP_TRIAL_DIRECTORY_UNAVAILABLE")

    try:
        initial_fence = json.loads(FENCE.read_text(encoding="utf-8"))
        if not fence_is_clear(initial_fence):
            return stop("STOP_INITIAL_FENCE_NOT_CLEAR")
        dump(folder / "fence-before.json", initial_fence)

        launch_fixture(source, cfg["source_window_position"], cfg["window_title"], cfg["effect_delay_ms"])
        if not await fixture_ready(source):
            return stop("STOP_SOURCE_TARGET_NOT_READY")
        targets = folder / "targets.json"
        dump(targets, {"fixture": source.hwnd})
        trial["target"] = {"hwnd": source.hwnd, "pid": source.process.pid, "title": cfg["window_title"], "region": cfg["capture_region"]}
        dump(folder / "window.json", trial["target"])
        if not await wait_until((folder / "computer-use-foreground-confirmed.json").exists, 120):
            return stop("STOP_SOURCE_ACTIVATION_MISSING")
        source_foreground = folder / "foreground-pre-dispatch.json"
        fg = fixture_foreground(source, cfg["window_title"], source_foreground)
        if fg.returncode:
            return stop("STOP_SOURCE_FOREGROUND_MISMATCH")
        if json.loads(source_foreground.read_text(encoding="utf-8")).get("window_rect", [])[:2] != cfg["source_window_position"]:
            return stop("STOP_SOURCE_POSITION_MISMATCH")

        def source_params(output):
            args = [
                "-m", "runtime.cli_v1.mcp_server", "--targets", str(targets),
                "--output-directory", str(folder / output), "--session-mode", "supervised-win32",
                "--supervised-win32-late-input-horizon-ms", str(cfg["horizon_ms"]),
            ]
            return StdioServerParameters(command=sys.executable, args=args, cwd=str(ROOT))

        def task_params(output):

                str(PACKET / "task_oracle_server.py"), str(targets), str(folder / output),
                str(fresh.folder / "fixture-state.json"), str(fresh.folder / "events.jsonl"),

                    return stop("STOP_SOURCE_ACTION_RECEIPT_UNVERIFIED")

                    "release_confirmed", "task_success", "replay_allowed",

                trial["source"] = source_proof
                record(folder / "source-action-summary.json", source_proof, required=True)
                trial["source_release_confirmed"] = action.get("release_confirmed") is True
                if action.get("status") == "completed" and trial["source_release_confirmed"]:
                    source.settlement = (action["operation_id"], source_generation)
                    await close_session("source", call_source)
                else:
                    record(folder / "source-session-close.json", {"status": "skipped_unresolved_source_custody"})

        if action.get("status") != "unknown" or action.get("release_confirmed") is not False:
            return stop("STOP_SOURCE_RECOVERY_NOT_ELIGIBLE")
        identity = cancel.get("producer_identity")
        absent, identity_check = producer_absent(identity)
        record(folder / "producer-identity-check.json", identity_check)
        if not absent:
            return stop("STOP_PRODUCER_ABSENCE_UNCONFIRMED")
        obligations = cancel.get("obligations")
        horizon_ns = cfg["horizon_ms"] * 1_000_000
        if not isinstance(obligations, list) or not obligations or cancel.get("late_input_horizon_ns") not in (None, 0, horizon_ns):
            return stop("STOP_SOURCE_CUSTODY_FIELDS_INVALID")
        custody = source_custody(action, obligations)
        dump(folder / "source-custody-record.json", custody)

        async with stdio_client(source_params("recovery-mcp")) as (reader, writer):
            async with ClientSession(reader, writer, read_timeout_seconds=timedelta(seconds=40)) as session:
                await session.initialize()

                async def call_recover_source(name, args):
                    result, _ = await call(session, "recovery", name, args)
                    return result

                observation = await call_recover_source("interface_observe", observe_args("fixture"))
                recovery_generation = observation_generation(observation)
                if recovery_generation is None:
                    return stop("STOP_RECOVERY_OBSERVATION_INVALID")
                fg = fixture_foreground(source, cfg["window_title"], folder / "foreground-pre-recovery.json")
                record(folder / "foreground-recovery-probe.json", {"returncode": fg.returncode, "stdout": fg.stdout, "stderr": fg.stderr})
                if fg.returncode:
                    return stop("STOP_RECOVERY_TARGET_FOREGROUND_MISMATCH")
                trial["recovery_attempted"] = True
                final = await call_recover_source("interface_recover_orphaned_input", {
                    "source_operation_id": action["operation_id"], "source_target_generation": source_generation,
                    "recovery_target_generation": recovery_generation, "obligations": obligations,
                    "source_custody_record": custody,
                })
                final = await poll_recovery(call_recover_source, final)
                record(folder / "source-recovery-final.json", final, required=True)
                trial["recovery_status"] = final.get("status")
                trial["recovery_release_confirmed"] = final.get("release_confirmed") is True
                receipt = final.get("recovery_receipt", {})
                trial["recovery_horizon_verified"] = receipt.get("late_input_horizon_verified") is True
                trial["recovery_receipt_summary"] = {
                    **{key: final.get(key) for key in (
                        "source_action_status", "source_operation_id", "source_target_generation",
                        "recovery_target_generation", "release_confirmed", "task_success", "replay_allowed",
                    )},
                    **{key: receipt.get(key) for key in (
                        "late_input_horizon_ns", "late_input_horizon_verified", "late_input_observation_count",
                        "late_input_max_observed_gap_ns", "late_input_max_sample_gap_ns",
                    )},
                    "readback_neutral": receipt.get("readback", {}).get("neutral"),
                }
                if not recovery_matches_action(final, action, recovery_generation, horizon_ns):
                    return stop("STOP_RECOVERY_UNVERIFIED")
                source.settlement = (action["operation_id"], source_generation)
                await close_session("recovery", call_recover_source)

        source_fence = json.loads(FENCE.read_text(encoding="utf-8"))
        trial["source_fence_verified"] = fence_matches_action(source_fence, *source.settlement)
        record(folder / "source-fence-before-followup.json", source_fence, required=True)
        if not trial["source_fence_verified"]:
            return stop("STOP_SOURCE_FENCE_UNVERIFIED")

        # Scoring and source-effect recording are diagnostic after custody is settled.
        effect = None
        prior_state = {}
        try:
            effect_path = folder / "effect.json"
            await wait_until(effect_path.exists, 10)
            if effect_path.exists():
                effect = json.loads(effect_path.read_text(encoding="utf-8-sig"))
            state_path = folder / "fixture-state.json"
            if state_path.exists():
                prior_state = json.loads(state_path.read_text(encoding="utf-8-sig"))
        except Exception as exc:
            trial["effect_error"] = exception_record(exc)
        trial["effect"] = effect
        trial["score"] = score_source(cfg, folder, record)
        trial["source_effect_observed"] = (
            isinstance(effect, dict) and effect.get("verified") is True
            and isinstance(prior_state, dict) and prior_state.get("text") == "a"
        )
        record(folder / "source-effect-observation.json", {
            "effect": effect, "state": prior_state, "verified": trial["source_effect_observed"],
            "used_as_successor_gate": False,
        })

        fresh.folder.mkdir()
        fresh_title = cfg["followup_window_title"]
        launch_fixture(fresh, cfg["followup_window_position"], fresh_title, 6000)
        if not await fixture_ready(fresh):
            return stop("STOP_FRESH_TARGET_NOT_READY")
        dump(fresh.folder / "target.json", {"hwnd": fresh.hwnd, "pid": fresh.process.pid, "title": fresh_title})
        dump(targets, {"fresh": fresh.hwnd})
        if not await wait_until((folder / "computer-use-followup-foreground-confirmed.json").exists, 120):
            return stop("STOP_FOLLOWUP_ACTIVATION_MISSING")
        fresh_foreground = folder / "foreground-pre-followup.json"
        fg = fixture_foreground(fresh, fresh_title, fresh_foreground)
        record(folder / "foreground-followup-probe.json", {"returncode": fg.returncode, "stdout": fg.stdout, "stderr": fg.stderr})
        if fg.returncode:
            return stop("STOP_FOLLOWUP_FOREGROUND_MISMATCH")
        if json.loads(fresh_foreground.read_text(encoding="utf-8")).get("window_rect", [])[:2] != cfg["followup_window_position"]:
            return stop("STOP_FOLLOWUP_POSITION_MISMATCH")

        async with stdio_client(task_params("task-mcp")) as (reader, writer):
            async with ClientSession(reader, writer, read_timeout_seconds=timedelta(seconds=40)) as session:
                await session.initialize()

                async def call3(name, args):
                    result, _ = await call(session, "followup", name, args)
                    return result

                gen3 = observation_generation(await call3("interface_observe", observe_args("fresh")))
                if gen3 is None or gen3 == source_generation:
                    return stop("STOP_FOLLOWUP_OBSERVE_INVALID")
                program = program_for("fresh", gen3, cfg["followup_program_id"], cfg["followup_lease_id"], 60_000, "B")
                dump(folder / "followup-program.json", program)
                fresh.dispatch_started = True
                dispatched = await call3("interface_dispatch", dispatch_args(program))
                cid3 = dispatched.get("call_id")
                if not isinstance(cid3, str) or not cid3:
                    return stop("STOP_FOLLOWUP_NO_CALL_ID")
                r3 = await poll_exact_action(call3, cid3)
                record(folder / "followup-action-result.json", r3, required=True)
                b_action = exact_action(r3, cid3, gen3)
                if b_action is None or b_action["operation_id"] == action["operation_id"]:
                    return stop("STOP_FOLLOWUP_ACTION_RECEIPT_UNVERIFIED")
                trial.update(
                    followup_call_id=cid3, followup_operation_id=b_action["operation_id"],
                    followup_status=b_action.get("status"), followup_release_confirmed=b_action.get("release_confirmed"),
                    followup_input_dispatched=b_action.get("input_dispatched"),
                )
                if b_action.get("status") == "completed" and b_action.get("release_confirmed") is True:
                    fresh.settlement = (b_action["operation_id"], gen3)
                    try:
                        v3 = await call3("interface_verify_task", {"call_id": cid3, "wait_for_result_ms": 5000})
                    except Exception as exc:
                        v3 = {"status": "oracle_error", "task_success": None, "replay_allowed": False, "error": exception_record(exc)}
                    record(folder / "followup-task-verification.json", v3, required=True)
                    trial["followup_task_success"] = verified_task_success(v3, cid3, b_action)
                    trial["outcome"] = (
                        "RECOVERY_AND_NEW_TASK_VERIFIED"
                        if b_action.get("input_dispatched") is True and trial["followup_task_success"] is True
                        else "STOP_FOLLOWUP_TASK_UNVERIFIED"
                    )
                    await close_session("followup", call3)
                    return trial
                record(folder / "followup-session-close.json", {"status": "skipped_unresolved_successor_custody"})
                trial["outcome"] = "STOP_FOLLOWUP_ACTION_NOT_RELEASED"
                if b_action.get("status") != "unknown" or b_action.get("release_confirmed") is not False:
                    return trial

        trial["followup_recovery_attempted"] = True
        try:
            async with stdio_client(task_params("followup-recovery-mcp")) as (reader, writer):
                async with ClientSession(reader, writer, read_timeout_seconds=timedelta(seconds=40)) as session:
                    await session.initialize()

                    async def call_recovery(name, args):
                        result, _ = await call(session, "followup-recovery", name, args)
                        return result

                    followup_recovery = await recover_successor_action(
                        call_recovery, r3, target_name="fresh", hwnd=fresh.hwnd, pid=fresh.process.pid,
                        title=fresh_title, folder=folder, cfg=cfg, record=record,
                    )
                    if followup_recovery is not None:
                        fresh.settlement = (b_action["operation_id"], gen3)
                        trial.update(
                            followup_recovery_status="recovered", followup_recovery_verified=True,
                            followup_recovery_release_confirmed=True, followup_recovery_horizon_verified=True,
                            followup_release_confirmed=True, outcome="STOP_FOLLOWUP_RECOVERED_TASK_UNVERIFIED",
                        )
                        await close_session("followup-recovery", call_recovery)
        except Exception as exc:
            trial["followup_recovery_error"] = exception_record(exc)
            record(folder / "followup-recovery-exception.json", trial["followup_recovery_error"])
        if fresh.settlement is None:
            trial.update(followup_recovery_status="unverified", followup_recovery_release_confirmed=False,
                         followup_recovery_horizon_verified=False)
        return trial
    except BaseException as exc:
        trial["exception"] = {**exception_record(exc), "traceback": traceback.format_exc()}
        record(folder / "trial-exception.json", trial["exception"])
        if not isinstance(exc, Exception):
            raise
        if trial["outcome"] in ("STOP_TRIAL_INCOMPLETE", "RECOVERY_AND_NEW_TASK_VERIFIED"):
            return stop("STOP_TRIAL_EXCEPTION")
        return trial
    finally:
        finalize_fixtures(trial, source, fresh, initial_fence, record)
