def sha(b):return hashlib.sha256(b).hexdigest()
def utc():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def save(path,value):
    with path.open('x',encoding='ascii') as f:json.dump(value,f,ensure_ascii=True,indent=2)

kernel=ctypes.WinDLL('kernel32',use_last_error=True)
kernel.OpenProcess.argtypes=[wintypes.DWORD,wintypes.BOOL,wintypes.DWORD];kernel.OpenProcess.restype=wintypes.HANDLE
kernel.CloseHandle.argtypes=[wintypes.HANDLE];kernel.CloseHandle.restype=wintypes.BOOL
kernel.WaitForSingleObject.argtypes=[wintypes.HANDLE,wintypes.DWORD];kernel.WaitForSingleObject.restype=wintypes.DWORD
kernel.GetExitCodeProcess.argtypes=[wintypes.HANDLE,ctypes.POINTER(wintypes.DWORD)];kernel.GetExitCodeProcess.restype=wintypes.BOOL
kernel.GetProcessTimes.argtypes=[wintypes.HANDLE]+[ctypes.POINTER(wintypes.FILETIME)]*4;kernel.GetProcessTimes.restype=wintypes.BOOL
def native(handle):
    code=wintypes.DWORD();a,b,c,d=[wintypes.FILETIME() for _ in range(4)]
    wait=kernel.WaitForSingleObject(handle,0)
    if not kernel.GetExitCodeProcess(handle,ctypes.byref(code)):raise ctypes.WinError(ctypes.get_last_error())
    if not kernel.GetProcessTimes(handle,*[ctypes.byref(x) for x in (a,b,c,d)]):raise ctypes.WinError(ctypes.get_last_error())
    number=lambda x:(x.dwHighDateTime<<32)|x.dwLowDateTime
    return {'wait_result':wait,'exit_code':code.value,'creation_filetime_100ns':number(a),
            'exit_filetime_100ns':number(b),'kernel_cpu_100ns':number(c),'user_cpu_100ns':number(d),'observed_utc':utc()}

OUT.mkdir(exist_ok=False)
source=STUDY/'source'

argv=[str(NODE),str(probe),str(module),str(OUT),NONCE]

process=subprocess.Popen(argv,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,creationflags=subprocess.CREATE_NO_WINDOW)
events=queue.Queue();stream_faults=[]
def pump(stream,path,enqueue):
    with path.open('xb') as out:
        total=0
        for line in iter(stream.readline,b''):
            total+=len(line)
            if total>65536:stream_faults.append('stream limit');events.put(('error','stream limit'));break
            out.write(line);out.flush()
            if enqueue:events.put(('line',line))
    if enqueue:events.put(('eof',None))
readers=[threading.Thread(target=pump,args=(process.stdout,OUT/'stdout.bin',True),daemon=True),
         threading.Thread(target=pump,args=(process.stderr,OUT/'stderr.bin',False),daemon=True)]
for reader in readers:reader.start()
handles={};native_rows=[];protocol=[];failure=None

try:
    while True:
        left=deadline-time.monotonic()
        if left<=0:raise TimeoutError('owned capture20s deadline')
        kind,data=events.get(timeout=left)
        if kind=='error':raise RuntimeError(data)
        if kind=='eof':break
        row=json.loads(data);protocol.append(row)
        if row['stage']=='prepared':
            for role in ('parent','descendant'):
                owner=row[role]
                if owner is None:continue
                if owner['token']!=NONCE+'-'+row['case']:raise RuntimeError('native owner token mismatch')
                handle=kernel.OpenProcess(0x100000|0x1000,False,owner['pid'])
                if not handle:raise ctypes.WinError(ctypes.get_last_error())
                handles[(row['case'],role)]=(handle,owner['pid'])
                native_rows.append({'stage':'prepared','case':row['case'],'role':role,'pid':owner['pid'],**native(handle)})
            go={'go':True,'case':row['case'],'token':NONCE+'-'+row['case']}
            process.stdin.write(json.dumps(go).encode('ascii')+b'\n');process.stdin.flush()
        elif row['stage']=='checkpoint':
            for (case,role),(handle,pid) in handles.items():
                if case==row['case']:native_rows.append({'stage':'checkpoint','case':case,'role':role,'pid':pid,**native(handle)})
            go={'checkpoint_captured':True,'case':row['case'],'token':NONCE+'-'+row['case']}
            process.stdin.write(json.dumps(go).encode('ascii')+b'\n');process.stdin.flush()
        elif row['stage']=='result':
            for (case,role),(handle,pid) in handles.items():

                    observed=kernel.WaitForSingleObject(handle,1000)
