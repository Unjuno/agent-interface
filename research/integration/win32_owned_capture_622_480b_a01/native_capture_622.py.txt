@PROC
def wndproc(window,msg,wp,lp):
    if msg in (0x0317,0x0318):
        try:
            assert int(window)==int(hwnd), 'Only the owned target may print'
            for rect,brush in zip((RECT(0,0,8,8),RECT(8,0,16,8),RECT(0,8,8,16),RECT(8,8,16,16)),brushes):
                assert u.FillRect(wp,c.byref(rect),brush)

    return u.DefWindowProcW(window,msg,wp,lp)

class Guard:
    def __init__(self,dll,allowed,history):self.dll=dll;self.allowed=allowed;self.history=history
    def __getattr__(self,name):
        if name not in self.allowed:raise RuntimeError('Forbidden native call: '+name)
        def call(*args):
            if name in ('IsWindow','GetClientRect','ClientToScreen','GetDC','ReleaseDC','PrintWindow'):
                assert int(args[0])==int(hwnd), 'No screen or other window capture'
            result=getattr(self.dll,name)(*args)
            row={'call':name,'result':int(result or 0)}
            if name=='GetDIBits' and result:
                row['raw_base64']=base64.b64encode(c.string_at(args[4],16*16*4)).decode()
            self.history.append(row)
            return result
        return call
