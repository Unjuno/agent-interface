"""Saved-only independent raster/crop/type/history oracle; executes no fixture."""
import base64
from collections import Counter
import hashlib
import json
from pathlib import Path
import sys

out=Path(sys.argv[1]).resolve();raw=json.loads((out/'RAW.json').read_bytes())
freeze=json.loads((out/'FREEZE.json').read_bytes());receipt=json.loads((out/'NATIVE_RECEIPT.json').read_bytes())
errors=[]
def need(ok,label):
    if not ok:errors.append(label)
need(receipt['exit_code']==0 and receipt['timed_out'] is False and receipt['attempts']==1,'native receipt')
need(raw['pid']==receipt['pid']==raw['owner_pid'] and raw['visible'] is False,'owned invisible HWND')
need(raw['callback_errors']==[] and len(raw['draws'])==4,'four owner renderings')
need(all(d['owned_hwnd']==raw['hwnd'] for d in raw['draws']),'only owned prints')
need(raw['cleanup']=={'destroy_window':True,'delete_brushes':[True]*4,'unregister_class':True},'explicit native cleanup')
for source in freeze['source_pins']:
    b=(out/'source'/source['path']).read_bytes();need(len(b)==source['bytes'] and hashlib.sha256(b).hexdigest()==source['sha256'],'literal source:'+source['path'])
cases=[('full',[0,0,16,16]),('top_left',[2,2,4,4]),('bottom_right',[12,12,4,4]),('cross_quadrants',[6,6,4,4]),('outside_right',[15,0,2,2]),('zero_width',[0,0,0,4])]
need([(r['case'],r['region']) for r in raw['rows']]==cases,'six complete ordered cases')
positive_counts={'IsWindow':2,'GetClientRect':1,'ClientToScreen':1,'GetDC':1,'CreateCompatibleDC':1,'CreateCompatibleBitmap':1,'SelectObject':2,'PrintWindow':1,'GetDIBits':1,'DeleteObject':1,'DeleteDC':1,'ReleaseDC':1}
for index,row in enumerate(raw['rows']):
    label=row['case'];counts=Counter(c['call'] for c in row['calls'])
    need(type(row['backend_emissions']) is int and row['backend_emissions']==0,label+':zero task emissions')
    need(type(row['started_ns']) is int and type(row['ended_ns']) is int and row['ended_ns']>=row['started_ns'],label+':record boundaries')
    if index<4:
        need(dict(counts)==positive_counts,label+':native call census')
        dib=[c for c in row['calls'] if c['call']=='GetDIBits'];need(len(dib)==1,label+':one DIB')
        pixels=base64.b64decode(dib[0]['raw_base64'],validate=True)
        need(len(pixels)==1024 and dib[0]['result']==16,label+':full raw geometry')
        # Independent coordinate oracle; alpha has no specified GDI meaning.
        for y in range(16):
            for x in range(16):
                expected=((0,0,255) if y<8 and x<8 else (0,255,0) if y<8 else (255,0,0) if x<8 else (0,255,255))
                i=(y*16+x)*4
                need(tuple(pixels[i:i+3])==expected,label+':pixel:'+str((x,y)))
        x,y,width,height=row['region']
        selected=bytes(v for yy in range(y,y+height) for xx in range(x,x+width) for v in pixels[(yy*16+xx)*4:(yy*16+xx+1)*4])
        report=row.get('report',{})
        need(report=={'bytes':len(selected),'sha256':hashlib.sha256(selected).hexdigest(),'width':width,'height':height},label+':exact public crop projection')
        need(all(c['result']!=0 for c in row['calls']),label+':successful native returns')
    else:
        need('report' not in row and row.get('exception',{}).get('type')=='Win32BackendError',label+':typed refusal')
        need(dict(counts)==({'IsWindow':2,'GetClientRect':1,'ClientToScreen':1} if index==4 else {}),label+':no capture entry')
payload_bytes=sum((out/f).stat().st_size for f in ['RAW.json','native.stdout','native.stderr','NATIVE_RECEIPT.json','NATIVE_STARTED.json'])
need(payload_bytes<=freeze['output_bound_bytes'],'actual output bound')
result={'status':'PASS_SCOPED_NATIVE_PIXEL_CROP' if not errors else 'FAIL_SCOPED_NATIVE_PIXEL_CROP','rows':len(raw['rows']),'positive_native_captures':4,'negative_pre_capture_refusals':2,'pixel_channels_checked':4*16*16*3,'errors':errors,'payload_bytes':payload_bytes,'scope':raw['scope'],'actual_native_pid':receipt['pid'],'cleanup':raw['cleanup'],'source_module_execution':0}
(out/'AUDIT.json').write_text(json.dumps(result,indent=2),encoding='utf-8');print(json.dumps(result));raise SystemExit(bool(errors))
