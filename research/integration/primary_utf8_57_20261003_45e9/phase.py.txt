from datetime import datetime,timezone

node=Path(shutil.which('node'));paths=sorted((root/'source/runtime/host_v1').glob('*.mjs'),key=lambda p:p.name)

def put(p,v):p.write_text(json.dumps(v,indent=2,sort_keys=True)+'\n',encoding='utf-8',newline='\n')

put(out/'FREEZE.json',{'frozen_utc':datetime.now(timezone.utc).isoformat(),'label':label,'source':[{'path':p.relative_to(root/'source').as_posix(),'sha256':sha(p.read_bytes()),'bytes':p.stat().st_size} for p in paths],
 'phase_source_sha256':sha(Path(__file__).read_bytes()),'node_version':subprocess.check_output([str(node),'--version']).decode().strip(),'node_binary_sha256':sha(node.read_bytes()),
 'python':sys.version,'platform':platform.platform(),'argv':command,'cwd':str(root/'source'),
 'H':'strict original byte admission before existing readline/exchange preserves valid input and observes same accepted promise on later malformed bytes',
 'T':'15 literal methods:5 malformed-byte forms,EOF truncation,ASCII/split multibyte/U+FFFD/text/JSON/BOM/owned listeners/accepted promise/corrupted chunk prefix',
 'D':'RED expected contract assertions retain first failures; fixed same15 pass,malformed neverexecute/consume,validvalues and acceptedwork preserved',
 'C':'validator aborts the malformed raw chunk before unaccepted valid prefix in that chunk; caller already-decoded text cannot reconstruct upstream original bytes',
 'U':'ordinary in-process controlled streams/inert exchange; no real backend/SDK/native/input/task/performance/formal allocation'})

for p in paths:assert sha(p.read_bytes())==next(x['sha256'] for x in json.loads((out/'FREEZE.json').read_bytes())['source'] if x['path']==p.relative_to(root/'source').as_posix())
print(json.dumps({'phase':label,'exit_code':r.returncode,'summary':[x for x in r.stdout.decode('utf-8',errors='replace').splitlines() if x.startswith('# tests') or x.startswith('# pass') or x.startswith('# fail')]}))
sys.exit(0 if (label=='red-first' and r.returncode==1) or (label=='green-first' and r.returncode==0) else 1)
