"""Exact five-file current public stdio import closure; bounded missing-blob fetch."""
from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, os, subprocess
ROOT = Path(__file__).resolve().parent
REPO = ROOT.parent / 'agent-interface'
REF = '9c0692dabbfc7fc2fa5bd111b6de4baee578800d'
PATHS = ['runtime/host_v1/' + n for n in ('primary_stdio.mjs','primary_exchange.mjs','primary_caller.mjs','relay_host.mjs','relay_client.mjs')]
ENV = dict(os.environ, GIT_NO_LAZY_FETCH='1')
def sha(b): return hashlib.sha256(b).hexdigest()
def git(*a, **kw): return subprocess.run(['git', *a], cwd=REPO, env=ENV, capture_output=True, check=True, **kw).stdout
def put(n, v): (ROOT/n).write_text(json.dumps(v, indent=2, sort_keys=True)+'\n', encoding='utf-8')
before=(REPO/'.git/index').read_bytes()
start=datetime.now(timezone.utc).isoformat()
cmd=['git','-c','fetch.negotiationAlgorithm=noop','fetch','origin','main','--depth=12','--no-tags','--no-write-fetch-head','--recurse-submodules=no','--filter=blob:none']
r=subprocess.run(cmd,cwd=REPO,env=ENV,capture_output=True,timeout=45)
(ROOT/'fetch.stdout.txt').write_bytes(r.stdout);(ROOT/'fetch.stderr.txt').write_bytes(r.stderr)
put('FETCH.json',{'argv':cmd,'started_utc':start,'ended_utc':datetime.now(timezone.utc).isoformat(),'exit_code':r.returncode})
assert r.returncode==0, r.stderr
tree=git('rev-parse',REF+'^{tree}').decode().strip()
records=[];missing=[]
for p in PATHS:
    meta,path=git('ls-tree',REF,'--',p).decode().strip().split('\t');mode,kind,oid=meta.split()
    assert path==p and kind=='blob'
    records.append({'path':p,'mode':mode,'blob_oid':oid})
    q=subprocess.run(['git','cat-file','-e',oid],cwd=REPO,env=ENV,capture_output=True)
    if q.returncode: missing.append(oid)
if missing:
    cmd=['git','-c','fetch.negotiationAlgorithm=noop','fetch','origin','--no-tags','--no-write-fetch-head','--recurse-submodules=no','--filter=blob:none','--stdin']
    r=subprocess.run(cmd,cwd=REPO,env=ENV,input=('\n'.join(sorted(set(missing)))+'\n').encode(),capture_output=True,timeout=45)
    (ROOT/'blob-fetch.stdout.txt').write_bytes(r.stdout);(ROOT/'blob-fetch.stderr.txt').write_bytes(r.stderr)
    put('BLOB_FETCH.json',{'argv':cmd,'explicit_oids':sorted(set(missing)),'exit_code':r.returncode})
    assert r.returncode==0, r.stderr
for row in records:
    b=git('cat-file','blob',row['blob_oid'])
    assert hashlib.sha1(b'blob '+str(len(b)).encode()+b'\0'+b).hexdigest()==row['blob_oid']
    row.update(bytes=len(b),sha256=sha(b))
    dst=ROOT/'source'/row['path'];dst.parent.mkdir(parents=True,exist_ok=True);dst.write_bytes(b)
assert before==(REPO/'.git/index').read_bytes()
put('SOURCE.json',{'main':REF,'tree':tree,'files':records,'primary_index_unchanged':True,'ended_utc':datetime.now(timezone.utc).isoformat()})
print(json.dumps({'status':'EXACT_CURRENT_SOURCE','main':REF,'files':len(records),'bytes':sum(x['bytes'] for x in records),'primary_index_unchanged':True}))
