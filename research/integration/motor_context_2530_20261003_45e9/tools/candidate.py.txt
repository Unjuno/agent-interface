"""Observe two production APIs; contains no expected-result oracle."""
import copy
import hashlib
import json
from pathlib import Path
import sys

source, corpus, output = map(Path, sys.argv[1:])
sys.path.insert(0, str(source.resolve()))
from runtime.motor_state_v1 import adapter, dispatch_bridge

raw = corpus.read_bytes()
cases = [json.loads(line) for line in raw.splitlines()]
counts = {"cases": 0, "exceptions": 0, "mutations": 0}
with output.open("x", encoding="utf-8", newline="\n") as target:
    for case in cases:
        supplied = copy.deepcopy(case)
        before = json.dumps(supplied, sort_keys=True)
        try:
            if supplied["api"] == "validate":
                result = adapter.validate(supplied["row"])
            else:
                result = dispatch_bridge.bridge_dispatch_result(supplied["result"], supplied["context"])
            actual = {"return": result}
        except Exception as error:
            actual = {"exception": type(error).__name__}
            counts["exceptions"] += 1
        unmutated = before == json.dumps(supplied, sort_keys=True)
        counts["mutations"] += not unmutated
        target.write(json.dumps({"case_id": case["case_id"], "actual": actual, "unmutated": unmutated}, sort_keys=True, separators=(",", ":")) + "\n")
        counts["cases"] += 1
        if target.tell() > 16777216:
            raise RuntimeError("predeclared output bound exceeded")
summary = {**counts, "cases_sha256": hashlib.sha256(raw).hexdigest(),
           "raw_sha256": hashlib.sha256(output.read_bytes()).hexdigest(),
           "source_sha256": {name: hashlib.sha256(Path(module.__file__).read_bytes()).hexdigest()
                             for name, module in (("adapter.py", adapter), ("dispatch_bridge.py", dispatch_bridge))}}
with output.with_suffix(".summary.json").open("x", encoding="utf-8", newline="\n") as target:
    json.dump(summary, target, indent=2); target.write("\n")
print(json.dumps(summary))
