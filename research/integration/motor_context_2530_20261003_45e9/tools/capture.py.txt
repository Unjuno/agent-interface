"""Append-only capture of an ordinary engineering subprocess."""
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

root = Path(__file__).resolve().parent
label, cwd, *command = sys.argv[1:]
out = root / "private" / label
out.mkdir(parents=True, exist_ok=False)
start = datetime.now(timezone.utc).isoformat()
env = dict(os.environ, PYTHONUTF8="1", PYTHONDONTWRITEBYTECODE="1")
result = subprocess.run(command, cwd=cwd, env=env, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
end = datetime.now(timezone.utc).isoformat()
(out / "stdout.txt").write_bytes(result.stdout)
(out / "stderr.txt").write_bytes(result.stderr)
receipt = {
    "run_id": "motor2530-45e9-" + label,
    "scope": "ordinary host engineering; not a formal allocation",
    "started_utc": start, "ended_utc": end, "cwd": cwd, "command": command,
    "exit_code": result.returncode,
    "stdout_sha256": hashlib.sha256(result.stdout).hexdigest(),
    "stderr_sha256": hashlib.sha256(result.stderr).hexdigest(),
}
(out / "receipt.json").write_text(json.dumps(receipt, indent=2) + "\n", encoding="utf-8", newline="\n")
print(json.dumps(receipt))
sys.stdout.reconfigure(encoding="utf-8")
print(result.stdout.decode("utf-8", errors="replace")[-700:])
print(result.stderr.decode("utf-8", errors="replace")[-1000:])
sys.exit(result.returncode)
