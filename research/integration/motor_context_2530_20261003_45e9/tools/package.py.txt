"""Create separate public derivatives without rewriting any private original."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path

root = Path(__file__).resolve().parent
workspace = root.parent.parent
public = workspace / "outputs/motor2530-v2"
public.mkdir(parents=True, exist_ok=False)
hashes = []
private_prefix = str(workspace)


def publish(source, relative):
    original = source.read_bytes()
    value = original
    if relative.endswith((".txt", ".json")):
        value = value.replace(private_prefix.encode(), b"<AUTHOR_WORKSPACE>")
        value = value.replace(private_prefix.replace("\\", "\\\\").encode(), b"<AUTHOR_WORKSPACE>")
        value = value.replace(private_prefix.replace("\\", "/").encode(), b"<AUTHOR_WORKSPACE>")
    target = public / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open("xb") as stream:
        stream.write(value)
    hashes.append({"artifact": relative, "original_sha256": hashlib.sha256(original).hexdigest(),
                   "public_sha256": hashlib.sha256(value).hexdigest(), "path_redacted": value != original,
                   "original_retained_and_read_back": source.read_bytes() == original})


for name in ("PLAN.json", "CORPUS_FREEZE.json", "cases.jsonl", "intake.json", "source-identities.json", "receipt-source-identities.json"):
    publish(root / name, name)
for name in ("adapter.py", "dispatch_bridge.py"):
    publish(root / "before-source/runtime/motor_state_v1" / name, "baseline/" + name + ".txt")
for name in ("capture.py", "prepare.py", "make_corpus.py", "candidate.py", "audit.py", "package.py", "receipt_seam.py", "materialize_receipt.py"):
    publish(root / name, "tools/" + name + ".txt")
for source in sorted((root / "private").rglob("*")):
    if source.is_file():
        publish(source, "observed/" + source.relative_to(root / "private").as_posix())
provenance = {"created_utc": datetime.now(timezone.utc).isoformat(), "artifacts": hashes,
              "scope": "Public path-redacted derivatives; every original above is retained unchanged in the author's private workspace.",
              "receipt_hashes": "Captured receipt stdout/stderr hashes identify original bytes; public hashes are recorded here and in SHA256SUMS.",
              "intake_limit": "First triage observations have exact source identities but no separately captured subprocess UTC/exit receipt; they are not the decisive corpus/test result.",
              "clock": "UTC wall timestamps mark actual subprocess start/end; no runtime performance claim.",
              "historical_runs": "Issue 2530's original allocation and cancelled/merged historical records were not rerun or relabeled."}
(public / "PROVENANCE.json").write_text(json.dumps(provenance, indent=2) + "\n", encoding="utf-8", newline="\n")
(public / ".gitattributes").write_text("* -text\n", encoding="utf-8", newline="\n")
(public / ".gitignore").write_text("__pycache__/\n*.py[cod]\n", encoding="utf-8", newline="\n")

entries = []
for source in sorted(public.rglob("*")):
    if source.is_file():
        value = source.read_bytes()
        entries.append(hashlib.sha256(value).hexdigest() + "  " + source.relative_to(public).as_posix())
(public / "SHA256SUMS").write_text("\n".join(entries) + "\n", encoding="utf-8", newline="\n")
assert all(item["original_retained_and_read_back"] for item in hashes)
size = sum(p.stat().st_size for p in public.rglob("*") if p.is_file())
assert size < 16777216
print(json.dumps({"public_files": len(entries) + 1, "bytes": size, "originals_retained": len(hashes)}))
