"""Freeze a finite decoded-JSON boundary corpus before invoking either source."""
import copy
import hashlib
import itertools
import json
from pathlib import Path

root = Path(__file__).resolve().parent
bad = [None, False, 0, 1.0, "", [], {}]
acks = ["ACKED", "PENDING", "UNKNOWN"] + bad
releases = ["NOT_TERMINAL", "VERIFIED_EMPTY", "UNVERIFIED", "FAILED"] + bad
uncertainties = ["NONE", "OS_UNCONFIRMED", "SURFACE_UNKNOWN", "FOCUS_UNKNOWN"] + bad
statuses = ["accepted", "rejected", "failed", "released"]


def state():
    return {"schema": "agent-interface/motor-state-v1", "state_id": "s1", "owner_id": "o1",
            "owner_revision": 1, "observation_id": "obs1", "surface_id": "surface1",
            "coordinate_frame": "screen", "commanded_pointer": {},
            "observed_pointer": {"x": 1, "y": 2}, "held_keys": [], "held_buttons": [],
            "input_ack": {"id": "ack1", "status": "ACKED"},
            "release": {"status": "NOT_TERMINAL", "retained": False},
            "uncertainty": "NONE", "events": [{"type": "RELEASE_TRANSITION"}]}


def context():
    value = state()
    del value["schema"]
    del value["release"]
    return value


cases = []
for ack, release, uncertainty in itertools.product(acks, releases, uncertainties):
    row = state()
    row["input_ack"]["status"] = ack
    row["release"]["status"] = release
    row["uncertainty"] = uncertainty
    cases.append({"api": "validate", "row": row, "group": "enum_product"})
    ctx = context()
    ctx["input_ack"]["status"] = ack
    ctx["uncertainty"] = uncertainty
    cases.append({"api": "bridge", "result": {"status": "accepted", "release": row["release"]},
                  "context": ctx, "group": "enum_product"})
for status in statuses + bad:
    cases.append({"api": "bridge", "result": {"status": status},
                  "context": context(), "group": "dispatch_status"})
for field in ("events", "commanded_pointer", "held_keys", "held_buttons", "input_ack"):
    if field in ("events", "held_keys", "held_buttons"):
        values = [None, False, 0, 1.0, "", {}, {"x": 1}, [], ["x"]]
    else:
        valid = {"id": "ack1", "status": "ACKED"} if field == "input_ack" else {"x": 1}
        values = [None, False, 0, 1.0, "", [], [["id", "ack1"], ["status", "ACKED"]], {}, valid]
    for status, value in itertools.product(statuses, values):
        ctx = context()
        ctx[field] = value
        cases.append({"api": "bridge", "result": {"status": status},
                      "context": ctx, "group": "container_shape"})
controls = [(None, context()), ({"status": "accepted"}, None)]
ctx = context(); del ctx["observation_id"]
controls.append(({"status": "accepted"}, ctx))
ctx = context(); ctx["owner_revision"] = False
controls.append(({"status": "accepted"}, ctx))
ctx = {key: context()[key] for key in ("state_id", "owner_id", "owner_revision", "observation_id", "surface_id", "coordinate_frame")}
controls.append(({"status": "accepted"}, ctx))
controls.append(({"status": "accepted", "release": None}, context()))
controls.append(({"status": "failed", "release": {"status": "FAILED", "retained": True, "error": "timeout"}}, context()))
controls.append(({"status": "accepted", "release": {"status": "NOT_TERMINAL", "retained": 0}}, context()))
for result, ctx in controls:
    cases.append({"api": "bridge", "result": result, "context": ctx, "group": "legacy_control"})
for i, case in enumerate(cases):
    case["case_id"] = f"m{i:04d}"
raw = "".join(json.dumps(case, sort_keys=True, separators=(",", ":")) + "\n" for case in cases).encode()
with (root / "cases.jsonl").open("xb") as target:
    target.write(raw)
assert len(cases) == 2619
freeze = {"cases": len(cases), "cases_sha256": hashlib.sha256(raw).hexdigest(),
          "groups": {name: sum(case["group"] == name for case in cases) for name in sorted({case["group"] for case in cases})},
          "code_sha256": {name: hashlib.sha256((root / name).read_bytes()).hexdigest() for name in ("make_corpus.py", "candidate.py", "audit.py")}}
with (root / "CORPUS_FREEZE.json").open("x", encoding="utf-8", newline="\n") as target:
    json.dump(freeze, target, indent=2); target.write("\n")
print(json.dumps(freeze))
