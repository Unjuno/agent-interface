"""Read-only public receipt boundary; no dispatch/session/backend invocation."""
import copy
import json
from pathlib import Path
import sys
import unittest

sys.path.insert(0, str(Path(sys.argv.pop(1)).resolve()))
from runtime.cli_v1.receipt import receipt_bytes
from runtime.cli_v1.test_receipt_motor_state import ReceiptMotorStateTests, state


class ReceiptMalformedMotorStateTests(unittest.TestCase):
    def test_wrong_enum_types_remain_read_only_validation_evidence(self):
        for field in ("input_ack", "release", "uncertainty"):
            for value in ([], {}, ["ACKED"]):
                with self.subTest(field=field, value=value):
                    motor = state()
                    if field == "uncertainty":
                        motor[field] = value
                    else:
                        motor[field]["status"] = value
                    report = {"status": "completed", "records": [], "motor_state": motor}
                    raw = json.dumps(report).encode()
                    original = copy.deepcopy(report)
                    view = receipt_bytes(raw)
                    self.assertEqual(view["motor_state_validation"],
                                     {"present": True, "accepted": False, "reason": field})
                    self.assertEqual(view["authority"], "none")
                    self.assertEqual(view["source"]["raw_report"], original)
                    self.assertEqual(receipt_bytes(raw, raw=True), original)
                    self.assertEqual(json.loads(raw), original)


if __name__ == "__main__":
    unittest.main()
