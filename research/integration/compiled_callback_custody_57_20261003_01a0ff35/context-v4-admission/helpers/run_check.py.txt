"""Capture actual ordinary child receipts and preserve the first RED."""
from pathlib import Path
import datetime
import hashlib
import json
import subprocess
import sys

here = Path(__file__).resolve().parent
stage, label, *names = sys.argv[1:]
folder = here / "runs" / stage
folder.mkdir(parents=True, exist_ok=False)
argv = [sys.executable, "-I", "-B"] + (["-O"] if "optimized" in stage else []) + [str(here / "probe_source.py"), label, *names]
receipt = {"argv": argv, "cwd": str(here), "kind": "ordinary inert author repair regression",
           "test_sha256": hashlib.sha256((here / "test_compiled_admission_custody_01a0ff35.py").read_bytes()).hexdigest(),
           "source_sha256": hashlib.sha256((here / label / "runtime/core_v1/compiled_gui.py").read_bytes()).hexdigest(),
           "started_utc": datetime.datetime.now(datetime.timezone.utc).isoformat()}
(folder / "START.json").write_text(json.dumps(receipt, indent=2) + "\n", encoding="utf-8")
with (folder / "stdout.txt").open("xb") as out, (folder / "stderr.txt").open("xb") as err:
    child = subprocess.Popen(argv, cwd=here, stdout=out, stderr=err)
    receipt["pid"] = child.pid
    (folder / "START.json").write_text(json.dumps(receipt, indent=2) + "\n", encoding="utf-8")
    receipt["exit_code"] = child.wait(timeout=30)
receipt["ended_utc"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
receipt["streams"] = {name: {"bytes": (folder / name).stat().st_size, "sha256": hashlib.sha256((folder / name).read_bytes()).hexdigest()}
                      for name in ("stdout.txt", "stderr.txt")}
(folder / "RESULT.json").write_text(json.dumps(receipt, indent=2) + "\n", encoding="utf-8")
print(json.dumps(receipt, sort_keys=True))
print((folder / "stderr.txt").read_text(encoding="utf-8"))
sys.exit(receipt["exit_code"])
