assert cwd.is_dir() and stage and argv
for suffix in ('.stdout','.stderr','.receipt.json'):
    assert not (here/(stage+suffix)).exists()
started = datetime.datetime.now(datetime.timezone.utc).isoformat()
child = subprocess.Popen(argv,cwd=cwd,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
try:
    stdout,stderr = child.communicate(timeout=120)
except subprocess.TimeoutExpired:
    child.kill(); stdout,stderr = child.communicate()
    raise

              pid=child.pid,exit_code=child.returncode,

              compiled_sha256=hashlib.sha256((cwd/'runtime/core_v1/compiled_gui.py').read_bytes()).hexdigest(),
              join_tests_sha256=hashlib.sha256((here/'test_compiled_custody_effect_join_01a0ff35.py').read_bytes()).hexdigest(),

              scope='ordinary author integration regression; inert fixtures; no formal allocation')
with (here/(stage+'.receipt.json')).open('x',encoding='utf-8') as f:
    json.dump(record,f,sort_keys=True,indent=2); f.write('\n')
