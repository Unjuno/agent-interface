"""One private, authenticated, loopback-only JupyterLab primary instance."""
import argparse
import hashlib
import importlib.metadata
import json
import os
import re
import secrets
import socket
import subprocess
import time
import urllib.request
from datetime import datetime,timezone
from pathlib import Path

parser=argparse.ArgumentParser()
parser.add_argument('phase')
parser.add_argument('--ttl-seconds',type=int,default=1200)
args=parser.parse_args()
if not re.fullmatch(r'[A-Za-z0-9-]+',args.phase) or not 60<=args.ttl_seconds<=1800:
    raise ValueError('finite owned phase')
root=Path(__file__).parent;run=root/args.phase
run.mkdir()
for name in ['files','runtime','config','data','settings','workspaces']:
    (run/name).mkdir()
if args.phase != 'primary-J01':
    raise ValueError('only the frozen primary namespace')
inputs=json.loads((root/'INPUTS.json').read_bytes())
for case in inputs['cases']:
    with (run/'files'/case['file']).open('xb') as f:f.write(case['initial'].encode('utf-8'))
with (run/'files'/inputs['sibling']['file']).open('xb') as f:f.write(inputs['sibling']['text'].encode('utf-8'))
prefs=run/'settings/@jupyterlab/docmanager-extension'
prefs.mkdir(parents=True)
(prefs/'plugin.jupyterlab-settings').write_bytes((root/'AUTOSAVE_OFF.jupyterlab-settings').read_bytes())
with socket.socket() as s:
    s.bind(('127.0.0.1',0));port=s.getsockname()[1]
token=secrets.token_hex(24)
config=run/'config/jupyter_server_config.py'
config.write_text(
    "c.ServerApp.ip = '127.0.0.1'\n"
    f"c.ServerApp.port = {port}\n"
    "c.ServerApp.port_retries = 0\n"
    "c.ServerApp.open_browser = False\n"
    "c.ServerApp.use_redirect_file = False\n"
    f"c.ServerApp.root_dir = {str(run/'files')!r}\n"
    "c.ServerApp.terminals_enabled = False\n"
    "c.ServerApp.log_level = 'INFO'\n",encoding='utf-8')
env=os.environ.copy()
for key,folder in [
    ('JUPYTER_CONFIG_DIR','config'),('JUPYTER_RUNTIME_DIR','runtime'),('JUPYTER_DATA_DIR','data'),
    ('JUPYTERLAB_SETTINGS_DIR','settings'),('JUPYTERLAB_WORKSPACES_DIR','workspaces')]:
    env[key]=str(run/folder)
env['JUPYTER_TOKEN']=token
python=root/'env/Scripts/python.exe'
command=[str(python),'-m','jupyterlab','--config',str(config)]
start=datetime.now(timezone.utc).isoformat()
stdout=(run/'server.stdout.private.txt').open('xb')
stderr=(run/'server.stderr.private.txt').open('xb')
child=subprocess.Popen(command,cwd=run,env=env,stdout=stdout,stderr=stderr,
    creationflags=subprocess.CREATE_NO_WINDOW if os.name=='nt' else 0)
control=dict(pid=child.pid,port=port,token=token,origin=f'http://127.0.0.1:{port}',
    started_utc=start,run=str(run),command=command)
with (run/'CONTROL_PRIVATE.json').open('x',encoding='utf-8') as f:json.dump(control,f,indent=2)
base=f'http://127.0.0.1:{port}'
headers={'Authorization':'token '+token}
warm=[]
deadline=time.monotonic()+60
ready=False
while time.monotonic()<deadline and child.poll() is None:
    when=datetime.now(timezone.utc).isoformat()
    try:
        req=urllib.request.Request(base+'/api/status',headers=headers)
        with urllib.request.urlopen(req,timeout=2) as response:
            body=response.read()
            warm.append(dict(utc=when,status=response.status,bytes=len(body),
                sha256=hashlib.sha256(body).hexdigest()))
            ready=response.status==200
            if ready:break
    except Exception as e:warm.append(dict(utc=when,error=type(e).__name__))
    time.sleep(.2)
public={k:v for k,v in control.items() if k!='token'}
public.update(dict(ready=ready,readiness=warm,authentication='private random token enabled',
    ttl_seconds=args.ttl_seconds,config_sha256=hashlib.sha256(config.read_bytes()).hexdigest(),
    installed_versions={name:importlib.metadata.version(name)
        for name in ['jupyterlab','jupyter-server','jupyterlab-server','tornado']},
    scope='owned local text editor; no notebook/kernel/terminal/native action; construction startup probes only'))
with (run/'START_PUBLIC.json').open('x',encoding='utf-8') as f:json.dump(public,f,indent=2)
print(json.dumps({k:v for k,v in public.items() if k not in ['readiness','command']}),flush=True)
reason='actual child exit'
if ready:
    try:child.wait(timeout=args.ttl_seconds)
    except subprocess.TimeoutExpired:reason='owned construction TTL'
else:reason='startup readiness failed'
if child.poll() is None:
    try:
        req=urllib.request.Request(base+'/api/shutdown',data=b'',method='POST',headers=headers)
        with urllib.request.urlopen(req,timeout=5) as response:response.read()
        child.wait(timeout=10)
    except Exception:
        child.terminate()
        try:child.wait(timeout=10)
        except subprocess.TimeoutExpired:child.kill();child.wait()
stdout.close();stderr.close()
closed=dict(started_utc=start,ended_utc=datetime.now(timezone.utc).isoformat(),
    pid=child.pid,returncode=child.returncode,reason=reason,ready=ready,launches=1,
    original_logs_private=True,auth_material_not_public=True)
with (run/'CLOSURE.json').open('x',encoding='utf-8') as f:json.dump(closed,f,indent=2)
print(json.dumps(closed),flush=True)
raise SystemExit(0 if ready and child.returncode==0 else 1)
