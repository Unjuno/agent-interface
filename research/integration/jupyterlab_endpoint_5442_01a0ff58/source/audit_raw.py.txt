"""Raw-only semantic oracle. Does not use claimed UI save success as disk truth."""
import base64,hashlib,json,sys
from pathlib import Path
def audit(root,run):
    inputs=json.loads((root/'INPUTS.json').read_bytes())
    if [c['id'] for c in inputs['cases']]!=['J001','J002','J003']:raise ValueError('case coverage')
    initial=json.loads((run/'observations/initial/disk.json').read_bytes())
    def rows(record):
        found={}
        for row in record['rows']:
            if row['file'] in found or row['stable'] is not True:raise ValueError('duplicate/unstable disk')
            data=base64.b64decode(row['content_b64'],validate=True)
            if type(row['bytes']) is not int or row['bytes']!=len(data) or row['sha256']!=hashlib.sha256(data).hexdigest():raise ValueError('byte custody')
            if type(row['mtime_ns']) is not int:raise ValueError('mtime type')
            found[row['file']]=(data,row['mtime_ns'])
        expected={c['file'] for c in inputs['cases']}|{inputs['sibling']['file']}
        if set(found)!=expected:raise ValueError('file coverage')
        return found
    base=rows(initial)
    for c in inputs['cases']:
        if base[c['file']][0]!=c['initial'].encode():raise ValueError('initial body')
    sibling=inputs['sibling'];sibling_bytes=sibling['text'].encode()
    ui=[json.loads(x) for x in (run/'ui.jsonl').read_text(encoding='utf-8').splitlines()]
    if len({x['seq'] for x in ui})!=len(ui):raise ValueError('UI sequence')
    summaries=[]
    for c in inputs['cases']:
        ident=c['id'];name=c['file'];stages={}
        for s in ['pre-save','first-save','final']:
            rec=json.loads((run/'observations'/f'{ident}-{s}'/'disk.json').read_bytes())
            if rec['stage']!=f'{ident}-{s}':raise ValueError('stage binding')
            observed=rows(rec)
            if observed[sibling['file']]!=(sibling_bytes,base[sibling['file']][1]):raise ValueError('sibling changed')
            stages[s]=observed
        external=c['actor']!='none'
        pre=(c['initial']+c['note']).encode()
        if stages['pre-save'][name][0]!=pre:raise ValueError('pre-save state')
        if external:
            actor=json.loads((run/'actors'/ident/'RESULT.json').read_bytes())
            if actor['id']!=ident or actor['file']!=name or type(actor['writes']) is not int or actor['writes']!=1 or actor['mtime_condition'] is not True:raise ValueError('actor binding/mtime')
            if actor['before_sha256']!=hashlib.sha256(c['initial'].encode()).hexdigest() or actor['after_sha256']!=hashlib.sha256(pre).hexdigest():raise ValueError('actor effect')
            if actor['before_mtime_ns']!=base[name][1] or actor['after_mtime_ns']!=stages['pre-save'][name][1]:raise ValueError('actor time join')
        case_ui=[r for r in ui if r['case']==ident]
        saves=[r for r in case_ui if r['action']=='save' and r['phase']=='before']
        if len(saves)!=(2 if external else 1):raise ValueError('save budget/coverage')
        expected_first=pre if external else c['goal'].encode()
        if stages['first-save'][name][0]!=expected_first:raise ValueError('first save semantic effect')
        conflict=[r for r in case_ui if r['phase']=='first-save-observed']
        if len(conflict)!=1 or ('File Changed' in conflict[0]['dom'])!=external:raise ValueError('conflict witness')
        if external:
            reverted=[r for r in case_ui if r['phase']=='revert-observed']
            if len(reverted)!=1 or any(line not in reverted[0]['dom'] for line in (c['initial']+c['note']).splitlines()):raise ValueError('fresh visible rebase')
        if stages['final'][name][0]!=(c['goal']+c['note']).encode():raise ValueError('goal or collateral loss')
        summaries.append(dict(id=ident,goal=True,concurrent_note_preserved=True,manual_save_intents=len(saves),conflict=external))
    final=rows(json.loads((run/'observations/J003-final/disk.json').read_bytes()))
    for c in inputs['cases']:
        if final[c['file']][0]!=(c['goal']+c['note']).encode():raise ValueError('earlier goal lost')
    return dict(decision='PASS_STOCK_FILE_CONFLICT_SCOPED',cases=summaries,scope='three serialized owned standard-app point observations, not arbitrary-app authenticity/physical input/continuous validity/performance')
if __name__=='__main__':
    root=Path(__file__).resolve().parent
    result=audit(root,Path(sys.argv[1]) if len(sys.argv)>1 else root/'primary-J01')
    print(json.dumps(result,ensure_ascii=False,indent=2))
