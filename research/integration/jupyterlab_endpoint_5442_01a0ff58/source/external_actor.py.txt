"""One serialized competing write per frozen external case; no score to controller."""
import hashlib,json,os,sys
from datetime import datetime,timezone
from pathlib import Path
root=Path(__file__).resolve().parent;run=root/'primary-J01'
inputs=json.loads((root/'INPUTS.json').read_bytes())
ident=sys.argv[1]
case=next(c for c in inputs['cases'] if c['id']==ident)
if case['actor']=='none':raise ValueError('no actor in this case')
folder=run/'actors'/ident;folder.mkdir(parents=True)
p=run/'files'/case['file'];before=p.stat();data=p.read_bytes()
if data!=case['initial'].encode():raise ValueError('actor precondition mismatch')
pending=dict(id=ident,file=p.name,actor=case['actor'],started_utc=datetime.now(timezone.utc).isoformat(),before_mtime_ns=before.st_mtime_ns,
    before_sha256=hashlib.sha256(data).hexdigest(),writes_planned=1)
with (folder/'REQUEST.json').open('x',encoding='utf-8') as f:json.dump(pending,f,indent=2)
with p.open('wb') as f:f.write((case['initial']+case['note']).encode())
if case['actor']=='samemtime':os.utime(p,ns=(before.st_atime_ns,before.st_mtime_ns))
after=p.stat();saved=p.read_bytes()
record=dict(**pending,ended_utc=datetime.now(timezone.utc).isoformat(),after_mtime_ns=after.st_mtime_ns,after_sha256=hashlib.sha256(saved).hexdigest(),writes=1,
    mtime_condition=after.st_mtime_ns==before.st_mtime_ns if case['actor']=='samemtime' else after.st_mtime_ns-before.st_mtime_ns>500_000_000)
with (folder/'RESULT.json').open('x',encoding='utf-8') as f:json.dump(record,f,indent=2)
print(json.dumps(dict(id=ident,operation_finished=True,record=str(folder/'RESULT.json'))))
