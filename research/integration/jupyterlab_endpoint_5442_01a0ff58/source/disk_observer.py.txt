"""Independent fresh process, read-only disk observation; no API/controller input."""
import base64, hashlib, json, sys
from datetime import datetime, timezone
from pathlib import Path
root=Path(__file__).resolve().parent
run=root/'primary-J01'
stage=sys.argv[1]
allowed={'initial'}|{c+'-'+s for c in ['J001','J002','J003'] for s in ['pre-save','first-save','final']}
if stage not in allowed:raise ValueError('frozen stage')
dest=run/'observations'/stage
dest.mkdir(parents=True)
start=datetime.now(timezone.utc).isoformat()
rows=[]
inputs=json.loads((root/'INPUTS.json').read_bytes())
names=[c['file'] for c in inputs['cases']]+[inputs['sibling']['file']]
for name in sorted(names):
    p=run/'files'/name
    before=p.stat();data=p.read_bytes();after=p.stat()
    stable=(before.st_size,before.st_mtime_ns)==(after.st_size,after.st_mtime_ns)
    rows.append(dict(file=p.name,bytes=len(data),sha256=hashlib.sha256(data).hexdigest(),
        content_b64=base64.b64encode(data).decode(),mtime_ns=after.st_mtime_ns,
        before_size=before.st_size,before_mtime_ns=before.st_mtime_ns,stable=stable))
aux=[]
for p in sorted((run/'files').rglob('*')):
    relative=p.relative_to(run/'files').as_posix()
    if relative in names:continue
    if not relative.startswith('.ipynb_checkpoints'):raise ValueError('unexpected auxiliary path')
    aux.append(dict(path=relative,directory=p.is_dir(),bytes=p.stat().st_size if p.is_file() else None,sha256=hashlib.sha256(p.read_bytes()).hexdigest() if p.is_file() else None))
out=dict(stage=stage,started_utc=start,ended_utc=datetime.now(timezone.utc).isoformat(),pid=__import__('os').getpid(),rows=rows,stock_checkpoints=aux,
    scope='point observation; stat check cannot prove atomicity against ABA or continuous validity')
with (dest/'disk.json').open('x',encoding='utf-8') as f:json.dump(out,f,indent=2)
print(json.dumps(dict(stage=stage,path=str(dest/'disk.json'),rows=len(rows))))
