"""Synthetic oracle construction checks, not an application experiment."""
import base64,copy,hashlib,json,tempfile,unittest
from pathlib import Path
from audit_raw import audit
ROOT=Path(__file__).resolve().parent
I=json.loads((ROOT/'INPUTS.json').read_bytes())
def fixture(run):
    values={c['file']:(c['initial'],1000) for c in I['cases']}
    values[I['sibling']['file']]=(I['sibling']['text'],1000)
    def snapshot(stage):
        folder=run/'observations'/stage;folder.mkdir(parents=True)
        rows=[]
        for name,(value,mtime) in values.items():
            b=value.encode();rows.append(dict(file=name,bytes=len(b),content_b64=base64.b64encode(b).decode(),sha256=hashlib.sha256(b).hexdigest(),mtime_ns=mtime,stable=True))
        (folder/'disk.json').write_text(json.dumps(dict(stage=stage,rows=rows)),encoding='utf-8')
    snapshot('initial');ui=[]
    def event(c,phase,action='observe',dom=''):
        ui.append(dict(seq=len(ui)+1,case=c['id'],phase=phase,action=action,dom=dom))
    for c in I['cases']:
        ident=c['id'];name=c['file'];external=c['actor']!='none'
        if external:
            mtime=600001000 if c['actor']=='newmtime' else 1000
            values[name]=(c['initial']+c['note'],mtime)
            folder=run/'actors'/ident;folder.mkdir(parents=True)
            (folder/'RESULT.json').write_text(json.dumps(dict(id=ident,file=name,writes=1,mtime_condition=True,before_mtime_ns=1000,after_mtime_ns=mtime,before_sha256=hashlib.sha256(c['initial'].encode()).hexdigest(),after_sha256=hashlib.sha256((c['initial']+c['note']).encode()).hexdigest())),encoding='utf-8')
        snapshot(ident+'-pre-save');event(c,'before','save')
        if not external:values[name]=(c['goal'],2000)
        event(c,'first-save-observed',dom='File Changed' if external else 'saved')
        snapshot(ident+'-first-save')
        if external:
            event(c,'revert-observed',dom=c['initial']+c['note']);event(c,'before','save')
            values[name]=(c['goal']+c['note'],3000)
        snapshot(ident+'-final')
    (run/'ui.jsonl').write_text('\n'.join(json.dumps(r) for r in ui)+'\n',encoding='utf-8')
def mutate(path,fn):
    v=json.loads(path.read_bytes());fn(v);path.write_text(json.dumps(v),encoding='utf-8')
class RawChecks(unittest.TestCase):
    def test_positive(self):
        with tempfile.TemporaryDirectory(prefix='jup-audit-',dir=ROOT) as t:
            run=Path(t);fixture(run);self.assertEqual(audit(ROOT,run)['decision'],'PASS_STOCK_FILE_CONFLICT_SCOPED')
    def test_six_semantic_corruptions(self):
        controls=['goal','note','sibling','duplicate','bytehash','boolwrites']
        for control in controls:
            with self.subTest(control=control),tempfile.TemporaryDirectory(prefix='jup-audit-',dir=ROOT) as t:
                run=Path(t);fixture(run)
                if control=='boolwrites':mutate(run/'actors/J003/RESULT.json',lambda v:v.update(writes=True))
                else:
                    path=run/'observations/J003-final/disk.json'
                    def change(v):
                        if control=='duplicate':v['rows'].append(copy.deepcopy(v['rows'][0]));return
                        target=I['sibling']['file'] if control=='sibling' else I['cases'][2]['file']
                        row=next(r for r in v['rows'] if r['file']==target)
                        if control=='bytehash':row['sha256']='0'*64;return
                        body=base64.b64decode(row['content_b64']).decode()
                        body=body.replace('14:00','10:30') if control=='goal' else body.replace(I['cases'][2]['note'],'') if control=='note' else body+'changed'
                        b=body.encode();row.update(content_b64=base64.b64encode(b).decode(),sha256=hashlib.sha256(b).hexdigest(),bytes=len(b))
                    mutate(path,change)
                with self.assertRaises(ValueError):audit(ROOT,run)
if __name__=='__main__':unittest.main(verbosity=2)
