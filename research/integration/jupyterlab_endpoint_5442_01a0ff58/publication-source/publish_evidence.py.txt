def entries(tree):
    values={}
    for item in git('ls-tree','-z',tree).split(b'\0'):
        if not item:continue

    return values
def mktree(values):
    b=b''.join((mode+' '+kind+' '+oid+'\t'+name).encode()+b'\0' for name,(mode,kind,oid) in sorted(values.items()))
    return git('mktree','-z',input=b).decode().strip()

    values={}

    return mktree(values)
