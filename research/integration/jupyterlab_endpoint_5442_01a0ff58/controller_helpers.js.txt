// Exact helper copied into CUA; all task decisions use the returned visible DOM.
var jpFs = await import('node:fs/promises');
var jpRoot = 'C:/Users/user/Documents/Codex/2026-10-03/new-chat/work/jupyterlab-5442/primary-J01';
var jpSeq = 0;
async function jpRecord(caseId, phase, action = 'observe', dom = '') {
  var row = {seq: ++jpSeq, utc: new Date().toISOString(), case: caseId, phase, action, dom};
  await jpFs.appendFile(jpRoot + '/ui.jsonl', JSON.stringify(row) + '\n');
  return row;
}
async function jpObserve(caseId, phase) {
  var dom = await jpTab.playwright.domSnapshot();
  await jpRecord(caseId, phase, 'observe', dom);
  nodeRepl.write(dom);
  return dom;
}
async function jpShot(caseId, phase) {
  await jpFs.writeFile(jpRoot + '/' + caseId + '-' + phase + '.jpg', await jpTab.screenshot({fullPage:false}));
}
