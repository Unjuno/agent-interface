"""Each case removes one real saved-provenance guarantee; no producer or mock."""
import argparse
import copy
import importlib.util
import json
import sys
import unittest
from pathlib import Path

p=argparse.ArgumentParser(); p.add_argument('--source',required=True); p.add_argument('--run',required=True); p.add_argument('--cases',required=True)
a=p.parse_args(); spec=importlib.util.spec_from_file_location('retained_audit',a.source)
audit=importlib.util.module_from_spec(spec); spec.loader.exec_module(audit)
data=audit.read(Path(a.run),json.loads(Path(a.cases).read_bytes()))

class SavedProvenance(unittest.TestCase):
    def rejected(self,change):
        copied=copy.deepcopy(data); change(copied)
        with self.assertRaises(ValueError): audit.audit(copied)

    def test_real_complete_custody_preserves_eight_goals_and_two_yields(self):
        result=audit.audit(copy.deepcopy(data))
        self.assertEqual((result['task_sends'],result['external_sends'],result['final_goal'],result['yield_budget'],result['ui_stages']),(20,16,8,2,148))

    def test_duplicate_fixture_cannot_replace_another_disturbance(self):
        self.rejected(lambda d:d['fixtures'].__setitem__(1,copy.deepcopy(d['fixtures'][0])))

    def test_sixteen_copies_cannot_claim_sixteen_external_sends(self):
        self.rejected(lambda d:d.update(fixtures=[copy.deepcopy(d['fixtures'][0]) for _ in range(16)]))

    def test_missing_external_receipt_is_not_complete_custody(self):
        self.rejected(lambda d:d['fixtures'].pop())

    def test_true_is_not_a_one_send_attempt_count(self):
        self.rejected(lambda d:d['fixtures'][0]['intent'].update(attempts=True))

    def test_declared_external_route_must_match_recorded_request(self):
        self.rejected(lambda d:d['fixtures'][0]['intent'].update(route='/apply'))

    def test_response_operation_identity_must_match_actual_external_receipt(self):
        self.rejected(lambda d:d['fixtures'][0]['response'].update(opid='other'))

    def test_missing_send_start_cannot_be_hidden_by_reindexing(self):
        def change(d):
            rows=d['ui']; rows.pop(next(i for i,r in enumerate(rows) if r['kind']=='send_started'))
            for i,r in enumerate(rows): r['index']=i
        self.rejected(change)

    def test_duplicate_send_start_cannot_be_hidden_by_reindexing(self):
        def change(d):
            rows=d['ui']; index=next(i for i,r in enumerate(rows) if r['kind']=='send_started')
            rows.insert(index,copy.deepcopy(rows[index]))
            for i,r in enumerate(rows): r['index']=i
        self.rejected(change)

    def test_send_nonce_must_equal_saved_pre_send_intent(self):
        self.rejected(lambda d:next(r for r in d['ui'] if r['kind']=='send_started')['fields'].update(nonce='other'))

    def test_inherited_ten_saved_field_corruptions_still_reject(self):
        controls=audit.controls(copy.deepcopy(data))
        self.assertEqual(len(controls),10)
        self.assertTrue(all(c['rejected'] for c in controls))

if __name__=='__main__': unittest.main(argv=[sys.argv[0]],verbosity=2)
