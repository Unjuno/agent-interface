"""Serialized independent actor; explicit intent before one HTTP send, no retries."""
import argparse
import json
from datetime import datetime, timezone
from pathlib import Path
from urllib.request import Request, urlopen

def main():
    p=argparse.ArgumentParser(); p.add_argument('--run',required=True); p.add_argument('--cases',required=True)
    p.add_argument('--phase',choices=['postinitial','repair1','repair2','shutdown'],required=True)
    p.add_argument('--case'); a=p.parse_args()
    run=Path(a.run); ready=json.loads((run/'READY.json').read_bytes()); cases=json.loads(Path(a.cases).read_bytes())
    out=run/('fixture-'+a.phase+('-'+a.case if a.case else '')); out.mkdir(exist_ok=False)
    operations=[]
    for c in cases['cases']:
        if a.case and c['case_id']!=a.case: continue
        values=[]
        if a.phase=='postinitial' and c['history']!='stable':
            values=['external postinitial '+c['case_id']]
            if c['history']=='aba_restore': values.append(c['goal'])
        elif a.phase=='repair1' and c['history'] in {'one_race','persistent_race'}:
            values=['external after form1 '+c['case_id']]
        elif a.phase=='repair2' and c['history']=='persistent_race':
            values=['external after form2 '+c['case_id']]
        for i,value in enumerate(values):
            identity='fixture.'+c['case_id']+'.'+a.phase+'.'+str(i)
            operations.append(dict(namespace=cases['namespace'],case_id=c['case_id'],target='doc:A',
                opid=identity,nonce=identity+'.nonce',value=value,goal_hash=c['goal_hash']))
    if a.phase=='shutdown': operations=[None]
    results=[]
    for i,item in enumerate(operations):
        route='/shutdown' if item is None else '/disturb'
        body=b'' if item is None else json.dumps(item,ensure_ascii=False).encode()
        intent=dict(utc=datetime.now(timezone.utc).isoformat(),route=route,body=body.decode(),attempts=1)
        (out/(str(i)+'-intent.json')).write_text(json.dumps(intent,ensure_ascii=False,indent=2),encoding='utf-8')
        try:
            with urlopen(Request('http://127.0.0.1:'+str(ready['port'])+route,data=body,
                headers={'Content-Type':'application/json'}),timeout=5) as response:
                result=dict(utc=datetime.now(timezone.utc).isoformat(),status=response.status,
                            body=response.read().decode(),opid=item['opid'] if item else 'shutdown')
        except BaseException as error:
            (out/(str(i)+'-ERROR.json')).write_text(json.dumps(dict(type=type(error).__name__,error=str(error))),encoding='utf-8')
            raise
        (out/(str(i)+'-response.json')).write_text(json.dumps(result,ensure_ascii=False,indent=2),encoding='utf-8')
        if result['status']!=200: raise ValueError('fixture response not accepted; no resend')
        results.append(result)
    print(json.dumps(dict(phase=a.phase,case=a.case,sends=len(results),status='recorded')))

if __name__=='__main__': main()
