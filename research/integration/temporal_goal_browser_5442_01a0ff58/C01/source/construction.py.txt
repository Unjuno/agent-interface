"""Ordinary pre-freeze guard checks on a separate private database."""
import copy
import json
import sqlite3
from datetime import datetime, timezone
from pathlib import Path
from app import Store

out=Path(__file__).parent/'construction-01'
out.mkdir(exist_ok=False)
start=datetime.now(timezone.utc).isoformat()
store=Store(out/'guard.sqlite','construction-not-C01')
store.create(['X'])
base=dict(namespace=store.namespace,case_id='X',target='doc:A',opid='initial',nonce='initial',
          value='日本語の目標',goal_hash='construction',expected_version=0)
checks=[]
def need(condition,label):
    if not condition: raise AssertionError(label)
    checks.append(label)
first=store.mutate('task',base)
need(first['outcome']=='APPLIED' and first['after_version']==1,'fresh version applies')
stale=base|dict(opid='stale',nonce='stale',value='wrong')
second=store.mutate('task',stale)
need(second['outcome']=='CONFLICT' and second['after_value']=='日本語の目標' and second['after_version']==1,'stale guarded write has zero target change')
for label,item in [('bool version',base|dict(opid='bool',nonce='bool',expected_version=True)),
                   ('wrong namespace',base|dict(namespace='foreign',opid='foreign',nonce='foreign')),
                   ('duplicate opid',base|dict(nonce='new')),
                   ('duplicate nonce',base|dict(opid='new')),
                   ('extra field',base|dict(extra='x')),
                   ('unknown target',base|dict(target='doc:C',opid='c',nonce='c'))]:
    try: store.mutate('task',item)
    except ValueError: checks.append(label+' rejected')
    else: raise AssertionError(label)
with sqlite3.connect(store.path) as db:
    rows=db.execute('SELECT target,value,version FROM docs ORDER BY target').fetchall()
    need(rows==[('doc:A','日本語の目標',1),('doc:B','untouched X',0)],'independent SQL target and sibling exact')
    need(db.execute('SELECT COUNT(*) FROM events').fetchone()[0]==2,'rejections add no events')
# Independent finite expected state/decision table, before actual browser construction.
golden={'stable':(1,0,True),'overwrite':(3,1,True),'one_race':(4,2,True),
        'persistent_race':(4,2,False),'aba_restore':(3,0,True)}
for history,(version,repairs,satisfied) in golden.items():
    v=1; value='goal'; attempts=0
    if history!='stable': v+=1; value='other'
    if history=='aba_restore': v+=1; value='goal'
    while value!='goal' and attempts<2:
        observed=v; attempts+=1
        if history=='persistent_race' or (history=='one_race' and attempts==1): v+=1; value='other'
        if observed==v: v+=1; value='goal'
    need((v,attempts,value=='goal')==(version,repairs,satisfied),'finite '+history)
result=dict(started_utc=start,ended_utc=datetime.now(timezone.utc).isoformat(),checks=checks,
            status='PASS',database='guard.sqlite',scope='ordinary separate construction, no C01 allocation')
(out/'RESULT.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
print(json.dumps(dict(status=result['status'],checks=len(checks))))
