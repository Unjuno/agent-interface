import hashlib
import json
from pathlib import Path

root = Path(__file__).resolve().parent
namespace = 'temporal-goal-5442-01a0ff58-C01'
strings = ['来週の設計レビューは火曜日の午後','備品リスト：鉛筆、紙、青い封筒']
histories = ['stable','overwrite','one_race','persistent_race','aba_restore']
cases = []
for group,goal in enumerate(strings):
    order = histories if group == 0 else list(reversed(histories))
    for history in order:
        identifier = 'C'+str(len(cases)+1).zfill(3)
        intent = dict(namespace=namespace,case_id=identifier,target='doc:A',goal=goal)
        goal_hash = hashlib.sha256(json.dumps(intent,sort_keys=True,separators=(',',':'),ensure_ascii=True).encode()).hexdigest()
        cases.append(dict(case_id=identifier,goal=goal,history=history,target='doc:A',goal_hash=goal_hash,
            initial_value='initial '+identifier,sibling_initial='untouched '+identifier,
            operations={phase:dict(opid=identifier+'.'+phase,nonce='C01.'+identifier+'.'+phase+'.nonce')
                        for phase in ('initial','repair1','repair2')}))
path = root/'cases.json'
with path.open('x',encoding='utf-8',newline='\n') as handle:
    json.dump(dict(namespace=namespace,cases=cases,max_repair_attempts=2),handle,ensure_ascii=False,indent=2)
print(json.dumps(dict(cases=len(cases),sha256=hashlib.sha256(path.read_bytes()).hexdigest())))
