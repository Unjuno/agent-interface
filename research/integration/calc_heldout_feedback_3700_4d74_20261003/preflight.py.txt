"""No task input: public admission/keymap preflight on a private Xvfb."""
import json,pathlib,subprocess,time
from Xlib import X,display
from runtime.backends.x11_v1.backend import X11Backend
from runtime.core_v1.contract import admit_program
root=pathlib.Path('/src');out=pathlib.Path('/out');plan=json.loads((root/'PLAN.json').read_text());log=(out/'Xvfb.log').open('wb')
p=subprocess.Popen(['Xvfb',':143','-screen','0','1280x800x24','-nolisten','tcp','-ac'],stdout=log,stderr=subprocess.STDOUT);d=b=None;rows=[]
try:
    deadline=time.monotonic()+3
    while d is None:
        try:d=display.Display(':143')
        except Exception:
            if time.monotonic()>deadline:raise
            time.sleep(.02)
    w=d.screen().root.create_window(1,1,100,100,0,d.screen().root_depth,X.InputOutput,X.CopyFromParent);w.map();d.sync();b=X11Backend(':143',{'owned':w.id})
    for spec in plan['rows']:
        program=json.loads(json.dumps(spec['program']));program['authority']['expires_at_ns']=time.monotonic_ns()+10_000_000_000
        a=admit_program(program,b.manifest(),now_ns=time.monotonic_ns(),current_observation_seq=1,current_binding_revision=1)
        if not a.accepted:raise RuntimeError('public admission refused: '+str(a.error))
        b.preflight(program);rows.append(dict(index=spec['index'],admission=True,backend_preflight=True,ops=len(program['ops'])))
    if b.emissions:raise RuntimeError('unexpected task emission')
finally:
    if b:b.close()
    if d:d.close()
    p.terminate();code=p.wait(3);log.close()
record=dict(rows=rows,task_inputs=0,formal_rows=0,Xvfb_exit=code)
(out/'PREFLIGHT.json').write_text(json.dumps(record,indent=2)+'\n');print(json.dumps(record))
