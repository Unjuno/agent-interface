import copy,json,pathlib,unittest
from auditor import saved_effect,verify
class Oracle(unittest.TestCase):
    def test_scalar_saved_effect(self):
        b=b'<office:document xmlns:office="urn:oasis:names:tc:opendocument:xmlns:office:1.0" xmlns:table="urn:oasis:names:tc:opendocument:xmlns:table:1.0"><table:table><table:table-row><table:table-cell/></table:table-row><table:table-row><table:table-cell office:value="29"/><table:table-cell office:value="31"/><table:table-cell office:value="899" table:formula="of:=[.A2]*[.B2]"/></table:table-row></table:table></office:document>'
        self.assertEqual(saved_effect(b),dict(a='29',b='31',result='899',formula='of:=[.A2]*[.B2]'))
    def test_repeated_cells_do_not_shift_effect(self):
        b=b'<office:document xmlns:office="urn:oasis:names:tc:opendocument:xmlns:office:1.0" xmlns:table="urn:oasis:names:tc:opendocument:xmlns:table:1.0"><table:table><table:table-row><table:table-cell/></table:table-row><table:table-row><table:table-cell office:value="29" table:number-columns-repeated="2"/><table:table-cell office:value="841" table:formula="of:=[.A2]*[.B2]"/></table:table-row></table:table></office:document>'
        self.assertEqual(saved_effect(b)['b'],'29');self.assertEqual(saved_effect(b)['result'],'841')
    def test_hand_complete_row_and_corruptions(self):
        p=json.loads((pathlib.Path(__file__).parent/'PLAN.json').read_text());s=p['rows'][0]
        image=dict(artifact=dict(sha256='hand-image'))
        program=copy.deepcopy(s['program']);program['authority']['expires_at_ns']=1000
        raw=dict(spec=s,source_hashes=p['source_hashes'],errors=[],program=program,dispatch_end_ns=900,receipt=dict(status='completed',admission='accepted',recovery_required=False,execution=dict(completed_ops=list(range(30)),observations=[image],waits=[dict(kind='fixed_delay',update_observed=None,requested_ms=50,completed=True)],releases=[dict(verified=True,keys_down=[],buttons_down=[])])),final_image=image,after_dispatch_physical=dict(keys=[],buttons=0),cleanup_physical=dict(keys=[],buttons=0),terminal=[dict(exit_code=c) for c in (255,0,0)],events=[dict(kind='primary_request',stage=x) for x in ('initial-review','final-review')],extra_observations=0,primary_first=dict(confirmed=True,values=[29,31,899],artifact_sha256='hand-image'))
        effect=dict(a='29',b='31',result='899',formula='of:=[.A2]*[.B2]')
        self.assertEqual(verify(p,raw,effect),[])
        changes=[lambda d:d['source_hashes'].update(bad='x'),lambda d:d['primary_first'].update(values=[29,31,900]),lambda d:d['primary_first'].update(artifact_sha256='other'),lambda d:d['after_dispatch_physical'].update(keys=[74]),lambda d:d['receipt']['execution']['completed_ops'].pop(),lambda d:d.update(extra_observations=2),lambda d:d['receipt']['execution']['waits'][0].update(update_observed=True),lambda d:d['terminal'].pop()]
        for i,change in enumerate(changes):
            with self.subTest(corruption=i):
                bad=copy.deepcopy(raw);change(bad);self.assertTrue(verify(p,bad,effect))
        wrong=dict(effect,result='900');self.assertTrue(verify(p,raw,wrong))
    def test_missing_raw_refuses(self):
        p=json.loads((pathlib.Path(__file__).parent/'PLAN.json').read_text());self.assertTrue(verify(p,{},{}))
if __name__=='__main__':unittest.main()
