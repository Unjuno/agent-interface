"""Saved-document/raw oracle. Never imports native runner or emits input."""
import hashlib,json,pathlib,sys,xml.etree.ElementTree as ET
NS={'table':'urn:oasis:names:tc:opendocument:xmlns:table:1.0','office':'urn:oasis:names:tc:opendocument:xmlns:office:1.0'}
def cells(blob):
    root=ET.fromstring(blob);table=root.find('.//table:table',NS)
    matrix=[]
    for row in table.findall('table:table-row',NS):
        entries=[]
        for cell in list(row):
            if cell.tag not in ('{'+NS['table']+'}table-cell','{'+NS['table']+'}covered-table-cell'):continue
            n=int(cell.get('{'+NS['table']+'}number-columns-repeated','1'));entries.extend([cell]*min(n,5))
        n=int(row.get('{'+NS['table']+'}number-rows-repeated','1'));matrix.extend([entries]*min(n,3))
        if len(matrix)>=3:break
    return matrix
def saved_effect(blob):
    m=cells(blob);row=m[1]
    return dict(a=row[0].get('{'+NS['office']+'}value'),b=row[1].get('{'+NS['office']+'}value'),result=row[2].get('{'+NS['office']+'}value'),formula=row[2].get('{'+NS['table']+'}formula'))
def verify(plan,raw,effect):
    errors=[]
    def need(v,m):
        if not v:errors.append(m)
    try:
        spec=raw['spec'];expected=plan['rows'][spec['index']]
        need(spec==expected,'row identity')
        need(raw['source_hashes']==plan['source_hashes'],'source pins')
        need(not raw['errors'],'producer errors')
        need(effect==dict(a=str(spec['a']),b=str(spec['b']),result=str(spec['a']*spec['b']),formula='of:=[.A2]*[.B2]'),'saved task effect')
        program=raw['program'];template=spec['program']
        need(program['ops']==template['ops'] and program['source']==template['source'] and program['program_id']==template['program_id'],'exact input program')
        need(program['authority']['expires_at_ns']>raw['dispatch_end_ns'],'lease during dispatch')
        receipt=raw['receipt'];execution=receipt['execution']
        need(receipt['status']=='completed' and receipt['admission']=='accepted' and not receipt['recovery_required'],'public admission/completion')
        need(execution['completed_ops']==list(range(len(program['ops']))),'complete operation list')
        need(len(execution['observations'])==1 and raw['final_image']==execution['observations'][0],'exact final capture/no hidden dispatch capture')
        wait=execution['waits'][-1]
        need(wait['kind']=='fixed_delay' and wait['update_observed'] is None and wait['requested_ms']==spec['wait_ms'] and wait['completed'],'final wait contract')
        need(all(r['verified'] is True and r['keys_down']==[] and r['buttons_down']==[] for r in execution['releases']),'public release')
        need(raw['after_dispatch_physical']['keys']==[] and raw['after_dispatch_physical']['buttons']==0 and raw['cleanup_physical']['keys']==[] and raw['cleanup_physical']['buttons']==0,'independent neutral')
        need(len(raw['terminal'])==3 and all(t['exit_code'] in (0,255) for t in raw['terminal']),'helper parents terminal')
        requests=[e for e in raw['events'] if e['kind']=='primary_request'];stages=[e['stage'] for e in requests]
        need(stages==(['initial-review','final-review'] if raw['extra_observations']==0 else ['initial-review','final-review','extra-review']),'exact review stages')
        need(raw['extra_observations'] in (0,1),'observation bound')
        decision=raw['primary_first'] if raw['extra_observations']==0 else raw['primary_extra']
        image=raw['final_image'] if raw['extra_observations']==0 else raw['extra']['image']
        need(decision['artifact_sha256']==image['artifact']['sha256'],'decision image identity')
        if decision.get('confirmed') is True:need(decision.get('values')==[spec['a'],spec['b'],spec['a']*spec['b']],'no false primary completion')
        else:errors.append('PRIMARY_UNCONFIRMED_AFTER_ALLOWED_OBSERVATION')
    except (KeyError,IndexError,TypeError,ValueError) as e:errors.append('malformed/missing: '+repr(e))
    return errors
if __name__=='__main__':
    src,data,out=map(pathlib.Path,sys.argv[1:4]);plan=json.loads((src/'PLAN.json').read_text());reports=[];errors=[]
    for spec in plan['rows']:
        d=data/('row'+str(spec['index']+1))
        if not (d/'raw.json').exists():errors.append('missing row '+str(spec['index']));continue
        raw=json.loads((d/'raw.json').read_text());effect=saved_effect((d/'heldout-comparison.fods').read_bytes());bad=verify(plan,raw,effect)
        for label in ['initial','final_image']+(['extra'] if 'extra' in raw else []):
            image=raw[label]['image'] if label in ('initial','extra') else raw[label]
            artifact=image['artifact'];p=d/'images'/pathlib.Path(artifact['path']).name
            if hashlib.sha256(p.read_bytes()).hexdigest()!=artifact['sha256']:bad.append('image byte mismatch '+label)
            if artifact['width']!=1280 or artifact['height']!=800 or artifact['source_raw_sha256']!=image['sha256']:bad.append('capture identity '+label)
        reports.append(dict(index=spec['index'],saved_effect=effect,extra=raw.get('extra_observations'),errors=bad,dispatch_ms=(raw['dispatch_end_ns']-raw['dispatch_start_ns'])/1e6))
        errors.extend(bad)
    signal=not errors and len(reports)==4 and reports[1]['extra']<reports[0]['extra'] and reports[2]['extra']<reports[3]['extra']
    result=dict(disposition='H_SCOPED_OBSERVATION_SIGNAL' if signal else 'HOLD_NO_HELDOUT_SIGNAL_OR_INCOMPLETE',errors=errors,rows=reports,adoption='HOLD_PROVIDER_HOST_ENDPOINT_USAGE_AND_GENERALIZATION_UNKNOWN')
    out.write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result));sys.exit(1 if errors else 0)
