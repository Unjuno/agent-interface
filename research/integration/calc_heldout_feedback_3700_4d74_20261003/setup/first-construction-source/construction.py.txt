"""One zero-task-input Calc readiness/PNG construction; no formal wait arms."""
import hashlib,json,os,pathlib,signal,subprocess,sys,time,traceback
from Xlib import X,display
from runtime.backends.x11_v1.backend import X11Backend
ROOT=pathlib.Path('/src');OUT=pathlib.Path('/out')
def now():return time.monotonic_ns()
def write(name,value):(OUT/name).write_text(json.dumps(value,indent=2,sort_keys=True)+'\n')
result=dict(schema='calc-readiness-construction-v1',task_inputs=0,formal_rows=0,started_ns=now(),errors=[],processes=[])
doc=OUT/'heldout-construction.fods'
doc.write_text('''<?xml version="1.0" encoding="UTF-8"?>
<office:document xmlns:office="urn:oasis:names:tc:opendocument:xmlns:office:1.0" xmlns:table="urn:oasis:names:tc:opendocument:xmlns:table:1.0" xmlns:text="urn:oasis:names:tc:opendocument:xmlns:text:1.0" office:version="1.2" office:mimetype="application/vnd.oasis.opendocument.spreadsheet"><office:body><office:spreadsheet><table:table table:name="Sheet1"><table:table-row><table:table-cell office:value-type="string"><text:p>a</text:p></table:table-cell><table:table-cell office:value-type="string"><text:p>b</text:p></table:table-cell><table:table-cell office:value-type="string"><text:p>product</text:p></table:table-cell></table:table-row><table:table-row table:number-rows-repeated="20"><table:table-cell table:number-columns-repeated="20"/></table:table-row></table:table></office:spreadsheet></office:body></office:document>''')
result['document_initial_sha256']=hashlib.sha256(doc.read_bytes()).hexdigest()
actors=[];handles=[];d=b=None
def start(name,args,env):
    h=(OUT/(name+'.log')).open('wb');handles.append(h)
    p=subprocess.Popen(args,env=env,stdout=h,stderr=subprocess.STDOUT,start_new_session=True)
    actors.append((name,p));result['processes'].append(dict(name=name,pid=p.pid,argv=args));return p
try:
    env=dict(os.environ,DISPLAY=':141',HOME='/out/home',XDG_CONFIG_HOME='/out/home/config',SAL_USE_VCLPLUGIN='gen')
    pathlib.Path(env['HOME']).mkdir(exist_ok=True)
    xvfb=start('Xvfb',['Xvfb',':141','-screen','0','1280x800x24','-nolisten','tcp','-ac'],env)
    deadline=now()+3_000_000_000
    while d is None:
        try:d=display.Display(':141')
        except Exception:
            if now()>deadline or xvfb.poll() is not None:raise
            time.sleep(.02)
    start('openbox',['openbox','--sm-disable'],env)
    time.sleep(.2)
    lo=start('Calc',['libreoffice','-env:UserInstallation=file:///out/profile','--calc','--nologo','--norestore','--nodefault','--nofirststartwizard',str(doc)],env)
    deadline=now()+20_000_000_000;found=[]
    def windows(w,depth=0):
        rows=[]
        if depth>5:return rows
        for c in w.query_tree().children:
            try:
                name=c.get_wm_name();attrs=c.get_attributes()
                rows.append(dict(id=c.id,title=str(name),map_state=attrs.map_state))
                rows.extend(windows(c,depth+1))
            except Exception:pass
        return rows
    while not found:
        rows=windows(d.screen().root)
        found=[w for w in rows if 'heldout-construction.fods' in w['title'] and w['map_state']==X.IsViewable]
        if len(found)>1:raise RuntimeError('ambiguous Calc document windows')
        if now()>deadline or lo.poll() is not None:raise RuntimeError('Calc readiness failed')
        if not found:time.sleep(.05)
    result['windows']=rows;result['target']=found[0];result['ready_ns']=now()
    time.sleep(1)
    b=X11Backend(':141',{'owned':found[0]['id']});b.configure_capture_artifacts(OUT/'images')
    result['capture']=b.observe_read_only('owned','screen_physical_px',[0,0,1280,800])
    result['backend_emissions']=b.emissions
    result['release_receipt']=b.release_all()
    bits=d.query_keymap();mask=d.screen().root.query_pointer().mask
    result['keys_down']=[k for k in range(256) if bits[k//8]&(1<<(k%8))]
    result['buttons']=int(mask)&(X.Button1Mask|X.Button2Mask|X.Button3Mask)
    result['document_after_sha256']=hashlib.sha256(doc.read_bytes()).hexdigest()
    result['packages_sha256']=hashlib.sha256(pathlib.Path('/opt/calc-package-pins.tsv').read_bytes()).hexdigest()
    (OUT/'package-pins.tsv').write_bytes(pathlib.Path('/opt/calc-package-pins.tsv').read_bytes())
    if 'artifact' not in result['capture'] or result['backend_emissions'] or result['keys_down'] or result['buttons'] or result['document_initial_sha256']!=result['document_after_sha256']:raise RuntimeError('readiness/capture/zero-input integrity')
except Exception:result['errors'].append(traceback.format_exc())
finally:
    if b:b.close()
    if d:d.close()
    for name,p in reversed(actors):
        try:
            if p.poll() is None:os.killpg(p.pid,signal.SIGTERM)
            code=p.wait(timeout=5)
        except subprocess.TimeoutExpired:
            os.killpg(p.pid,signal.SIGKILL);code=p.wait(timeout=3);result['errors'].append(name+' required SIGKILL')
        result.setdefault('terminal',[]).append(dict(name=name,pid=p.pid,exit_code=code,cleanup='process group TERM; parent waited'))
    for h in handles:h.close()
    result['ended_ns']=now();write('CONSTRUCTION.json',result)
print(json.dumps(dict(errors=result['errors'],capture=result.get('capture'),task_inputs=0,formal_rows=0)))
sys.exit(2 if result['errors'] else 0)
