from pathlib import Path
import subprocess,json,datetime
O=Path(__file__).resolve().parent;R=O.parent.parent/'work/effect-boundary'
p=subprocess.run(['powershell','-NoProfile','-Command','Get-Process -Id 37608 -ErrorAction SilentlyContinue | Select-Object Id,ProcessName,StartTime | ConvertTo-Json -Compress'],capture_output=True)
g=subprocess.run(['git','-c','gc.auto=0','-c','maintenance.auto=false','status','--porcelain','--untracked-files=no'],cwd=R,capture_output=True)

(O/'FINISH-STATE.json').write_text(json.dumps(r,indent=2)+'\n',encoding='utf-8');print(json.dumps(r))
