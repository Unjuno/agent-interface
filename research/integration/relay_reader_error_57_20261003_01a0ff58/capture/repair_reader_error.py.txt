from pathlib import Path
import hashlib, json
root = Path(__file__).resolve().parent / 'relay-output-read-error'
path = root / 'relay_client.mjs'
old = path.read_bytes()
needle = b'  const reader = createInterface({ input: child.stdout, crlfDelay: Infinity });\n'
if old.count(needle) != 1:
    raise ValueError('one exact baseline reader required')
path.write_bytes(old.replace(needle, needle + b"  reader.on('error', fail);\n"))
capture = root.parent / 'capture_reader_checks.py'
first = capture.read_bytes()
fixed = first.decode().replace("print(stdout.decode(errors='replace')[-14000:])", "print(json.dumps({'saved_node_stdout_tail': stdout.decode(errors='replace')[-3500:]}, ensure_ascii=True))")
with (root.parent / 'capture_reader_checks_v2.py').open('x', encoding='utf-8') as f:
    f.write(fixed)
record = {'runtime_change': "reader.on('error', fail);", 'original_sha256': hashlib.sha256(old).hexdigest(),
          'candidate_sha256': hashlib.sha256(path.read_bytes()).hexdigest(),
          'capture_v1_sha256': hashlib.sha256(first).hexdigest(),
          'capture_v1_diagnostic_failure': 'Console cp932 could not print Node reporter U+2716 after complete raw/log/actual-exit receipt saved. Node first RED remained3fail/1pass; no experiment replay.',
          'regressions_unchanged_sha256': hashlib.sha256((root / 'test_relay_client.mjs').read_bytes()).hexdigest()}
with (root / 'REPAIR.json').open('x', encoding='utf-8') as f:
    json.dump(record, f, indent=2); f.write('\n')
print(json.dumps(record))
