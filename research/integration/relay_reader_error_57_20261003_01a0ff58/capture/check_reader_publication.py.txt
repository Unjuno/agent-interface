from pathlib import Path
import hashlib, json, subprocess, re, fnmatch

root = Path(__file__).resolve().parent
repo = root / 'relay-reader-checkout'
prefix = 'research/integration/relay_reader_error_57_20261003_01a0ff58'
package = repo / prefix
def git(*args):
    return subprocess.run(['git', *args], cwd=repo, capture_output=True, check=True).stdout
def sha(b): return hashlib.sha256(b).hexdigest()
entries = []
for line in (package / 'SHA256SUMS').read_text().splitlines():
    digest, path = line.split(None, 1)
    path = path.strip()
    assert sha((package / path).read_bytes()) == digest, path
    entries.append(path)
files = sorted(p.relative_to(package).as_posix() for p in package.rglob('*') if p.is_file())
assert sorted(entries + ['SHA256SUMS']) == files
assert len(files) == 52
for path in files:
    b = (package / path).read_bytes()
    assert b'<USER_HOME>' not in b and b'<USER_HOME>' not in b, path
    assert b'<USER_HOME>' not in b, path
    assert not path.endswith(('.py', '.mjs', '.js', '.yml', '.yaml')), path
publication = json.loads((package / 'PUBLICATION.json').read_bytes())
assert len(publication['mappings']) == 46
source = json.loads((package / 'SOURCE.json').read_bytes())
for path, pin in source['source_pins'].items():
    assert sha(git('show', 'HEAD:' + path)) == pin['sha256']
    assert git('rev-parse', 'HEAD:' + path).decode().strip() == pin['git_blob']
changed = git('diff', '--name-only', source['base_commit'], 'HEAD').decode().splitlines()
assert changed == ['runtime/host_v1/relay_client.mjs', 'runtime/host_v1/test_relay_client.mjs']
workflow_paths = git('ls-tree', '-r', '--name-only', 'HEAD', '.github/workflows').decode().splitlines()
headers = {}
possible = {}
for path in workflow_paths:
    if not path.endswith(('.yml', '.yaml')): continue
    text = git('show', 'HEAD:' + path).decode()
    header = re.split(r'^(?:permissions|jobs|concurrency|env|defaults):', text, maxsplit=1, flags=re.M)[0]
    if not re.search(r'^  (push|pull_request):', header, re.M): continue
    headers[path] = header
    patterns = re.findall(r"^\s+- ['\"]([^'\"]+)['\"]\s*$", header, re.M)
    if not re.search(r'^\s+paths:', header, re.M) or any(fnmatch.fnmatchcase(p, pat) for p in changed + [prefix + '/README.md'] for pat in patterns):
        possible[path] = header
out = root / 'relay-output-read-error' / 'publication-check-01'
out.mkdir(exist_ok=False)
(out / 'automatic-workflow-headers.json').write_text(json.dumps(headers, indent=2) + '\n')
(out / 'potentially-matching-headers.json').write_text(json.dumps(possible, indent=2) + '\n')
report = {'manifest_entries': len(entries), 'package_files': len(files),
          'manifest_sha256': sha((package / 'SHA256SUMS').read_bytes()),
          'original_mappings': len(publication['mappings']), 'source_pins': source['source_pins'],
          'automatic_workflows': len(headers), 'conservative_matching_workflows': sorted(possible),
          'workflow_classifier_scope': 'Conservative header match; exact matching headers reviewed separately. No dispatch/execution.'}
(out / 'REPORT.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps(report))
for path, header in possible.items(): print(path + '\n' + header)
