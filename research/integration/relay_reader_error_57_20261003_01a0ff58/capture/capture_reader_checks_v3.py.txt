from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, subprocess, sys
root = Path(__file__).resolve().parent
checkout = root / 'relay-reader-checkout'
phase = sys.argv[1]
out = root / 'relay-output-read-error' / phase
out.mkdir(exist_ok=False)
node = Path(r'<USER_HOME>\.cache\codex-runtimes\codex-primary-runtime\dependencies\node\bin\node.exe')
sources = {}
paths = ['runtime/host_v1/relay_client.mjs', 'runtime/host_v1/relay_host.mjs',
         'runtime/host_v1/test_relay_client.mjs', 'runtime/host_v1/test_relay_host.mjs',
         'runtime/integration_checks/host_timing.py']
for path in paths:
    b = (checkout / path).read_bytes()
    (out / (path.replace('/', '__') + '.txt')).write_bytes(b)
    sources[path] = hashlib.sha256(b).hexdigest()
command = [str(node), '--test', 'runtime/host_v1/test_relay_client.mjs', 'runtime/host_v1/test_relay_host.mjs']
start = datetime.now(timezone.utc).isoformat()
try:
    p = subprocess.run(command, cwd=checkout, capture_output=True, timeout=35)
    stdout, stderr, code, timeout = p.stdout, p.stderr, p.returncode, False
except subprocess.TimeoutExpired as error:
    stdout, stderr, code, timeout = error.stdout or b'', error.stderr or b'', None, True
(out / 'stdout.log').write_bytes(stdout)
(out / 'stderr.log').write_bytes(stderr)
py = subprocess.run(['python3', '-c', 'import sys;print(sys.version);print(sys.executable)'], cwd=checkout, capture_output=True, check=True).stdout.decode().strip()
for path, wanted in sources.items():
    if hashlib.sha256((checkout / path).read_bytes()).hexdigest() != wanted:
        raise ValueError('source changed ' + path)
record = {'phase': phase, 'start_utc': start, 'end_utc': datetime.now(timezone.utc).isoformat(),
          'node': subprocess.run([str(node), '--version'], capture_output=True, check=True).stdout.decode().strip(),
          'node_binary_sha256': hashlib.sha256(node.read_bytes()).hexdigest(), 'helper_python': py,
          'command': command, 'cwd': str(checkout), 'exit_code': code, 'timeout': timeout,
          'source': sources, 'source_readback_unchanged': True,
          'stdout_sha256': hashlib.sha256(stdout).hexdigest(), 'stderr_sha256': hashlib.sha256(stderr).hexdigest(),
          'stdout_bytes': len(stdout), 'stderr_bytes': len(stderr),
          'scope': 'ordinary actual Node24 Windows inert child/stream checks; timing helper uses independently reported Python3.13.14; no SDK/GUI/input/formal run'}
with (out / 'receipt.json').open('x', encoding='utf-8') as f:
    json.dump(record, f, indent=2); f.write('\n')
print(json.dumps(record))
print(json.dumps({'node_stdout_tail': stdout.decode(errors='replace')[-1100:]}, ensure_ascii=True))
sys.exit(1 if timeout else code)
