from pathlib import Path
import hashlib, json, subprocess
root = Path(__file__).resolve().parent
repo = root / 'agent-interface'
base = '3e93df3755b8ae8e2e063ed8f61524989e4bf3df'
out = root / 'relay-output-read-error'
out.mkdir(exist_ok=False)
records = {}
for name in ['relay_client.mjs', 'relay_host.mjs', 'test_relay_client.mjs', 'test_relay_host.mjs']:
    path = 'runtime/host_v1/' + name
    data = subprocess.run(['git', 'show', base + ':' + path], cwd=repo, capture_output=True, check=True).stdout
    (out / name).write_bytes(data)
    records[path] = {'sha256': hashlib.sha256(data).hexdigest(), 'bytes': len(data)}
with (out / 'BASE.json').open('x', encoding='utf-8') as f:
    json.dump({'base': base, 'source': records}, f, indent=2); f.write('\n')
print(json.dumps({'base': base, 'exported': len(records)}))
