from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, subprocess, sys
root = Path(__file__).resolve().parent
checkout = root / 'relay-reader-checkout'
out = root / 'relay-output-read-error/coupled-01'
out.mkdir(exist_ok=False)
node = Path(r'<USER_HOME>\.cache\codex-runtimes\codex-primary-runtime\dependencies\node\bin\node.exe')
names = ['relay_client.mjs', 'relay_host.mjs', 'primary_caller.mjs', 'primary_exchange.mjs',
         'primary_stdio.mjs', 'test_primary_caller.mjs', 'test_primary_exchange.mjs', 'test_primary_stdio.mjs']
sources = {}
for name in names:
    b = (checkout / 'runtime/host_v1' / name).read_bytes()
    (out / (name + '.txt')).write_bytes(b)
    sources['runtime/host_v1/' + name] = hashlib.sha256(b).hexdigest()
command = [str(node), '--test'] + ['runtime/host_v1/' + x for x in names if x.startswith('test_')]
start = datetime.now(timezone.utc).isoformat()
p = subprocess.run(command, cwd=checkout, capture_output=True, timeout=35)
(out / 'stdout.log').write_bytes(p.stdout)
(out / 'stderr.log').write_bytes(p.stderr)
for path, wanted in sources.items():
    if hashlib.sha256((checkout / path).read_bytes()).hexdigest() != wanted:
        raise ValueError('source changed ' + path)
head = subprocess.run(['git', 'rev-parse', 'HEAD'], cwd=checkout, capture_output=True, check=True).stdout.decode().strip()
record = {'start_utc': start, 'end_utc': datetime.now(timezone.utc).isoformat(), 'command': command,
          'cwd': str(checkout), 'head': head, 'exit_code': p.returncode,
          'node': 'v24.19.0', 'node_binary_sha256': hashlib.sha256(node.read_bytes()).hexdigest(),
          'capture_python': sys.version, 'source': sources, 'source_readback_unchanged': True,
          'stdout_sha256': hashlib.sha256(p.stdout).hexdigest(), 'stderr_sha256': hashlib.sha256(p.stderr).hexdigest(),
          'stdout_bytes': len(p.stdout), 'stderr_bytes': len(p.stderr),
          'scope': 'Necessary existing caller/exchange/stdio contract checks on exact reader repair. No #6919 proposed source or frozen comparison, SDK/backend/input or formal run.'}
with (out / 'receipt.json').open('x', encoding='utf-8') as f:
    json.dump(record, f, indent=2); f.write('\n')
print(json.dumps(record))
print(json.dumps({'stdout_tail': p.stdout.decode(errors='replace')[-1500:]}, ensure_ascii=True))
sys.exit(p.returncode)
