import {PassThrough,Writable} from 'node:stream';
import {readFile,writeFile} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {servePrimaryLines} from './source/runtime/host_v1/primary_stdio.mjs';
const limit=65536,rows=[],tick=()=>new Promise(r=>setImmediate(r)),hash=b=>createHash('sha256').update(b).digest('hex');
const plan=JSON.parse(await readFile(new URL('./CASES.json',import.meta.url)));
function frontier(input){
 let parts=[],size=0,peak=0,rejected=false,forwarded=0;
 function stop(){rejected=true;parts=[];size=0;input.end();}
 return {
  feed(chunk){if(rejected)return;let start=0;
   while(start<chunk.length){const lf=chunk.indexOf(10,start),end=lf<0?chunk.length:lf,part=chunk.subarray(start,end);
    if(size+part.length>limit){const available=limit-size;if(available){parts.push(part.subarray(0,available));size+=available;}peak=Math.max(peak,size);stop();return;}
    if(part.length){parts.push(part);size+=part.length;peak=Math.max(peak,size);}
    if(lf>=0){const frame=Buffer.concat(parts,size);input.write(Buffer.concat([frame,Buffer.from('\n')]));forwarded++;parts=[];size=0;start=lf+1;}else break;
   }
  },end(){if(!rejected){if(size)input.write(Buffer.concat(parts,size));parts=[];size=0;input.end();}},state(){return {typed_frontier:rejected?'FRAME_TOO_LARGE':'none',buffer_peak_bytes:peak,forwarded_frames:forwarded};}
 };
}
for(const c of plan.cases)for(const policy of ['current','byte-frontier']){
 const wire=await readFile(new URL('./inputs/'+c.file,import.meta.url));const input=new PassThrough(),out=[];let calls=[],release=null,settled=0,outcome='pending',identity=false;
 const original={original:'same-result'};
 const exchange={state:()=>({next_id:calls.length+1}),execute:async request=>{calls.push({id:request.id,padding_sha256:hash(Buffer.from(request.padding??''))});if(c.held&&request.id===1){const result=await new Promise(r=>release=r);identity=result===original;return result;}return {id:request.id};}};
 const output=new Writable({write(b,e,callback){out.push(JSON.parse(b));callback();}});
 const promise=servePrimaryLines({exchange,input,output}).then(()=>{settled++;outcome='resolved';},e=>{settled++;outcome='rejected:'+e.code;});
 const gate=policy==='byte-frontier'?frontier(input):null;
 const send=b=>gate?gate.feed(b):input.write(b);let offset=0;
 if(c.held){const n=wire.indexOf(10)+1;send(wire.subarray(0,n));offset=n;for(let n=0;n<3;n++)await tick();if(!release)throw Error('original not accepted');}
 const widths=[1,2,3,7,64,4096];let step=0;
 while(offset<wire.length){const end=Math.min(wire.length,offset+widths[step++%widths.length]);send(wire.subarray(offset,end));offset=end;}
 if(c.has_newline){if(gate)gate.end();else input.end();}
 for(let n=0;n<3;n++)await tick();
 const before={outcome,settled,calls:calls.length,gate:gate?.state()??null,output_statuses:out.map(x=>x.status)};
 // Same original result only; no cancellation/replacement/resend.
 if(c.held)release(original);
 if(gate)gate.end();else input.end();
 await promise;for(let n=0;n<2;n++)await tick();
 const row={case:c.name,policy,wire_bytes:wire.length,wire_sha256:hash(wire),chunks:step,before,final:{outcome,settled,calls,original_result_identity:c.held?identity:null,output:out,gate:gate?.state()??null,input_error_listeners:input.listenerCount('error'),output_error_listeners:output.listenerCount('error')}};
 rows.push(row);console.log(JSON.stringify(row));input.destroy();output.destroy();
}
await writeFile(new URL('./ROWS.json',import.meta.url),JSON.stringify(rows,null,2)+'\n');
