"""Fresh no-input generation observation for a frozen owned fixture."""
import ctypes
from ctypes import wintypes as W
import hashlib,json,subprocess,sys
from pathlib import Path
P=Path(__file__).resolve().parent

def main():
    c=json.loads((P/'conditions.json').read_text())
    repo=Path(c['repo']);sys.path.insert(0,str(repo))
    from runtime.cli_v1.observe import observe
    from runtime.cli_v1.win32_custody_fence import _win32_logon_namespace_digest
    import os
    if subprocess.check_output(['git','rev-parse','HEAD'],cwd=repo,text=True).strip()!=c['source_commit']:raise RuntimeError('source changed')
    for name,sha in c['file_hashes'].items():
        if hashlib.sha256((P/name).read_bytes()).hexdigest()!=sha:raise RuntimeError('frozen file changed')
    journal=Path(os.environ['LOCALAPPDATA'])/'Unjuno/AgentInterface/custody-v1'/(_win32_logon_namespace_digest()+'.json')
    if hashlib.sha256(journal.read_bytes()).hexdigest()!=c['custody_before']['sha256']:raise RuntimeError('custody changed')
    meta=json.loads((P/'fixture-meta.json').read_text())
    if meta['nonce']!=c['nonce'] or meta['source_sha256']!=c['file_hashes']['fixture_app.py']:raise RuntimeError('fixture provenance mismatch')
    u=ctypes.WinDLL('user32',use_last_error=True)
    u.GetWindowThreadProcessId.argtypes=[W.HWND,ctypes.POINTER(W.DWORD)];u.GetWindowThreadProcessId.restype=W.DWORD
    owner=W.DWORD()
    if not u.GetWindowThreadProcessId(meta['hwnd'],ctypes.byref(owner)) or owner.value!=meta['pid']:raise RuntimeError('fixture owner changed')
    rect=meta['rect'];region=[rect[0],rect[1],rect[2]-rect[0],rect[3]-rect[1]]
    result=observe({'fixture':meta['hwnd']},target='fixture',frame='screen_physical_px',region=region,capture_directory=P/'preflight-images')
    with (P/'preflight-observation.json').open('x',encoding='utf-8') as f:json.dump(result,f,indent=2)
    if result.get('status')!='returned' or result.get('target_generation_consistency')!='matched' or result.get('input_dispatched') is not False:raise RuntimeError('preflight observation failed')
    if hashlib.sha256(journal.read_bytes()).hexdigest()!=c['custody_before']['sha256']:raise RuntimeError('custody changed during observation')
    config={'repo':str(repo),'packet':str(P),'nonce':c['nonce'],'generation':result['target_generation'],'targets':{'fixture':meta['hwnd']}}
    with (P/'oracle-server-config.json').open('x',encoding='utf-8') as f:json.dump(config,f,indent=2)
    print(json.dumps({'preflight':'matched','input_dispatched':False,'generation':result['target_generation']}))

if __name__=='__main__':main()
