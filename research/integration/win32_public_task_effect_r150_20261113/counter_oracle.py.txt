"""Read-only fixture oracle candidate; not yet qualified by a native live run."""
import hashlib
import json
from pathlib import Path
import time
import uuid

def make_counter_oracle(packet, nonce, expected_generation):
    packet=Path(packet)
    def observe(action):
        if action['target_generation']!=expected_generation:
            raise ValueError('target generation mismatch')
        with (packet/'effects.jsonl').open('rb') as source:
            raw=source.read(65537)
        if len(raw)>65536:raise ValueError('effect evidence exceeds bound')
        rows=[json.loads(line) for line in raw.splitlines() if line]
        observed=time.monotonic_ns()
        valid=(len(rows)==1 and type(rows[0]) is dict and rows[0].get('nonce')==nonce
          and rows[0].get('kind')=='task_effect'
          and type(rows[0].get('counter')) is int and rows[0]['counter']==1
          and rows[0].get('first_transition') is True
          and rows[0].get('phase')=='resumed'
          and type(rows[0].get('monotonic_ns')) is int
          and 0<rows[0]['monotonic_ns']<=observed)
        oracle_id=uuid.uuid4().hex
        artifact=packet/('oracle-evidence-'+oracle_id+'.json')
        evidence={'call_id':action['call_id'],'operation_id':action['operation_id'],
          'target_generation':expected_generation,'observed_monotonic_ns':observed,
          'source_sha256':hashlib.sha256(raw).hexdigest(),'source_rows':rows,
          'scope':'one observed counter effect; no application quiescence claim'}
        with artifact.open('x',encoding='utf-8') as f:json.dump(evidence,f,sort_keys=True)
        return {'call_id':action['call_id'],'operation_id':action['operation_id'],
          'target_generation':expected_generation,'oracle_id':oracle_id,
          'outcome':'success' if valid else 'failure',
          'observed_monotonic_ns':observed,'evidence_ref':str(artifact)}
    return observe
