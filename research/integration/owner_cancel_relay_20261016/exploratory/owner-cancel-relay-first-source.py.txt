"""Explicit private cancellation lane; never grants input authority.

An accepted cancellation and a verified native release are separate outcomes.
The first bounded outcome remains immutable even if the owner settles later.
"""
import copy
import threading
import time

import input_owner_rpc_v2 as rpc

SCHEMA='input-owner-cancel-v4'


class LeaseRegistry:
    def __init__(self):
        self._lock=threading.Lock();self._leases={};self.error=None

    def decode(self,dto):
        with self._lock:
            if self.error is not None:raise RuntimeError('cancellation authority retired: '+self.error)
            return rpc.decode_lease(dto,self._leases)

    def retire(self,reason):
        with self._lock:
            if self.error is None:self.error=str(reason)
            for lease in self._leases.values():lease.set()


def validate_cause(cause,dto):
    if cause is None:return False
    if type(cause) is not dict or set(cause)!={'intent_token','record'} or cause['intent_token']!=dto['intent_token']:
        raise ValueError('cancellation cause intent mismatch')
    row=cause['record']
    if (type(row) is not dict or row.get('event')!='owner_release'
            or row.get('reason') not in ('cancelled','expired','stop_requested','target_generation_invalidated','target_query_invalidated','focus_changed','surface_changed')
            or row.get('intent_token')!=dto['intent_token'] or row.get('valid_until_ns')!=dto['deadline_ns']
            or row.get('target_process_identity')!=dto['target_process_identity']
            or type(row.get('verified')) is not bool
            or any(type(row.get(k)) is not list for k in ('keys_down','buttons_down','keys_unknown','key_state_errors'))):
        raise ValueError('cancellation cause source/outcome mismatch')
    if row['verified'] and (row.get('key_state_source')!='x11_query_keymap'
            or any(row[k] for k in ('keys_down','buttons_down','keys_unknown','key_state_errors'))):
        raise ValueError('verified cancellation contradicts release state')
    return row['verified']


class CancellationService:
    def __init__(self,sock,*,display,service_id,owner_id,cause_timeout=1):
        self.sock=sock;self.display=display;self.service_id=service_id;self.owner_id=owner_id
        self.cause_timeout=rpc._timeout(cause_timeout);self.registry=LeaseRegistry()
        self.stop=threading.Event();self.records=[]
        self.thread=threading.Thread(target=self._run,name='owned-cancel-control',daemon=False)
        self.thread.start()

    @property
    def error(self):return self.registry.error

    def _run(self):
        buffer=bytearray();next_id=1
        try:
            rpc._send_frame(self.sock,dict(schema=SCHEMA+'-ready',display=self.display,
                service_id=self.service_id,owner_id=self.owner_id),time.monotonic()+2)
            while not self.stop.is_set():
                try:request=rpc._read_frame(self.sock,buffer,time.monotonic()+.2)
                except TimeoutError:continue
                if (type(request) is not dict or set(request)!={'schema','service_id','request_id','lease'}
                        or request['schema']!=SCHEMA or request['service_id']!=self.service_id
                        or type(request['request_id']) is not int or request['request_id']!=next_id
                        or not 1<=next_id<=rpc.MAX_LEASES or buffer):
                    raise ValueError('cancellation identity/schema invalid')
                dto=request['lease'];rpc._validate_lease(dto)
                if dto['cancel_requested'] is not True:raise ValueError('cancellation must revoke authority')
                next_id+=1
                lease=self.registry.decode(dto);lease.set()
                cause=lease.wait_interruption(self.cause_timeout)
                response=dict(schema=SCHEMA+'-response',service_id=self.service_id,request_id=request['request_id'],
                              intent_token=dto['intent_token'],accepted=True,cause=cause)
                self.records.append(dict(request=copy.deepcopy(request),response=copy.deepcopy(response)))
                rpc._send_frame(self.sock,response,time.monotonic()+2)
        except BaseException as error:
            if not self.stop.is_set():self.registry.retire(type(error).__name__+': '+str(error))
        finally:
            try:rpc._detach_socket(self.sock)
            except BaseException as error:
                self.registry.retire('control detach: '+type(error).__name__+': '+str(error))

    def close(self):
        self.stop.set();rpc._detach_socket(self.sock);self.thread.join(3)
        if self.thread.is_alive():raise RuntimeError('cancellation service thread remains owned')
        return dict(schema=SCHEMA+'-service',error=self.error,records=copy.deepcopy(self.records),
                    thread_stopped=True,grants_input_authority=False)


class CancellationClient:
    def __init__(self,sock,*,startup,timeout=2):
        self.sock=sock;self.startup=copy.deepcopy(startup);self.timeout=rpc._timeout(timeout)
        self._gate=threading.Lock();self._lock=threading.Lock();self._stop=threading.Event()
        self._entries={};self._next_id=1;self._retired=False;self._buffer=bytearray()
        try:
            ready=rpc._read_frame(sock,self._buffer,time.monotonic()+self.timeout)
            expected=dict(schema=SCHEMA+'-ready',display=startup['display'],service_id=startup['service_id'],owner_id=startup['owner_id'])
            if startup.get('cancellation_enabled') is not True or ready!=expected or self._buffer:
                raise ValueError('cancellation readiness mismatch')
        except BaseException:
            self._retired=True;rpc._detach_socket(sock);raise

    @property
    def outcomes(self):
        with self._lock:return {token:copy.deepcopy(entry['outcome']) for token,entry in self._entries.items()}

    @property
    def thread_alive(self):return any(entry['thread'].is_alive() for entry in self._entries.values())

    def watch(self,lease):
        dto=rpc.encode_lease(lease);token=dto['intent_token']
        with self._lock:
            if self._stop.is_set() or self._retired:raise RuntimeError('cancellation client retired')
            existing=self._entries.get(token)
            if existing is not None:
                if existing['lease'] is not lease:raise ValueError('intent belongs to another original lease')
                original=dict(existing['dto']);original['cancel_requested']=dto['cancel_requested'];original['focus_invalid']=dto['focus_invalid']
                if original!=dto:raise ValueError('watched source binding changed')
                return
            if len(self._entries)>=rpc.MAX_LEASES:raise ValueError('cancellation inventory exhausted')
            entry=dict(lease=lease,dto=copy.deepcopy(dto),done=threading.Event(),outcome=None)
            thread=threading.Thread(target=self._watch,args=(entry,),name='parent-cancel-'+token[:8],daemon=False)
            entry['thread']=thread;self._entries[token]=entry;thread.start()

    def _watch(self,entry):
        while not self._stop.wait(.005):
            if entry['lease'].cancel.is_set():break
        else:return
        outcome=dict(accepted=False,release_verified=False,cause=None,error=None)
        acquired=False
        try:
            acquired=self._gate.acquire(timeout=self.timeout)
            if not acquired:raise TimeoutError('cancellation sender slot deadline')
            if self._retired or self._stop.is_set():raise RuntimeError('cancellation client retired')
            dto=copy.deepcopy(entry['dto']);dto['cancel_requested']=True
            request=dict(schema=SCHEMA,service_id=self.startup['service_id'],request_id=self._next_id,lease=dto)
            self._next_id+=1
            rpc._send_frame(self.sock,request,time.monotonic()+self.timeout)
            response=rpc._read_frame(self.sock,self._buffer,time.monotonic()+self.timeout)
            if (type(response) is not dict or set(response)!={'schema','service_id','request_id','intent_token','accepted','cause'}
                    or response['schema']!=SCHEMA+'-response' or response['service_id']!=self.startup['service_id']
                    or type(response['request_id']) is not int or response['request_id']!=request['request_id']
                    or response['intent_token']!=dto['intent_token'] or response['accepted'] is not True or self._buffer):
                raise ValueError('cancellation response identity/schema mismatch')
            verified=validate_cause(response['cause'],dto)
            outcome.update(accepted=True,release_verified=verified,cause=copy.deepcopy(response['cause']))
            if response['cause'] is not None:entry['lease'].record_interruption(response['cause']['record'])
        except BaseException as error:
            self._retired=True;outcome['error']=type(error).__name__+': '+str(error)
            try:rpc._detach_socket(self.sock)
            except BaseException as cleanup:outcome['cleanup_error']=type(cleanup).__name__+': '+str(cleanup)
        finally:
            if acquired:self._gate.release()
            with self._lock:entry['outcome']=copy.deepcopy(outcome)
            entry['done'].set()

    def wait(self,token,timeout=2):
        timeout=rpc._timeout(timeout)
        with self._lock:entry=self._entries[token]
        if not entry['done'].wait(timeout):raise TimeoutError('no cancellation outcome yet')
        with self._lock:return copy.deepcopy(entry['outcome'])

    def close(self):
        self._stop.set();self._retired=True;rpc._detach_socket(self.sock)
        for entry in list(self._entries.values()):entry['thread'].join(self.timeout+1)
        if self.thread_alive:raise RuntimeError('parent cancellation thread remains owned')
