@contextmanager
def fake_owner(*,display,out,service_id):
    # Spawn imports this function; no real display is touched before this fixture.
    with backend_fixture() as (backend,events,generation,fake):
        if display!=':fake':raise ValueError('Fake factory accepts only :fake')
        stop=threading.Event()
        def crash_watch():
            while not stop.wait(.005):
                if (out/'fixture-crash').exists():os._exit(71)
        watcher=threading.Thread(target=crash_watch,name='owned-fake-crash-watch',daemon=True)
        watcher.start()
        previous=native.xtest.fake_input.side_effect
        def emit(connection,event,code):
            if event==native.X.KeyRelease and (out/'fixture-block-release').exists():
                while not (out/'fixture-resume-release').exists():
                    if stop.wait(.005):raise RuntimeError('fake release test retired')
            result=previous(connection,event,code)
            row=dict(event=event,code=code,down=sorted(fake.down),pid=os.getpid(),thread=threading.get_ident(),at_ns=time.perf_counter_ns())
            with (out/'fixture-edges.jsonl').open('a',encoding='utf-8') as stream:
                stream.write(json.dumps(row)+'\n');stream.flush();os.fsync(stream.fileno())
            return result
        native.xtest.fake_input.side_effect=emit
        try:yield backend.owner
        finally:stop.set();watcher.join(1)

def client_process(sock,ready,control,service_id):
    client=None
    try:
        client=OwnerRpcClient(sock,display=':fake',service_id=service_id,timeout=2)
        lease=make_lease()
        admission=client.call('down',lease,'a')
        ready.send(dict(pid=os.getpid(),parent_pid=os.getppid(),admission=admission))
        command=control.recv()
        if command=='crash':os._exit(73)
        if command!='close':raise ValueError('test control command')
    finally:
        if client is not None:client.close()
        else:sock.close()
        ready.close();control.close()


def make_lease():
    lease=Lease(time.perf_counter_ns()+10_000_000_000)
    lease.intent_token='owned-process-test-intent'
    lease.expected_focus=lease.expected_surface=41
    lease.expected_geometry=[0,0,32,32]
    lease.target_process_identity=(1,41,'a'*64,lease.deadline)
    return lease


def edges(out):
    path=out/'fixture-edges.jsonl'
    return [json.loads(line) for line in path.read_text().splitlines()] if path.exists() else []


class OwnedOwnerProcessTests(unittest.TestCase):
    def test_module_available(self):
        self.assertIsNotNone(importlib.util.find_spec('input_owner_process_v3'),
                             'Independent owner process and bound receipt implementation absent')

    def launch(self,temp):
        return process_api().launch_owner_service(display=':fake',out=Path(temp)/'service',_owner_factory=fake_owner)

    def finish(self,handle):
        handle.close_channel()
        (handle.out/'fixture-resume-release').touch()
        (handle.out/'fixture-resume-exit').touch()
        handle.process.join(6)
        forced=False
        if handle.process.is_alive():
            forced=True;(handle.out/'fixture-crash').touch();handle.process.join(4)
        matched=[r for r in ROWS if r.get('service_id')==handle.service_id or (r.get('startup') or {}).get('service_id')==handle.service_id]
        if not matched:
            row=dict(case='failure-cleanup-capture',service_id=handle.service_id,startup=handle.startup,
                spawn_pid=handle.process.pid,report=handle.wait(.1),files=handle.saved_files())
            ROWS.append(row);matched=[row]
        for row in matched:row['cleanup']=dict(forced=forced,process_alive=handle.process.is_alive(),exit_code=handle.process.exitcode)
        self.assertFalse(handle.process.is_alive(),'owned fake service still running')
        self.assertFalse(forced,'test required crash fallback')

    def test_normal_close_binds_saved_receipts_and_owned_process_exit(self):
        with tempfile.TemporaryDirectory() as temp:
            handle=self.launch(temp);client=None
            try:
                client=OwnerRpcClient(handle.take_socket(),display=':fake',service_id=handle.service_id)
                admission=client.call('down',make_lease(),'a')
                close=client.close();report=handle.wait(5)
                self.assertTrue(close['verified'])
                self.assertTrue(report['verified'],report)
                self.assertFalse(handle.process.is_alive())
                self.assertEqual([r['event'] for r in edges(handle.out)],[2,3])
                self.assertNotEqual(handle.startup['pid'],os.getpid())
                for name,metadata in handle.saved_files().items():
                    self.assertIn('bytes_b64',metadata,'Saved raw must survive newline projection')
                    self.assertEqual(base64.b64decode(metadata['bytes_b64']),(handle.out/name).read_bytes())
                with self.assertRaises(RuntimeError):handle.take_socket()
                ROWS.append(dict(case='normal',startup=handle.startup,spawn_pid=handle.process.pid,
                                 admission=admission,report=report,edges=edges(handle.out),files=handle.saved_files()))
            finally:
                if client:client.close()
                self.finish(handle)

    def test_actual_client_process_abrupt_exit_leaves_service_to_release(self):
        with tempfile.TemporaryDirectory() as temp:
            handle=self.launch(temp);ctx=multiprocessing.get_context('spawn')
            ready_parent,ready_child=ctx.Pipe(duplex=False)
            control_child,control_parent=ctx.Pipe(duplex=False)
            channel=handle.take_socket()
            child=ctx.Process(target=client_process,args=(channel,ready_child,control_child,handle.service_id))
            child.start();channel.close();ready_child.close();control_child.close()
            client_ready=None
            try:
                self.assertTrue(ready_parent.poll(5),'client admission not reported')
                client_ready=ready_parent.recv()
                self.assertTrue(handle.process.is_alive())
                control_parent.send('crash');child.join(5)
                self.assertFalse(child.is_alive());self.assertNotEqual(child.exitcode,0)
                report=handle.wait(5)
                self.assertTrue(report['verified'],report)
                self.assertEqual([r['event'] for r in edges(handle.out)],[2,3])
                self.assertNotEqual(client_ready['pid'],handle.startup['pid'])
                ROWS.append(dict(case='client-death',startup=handle.startup,spawn_pid=handle.process.pid,
                                 client_spawn_pid=child.pid,client_ready=client_ready,client_exit=child.exitcode,
                                 report=report,edges=edges(handle.out),files=handle.saved_files()))
            finally:
                if child.is_alive():
                    try:control_parent.send('crash')
                    except (BrokenPipeError,OSError):pass
                child.join(5);ready_parent.close();control_parent.close()
                self.finish(handle)
                self.assertFalse(child.is_alive())

    def test_export_hash_is_bound_to_one_observation_of_each_saved_file(self):

            target=handle.out/'fixture-edges.jsonl';target.write_bytes(b'old\n')
            original=Path.read_bytes;reads=[]
            def changing(path):
                if path==target:
                    reads.append(1)

    def test_startup_interrupt_cleanup_pair_and_unstarted_handle_preserve_custody(self):
        api=process_api();outcomes=[]
        for fault in ('interrupt','interrupt-and-close','start-failure'):
            with self.subTest(fault=fault), tempfile.TemporaryDirectory() as temp:
                primary=KeyboardInterrupt('startup interrupt');cleanup=OSError('ready pipe close')
                class Ready:
                    def poll(self,_):raise primary
                    def close(self):
                        if fault=='interrupt-and-close':raise cleanup
                class ChildReady:
                    def close(self):pass
                class FakeProcess:
                    pid=None;exitcode=None
                    def start(self):
                        if fault=='start-failure':raise RuntimeError('spawn construction failed')
                        self.pid=12345
                    def join(self,_):raise AssertionError('unstarted/fake process must not be joined')
                    def is_alive(self):return False
                process=FakeProcess()
                class Context:
                    def Pipe(self,**_):return Ready(),ChildReady()
                    def Process(self,**_):return process
                expected=api.ServiceStartupError if fault=='start-failure' else (BaseExceptionGroup if fault=='interrupt-and-close' else KeyboardInterrupt)
                with patch.object(api.multiprocessing,'get_context',return_value=Context()):
                    with self.assertRaises(expected) as caught:
                        api.launch_owner_service(display=':fake',out=Path(temp)/'service',_owner_factory=fake_owner)
                if fault=='start-failure':
                    handle=caught.exception.handle;report=handle.wait(.1)
                    self.assertFalse(report['verified']);self.assertFalse(report['process_alive'])
                    self.assertIsNone(report['exit_code'])
                else:
                    handle=getattr(caught.exception,'owner_service_handle',None)
                    self.assertIsNotNone(handle,'Fatal startup must retain its owned handle')
                    if fault=='interrupt':self.assertIs(caught.exception,primary)
                    else:
                        self.assertIs(caught.exception.exceptions[0],primary)
                        self.assertIs(caught.exception.exceptions[1],cleanup)
                self.assertIs(handle.process,process)
                outcomes.append(dict(fault=fault,owned_handle_retained=True,primary_preserved=True))
        ROWS.append(dict(case='startup-custody',outcomes=outcomes,scope='inert process factory/private sockets; no child'))
