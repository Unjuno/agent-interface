"""Run each newly frozen ordinary job once; capture full original output and actual endpoints."""
from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,os,subprocess,sys,psutil
r=Path(__file__).resolve().parent;f=json.loads((r/'FREEZE.json').read_bytes());job=next(x for x in f['jobs'] if x['name']==sys.argv[1])
def sha(b):return hashlib.sha256(b).hexdigest()
def put(p,x):p.write_text(json.dumps(x,sort_keys=True,indent=2)+'\n',encoding='utf-8',newline='\n')
for x in f['source']:assert sha((r/x['variant']/x['path']).read_bytes())==x['sha256']
for x in f['helpers']:assert sha((r/x['path']).read_bytes())==x['sha256']
out=Path(job['output'])
if 'config_sha256' in job:assert sha((out/'config.json').read_bytes())==job['config_sha256']
else:out.mkdir(parents=True,exist_ok=False)
assert not (out/'START.json').exists()
(r/'tmp').mkdir(exist_ok=True)
env=dict(os.environ,**job['environment_delta']);started=datetime.now(timezone.utc).isoformat();code=None;live=False
with (out/'stdout.txt').open('wb') as stdout,(out/'stderr.txt').open('wb') as stderr:
 p=subprocess.Popen(job['argv'],cwd=job['cwd'],env=env,stdout=stdout,stderr=stderr)
 put(out/'START.json',{'job':job,'pid':p.pid,'ppid':os.getpid(),'started_utc':started,'freeze_sha256':sha((r/'FREEZE.json').read_bytes())})
 try:code=p.wait(timeout=f['job_timeout_seconds'])
 except subprocess.TimeoutExpired:live=True
ended=datetime.now(timezone.utc).isoformat();absent=not psutil.pid_exists(p.pid)
put(out/'RECEIPT.json',{'job':job,'pid':p.pid,'ppid':os.getpid(),'started_utc':started,'ended_or_observed_utc':ended,'exit_code':code,'confirmed_live_at_timeout':live,'original_pid_observed_absent':absent,'stdout_bytes':(out/'stdout.txt').stat().st_size,'stderr_bytes':(out/'stderr.txt').stat().st_size,'stdout_sha256':sha((out/'stdout.txt').read_bytes()),'stderr_sha256':sha((out/'stderr.txt').read_bytes())})
print(json.dumps({'job':job['name'],'pid':p.pid,'exit_code':code,'pid_absent':absent,'live':live,'stdout_bytes':(out/'stdout.txt').stat().st_size,'stderr_bytes':(out/'stderr.txt').stat().st_size}))
assert code==0 and absent and not live,('inspect original PID, do not restart',p.pid,code,absent,live)
