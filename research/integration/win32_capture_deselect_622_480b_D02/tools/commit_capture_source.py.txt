import hashlib
import json
import os
import pathlib
import subprocess
root = pathlib.Path(__file__).resolve().parent
repo = root / 'agent-interface'
source = root / 'capture-contract-source'
out = root / 'win32-capture-deselection-622-D02'
base = '951beae4caf0cad4e788d329f4db670348afee26'
branch = 'research/622-capture-deselect-480b-20261003'
index = out / 'SOURCE_INDEX'
assert not index.exists()
env = dict(os.environ, GIT_INDEX_FILE=str(index))
def git(*args, data=None):
    return subprocess.check_output(['git', *args], cwd=repo, env=env, input=data).decode().strip()
git('read-tree', base)
records = []
for path in ('runtime/backends/win32_v1/backend.py', 'runtime/backends/win32_v1/test_integration.py'):
    data = (source / path).read_bytes()
    blob = git('hash-object', '-w', '--stdin', data=data)
    git('update-index', '--add', '--cacheinfo', f'100644,{blob},{path}')
    records.append({'path': path, 'blob': blob, 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()})
tree = git('write-tree', '--missing-ok')
head = git('commit-tree', tree, '-p', base, data=b'Fix Win32 capture bitmap deselection before GetDIBits\n')
git('diff', '--check', base, head)
names = git('diff', '--name-only', base, head).splitlines()
assert names == [row['path'] for row in records]
git('update-ref', 'refs/heads/' + branch, head, '0' * 40)
patch = subprocess.check_output(['git', 'diff', '--binary', base, head], cwd=repo)
(out / 'SOURCE.diff').write_bytes(patch)
result = {'base': base, 'head': head, 'tree': tree, 'branch': branch,
          'files': records, 'diff_bytes': len(patch), 'diff_sha256': hashlib.sha256(patch).hexdigest(),
          'main_sends': 0}
(out / 'SOURCE_COMMIT.json').write_text(json.dumps(result, indent=2), encoding='utf-8')
print(json.dumps(result))
