import hashlib
import json
import pathlib
import subprocess
import sys
from datetime import datetime, timezone
out = pathlib.Path(sys.argv[1]).resolve()
source = pathlib.Path(__file__).with_name('native_capture_d02.py').resolve()
freeze = json.loads((out / 'FREEZE.json').read_bytes())
for path, key in [(source, 'runner_sha256'), (pathlib.Path(__file__), 'launcher_sha256'),
                  (source.with_name('audit_capture_d02.py'), 'auditor_sha256'),
                  (out / 'backend.candidate.py.txt', 'candidate_sha256')]:
    assert hashlib.sha256(path.read_bytes()).hexdigest() == freeze[key]
for item in freeze['binary_pins']:
    assert hashlib.sha256(pathlib.Path(item['path']).read_bytes()).hexdigest() == item['sha256']
start = {'start': datetime.now(timezone.utc).isoformat(), 'argv': [sys.executable, str(source), str(out)],
         'attempts': 1, 'timeout_seconds': 12}
with (out / 'NATIVE_STARTED.json').open('x') as file:
    json.dump(start, file, indent=2)
child = subprocess.Popen(start['argv'], stdout=subprocess.PIPE, stderr=subprocess.PIPE)
start['pid'] = child.pid
(out / 'NATIVE_STARTED.json').write_text(json.dumps(start, indent=2))
timeout = False
try:
    stdout, stderr = child.communicate(timeout=12)
except subprocess.TimeoutExpired:
    timeout = True
    child.kill()
    stdout, stderr = child.communicate(timeout=5)
(out / 'native.stdout').write_bytes(stdout)
(out / 'native.stderr').write_bytes(stderr)
result = {**start, 'end': datetime.now(timezone.utc).isoformat(), 'exit_code': child.returncode,
          'timed_out': timeout, 'stdout_sha256': hashlib.sha256(stdout).hexdigest(),
          'stderr_sha256': hashlib.sha256(stderr).hexdigest()}
(out / 'NATIVE_RECEIPT.json').write_text(json.dumps(result, indent=2))
print(json.dumps(result))
raise SystemExit(bool(timeout or child.returncode))
