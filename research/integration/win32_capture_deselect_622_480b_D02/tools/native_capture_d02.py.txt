"""Owned memory-only BitBlt/GetDIBits path; no window, screen pixels or input."""
import ast
import base64
import ctypes
import hashlib
import json
import os
import pathlib
import sys
from ctypes import wintypes

out = pathlib.Path(sys.argv[1]).resolve()
freeze = json.loads((out / 'FREEZE.json').read_bytes())
candidate_path = out / 'backend.candidate.py.txt'
candidate = candidate_path.read_bytes()
assert hashlib.sha256(candidate).hexdigest() == freeze['candidate_sha256']
tree = ast.parse(candidate)
nodes = []
for node in tree.body:
    if isinstance(node, ast.ClassDef) and node.name in {
        'Win32BackendError', 'BITMAPINFOHEADER', 'RGBQUAD', 'BITMAPINFO'
    }:
        nodes.append(node)
    if isinstance(node, ast.ClassDef) and node.name == 'Win32Backend':
        nodes += [n for n in node.body if isinstance(n, ast.FunctionDef) and n.name == '_capture_hdc']
assert len(nodes) == 5
ns = {'ctypes': ctypes, 'wintypes': wintypes, 'PW_CLIENTONLY': 1,
      'SRCCOPY': 0x00CC0020, 'BI_RGB': 0, 'DIB_RGB_COLORS': 0}
exec(compile(ast.Module(body=nodes, type_ignores=[]), str(candidate_path), 'exec'), ns)
method, Info, Header = ns['_capture_hdc'], ns['BITMAPINFO'], ns['BITMAPINFOHEADER']
u = ctypes.WinDLL('user32', use_last_error=True)
g = ctypes.WinDLL('gdi32', use_last_error=True)
class Rect(ctypes.Structure):
    _fields_ = [('left', wintypes.LONG), ('top', wintypes.LONG),
                ('right', wintypes.LONG), ('bottom', wintypes.LONG)]
class Bitmap(ctypes.Structure):
    _fields_ = [('bmType', wintypes.LONG), ('bmWidth', wintypes.LONG),
                ('bmHeight', wintypes.LONG), ('bmWidthBytes', wintypes.LONG),
                ('bmPlanes', wintypes.WORD), ('bmBitsPixel', wintypes.WORD),
                ('bmBits', ctypes.c_void_p)]
specs = [
    (u, 'GetDC', [wintypes.HWND], wintypes.HDC),
    (u, 'ReleaseDC', [wintypes.HWND, wintypes.HDC], ctypes.c_int),
    (u, 'FillRect', [wintypes.HDC, ctypes.POINTER(Rect), wintypes.HBRUSH], ctypes.c_int),
    (g, 'CreateCompatibleDC', [wintypes.HDC], wintypes.HDC),
    (g, 'CreateCompatibleBitmap', [wintypes.HDC, ctypes.c_int, ctypes.c_int], wintypes.HBITMAP),
    (g, 'SelectObject', [wintypes.HDC, wintypes.HGDIOBJ], wintypes.HGDIOBJ),
    (g, 'GetCurrentObject', [wintypes.HDC, wintypes.UINT], wintypes.HGDIOBJ),
    (g, 'GetObjectW', [wintypes.HANDLE, ctypes.c_int, wintypes.LPVOID], ctypes.c_int),
    (g, 'CreateSolidBrush', [wintypes.DWORD], wintypes.HBRUSH),
    (g, 'DeleteObject', [wintypes.HGDIOBJ], wintypes.BOOL),
    (g, 'DeleteDC', [wintypes.HDC], wintypes.BOOL),
    (g, 'GdiFlush', [], wintypes.BOOL),
    (g, 'BitBlt', [wintypes.HDC, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_int,
                  wintypes.HDC, ctypes.c_int, ctypes.c_int, wintypes.DWORD], wintypes.BOOL),
    (g, 'GetDIBits', [wintypes.HDC, wintypes.HBITMAP, wintypes.UINT, wintypes.UINT,
                     wintypes.LPVOID, ctypes.POINTER(Info), wintypes.UINT], ctypes.c_int),
]
for dll, name, args, returns in specs:
    fn = getattr(dll, name)
    fn.argtypes, fn.restype = args, returns
events = []
raw = {'schema': 'owned-memory-capture-622-D02', 'pid': os.getpid(),
       'candidate_sha256': hashlib.sha256(candidate).hexdigest(), 'events': events,
       'cases': [], 'input_calls': 0, 'screen_pixel_reads': 0, 'window_handles': 0,
       'PrintWindow_calls': 0, 'adapter': 'GetDC/ReleaseDC route only to owned source memory DC; other capture APIs are real GDI.'}
def save():
    (out / 'RAW.json').write_text(json.dumps(raw, indent=2), encoding='utf-8')
def call(name, fn, *args):
    events.append({'phase': 'begin', 'call': name})
    value = fn(*args)
    events.append({'phase': 'end', 'call': name, 'return': int(value or 0)})
    return value
display = source = source_bitmap = old_source = None
brushes = []
try:
    # Display DC supplies only bitmap format; it is never copied/read as pixels.
    display = call('fixture.GetDC_format_only', u.GetDC, None)
    assert display
    source = call('fixture.CreateCompatibleDC', g.CreateCompatibleDC, display)
    assert source
    source_bitmap = call('fixture.CreateCompatibleBitmap', g.CreateCompatibleBitmap, display, 16, 16)
    assert source_bitmap
    desc = Bitmap()
    assert call('fixture.GetObjectW', g.GetObjectW, source_bitmap, ctypes.sizeof(desc), ctypes.byref(desc)) == ctypes.sizeof(desc)
    raw['source_descriptor'] = {'width': desc.bmWidth, 'height': desc.bmHeight,
                                'planes': desc.bmPlanes, 'bits_pixel': desc.bmBitsPixel,
                                'bmBits_is_null': desc.bmBits is None}
    old_source = call('fixture.SelectObject', g.SelectObject, source, source_bitmap)
    assert old_source and old_source != ctypes.c_void_p(-1).value
    for rect, color in zip([(0, 0, 8, 8), (8, 0, 16, 8), (0, 8, 8, 16), (8, 8, 16, 16)],
                           [0x0000ff, 0x00ff00, 0xff0000, 0x00ffff]):
        brush = call('fixture.CreateSolidBrush', g.CreateSolidBrush, color)
        assert brush
        brushes.append(brush)
        assert call('fixture.FillRect', u.FillRect, source, ctypes.byref(Rect(*rect)), brush)
    assert call('fixture.GdiFlush', g.GdiFlush)
    # Proper direct baseline readout before reselecting source for native BitBlt.
    restored = call('fixture.restore_for_baseline', g.SelectObject, source, old_source)
    assert restored == source_bitmap
    old_source = None
    info = Info()
    info.bmiHeader = Header(ctypes.sizeof(Header), 16, -16, 1, 32, 0, 1024, 0, 0, 0, 0)
    buffer = ctypes.create_string_buffer(1024)
    baseline_rows = call('fixture.GetDIBits', g.GetDIBits, source, source_bitmap, 0, 16,
                         buffer, ctypes.byref(info), 0)
    raw['baseline'] = {'rows': int(baseline_rows), 'base64': base64.b64encode(buffer.raw).decode(),
                       'sha256': hashlib.sha256(buffer.raw).hexdigest()}
    assert baseline_rows == 16
    old_source = call('fixture.reselect_source', g.SelectObject, source, source_bitmap)
    assert old_source and old_source != ctypes.c_void_p(-1).value
    save()
    class GDI:
        def __getattr__(self, name):
            fn = getattr(g, name)
            def invoke(*args):
                if name == 'GetDIBits':
                    selected = call('candidate.GetCurrentObject', g.GetCurrentObject, args[0], 7)
                    events.append({'phase': 'guard', 'call': 'candidate.bitmap_deselected',
                                   'selected': int(selected or 0), 'bitmap': int(args[1]),
                                   'ok': selected != args[1]})
                    assert selected != args[1], 'candidate violated GetDIBits precondition'
                return call('candidate.' + name, fn, *args)
            return invoke
    class User:
        def GetDC(self, token):
            assert token == 622
            events.append({'phase': 'adapter', 'call': 'candidate.GetDC_owned_source'})
            return source
        def ReleaseDC(self, token, dc):
            assert token == 622 and dc == source
            events.append({'phase': 'adapter', 'call': 'candidate.ReleaseDC_owned_source'})
            return 1
        def PrintWindow(self, *args):
            raise AssertionError('PrintWindow excluded')
    class Backend:
        user32 = User()
        gdi32 = GDI()
    for name, x, y, width, height in freeze['cases']:
        start = len(events)
        data = method(Backend(), 622, x, y, width, height, print_window=False)
        raw['cases'].append({'name': name, 'x': x, 'y': y, 'width': width, 'height': height,
                             'base64': base64.b64encode(data).decode(),
                             'sha256': hashlib.sha256(data).hexdigest(),
                             'event_start': start, 'event_end': len(events)})
        save()
    raw['terminal'] = 'FOUR_OWNED_MEMORY_CAPTURES_COMPLETE'
finally:
    cleanup = {}
    if old_source and source:
        value = call('fixture.restore_final', g.SelectObject, source, old_source)
        cleanup['restore_source'] = bool(value) and value != ctypes.c_void_p(-1).value
    if source_bitmap:
        cleanup['delete_source_bitmap'] = bool(call('fixture.DeleteObject_bitmap', g.DeleteObject, source_bitmap))
    if source:
        cleanup['delete_source_dc'] = bool(call('fixture.DeleteDC', g.DeleteDC, source))
    cleanup['delete_brushes'] = [bool(call('fixture.DeleteObject_brush', g.DeleteObject, b)) for b in brushes]
    if display:
        cleanup['release_format_dc'] = bool(call('fixture.ReleaseDC', u.ReleaseDC, None, display))
    raw['cleanup'] = cleanup
    save()
print(json.dumps({'terminal': raw.get('terminal'), 'cases': len(raw['cases']), 'cleanup': raw['cleanup']}))
