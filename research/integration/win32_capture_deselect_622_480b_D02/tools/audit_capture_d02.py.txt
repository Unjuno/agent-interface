"""Saved JSON/base64-only oracle; no native fixture/backend/DLL imports."""
import base64
import collections
import hashlib
import json
import pathlib
import sys

def classify(raw, receipt, freeze):
    errors = []
    def need(ok, label):
        if not ok:
            errors.append(label)
    def integer(value):
        return type(value) is int
    def nonzero(value):
        return integer(value) and value != 0
    need(integer(receipt['exit_code']) and receipt['exit_code'] == 0 and
         receipt['timed_out'] is False and receipt['attempts'] == 1, 'actual native completion')
    need(integer(raw['pid']) and raw['pid'] == receipt['pid'], 'actual native PID')
    need(raw['schema'] == 'owned-memory-capture-622-D02' and
         raw['candidate_sha256'] == freeze['candidate_sha256'], 'source identity')
    for key in ('input_calls', 'screen_pixel_reads', 'window_handles', 'PrintWindow_calls'):
        need(integer(raw[key]) and raw[key] == 0, key)
    need(raw.get('terminal') == 'FOUR_OWNED_MEMORY_CAPTURES_COMPLETE', 'terminal capture completion')
    desc = raw['source_descriptor']
    need(desc['width'] == 16 and desc['height'] == 16 and desc['planes'] == 1 and
         desc['bits_pixel'] in (24, 32) and desc['bmBits_is_null'] is True, 'known owned color DDB')
    def pixels(row, x, y, width, height, label):
        data = base64.b64decode(row['base64'], validate=True)
        need(len(data) == width * height * 4, label + ' complete buffer')
        need(hashlib.sha256(data).hexdigest() == row['sha256'], label + ' literal hash')
        if len(data) != width * height * 4:
            return
        for yy in range(height):
            for xx in range(width):
                sx, sy = xx + x, yy + y
                color = (0, 0, 255) if sx < 8 and sy < 8 else (0, 255, 0) if sy < 8 else (255, 0, 0) if sx < 8 else (0, 255, 255)
                offset = (yy * width + xx) * 4
                need(tuple(data[offset:offset + 3]) == color, label + ' pixel:' + str((xx, yy)))
    need(integer(raw['baseline']['rows']) and raw['baseline']['rows'] == 16, 'proper baseline rows')
    pixels(raw['baseline'], 0, 0, 16, 16, 'baseline')
    need(len(raw['cases']) == len(freeze['cases']) == 4, 'four exact cases')
    for index, row in enumerate(raw['cases']):
        if index >= len(freeze['cases']):
            errors.append('extra case')
            continue
        name, x, y, width, height = freeze['cases'][index]
        need([row['name'], row['x'], row['y'], row['width'], row['height']] ==
             [name, x, y, width, height], 'exact ordered case:' + name)
        need(all(integer(row[k]) for k in ('x', 'y', 'width', 'height', 'event_start', 'event_end')),
             'typed case:' + name)
        pixels(row, x, y, width, height, name)
        events = raw['events'][row['event_start']:row['event_end']]
        ends = [e for e in events if e['phase'] == 'end']
        counts = collections.Counter(e['call'] for e in ends)
        need(counts == {'candidate.CreateCompatibleDC': 1, 'candidate.CreateCompatibleBitmap': 1,
                        'candidate.SelectObject': 2, 'candidate.BitBlt': 1,
                        'candidate.GetCurrentObject': 1, 'candidate.GetDIBits': 1,
                        'candidate.DeleteObject': 1, 'candidate.DeleteDC': 1}, 'exact GDI call census:' + name)
        guards = [e for e in events if e['phase'] == 'guard']
        need(len(guards) == 1 and guards[0]['ok'] is True and integer(guards[0]['selected']) and
             integer(guards[0]['bitmap']) and guards[0]['selected'] != guards[0]['bitmap'],
             'actual bitmap deselected:' + name)
        selects = [e for e in ends if e['call'] == 'candidate.SelectObject']
        need(len(selects) == 2 and all(nonzero(e['return']) and e['return'] != (2**64 - 1) for e in selects),
             'native select/restoration succeeds:' + name)
        for e in ends:
            if e['call'] in {'candidate.BitBlt', 'candidate.DeleteObject', 'candidate.DeleteDC'}:
                need(nonzero(e['return']), 'native draw/cleanup:' + name + ':' + e['call'])
            if e['call'] == 'candidate.GetDIBits':
                need(integer(e['return']) and e['return'] == height, 'native rows:' + name)
    cleanup = raw['cleanup']
    need(set(cleanup) == {'restore_source', 'delete_source_bitmap', 'delete_source_dc',
                         'delete_brushes', 'release_format_dc'}, 'complete fixture cleanup')
    need(all(cleanup.get(key) is True for key in ('restore_source', 'delete_source_bitmap',
                                                'delete_source_dc', 'release_format_dc')) and
         cleanup.get('delete_brushes') == [True] * 4 and
         all(type(v) is bool for v in cleanup.get('delete_brushes', [])), 'typed fixture cleanup')
    return errors

if __name__ == '__main__':
    out = pathlib.Path(sys.argv[1])
    freeze = json.loads((out / 'FREEZE.json').read_bytes())
    raw = json.loads((out / 'RAW.json').read_bytes())
    receipt = json.loads((out / 'NATIVE_RECEIPT.json').read_bytes())
    errors = classify(raw, receipt, freeze)
    for key, name in [('runner_sha256', 'native_capture_d02.py'), ('launcher_sha256', 'launch_capture_d02.py'),
                      ('auditor_sha256', 'audit_capture_d02.py')]:
        assert hashlib.sha256(pathlib.Path('work', name).read_bytes()).hexdigest() == freeze[key]
    for key, path in [('candidate_sha256', out / 'backend.candidate.py.txt')]:
        assert hashlib.sha256(path.read_bytes()).hexdigest() == freeze[key]
    result = {'status': 'PASS_OWNED_MEMORY_CAPTURE_CANDIDATE' if not errors else 'FAIL_OWNED_MEMORY_CAPTURE_CANDIDATE',
              'errors': errors, 'baseline_coordinates': 256, 'candidate_coordinates': 304,
              'native_replay': 0, 'received_source_execution': 0,
              'scope': 'Four candidate memory-source BitBlt captures with actual native DDB/deselection/cleanup; no window/screen capture or input/task claim; original blank PrintWindow cause unknown.'}
    (out / 'AUDIT.json').write_text(json.dumps(result, indent=2), encoding='utf-8')
    print(json.dumps(result))
    raise SystemExit(bool(errors))
