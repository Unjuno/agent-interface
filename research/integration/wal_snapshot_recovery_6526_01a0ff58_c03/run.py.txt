import datetime,hashlib,json,pathlib,subprocess,sys
P=pathlib.Path(__file__).resolve().parent; label=sys.argv[1]; directory=P/('command-'+label); directory.mkdir(exist_ok=False)
sha=lambda b:hashlib.sha256(b).hexdigest();utc=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
freeze=json.loads((P/'FREEZE.json').read_text(encoding='utf-8'))
for pin in freeze['sources']:
    b=(P/pin['path']).read_bytes()
    if len(b)!=pin['bytes'] or sha(b)!=pin['sha256']:raise RuntimeError('frozen source changed')
argv=[sys.executable,'-X','utf8','-B',str(P/(label+'.py'))]
start=utc(); child=subprocess.Popen(argv,stdout=subprocess.PIPE,stderr=subprocess.PIPE);o,e=child.communicate(timeout=30);end=utc()
(directory/'stdout.txt').write_bytes(o);(directory/'stderr.txt').write_bytes(e)
receipt={'argv':['python','-X','utf8','-B',label+'.py'],'started_utc':start,'ended_utc':end,'child_pid':child.pid,'exit_code':child.returncode,'stdout_bytes':len(o),'stdout_sha256':sha(o),'stderr_bytes':len(e),'stderr_sha256':sha(e),'formal_replays':0,'main_requests':0}
(directory/'receipt.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8');print(json.dumps(receipt));print(o.decode('utf-8'));print(e.decode('utf-8'));sys.exit(child.returncode)
