import ast,copy,datetime,hashlib,json,pathlib,sqlite3,tempfile
P=pathlib.Path(__file__).resolve().parent
source=(P/'audit.py').read_bytes();tree=ast.parse(source)
definitions=[n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name in ('check','verify')]
ns={'P':P,'sha':lambda b:hashlib.sha256(b).hexdigest(),'json':json,'sqlite3':sqlite3,'tempfile':tempfile,'pathlib':pathlib,'base64':__import__('base64')}
exec(compile(ast.Module(body=definitions,type_ignores=[]),'<OWN_FIRST_AUDIT_FUNCTIONS_ONLY>','exec'),ns)
raw=json.loads((P/'native-first/RAW.json').read_text(encoding='utf-8'));witnesses=[]
for kind in ('WRONG_TARGET','WRONG_BEGIN_MODE'):
    data=copy.deepcopy(raw);row=data['rows'][0]
    original='UPDATE counter SET revision=revision+1 WHERE id=1' if kind=='WRONG_TARGET' else 'BEGIN'
    altered='UPDATE counter SET revision=revision+1 WHERE id=0' if kind=='WRONG_TARGET' else 'BEGIN IMMEDIATE'
    for e in row['events']:
        if e['actor']=='reader' and e['sql']==original:e['sql']=altered
    row['traces']['reader']=[altered if x==original else x for x in row['traces']['reader']]
    accepted=True
    try:ns['verify'](data)
    except ValueError:accepted=False
    witnesses.append({'kind':kind,'effective_change':original!=altered,'first_verifier_accepted':accepted,'record':data})
result={'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'first_audit_source_sha256':hashlib.sha256(source).hexdigest(),'scope':'Own first verifier functions only against modified saved copies. No original auditor main invocation/producer/native fixture reexecution.','witnesses':witnesses}
(P/'SQL_CONTROL_WITNESS.json').write_text(json.dumps(result,sort_keys=True,indent=2)+'\n',encoding='utf-8');print(json.dumps({'witnesses':[(w['kind'],w['first_verifier_accepted']) for w in witnesses],'source_sha256':result['first_audit_source_sha256']}))
