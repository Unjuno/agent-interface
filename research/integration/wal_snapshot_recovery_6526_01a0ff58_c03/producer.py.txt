import base64,datetime,hashlib,json,os,pathlib,sqlite3,subprocess,sys
P=pathlib.Path(__file__).resolve().parent
utc=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
sha=lambda b:hashlib.sha256(b).hexdigest()
out=P/'native-first';out.mkdir(exist_ok=False)
conditions=json.loads((P/'PROTOCOL.json').read_text(encoding='utf-8'))['conditions']
allrows=[]
for index,condition in enumerate(conditions):
    db=out/('cell-'+str(index)+'.sqlite'); require_new=not db.exists()
    if not require_new:raise RuntimeError('DB already exists')
    setup=sqlite3.connect(db,isolation_level=None,timeout=0)
    journal=setup.execute('PRAGMA journal_mode=WAL').fetchone()[0]
    setup.executescript('CREATE TABLE counter(id INTEGER PRIMARY KEY, revision INTEGER NOT NULL); INSERT INTO counter VALUES(1,0);')
    setup.close()
    reader=sqlite3.connect(db,isolation_level=None,timeout=0); writer=sqlite3.connect(db,isolation_level=None,timeout=0)
    reader.execute('PRAGMA wal_autocheckpoint=0');writer.execute('PRAGMA wal_autocheckpoint=0')
    traces={'reader':[],'writer':[]};reader.set_trace_callback(traces['reader'].append);writer.set_trace_callback(traces['writer'].append)
    events=[]
    def external():
        c=sqlite3.connect(db.resolve().as_uri()+'?mode=ro',uri=True,isolation_level=None,timeout=0)
        v=c.execute('SELECT revision FROM counter WHERE id=1').fetchone()[0];c.close();return v
    def sql(conn,actor,label,statement):
        before=conn.in_transaction; changes=conn.total_changes; error=None; returned=None; affected=None
        try:
            cursor=conn.execute(statement); affected=cursor.rowcount
            returned=cursor.fetchall()
        except sqlite3.Error as e:error={'class':type(e).__name__,'code':getattr(e,'sqlite_errorcode',None),'name':getattr(e,'sqlite_errorname',None),'message':str(e)}
        after=conn.in_transaction
        local=conn.execute('SELECT revision FROM counter WHERE id=1').fetchone()[0]
        events.append({'actor':actor,'label':label,'sql':statement,'tx_before':before,'tx_after':after,'changes_before':changes,'changes_after':conn.total_changes,'error':error,'returned':returned,'rowcount':affected,'local_after':local,'external_after':external()})
    if condition['snapshot']=='OLD_READ_BEFORE_WRITER':
        sql(reader,'reader','begin','BEGIN');sql(reader,'reader','snapshot','SELECT revision FROM counter WHERE id=1')
    sql(writer,'writer','writer_begin','BEGIN IMMEDIATE')
    sql(writer,'writer','writer_update','UPDATE counter SET revision=revision+1 WHERE id=1')
    sql(writer,'writer','writer_commit','COMMIT');writer.close()
    if condition['snapshot']=='FRESH_READ_AFTER_WRITER':
        sql(reader,'reader','begin','BEGIN');sql(reader,'reader','snapshot','SELECT revision FROM counter WHERE id=1')
    sql(reader,'reader','first_update','UPDATE counter SET revision=revision+1 WHERE id=1')
    if condition['policy']=='ONE_SAME_TRANSACTION_WRITE_REPEAT':
        sql(reader,'reader','planned_repeat','UPDATE counter SET revision=revision+1 WHERE id=1')
    elif condition['policy']=='RESET_AND_NEW_WRITE':
        sql(reader,'reader','reset','ROLLBACK');sql(reader,'reader','new_begin','BEGIN');sql(reader,'reader','new_update','UPDATE counter SET revision=revision+1 WHERE id=1')
    sql(reader,'reader','diagnostic_commit','COMMIT');reader.close()
    siblings=sorted(x.name for x in out.glob(db.name+'*'))
    blob=db.read_bytes(); child_start=utc(); child=subprocess.Popen([sys.executable,'-X','utf8','-B',str(P/'endpoint.py'),str(db)],stdout=subprocess.PIPE,stderr=subprocess.PIPE)
    stdout,stderr=child.communicate(timeout=10); child_end=utc()
    (out/('endpoint-'+str(index)+'.stdout.txt')).write_bytes(stdout);(out/('endpoint-'+str(index)+'.stderr.txt')).write_bytes(stderr)
    endpoint=json.loads(stdout.decode('utf-8')) if child.returncode==0 else None
    row={'condition':condition,'journal':journal,'events':events,'traces':traces,'writer_connection_closed_before_first_update':True,'closed_file_roster':siblings,'db':{'name':db.name,'bytes':len(blob),'sha256':sha(blob),'base64':base64.b64encode(blob).decode()},'endpoint':endpoint,'endpoint_native':{'argv':['python','-X','utf8','-B','endpoint.py',db.name],'pid':child.pid,'started_utc':child_start,'ended_utc':child_end,'exit_code':child.returncode,'stdout_bytes':len(stdout),'stdout_sha256':sha(stdout),'stderr_bytes':len(stderr),'stderr_sha256':sha(stderr)}}
    allrows.append(row)
    (out/'RAW.json').write_text(json.dumps({'utc':utc(),'pid':os.getpid(),'conditions':conditions,'rows':allrows},sort_keys=True,indent=2)+'\n',encoding='utf-8')
    if child.returncode!=0:raise RuntimeError('endpoint failed; preserve prefix')
print(json.dumps({'rows':len(allrows),'endpoints':[r['endpoint']['revision'] for r in allrows],'first_errors':[next(e for e in r['events'] if e['label']=='first_update')['error'] for r in allrows]},sort_keys=True))
