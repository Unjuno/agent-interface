import base64,copy,datetime,hashlib,json,pathlib,sqlite3,tempfile
P=pathlib.Path(__file__).resolve().parent;sha=lambda b:hashlib.sha256(b).hexdigest()
def check(x,m):
    if not x:raise ValueError(m)
def verify(raw):
    conditions=json.loads((P/'PROTOCOL.json').read_text(encoding='utf-8'))['conditions'];check(raw['conditions']==conditions,'conditions');check(len(raw['rows'])==6,'all six')
    total=0
    for i,row in enumerate(raw['rows']):
        check(row['condition']==conditions[i] and row['journal']=='wal','condition order/journal')
        old=row['condition']['snapshot']=='OLD_READ_BEFORE_WRITER';policy=row['condition']['policy']
        wanted=['begin','snapshot','writer_begin','writer_update','writer_commit'] if old else ['writer_begin','writer_update','writer_commit','begin','snapshot']
        wanted+=['first_update']
        if policy=='ONE_SAME_TRANSACTION_WRITE_REPEAT':wanted+=['planned_repeat']
        if policy=='RESET_AND_NEW_WRITE':wanted+=['reset','new_begin','new_update']
        wanted+=['diagnostic_commit'];check([e['label'] for e in row['events']]==wanted,'event sequence')
        # Independently replay the declared SQL into a scalar transaction model.
        committed=0;local={'reader':0,'writer':0};active={'reader':False,'writer':False};snapshot={'reader':None,'writer':None};changes={'reader':0,'writer':0};trace={'reader':[],'writer':[]}
        for event in row['events']:
            actor=event['actor'];sql=event['sql'];before=active[actor];priorchanges=changes[actor];expected_error=None;expected_return=[];rowcount=-1
            declared={'begin':'BEGIN','snapshot':'SELECT revision FROM counter WHERE id=1','writer_begin':'BEGIN IMMEDIATE','writer_update':'UPDATE counter SET revision=revision+1 WHERE id=1','writer_commit':'COMMIT','first_update':'UPDATE counter SET revision=revision+1 WHERE id=1','planned_repeat':'UPDATE counter SET revision=revision+1 WHERE id=1','reset':'ROLLBACK','new_begin':'BEGIN','new_update':'UPDATE counter SET revision=revision+1 WHERE id=1','diagnostic_commit':'COMMIT'}
            check(sql==declared[event['label']], 'exact declared SQL by label')
            if sql in ('BEGIN','BEGIN IMMEDIATE'):active[actor]=True;snapshot[actor]=None;local[actor]=committed
            elif sql.startswith('SELECT '):
                if snapshot[actor] is None:snapshot[actor]=committed;local[actor]=committed
                expected_return=[[local[actor]]]
            elif sql.startswith('UPDATE '):
                if actor=='reader' and snapshot[actor] is not None and snapshot[actor]<committed:
                    expected_error={'class':'OperationalError','code':517,'name':'SQLITE_BUSY_SNAPSHOT','message':'database is locked'};rowcount=None;expected_return=None
                else:
                    if snapshot[actor] is None:snapshot[actor]=committed;local[actor]=committed
                    local[actor]+=1;changes[actor]+=1;rowcount=1
            elif sql=='ROLLBACK':active[actor]=False;snapshot[actor]=None;local[actor]=committed
            elif sql=='COMMIT':
                if actor=='writer' or changes[actor]>0 and not (old and policy!='RESET_AND_NEW_WRITE'):committed=local[actor]
                active[actor]=False;snapshot[actor]=None;local[actor]=committed
            else:raise ValueError('undeclared SQL')
            # The producer's post-operation local SELECT establishes the read
            # snapshot after BEGIN, even before the explicitly labelled SELECT.
            if active[actor] and snapshot[actor] is None:snapshot[actor]=committed;local[actor]=committed
            if not active[actor]:local[actor]=committed
            check(event['tx_before'] is before and event['tx_after'] is active[actor],'transaction state')
            check(event['changes_before']==priorchanges and event['changes_after']==changes[actor],'effect accounting')
            check(event['error']==expected_error and event['returned']==expected_return and event['rowcount']==rowcount,'returned SQL result')
            check(type(event['local_after']) is int and event['local_after']==local[actor] and type(event['external_after']) is int and event['external_after']==committed,'value visibility')
            trace[actor]+=[sql,'SELECT revision FROM counter WHERE id=1'];total+=1
        check(trace==row['traces'],'full SQL trace')
        expected=1 if old and policy!='RESET_AND_NEW_WRITE' else (3 if not old and policy=='ONE_SAME_TRANSACTION_WRITE_REPEAT' else 2)
        check(committed==expected and row['writer_connection_closed_before_first_update'] is True,'declared diagnostic endpoint')
        check(row['closed_file_roster']==[row['db']['name']],'closed complete DB roster')
        blob=base64.b64decode(row['db']['base64']);check(len(blob)==row['db']['bytes'] and sha(blob)==row['db']['sha256'],'DB capsule')
        check(blob==(P/'native-first'/row['db']['name']).read_bytes(),'actual original DB image')
        with tempfile.TemporaryDirectory(prefix='own-c03-audit-') as directory:
            target=pathlib.Path(directory)/'copy.sqlite';target.write_bytes(blob);connection=sqlite3.connect(target.as_uri()+'?mode=ro&immutable=1',uri=True,isolation_level=None)
            check(connection.execute('SELECT revision FROM counter WHERE id=1').fetchall()==[(expected,)],'closed DB revision');check(connection.execute('PRAGMA integrity_check').fetchone()[0]=='ok','closed integrity');connection.close();check(target.read_bytes()==blob,'copy unchanged')
        endpoint=row['endpoint'];check(endpoint['revision']==expected and endpoint['integrity']=='ok' and endpoint['unchanged'] is True,'fresh endpoint value')
        check(endpoint['db_sha256']==sha(blob) and endpoint['db_bytes']==len(blob),'endpoint DB custody')
        native=row['endpoint_native'];check(native['exit_code']==0 and native['pid']>0 and native['started_utc']<=native['ended_utc'],'native terminal')
        for stream in ('stdout','stderr'):
            b=(P/'native-first'/('endpoint-'+str(i)+'.'+stream+'.txt')).read_bytes();check(len(b)==native[stream+'_bytes'] and sha(b)==native[stream+'_sha256'],'native stream')
            if stream=='stdout':check(json.loads(b)==endpoint,'full endpoint stdout')
    return total
raw=json.loads((P/'native-first/RAW.json').read_text(encoding='utf-8'));total=verify(raw);controls=[]

    try:verify(altered)
    except (ValueError,KeyError):controls.append({'label':label,'refused':True,'record':altered});return
    raise RuntimeError('control accepted '+label)
