from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

WORK = Path(__file__).resolve().parent
REPO = WORK / "agent-interface"
P = REPO / "research/integration/core_admission_composition_57_20261003_01a0ff59"
def utc():
    return datetime.now(timezone.utc).isoformat()
def write_json(path, value):
    path.write_bytes(json.dumps(value, indent=2).encode() + b"\n")

if sys.argv[1] == "freeze":
    if (P / "FREEZE.json").exists():
        raise RuntimeError("freeze exists; preserve first freeze")
    files = sorted(f for f in P.rglob("*") if f.is_file())
    freeze = {"schema": "core-admission-composition-freeze-v1", "frozen_at_utc": utc(),
              "worker_id": "01a0ff59-2f23-7ff0-ba99-cc5cb8d494fc", "policy_version": "FINAL-v5",
              "allocation_type": "finite analytical / ordinary engineering; no formal allocation",
              "case_count_per_arm": 9000, "rows": 36000, "invocations": 1,
              "environment": {"platform": sys.platform, "python": sys.version,
                              "processor": os.environ.get("PROCESSOR_IDENTIFIER"),
                              "threads": "one Python main thread; no worker pool",
                              "shared_resources": "none acquired", "fleet_deadline": "unavailable"},
              "source_sha256": {f.relative_to(P).as_posix(): hashlib.sha256(f.read_bytes()).hexdigest() for f in files},
              "decision": "combined 9000/9000 strict oracle match; no exception/mutation; effective negative controls; core regressions"}
    write_json(P / "FREEZE.json", freeze)
    print(json.dumps(freeze, indent=2))
else:
    label = sys.argv[1]
    commands = {
        "matrix": (P, ["-B", "run_matrix.py"]),
        "audit": (P, ["-B", "audit_matrix.py"]),
        "controls": (P, ["-B", "check_controls.py"]),
        "core": (WORK / "composition-suite", ["-B", "-m", "unittest", "discover", "-s", "runtime/core_v1", "-t", ".", "-v"]),
        "core-optimized": (WORK / "composition-suite", ["-B", "-O", "-m", "unittest", "discover", "-s", "runtime/core_v1", "-t", ".", "-v"]),
        "doctor": (WORK / "composition-suite", ["-B", "-m", "runtime.core_v1.doctor"]),
    }
    cwd, args = commands[label]
    log = P / "logs"
    log.mkdir(exist_ok=True)
    receipt = log / (label + ".json")
    if receipt.exists():
        raise RuntimeError("record exists; do not overwrite: " + label)
    start = utc()
    env = dict(os.environ, PYTHONDONTWRITEBYTECODE="1", OMP_NUM_THREADS="1", OPENBLAS_NUM_THREADS="1")
    result = subprocess.run([sys.executable, *args], cwd=cwd, env=env, capture_output=True, timeout=120)
    end = utc()
    metadata = {"command": ["python", *args], "cwd_role": "package" if cwd == P else "private exact-Git-byte core export",

    private = WORK / "command-originals"
    private.mkdir(exist_ok=True)
    metadata["logs"] = {}
    for stream, data in [("stdout", result.stdout), ("stderr", result.stderr)]:
        (private / (label + "." + stream)).write_bytes(data)
        public = data.replace(str(WORK).encode(), b"<private-work>").replace(str(Path.home()).encode(), b"<private-home>")
        (log / (label + "." + stream)).write_bytes(public)
        metadata["logs"][stream] = {"original_sha256": hashlib.sha256(data).hexdigest(),
                                    "published_sha256": hashlib.sha256(public).hexdigest(),
                                    "private_prefix_redaction": data != public}
    write_json(receipt, metadata)
    print(json.dumps(metadata, indent=2))
    print(result.stdout.decode(errors="replace")[-6500:])
    print(result.stderr.decode(errors="replace")[-4500:])
    sys.exit(result.returncode)
