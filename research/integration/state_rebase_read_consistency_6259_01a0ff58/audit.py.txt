"""Independent raw truth check, without importing producer."""
import json
from pathlib import Path
p=Path(__file__).resolve().parent
d=json.loads((p/'RAW.json').read_text(encoding='utf-8'))
assert len(d['rows'])==6
assert len({(r['policy'],r['mutate']) for r in d['rows']})==6
for r in d['rows']:
 expected=r['policy']=='split' and r['mutate']
 assert r['accepted'] is expected
 assert r['first']==['X',0]
 assert r['second']==([0,1] if r['mutate'] and r['policy']!='snapshot' else [1,0])
 assert r['actual']==(['Y',0,1] if r['mutate'] else ['X',1,0])
 assert not (r['actual'][0]=='X' and r['actual'][1]==0)
result={'verdict':'CONSTRUCTION_PASS_SCOPED','false_acceptances':{'split':1,'snapshot':0,'generation_join':0},'real_satisfied_states':0,'rows':6,'limits':'No real task success or replay permission; assumes writer updates generation atomically.'}
(p/'AUDIT.json').write_text(json.dumps(result,indent=2),encoding='utf-8')
print(json.dumps(result))
