from pathlib import Path
import sys,types,unittest
archive=Path(sys.argv[1]).resolve(); sys.path.insert(0,str(archive/'runtime.pyz'))
suite=unittest.TestSuite()
for name in ('test_native_reference_pointer','test_event_reference_marker','test_receipt_pointer_escapes'):
    module=types.ModuleType('archive_check_'+name); sys.modules[module.__name__]=module
    path=archive/(name+'.py.txt'); exec(compile(path.read_bytes(),str(path),'exec'),module.__dict__)
    suite.addTests(unittest.defaultTestLoader.loadTestsFromModule(module))
for name,module in tuple(sys.modules.items()):
    if name=='runtime' or name.startswith('runtime.'):
        origin=getattr(module,'__file__',None)
        if origin and not origin.startswith(str(archive/'runtime.pyz')+'/'):
            raise RuntimeError('nonarchive runtime import: '+name+' '+origin)
result=unittest.TextTestRunner(verbosity=2).run(suite)
if result.testsRun != 21: raise RuntimeError('incorrect archive coverage')
raise SystemExit(0 if result.wasSuccessful() else 1)
