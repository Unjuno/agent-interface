import ast
import hashlib
import json
from pathlib import Path


ROOT = Path(__file__).resolve().parent
raw = (ROOT / 'client.py.txt').read_bytes()
tree = ast.parse(raw.decode('utf-8'))
subject = next(v for v in tree.body if isinstance(v, ast.ClassDef) and v.name == 'CodexAppServerClient')
original_methods = {v.name: ast.dump(v, include_attributes=False) for v in subject.body if isinstance(v, ast.FunctionDef)}
for method in subject.body:
    if not isinstance(method, ast.FunctionDef):
        continue
    if method.name == '__init__':
        index = next(i for i, v in enumerate(method.body) if isinstance(v, ast.Assign) and any(isinstance(t, ast.Attribute) and t.attr == '_responses' for t in v.targets))
        method.body[index:index] = ast.parse('self._pending = set()').body
    elif method.name == '_read':
        loop = method.body[0].body[0]
        condition = next(v for v in loop.body if isinstance(v, ast.With))
        gate = condition.body[0]
        original_assignment = gate.body[0]
        owned = ast.parse('if message["id"] in self._pending:\n    pass').body[0]
        owned.body = [original_assignment]
        gate.body = [owned]
    elif method.name == 'request':
        allocation = method.body[0]
        allocation.body.extend(ast.parse('self._pending.add(identifier)').body)
        final = ast.parse('with self._condition:\n    self._pending.discard(identifier)\n    self._responses.pop(identifier, None)').body
        method.body = [allocation, ast.Try(body=method.body[1:], handlers=[], orelse=[], finalbody=final)]
ast.fix_missing_locations(tree)
changed = [v.name for v in subject.body if isinstance(v, ast.FunctionDef) and ast.dump(v, include_attributes=False) != original_methods[v.name]]
if set(changed) != {'__init__', '_read', 'request'}:
    raise ValueError('private comparison expanded outside pending ownership')
control = ('# Private diagnostic comparator; not shared/runtime adoption.\n' + ast.unparse(tree) + '\n').encode('utf-8')
(ROOT / 'pending_control.py.txt').write_bytes(control)
inputs = {'id': 'appserver-late-reply-59-df63-a01-20261003',
          'source_main': '10fba1d3c9aa0a6fbf39983cdcda099a10df497c',
          'source_sha256': hashlib.sha256(raw).hexdigest(),
          'control_sha256': hashlib.sha256(control).hexdigest(),
          'late_counts': [0, 1, 7, 23], 'arms': ['current', 'pending'],
          'orphan_id': 104729, 'result_blob_characters': 4097,
          'request_blob_characters': 2053, 'timeout_for_retirement': 0,
          'healthy_timeout_seconds': 1, 'fixture_join_seconds': 2,
          'cells': 10, 'original_or_formal_producer_replay': False,
          'max_active_fixtures': 1, 'native_peers': 0, 'output_cap_bytes': 4194304}
(ROOT / 'INPUT.json').write_text(json.dumps(inputs, sort_keys=True, ensure_ascii=True, indent=2) + '\n', encoding='utf-8')
(ROOT / 'COMPARATOR_AST_PROOF.json').write_text(json.dumps({'changed_methods': changed,
            'unchanged_methods': sorted(set(original_methods) - set(changed)),
            'constructor_change': 'initialize pending set before reader starts',
            'reader_change': 'cache only IDs in pending set; record every frame first',
            'request_change': 'register allocated ID under condition; retire/cache-pop in finally',
            'source_sha256': inputs['source_sha256'], 'control_sha256': inputs['control_sha256'],
            'subject_or_control_executed': False}, indent=2) + '\n', encoding='utf-8')
print(json.dumps({'source_bytes': len(raw), 'control_bytes': len(control), 'changed_methods': changed, 'subject_executed': False}))
