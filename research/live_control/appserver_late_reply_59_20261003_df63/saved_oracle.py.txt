import copy

ROOT = Path(__file__).resolve().parent
CONFIG = json.loads((ROOT / 'INPUT.json').read_text(encoding='utf-8'))

def literal(value):
    return json.dumps(value, sort_keys=True, ensure_ascii=True, separators=(',', ':'))

def equal(actual, expected, label):
    if literal(actual) != literal(expected):
        raise AssertionError(label)

def rich(kind, identifier):
    return {'kind': kind, 'identifier': identifier, 'blob': 'R' * 4097,
            'meaning': '\u4fdd\u6301 caf\u00e9', 'integer': 37}

def timeout(method):
    return {'error_type': 'TimeoutError', 'error_text': 'app-server request timed out: ' + method, 'result': None}

def response(kind, identifier):
    return {'id': identifier, 'result': rich(kind, identifier)}

def request(method, identifier):
    message = {'method': method, 'id': identifier,
               'params': {'rich_input': 'M' * 2053, 'semantic_marker': 37}}
    return {'literal': json.dumps(message, separators=(',', ':')) + '\n', 'message': message}

NOTICE = {'method': 'study/notice', 'params': {'meaning': '\u4fdd\u6301', 'token': 'notice-37', 'value': 37}}

def expected(arm, case, count):
    requests, frames, journal = [], [], []
    def sent(method, identifier):
        record = request(method, identifier)
        requests.append(record)
        journal.append({'direction': 'sent', 'message': record['message']})
    def received(kind, identifier, stage):
        message = response(kind, identifier)
        frames.append({'stage': stage, 'message': message})
        journal.append({'direction': 'received', 'message': message})
    cache = []
    if case == 'future':
        received('unowned-before-invocation', 1, 'before_request')
        sent('study/held-fresh', 1)
        outcomes = [{'error_type': None, 'result': rich('unowned-before-invocation', 1)}] if arm == 'current' else [timeout('study/held-fresh')]
        received('fresh-after-retirement', 1, 'after_first_outcome')
        if arm == 'current':
            cache.append({'id': 1, 'message': response('fresh-after-retirement', 1)})
        healthy_id = 2
    else:
        for identifier in range(1, count + 1):
            sent('study/retire', identifier)
        outcomes = [timeout('study/retire') for _ in range(count)]
        for identifier in range(1, count + 1):
            received('late', identifier, 'after_all_timeouts')
            if arm == 'current':
                cache.append({'id': identifier, 'message': response('late', identifier)})
        healthy_id = count + 1
    sent('study/healthy', healthy_id)
    received('healthy', healthy_id, 'healthy_reply')
    if count:
        received('duplicate-completed', healthy_id, 'after_healthy_outcome')
        received('orphan', 104729, 'unowned_orphan')
        if arm == 'current':
            cache.extend([{'id': healthy_id, 'message': response('duplicate-completed', healthy_id)},
                          {'id': 104729, 'message': response('orphan', 104729)}])
    frames.append({'stage': 'notice', 'message': NOTICE})
    journal.append({'direction': 'received', 'message': NOTICE})
    return {'requests': requests, 'frames': frames, 'journal': journal, 'outcomes': outcomes,
            'healthy': {'error_type': None, 'result': rich('healthy', healthy_id)}, 'notice': NOTICE,
            'before_close': {'closed': False, 'cache': cache, 'notification_count': 0,
                             'pending': [] if arm == 'pending' else None, 'next_id': healthy_id + 1},
            'cached_payload_bytes': len(literal(cache).encode('utf-8'))}

def check(row, journal, arm, case, count):
    equal([row['arm'], row['case'], row['count']], [arm, case, count], 'cell identity/type')
    if 'harness_error' in row or 'cleanup_sync_error' in row:
        raise AssertionError('first construction or cleanup failure')
    equal(row['source_sha256'], CONFIG['source_sha256'] if arm == 'current' else CONFIG['control_sha256'], 'source digest')
    equal(row['source_bytes'], 6194 if arm == 'current' else 6439, 'source bytes')
    equal(row['source_unchanged'], True, 'source after execution')
    if type(row['driver_pid']) is not int or row['driver_pid'] <= 0:
        raise AssertionError('CPU driver PID')
    begin, end = [datetime.datetime.fromisoformat(row[key]) for key in ('started_utc', 'ended_utc')]
    if begin.tzinfo is None or end < begin:
        raise AssertionError('UTC interval')
    equal(row['factory'], [{'command': ['owned-inert-fifo'], 'kwargs': {'cwd': None, 'stdin': -1, 'stdout': -1,
        'stderr': -1, 'text': True, 'bufsize': 1}, 'native_process_created': False}], 'literal fake factory')
    equal(row['terminal'], {'reader_alive': False, 'journal_closed': True, 'stdin_closed': True,
        'stdout_closed': True, 'stderr_closed': True, 'fake_terminate_calls': 0, 'native_processes_created': 0}, 'terminal resources')
    exp = expected(arm, case, count)
    for key in exp:
        if key != 'journal':
            equal(row[key], exp[key], 'literal ' + key)
    projected = []
    clocks = []
    for entry in journal:
        equal(sorted(entry), ['direction', 'message', 'observed_ns'], 'journal schema')
        if type(entry['observed_ns']) is not int or entry['observed_ns'] <= 0:
            raise AssertionError('journal integer clock')
        clocks.append(entry['observed_ns'])
        projected.append({key: entry[key] for key in ('direction', 'message')})
    if clocks != sorted(set(clocks)):
        raise AssertionError('journal complete increasing clocks')
    equal(projected, exp['journal'], 'whole sent/received journal order')
