import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys

sys.stdout.reconfigure(encoding='utf-8')
ROOT=Path(__file__).resolve().parent
REPO=ROOT.parent/'agent-interface'
BASE='a878df7886c5901f86806aed5a4696c81457c9e7'
BRANCH='research/appserver-late-reply-59-df63-20261003'
PREFIX='research/live_control/appserver_late_reply_59_20261003_df63'
INDEX=ROOT/'owned_archive_v2.index'
assert not INDEX.exists()
env=dict(os.environ,GIT_INDEX_FILE=str(INDEX),GIT_TERMINAL_PROMPT='0',GIT_OPTIONAL_LOCKS='0')
def git(*args,data=None,env_override=None):
    return subprocess.check_output(['git','-C',str(REPO),*args],input=data,env=env_override or env)
root_head=git('rev-parse','HEAD').decode().strip()
root_status=git('status','--porcelain',env_override=os.environ.copy()).decode()
assert root_head=='2cb3873bf54882904b1818467a77eb55d167906b' and root_status==''
assert not git('ls-remote','--heads','origin','refs/heads/'+BRANCH).strip(), 'remote owned branch exists; stop'
assert subprocess.run(['git','-C',str(REPO),'show-ref','--verify','--quiet','refs/heads/'+BRANCH]).returncode==1
public=ROOT/'public'
manifest={line.split('  ',1)[1]:line.split('  ',1)[0] for line in (public/'SHA256SUMS').read_text().splitlines()}
assert len(manifest)==53 and 'SHA256SUMS' not in manifest
allpaths={p.relative_to(public).as_posix():p for p in public.rglob('*') if p.is_file()}
assert set(allpaths)==set(manifest)|{'SHA256SUMS'}
for path,digest in manifest.items():
    assert hashlib.sha256(allpaths[path].read_bytes()).hexdigest()==digest
assert sum(p.stat().st_size for p in (ROOT/'a01').rglob('*') if p.is_file())<4194304
git('read-tree',BASE)
images=[]
for name,path in sorted(allpaths.items()):
    data=path.read_bytes()
    oid=git('hash-object','-w','--stdin',data=data).decode().strip()
    target=PREFIX+'/'+name
    git('update-index','--add','--cacheinfo','100644',oid,target)
    images.append({'path':target,'mode':'100644','blob':oid,'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest()})
nav=git('show',BASE+':research/live_control/README.md')
addition=b'| #59 late/unowned app-server response lifetime (controlled FIFO) | [First ten cells and limits](appserver_late_reply_59_20261003_df63/README.md) |\n'
anchor=b'\n## Track map\n'
assert nav.count(anchor)==1
newnav=nav.replace(anchor,b'\n'+addition+anchor)
oid=git('hash-object','-w','--stdin',data=newnav).decode().strip()
git('update-index','--cacheinfo','100644',oid,'research/live_control/README.md')
images.append({'path':'research/live_control/README.md','mode':'100644','blob':oid,'bytes':len(newnav),'sha256':hashlib.sha256(newnav).hexdigest()})
tree=git('write-tree').decode().strip()
delta=git('diff','--name-status',BASE,tree).decode('utf-8').splitlines()
assert len(delta)==55 and sum(line.startswith('M\t') for line in delta)==1
checks=subprocess.run(['git','-C',str(REPO),'diff','--check',BASE,tree],env=env,capture_output=True)
(ROOT/'BUILD_DIFF_CHECK_V2.stdout').write_bytes(checks.stdout)
(ROOT/'BUILD_DIFF_CHECK_V2.stderr').write_bytes(checks.stderr)
record={'base':BASE,'tree':tree,'branch':BRANCH,'images':images,'delta':delta,
    'diff_check_exit':checks.returncode,'root_head':root_head,'root_delivery_unchanged':True,
    'recorded_utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
(ROOT/'LOCAL_TREE_V2.json').write_text(json.dumps(record,sort_keys=True,indent=2)+'\n',encoding='utf-8')
if checks.returncode:
    print(checks.stdout.decode('utf-8'))
    raise SystemExit(checks.returncode)
head=git('commit-tree',tree,'-p',BASE,data=b'research(#59): retain late and unowned app-server response boundary [skip ci]\n').decode().strip()
git('update-ref','refs/heads/'+BRANCH,head,'0'*40)
assert git('rev-parse','HEAD').decode().strip()==root_head
assert git('status','--porcelain',env_override=os.environ.copy()).decode()==root_status
record['head']=head
(ROOT/'LOCAL_COMMIT.json').write_text(json.dumps(record,sort_keys=True,indent=2)+'\n',encoding='utf-8')
print(json.dumps({'head':head,'tree':tree,'base':BASE,'paths':len(images),'manifest_targets':len(manifest),'diff_check_exit':checks.returncode,'root_delivery_unchanged':True}))
