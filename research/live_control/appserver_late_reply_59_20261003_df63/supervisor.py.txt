import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import time

ROOT = Path(__file__).resolve().parent
phase = sys.argv[1]
assert phase in ('driver','saved_oracle')
freeze = json.loads((ROOT/'FREEZE.json').read_text(encoding='utf-8'))
def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def pins():
    for row in freeze['pins']:
        if sha(row['path']) != row['sha256']:
            raise AssertionError('frozen image changed: '+row['label'])
pins()
target = ROOT/(phase+'.py')
prefix = ROOT/(phase+'_PROCESS')
assert not prefix.with_suffix('.json').exists(), 'first execution receipt already exists; no rerun'
env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1', PYTHONIOENCODING='utf-8')
started = datetime.datetime.now(datetime.timezone.utc).isoformat()
clock = time.monotonic()
process = subprocess.Popen([sys.executable,'-B',str(target)], cwd=ROOT, env=env, stdout=subprocess.PIPE,stderr=subprocess.PIPE)
timed_out=False
try:
    stdout,stderr=process.communicate(timeout=20)
except subprocess.TimeoutExpired:
    timed_out=True
    process.kill()
    stdout,stderr=process.communicate(timeout=5)
ended = datetime.datetime.now(datetime.timezone.utc).isoformat()
prefix.with_suffix('.stdout').write_bytes(stdout)
prefix.with_suffix('.stderr').write_bytes(stderr)
pin_error=None
try:
    pins()
except Exception as error:
    pin_error=str(error)
receipt={'phase':phase,'command':[sys.executable,'-B',str(target)],'supervisor_pid':os.getpid(),
    'pid':process.pid,'started_utc':started,'ended_utc':ended,'elapsed_seconds':time.monotonic()-clock,
    'returncode':process.returncode,'timed_out':timed_out,'stdout_bytes':len(stdout),'stderr_bytes':len(stderr),
    'stdout_sha256':hashlib.sha256(stdout).hexdigest(),'stderr_sha256':hashlib.sha256(stderr).hexdigest(),
    'freeze_sha256':sha(ROOT/'FREEZE.json'),'post_pin_error':pin_error,'terminal':process.poll() is not None}
prefix.with_suffix('.json').write_text(json.dumps(receipt,sort_keys=True,indent=2)+'\n',encoding='utf-8')
print(json.dumps(receipt,sort_keys=True))
if process.returncode or timed_out or pin_error or len(stdout)+len(stderr)>4194304:
    raise SystemExit(1)
