import ast
import datetime
import hashlib
import importlib.machinery
import inspect
import io
import json
import os
from pathlib import Path
import queue
import shutil
import subprocess
import sys
import threading
ROOT=Path(__file__).resolve().parent
REPO=ROOT.parent/'agent-interface'
assert not (ROOT/'a01').exists(), 'subject execution already exists'
assert not (ROOT/'FREEZE.json').exists(), 'freeze already exists'
cfg=json.loads((ROOT/'INPUT.json').read_text(encoding='utf-8'))
readback=json.loads((ROOT/'PRE_FREEZE_READBACK.json').read_text(encoding='utf-8'))
claim=json.loads(readback['claim_readback']['value']['structuredContent']['content'])
local=(ROOT/'CLAIM_REQUEST.txt').read_text(encoding='utf-8').strip()
assert local == claim['body'], 'whole actual claim/request mismatch'
g=lambda *args:subprocess.check_output(['git','-C',str(REPO),*args])
main=g('rev-parse','origin/main').decode().strip()
assert main=='a878df7886c5901f86806aed5a4696c81457c9e7'
source=g('show',main+':research/live_control/codex_app_server_client_v2.py')
assert source==(ROOT/'client.py.txt').read_bytes()
assert hashlib.sha256(source).hexdigest()==cfg['source_sha256']
images=g('ls-tree','-r','--full-tree',main).decode('utf-8')
assert not any(line.rsplit('\t',1)[1].rsplit('/',1)[-1]=='AGENTS.md' for line in images.splitlines())
definitions=['README.md','ROADMAP.md','CURRENT_GOAL.md','WORKER_QUICKSTART.md','RESEARCH_METHOD.md','ISSUE_FAILURE_CLASS.md']
definition_rows=[]
for name in definitions:
    path=next(line.rsplit('\t',1)[1] for line in images.splitlines() if line.rsplit('\t',1)[1]==name)
    blob=g('show',main+':'+path)
    definition_rows.append({'path':path,'blob':g('rev-parse',main+':'+path).decode().strip(),'sha256':hashlib.sha256(blob).hexdigest(),'bytes':len(blob)})
pins=[]
def pin(path,label):
    path=Path(path).resolve()
    data=path.read_bytes()
    pins.append({'label':label,'path':str(path),'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest()})
for path in sorted(ROOT.iterdir()):
    if path.is_file():
        if path.suffix=='.py':
            ast.parse(path.read_text(encoding='utf-8'))
        pin(path,path.name)
pin(sys.executable,'Python executable')
for module in [json,queue,threading,subprocess,importlib.machinery,inspect,shutil]:
    pin(inspect.getsourcefile(module),'stdlib '+module.__name__)
record={'id':cfg['id'],'frozen_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
 'recorded_main':main,'source_context':cfg['source_main'],'source_same_bytes_at_recheck':True,
 'governing_definitions':definition_rows,'no_tracked_AGENTS_complete_tree':True,
 'claim_full_body_readback':claim['id'],'positive_id_only':True,'native_peers':0,
 'no_subject_execution_before_freeze':True,'pins':pins,'python_version':sys.version,
 'supervisor_seconds':20,'original_output_cap_bytes':4194304,
 'max_active_fixtures':1,'N':None,'common_deadline':None,'new_workers':0,
 'disk_free_bytes':shutil.disk_usage(ROOT).free}
(ROOT/'FREEZE.json').write_text(json.dumps(record,sort_keys=True,indent=2)+'\n',encoding='utf-8')
print(json.dumps({'frozen_utc':record['frozen_utc'],'source':cfg['source_sha256'],'pins':len(pins),'disk_free_bytes':record['disk_free_bytes'],'subject_executed':False}))
