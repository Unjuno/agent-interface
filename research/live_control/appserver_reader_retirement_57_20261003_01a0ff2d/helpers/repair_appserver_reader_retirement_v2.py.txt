import ast, hashlib, json
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parent.parent
private = root / 'work/repairs/appserver-journal-close-02'
out = root / 'outputs/appserver-journal-close-02'
path = private / 'codex_app_server_client_v2.py'
before = path.read_bytes()
if hashlib.sha256(before).hexdigest() != '464c770e357b664c35866e271740380307bfcd52dbaa8573e258d5600c006531':
    raise ValueError('V1 identity differs')
red = json.loads((out / 'red/receipt.json').read_bytes())
if red['exit_code'] != 1:
    raise ValueError('expected preserved first regression failure')
old = b'        self._reader.join(timeout=timeout)\n'
new = old + b'        if self._reader.is_alive():\n            raise TimeoutError("app-server reader close timed out")\n'
if before.count(old) != 1:
    raise ValueError('reader join location differs')
after = before.replace(old, new, 1)
def methods(data):
    return {n.name: ast.dump(n, include_attributes=False) for n in ast.walk(ast.parse(data)) if isinstance(n, ast.FunctionDef)}
first, second = methods(before), methods(after)
if first.keys() != second.keys() or [k for k in first if first[k] != second[k]] != ['close']:
    raise ValueError('changed method scope differs')
record = {'created_before_green_utc': datetime.now(timezone.utc).isoformat(),
          'before_sha256': hashlib.sha256(before).hexdigest(), 'after_sha256': hashlib.sha256(after).hexdigest(),
          'after_bytes': len(after), 'only_changed_method': 'close', 'exact_old': old.decode(), 'exact_new': new.decode(),
          'scope': 'report actual live reader after join before touching journal; leave reader _closed/cache state for actual _read finally; retain earlier mutex and None/error semantics'}
(out / 'FIX_BEFORE_GREEN.json').write_text(json.dumps(record, indent=2) + '\n', encoding='utf-8')
path.write_bytes(after)
print(json.dumps(record, indent=2))
