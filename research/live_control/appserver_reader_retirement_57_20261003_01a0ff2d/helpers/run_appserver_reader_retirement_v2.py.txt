import hashlib, json, os, subprocess, sys
from datetime import datetime, timezone
from pathlib import Path

started = datetime.now(timezone.utc).isoformat()
root = Path(__file__).resolve().parent.parent
private = root / 'work/repairs/appserver-journal-close-02'
out = root / 'outputs/appserver-journal-close-02'
stage = sys.argv[1]
if stage not in ('red', 'green-normal', 'green-optimized'):
    raise ValueError(stage)
target = out / stage
target.mkdir(exist_ok=False)
cases = target / 'cases'
cases.mkdir()
freeze = json.loads((out / 'FREEZE_BEFORE_RED.json').read_bytes())
identities = {}
for name, expected in [('test_appserver_journal_close_01a0ff2d.py', freeze['adapted_test_sha256']),
                       ('test_appserver_reader_retirement_01a0ff2d.py', freeze['new_test_sha256']),
                       ('codex_app_server_client_v2.py', None)]:
    data = (private / name).read_bytes()
    digest = hashlib.sha256(data).hexdigest()
    if expected is not None and digest != expected:
        raise ValueError('frozen test changed')
    identities[name] = {'sha256': digest, 'bytes': len(data)}
    (target / ('executed-' + name + '.txt')).write_bytes(data)
selected = ['test_appserver_reader_retirement_01a0ff2d'] if stage == 'red' else ['test_appserver_journal_close_01a0ff2d', 'test_appserver_reader_retirement_01a0ff2d']
command = [sys.executable, '-B'] + (['-O'] if stage == 'green-optimized' else []) + ['-m', 'unittest', '-v', *selected]
intent = {'driver_pid': os.getpid(), 'driver_started_utc': started, 'stage': stage, 'argv': command,
          'cwd': str(private), 'identities': identities, 'ordinary_repair_regression': True,
          'peer_or_formal_replay': False, 'written_before_native_start_utc': datetime.now(timezone.utc).isoformat()}
(target / 'INTENT.json').write_text(json.dumps(intent, indent=2) + '\n', encoding='utf-8')
env = dict(os.environ, PYTHONDONTWRITEBYTECODE='1', JOURNAL_CLOSE_EVIDENCE_DIR=str(cases))
begin = datetime.now(timezone.utc).isoformat()
process = subprocess.Popen(command, cwd=private, env=env, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
stdout, stderr = process.communicate(timeout=25)
end = datetime.now(timezone.utc).isoformat()
for name, data in [('stdout.log', stdout), ('stderr.log', stderr)]:
    (target / name).write_bytes(data)
receipt = dict(intent, pid=process.pid, started_utc=begin, ended_utc=end, exit_code=process.returncode,
               stdout_bytes=len(stdout), stdout_sha256=hashlib.sha256(stdout).hexdigest(),
               stderr_bytes=len(stderr), stderr_sha256=hashlib.sha256(stderr).hexdigest())
(target / 'receipt.json').write_text(json.dumps(receipt, indent=2) + '\n', encoding='utf-8')
print(json.dumps(receipt, indent=2))
print(stderr.decode(errors='replace'))
sys.exit(process.returncode)
