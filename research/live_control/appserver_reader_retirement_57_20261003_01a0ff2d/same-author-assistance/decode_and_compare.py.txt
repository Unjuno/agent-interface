"""Decode public evidence data and inspect ASTs only; never imports/executes peer code."""
import ast
import base64
import datetime
import gzip
import hashlib
import json
from pathlib import Path, PurePosixPath
import traceback

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
TRIGGER = ROOT/'outputs/6955-reader-retirement-trigger.json'
OWN = ROOT/'work/repairs/appserver-journal-close-01'

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def method(raw, name):
    tree = ast.parse(raw)
    cls = next(n for n in tree.body if isinstance(n,ast.ClassDef) and n.name=='CodexAppServerClient')
    return next(n for n in cls.body if isinstance(n,ast.FunctionDef) and n.name==name)

def main():
    trigger_raw = TRIGGER.read_bytes()
    trigger = json.loads(trigger_raw)
    body = trigger['native_source_outcome']['body']
    blocks = body.split(chr(96)*3)
    assert len(blocks)==3
    language, encoded = blocks[1].split('\n',1)
    assert language=='base64-gzip-json'
    compressed = base64.b64decode(''.join(encoded.split()),validate=True)
    assert len(compressed)==24144 and sha(compressed)=='4a3b2d239995d1f4981ec6ce7ba9cb55bafbc404e53aa722be9066bd176b6e27'
    payload = gzip.decompress(compressed)
    assert len(payload)==126551 and sha(payload)=='f78eb859e84247c0165d3b8d6740f1361f2efa9c162c23104379a197c45adbd0'
    envelope=json.loads(payload)
    assert envelope['schema']=='b04b/client-close-reader-native-v1'
    assert envelope['worker']=='01a0ff33-b04b-7b51-8194-a60b82fed8a2'
    assert len(envelope['files'])==58
    data=HERE/'peer-data'
    records=[]; files={}
    for row in envelope['files']:
        path=PurePosixPath(row['path'])
        assert not path.is_absolute() and path.parts and all(p not in ('..','.') for p in path.parts)
        assert ':' not in row['path'] and '\\' not in row['path'] and row['path'] not in files
        raw=row['utf8'].encode('utf-8')
        assert len(raw)==row['public_bytes'] and sha(raw)==row['public_sha256']
        assert row['changed'] is (row['original_bytes']!=row['public_bytes'] or row['original_sha256']!=row['public_sha256'])
        if not row['changed']:
            assert row['projection']=='identity'
        target=data/row['path']
        target.parent.mkdir(parents=True,exist_ok=True)
        with target.open('xb') as handle:handle.write(raw)
        files[row['path']]=raw
        records.append({k:v for k,v in row.items() if k!='utf8'})
    assert sum(r['changed'] for r in records)==23
    original=files['original-client.py']; comparator=files['reader-refusal-client.py']
    old=b'        self._reader.join(timeout=timeout)\n'
    new=old+b'        if self._reader.is_alive():\n            raise TimeoutError("app-server reader close timed out")\n'
    assert original.count(old)==1 and original.replace(old,new,1)==comparator
    own_client=(OWN/'codex_app_server_client_v2.py').read_bytes()
    own_test=(OWN/'test_appserver_journal_close_01a0ff2d.py').read_bytes()
    for name,raw in [('owned-v1-client.py.txt',own_client),('owned-v1-test.py.txt',own_test)]:
        with (HERE/name).open('xb') as handle:handle.write(raw)
    baseline_close=method(original,'close'); own_close=method(own_client,'close')
    same_prefix=ast.dump(ast.Module(body=baseline_close.body[:2],type_ignores=[]),include_attributes=False)==ast.dump(ast.Module(body=own_close.body[:2],type_ignores=[]),include_attributes=False)
    same_read=ast.dump(method(original,'_read'),include_attributes=False)==ast.dump(method(own_client,'_read'),include_attributes=False)
    assert same_prefix and same_read
    raw_rows=[]
    for name,raw in files.items():
        if name.endswith('/raw.json'):
            row=json.loads(raw)
            fields=['status','arm','condition','source_sha256','reader_alive_at_checkpoint','journal_closed_at_checkpoint',
                    'client_closed_at_checkpoint','checkpoint_outcome','notifications_after_release',
                    'construction_error_type','final_reader_alive','final_caller_alive']
            raw_rows.append({'path':name,**{k:row[k] for k in fields if k in row},
                             'reader_error_types':[r['type'] for r in row.get('reader_exceptions',[])]})
    assert len(raw_rows)==5
    receipt={'recorded_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
             'role':'Author-session auxiliary static inspection; no independent vote, nonauthor application certificate, or native remeasurement',
             'trigger':{'bytes':len(trigger_raw),'sha256':sha(trigger_raw)},
             'gzip':{'bytes':len(compressed),'sha256':sha(compressed)},
             'payload':{'bytes':len(payload),'sha256':sha(payload)},
             'files':records,'public_files':58,'identity_files':35,'projected_files':23,
             'private_original_authentication':False,'peer_helpers_imported_or_executed':False,
             'producer_or_saved_checker_replays':0,'native_model_ui_formal_experiment_allocations':0,
             'owned_client':{'bytes':len(own_client),'sha256':sha(own_client)},
             'owned_test':{'bytes':len(own_test),'sha256':sha(own_test)},
             'owned_reader_and_close_prefix_AST_equal_to_peer_baseline':True,
             'peer_comparator_exact_single_guard_insertion':True,'retained_rows_literal_read':raw_rows,
             'limit':'Data/image identity and literal source/row inspection only. Original projected private bytes and actual runtime identities not independently authenticated.'}
    with (HERE/'data-source-receipt.json').open('x',encoding='utf-8',newline='\n') as handle:
        json.dump(receipt,handle,sort_keys=True,ensure_ascii=True,indent=2);handle.write('\n')
    print(json.dumps({k:v for k,v in receipt.items() if k not in ('files','retained_rows_literal_read')},sort_keys=True))
    print(json.dumps(raw_rows,ensure_ascii=True,sort_keys=True))

try:
    main()
except Exception:
    with (HERE/'decoder-first-error.txt').open('x',encoding='utf-8',newline='\n') as handle:traceback.print_exc(file=handle)
    raise
