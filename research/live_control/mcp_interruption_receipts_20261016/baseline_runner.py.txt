"""Ordinary pinned-base MCP regression. Excludes child probes (separate imports)."""
import importlib
import json
import subprocess
import sys
import types
import unittest

BASE='1b1c0064e4acc32f4aa451218130db3ac33a4e55'
name='runtime.cli_v1.mcp_server'
source=subprocess.check_output(['git','show',f'{BASE}:runtime/cli_v1/mcp_server.py'])
module=types.ModuleType(name)
module.__package__='runtime.cli_v1'
module.__file__=f'{BASE}/runtime/cli_v1/mcp_server.py'
sys.modules[name]=module
exec(compile(source,module.__file__,'exec'),module.__dict__)
setattr(importlib.import_module('runtime.cli_v1'),'mcp_server',module)
tests=importlib.import_module('runtime.cli_v1.test_mcp_interruption_receipt')
names=[n for n in unittest.defaultTestLoader.getTestCaseNames(tests.InterruptionReceiptTests)
       if n!='test_keyboardinterrupt_and_systemexit_are_preserved_in_owned_process']
suite=unittest.TestSuite(tests.InterruptionReceiptTests(n) for n in names)
result=unittest.TextTestRunner(verbosity=2).run(suite)
print(json.dumps(dict(base=BASE,methods=result.testsRun,failures=len(result.failures),errors=len(result.errors),
    excluded_method='owned process probes import checkout source, so not a base-source control',
    scope='complete pinned original MCP module, current unchanged dependencies and new Fake X regression methods')))
raise SystemExit(1 if not result.wasSuccessful() else 0)
