"""Actual Python bridge with Fake X; gated remaining synchronous boundaries."""
import argparse
import hashlib
import importlib.metadata
import json
import platform
from pathlib import Path
import subprocess
import sys
import threading
import unittest
from unittest.mock import patch

from runtime.cli_v1.test_mcp_process_identity import FakeX
from runtime.guarded_x11_v1.bridge import NativeHandleBridge

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[2]
MATRIX = [('healthy','none')] + [(boundary,trigger) for boundary in
          ('capture','persistence','callback') for trigger in ('cancel','expiry')]


def digest(path): return hashlib.sha256(path.read_bytes()).hexdigest()
def write(path, value):
    with path.open('x',encoding='utf-8',newline='\n') as output:
        json.dump(value,output,indent=2);output.write('\n')


def sources():
    files = sorted((ROOT/'runtime').rglob('*.py')) + sorted(HERE.glob('*.py')) + [HERE/'PLAN.md']
    return {p.relative_to(ROOT).as_posix():digest(p) for p in files}


def run_case(directory, boundary, trigger):
    case = unittest.TestCase()
    fake = FakeX(case)
    fake.backend.d.get_display_name = lambda:':fake-only'
    entered,resume,finished,closed = [threading.Event() for _ in range(4)]
    now = [1_000_000]
    result, errors = [], []
    class Reader:
        def read(self, operation, xid):
            return True if operation == 'focus' else fake.generation
        def close(self): closed.set()
    def gate():
        entered.set()
        if not resume.wait(5): raise TimeoutError('harness gate not resumed')
    capture = fake.backend.observe_read_only
    def guarded_capture(*args, **kwargs):
        if boundary == 'capture' and 1 in fake.down: gate()
        return capture(*args, **kwargs)
    fake.backend.observe_read_only = guarded_capture
    bridge = None
    worker = None
    try:
        with patch('time.monotonic_ns',side_effect=lambda:now[0]), \
                patch('runtime.guarded_x11_v1.target_queries.X11TargetQueries',return_value=Reader()), \
                patch('Xlib.display.Display',side_effect=AssertionError('real display forbidden')):
            bridge = NativeHandleBridge(':fake-only',{'app':20},'app',directory,
                                        process_identity=True,query_worker=True)
            source = bridge.observe()
            offset = bridge.mint('field',source['sequence'],[8,8],region_size=[12,12])
            save = bridge._save
            def guarded_save(name, value):
                if boundary == 'persistence' and name.startswith('observation-') and 1 in fake.down: gate()
                return save(name,value)
            bridge._save = guarded_save
            stages = []
            def verify(stage, observation, image):
                stages.append(stage)
                if boundary == 'callback' and 1 in fake.down: gate()
                return True
            tail = [{'op':'key_chord','keys':['CTRL','a']}]
            token = 'this-frozen-call'
            def invoke():
                try:
                    with bridge.input_guard('field',offset,tail=tail,verify=verify),bridge.cancellation_scope(token):
                        result.append(bridge.keyboard('field',offset,tail=tail))
                except BaseException as error: errors.append(repr(error))
                finally: finished.set()
            worker = threading.Thread(target=invoke,name='fake-bridge-input')
            worker.start()
            blocked = None
            if boundary != 'healthy':
                if not entered.wait(5): raise RuntimeError('gate not entered: '+repr((result,errors)))
                original_deadline = bridge.deadline
                ack = bridge.request_cancel(token) if trigger == 'cancel' else None
                if trigger == 'expiry': now[0] = original_deadline
                blocked = dict(fake_keys_down=sorted(fake.down),returned=finished.is_set(),
                    original_deadline_ns=original_deadline,clock_ns=now[0],ack=ack,
                    gate_open=resume.is_set(),emissions=list(fake.emissions),
                    query_worker=bridge._target_queries.snapshot(),
                    release_receipt_returned=bool(result))
                resume.set()
            if not finished.wait(5): raise TimeoutError('input caller did not finish')
            worker.join(1)
            if errors: raise RuntimeError(repr(errors))
            row = dict(boundary=boundary,trigger=trigger,blocked=blocked,result=result[0],
                emissions=list(fake.emissions),fake_keys_down=sorted(fake.down),
                guard_stages=stages,review_required=bridge.review_required,
                query_worker=bridge._target_queries.snapshot(),
                consumed_live_allocations=0)
            bridge.close()
            if not closed.wait(5): raise TimeoutError('read client not closed')
            row['reader_closed'] = True
            return row
    finally:
        resume.set()
        if worker is not None: worker.join(5)
        if worker is None or not worker.is_alive():
            if bridge is not None: bridge.stop_target_queries()
            case.doCleanups()


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--freeze',action='store_true')
    parser.add_argument('--output-dir',type=Path)
    args = parser.parse_args()
    freeze_path = HERE/'SOURCE_FREEZE.json'
    if args.freeze:
        write(freeze_path,dict(base=subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip(),
              files=sources(),matrix=MATRIX,scope='Fake X only; actual Python bridge; no live allocation',
              dependencies={name:importlib.metadata.version(name) for name in ('python-xlib','Pillow','mcp')}))
        return
    frozen = json.loads(freeze_path.read_bytes())
    if sources() != frozen['files']: raise RuntimeError('source differs from freeze')
    if args.output_dir is None: parser.error('fresh output directory required')
    args.output_dir.mkdir()
    rows = []
    for boundary,trigger in MATRIX:
        directory = args.output_dir/(boundary+'-'+trigger)
        directory.mkdir()
        row = run_case(directory,boundary,trigger)
        write(directory/'cell.json',row)
        rows.append(row)
    write(args.output_dir/'raw.json',dict(scope='FAKE_X_SYNC_BOUNDARY_CHARACTERIZATION',
          freeze_sha256=digest(freeze_path),python=sys.version,platform=platform.platform(),
          optimized=sys.flags.optimize,cases=rows,source_postcheck=sources()==frozen['files']))
    write(args.output_dir/'FILES.json',{p.relative_to(args.output_dir).as_posix():digest(p)
          for p in sorted(args.output_dir.rglob('*')) if p.is_file()})


if __name__ == '__main__': main()
