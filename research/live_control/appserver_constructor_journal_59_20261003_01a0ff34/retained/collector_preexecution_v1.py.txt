"""One bounded twelve-cell constructor journal ownership comparison."""
import datetime as dt
import hashlib
import io
import json
import os
from pathlib import Path
import queue
import subprocess
import sys
import time
import types
import weakref

ROOT = Path(__file__).resolve().parent / "appserver-constructor-journal-59-01a0ff34-a01"
SPEC = json.loads((ROOT / "SPEC.json").read_text(encoding="ascii"))

class Output:
    def __init__(self):
        self.rows = queue.Queue()
        self.closed = False
    def __iter__(self):
        while True:
            row = self.rows.get(timeout=SPEC["reader_queue_guard_seconds"])
            if row is None:
                return
            yield row
    def close(self):
        self.closed = True

class Input(io.StringIO):
    def __init__(self, process):
        super().__init__()
        self.process = process
    def write(self, text):
        count = super().write(text)
        message = json.loads(text)
        if "id" in message:
            self.process.stdout.rows.put(json.dumps({"id": message["id"], "result": "healthy"}) + "\n")
            self.process.stdout.rows.put(json.dumps({"method": "fixture/notice", "params": {"value": "preserved"}}) + "\n")
        return count

class Process:
    def __init__(self):
        self.stdout = Output()
        self.stdin = Input(self)
        self.stderr = io.StringIO()
        self.returncode = None
    def poll(self):
        return self.returncode
    def terminate(self):
        self.returncode = 0
        self.stdout.rows.put(None)
    def wait(self, timeout=None):
        return self.returncode
    def kill(self):
        self.terminate()

def fd_open(fd):
    if fd is None:
        return None
    try:
        os.fstat(fd)
        return True
    except OSError as error:
        if error.errno != 9:
            raise
        return False

def make_cell(variant, case):
    cell_dir = ROOT / "cells" / (variant + "-" + case)
    cell_dir.mkdir(parents=True, exist_ok=False)
    journal_path = cell_dir / "journal.jsonl"
    prior = b"already-owned fixture bytes\n" if case == "journal_open_failure" else None
    if prior is not None:
        journal_path.write_bytes(prior)
    missing = cell_dir / "absent-owned-executable.exe"
    if missing.exists():
        raise RuntimeError("owned absent executable unexpectedly exists")
    module = types.ModuleType("constructor_cell_" + variant + "_" + case)
    source_path = ROOT / "source" / ("current_client.py.txt" if variant == "current" else "private_reference.py.txt")
    exec(compile(source_path.read_bytes(), str(source_path), "exec"), module.__dict__)
    observed = {"journal_weak": None, "fd": None, "original_error": None, "factory_calls": 0, "native_popen_attempts": 0, "returned_process": None}
    def recorded_open(*args, **kwargs):
        try:
            handle = open(*args, **kwargs)
        except BaseException as error:
            observed["original_error"] = error
            raise
        observed["journal_weak"] = weakref.ref(handle)
        observed["fd"] = handle.fileno()
        return handle
    def factory(*args, **kwargs):
        observed["factory_calls"] += 1
        try:
            if case in ("missing_executable", "no_journal_missing_executable"):
                observed["native_popen_attempts"] += 1
                return subprocess.Popen(*args, **kwargs)
            if case == "factory_type_error":
                raise TypeError("owned factory refuses construction")
            if case == "factory_keyboard_interrupt":
                raise KeyboardInterrupt("owned inert interrupt control")
            process = Process()
            observed["returned_process"] = process
            return process
        except BaseException as error:
            observed["original_error"] = error
            raise
    module.open = recorded_open
    client = None
    failure = None
    started = dt.datetime.now(dt.timezone.utc).isoformat()
    start_ns = time.perf_counter_ns()
    try:
        client = module.CodexAppServerClient([str(missing)] if "missing_executable" in case else ["owned-inert-factory"], process_factory=factory, journal_path=None if case == "no_journal_missing_executable" else str(journal_path))
    except BaseException as error:
        failure = error
    call_end_ns = time.perf_counter_ns()
    handle = None if observed["journal_weak"] is None else observed["journal_weak"]()
    retained_partial = None
    if failure is not None:
        traceback = failure.__traceback__
        while traceback is not None:
            possible = traceback.tb_frame.f_locals.get("self")
            if isinstance(possible, module.CodexAppServerClient):
                retained_partial = possible
                break
            traceback = traceback.tb_next
    row = {"variant": variant, "case": case, "started_utc": started,
           "constructor_start_ns": start_ns, "constructor_end_ns": call_end_ns,
           "exception_type": None if failure is None else type(failure).__name__,
           "exception_message": None if failure is None else str(failure),
           "same_original_exception": None if failure is None else failure is observed["original_error"],
           "failure_traceback_retains_client": None if failure is None else retained_partial is not None,
           "constructor_returned": client is not None,
           "factory_calls": observed["factory_calls"], "native_popen_attempts": observed["native_popen_attempts"],
           "process_returned": observed["returned_process"] is not None,
           "journal_observer_kind": "weakref plus numeric native fd only before observation",
           "journal_acquired": handle is not None,
           "journal_closed_before_driver_cleanup": None if handle is None else handle.closed,
           "native_fd_open_before_driver_cleanup": fd_open(observed["fd"]),
           "reader_created_after_failure": None if failure is None else hasattr(retained_partial, "_reader"),
           "healthy_result": None, "healthy_notice": None, "source_close_completed": None}
    try:
        if client is not None:
            row["healthy_result"] = client.request("fixture/echo", timeout=SPEC["healthy_wait_timeout_seconds"])
            row["healthy_notice"] = client.wait_notification(lambda value: value.get("method") == "fixture/notice", timeout=SPEC["healthy_wait_timeout_seconds"])
            client.close(timeout=1)
            row["source_close_completed"] = True
            row["reader_alive_after_source_close"] = client._reader.is_alive()
            row["journal_closed_after_source_close"] = client._journal.closed
    finally:
        if handle is not None and not handle.closed:
            handle.close()
        process = observed["returned_process"]
        if process is not None:
            if process.poll() is None:
                process.terminate()
            client._reader.join(timeout=2)
            for stream in (process.stdin, process.stdout, process.stderr):
                stream.close()
            row["driver_streams_closed"] = all(stream.closed for stream in (process.stdin, process.stdout, process.stderr))
            row["reader_alive_at_end"] = client._reader.is_alive()
        else:
            row["driver_streams_closed"] = None
            row["reader_alive_at_end"] = False
        row["journal_closed_at_end"] = None if handle is None else handle.closed
        row["native_fd_open_at_end"] = fd_open(observed["fd"])
        data = journal_path.read_bytes() if journal_path.exists() else None
        row["journal_bytes"] = None if data is None else len(data)
        row["journal_sha256"] = None if data is None else hashlib.sha256(data).hexdigest()
        row["preexisting_journal_unchanged"] = None if prior is None else data == prior
        row["journal_rows"] = [] if data is None or prior is not None or not data else [json.loads(line) for line in data.splitlines()]
        row["ended_utc"] = dt.datetime.now(dt.timezone.utc).isoformat()
        with (cell_dir / "OBSERVATION.json").open("x", encoding="ascii") as file:
            file.write(json.dumps(row, ensure_ascii=True, indent=2) + "\n")
    return row

def main():
    freeze = json.loads((ROOT / "FREEZE.json").read_text(encoding="ascii"))
    for entry in freeze["files"]:
        data = Path(entry["path"]).read_bytes()
        if len(data) != entry["bytes"] or hashlib.sha256(data).hexdigest() != entry["sha256"]:
            raise RuntimeError("frozen file drift")
    rows = []
    for variant in SPEC["variants"]:
        for case in SPEC["cases"]:
            rows.append(make_cell(variant, case))
    result = {"rows": rows, "runtime": {"python": sys.version, "platform": sys.platform, "perf_clock": vars(time.get_clock_info("perf_counter"))}, "qualification": SPEC["U"]}
    with (ROOT / "RAW.json").open("x", encoding="ascii") as file:
        file.write(json.dumps(result, ensure_ascii=True, indent=2) + "\n")
    print(json.dumps({"status": "TWELVE_FIRST_CELLS_RETAINED", "rows": len(rows), "native_popen_failure_attempts": sum(row["native_popen_attempts"] for row in rows), "healthy_local_readers": sum(row["constructor_returned"] for row in rows), "owned_readers_live_at_end": sum(row["reader_alive_at_end"] for row in rows)}, ensure_ascii=True))

if __name__ == "__main__":
    main()
