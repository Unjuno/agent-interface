"""Independent literal saved-file audit, without subject/factory execution."""
import copy
import hashlib
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parent / "appserver-constructor-journal-59-01a0ff34-a01"

def strict_equal(left, right):
    if type(left) is not type(right):
        return False
    if type(left) is dict:
        return left.keys() == right.keys() and all(strict_equal(left[key], right[key]) for key in left)
    if type(left) is list:
        return len(left) == len(right) and all(strict_equal(a, b) for a, b in zip(left, right))
    return left == right

CASES = ["healthy", "missing_executable", "no_journal_missing_executable", "factory_type_error", "factory_keyboard_interrupt", "journal_open_failure"]
ERRORS = {"missing_executable": "FileNotFoundError", "no_journal_missing_executable": "FileNotFoundError", "factory_type_error": "TypeError", "factory_keyboard_interrupt": "KeyboardInterrupt", "journal_open_failure": "FileExistsError"}

def audit(raw):
    rows = raw["rows"]
    if len(rows) != 12 or [(r["variant"], r["case"]) for r in rows] != [(v,c) for v in ("current", "private_reference") for c in CASES]:
        raise ValueError("complete ordered twelve identities required")
    for row in rows:
        case, variant = row["case"], row["variant"]
        acquired = case not in ("no_journal_missing_executable", "journal_open_failure")
        healthy = case == "healthy"
        before_open = acquired and (healthy or variant == "current")
        expected = {"exception_type": ERRORS.get(case), "constructor_returned": healthy,
                    "same_original_exception": None if healthy else True,
                    "failure_traceback_retains_client": None if healthy else True,
                    "factory_calls": 0 if case == "journal_open_failure" else 1,
                    "native_popen_attempts": 1 if "missing_executable" in case else 0,
                    "process_returned": healthy, "journal_acquired": acquired,
                    "journal_closed_before_driver_cleanup": not before_open if acquired else None,
                    "native_fd_open_before_driver_cleanup": before_open if acquired else None,
                    "reader_created_after_failure": None if healthy else False,
                    "healthy_result": "healthy" if healthy else None,
                    "healthy_notice": {"method": "fixture/notice", "params": {"value": "preserved"}} if healthy else None,
                    "source_close_completed": True if healthy else None,
                    "driver_streams_closed": True if healthy else None,
                    "reader_alive_at_end": False, "journal_closed_at_end": True if acquired else None,
                    "native_fd_open_at_end": False if acquired else None,
                    "preexisting_journal_unchanged": True if case == "journal_open_failure" else None}
        if not strict_equal({key: row[key] for key in expected}, expected):
            raise ValueError("literal ownership/error/control values differ")
        if type(row["constructor_start_ns"]) is not int or type(row["constructor_end_ns"]) is not int or row["constructor_end_ns"] < row["constructor_start_ns"]:
            raise ValueError("constructor clock ordering")
        journal = row["journal_rows"]
        if healthy:
            semantic = [{"direction": r["direction"], "message": r["message"]} for r in journal]
            expected_semantic = [{"direction":"sent", "message":{"method":"fixture/echo","id":1}}, {"direction":"received", "message":{"id":1,"result":"healthy"}}, {"direction":"received", "message":{"method":"fixture/notice","params":{"value":"preserved"}}}]
            if not strict_equal(semantic, expected_semantic) or row["reader_alive_after_source_close"] is not False or row["journal_closed_after_source_close"] is not True:
                raise ValueError("healthy journal/source close")
            if any(type(r["observed_ns"]) is not int for r in journal):
                raise ValueError("journal clock type")
        elif journal != []:
            raise ValueError("failure frame inventory")
    return "COUNTEREXAMPLE_RETAINED_TRACEBACK_OWNS_OPEN_JOURNAL"

raw = json.loads((ROOT / "RAW.json").read_text(encoding="ascii"))
disposition = audit(raw)
for row in raw["rows"]:
    path = ROOT / "cells" / (row["variant"] + "-" + row["case"])
    if not strict_equal(json.loads((path / "OBSERVATION.json").read_text(encoding="ascii")), row):
        raise ValueError("per-cell saved result differs")
    journal = path / "journal.jsonl"
    data = journal.read_bytes() if journal.exists() else None
    if (None if data is None else len(data)) != row["journal_bytes"] or (None if data is None else hashlib.sha256(data).hexdigest()) != row["journal_sha256"]:
        raise ValueError("actual journal byte custody")
    if row["case"] == "healthy" and not strict_equal([json.loads(line) for line in data.splitlines()], row["journal_rows"]):
        raise ValueError("healthy actual journal")
controls = [
    ("invent_source_cleanup", 1, "journal_closed_before_driver_cleanup", True),
    ("invent_closed_fd", 1, "native_fd_open_before_driver_cleanup", False),
    ("reference_leak", 7, "native_fd_open_before_driver_cleanup", True),
    ("replace_original_error", 1, "same_original_exception", False),
    ("bool_alias", 1, "native_popen_attempts", True),
    ("invent_reader", 1, "reader_created_after_failure", True),
    ("lost_healthy_reply", 0, "healthy_result", "lost"),
    ("corrupt_prior_file", 5, "preexisting_journal_unchanged", False),
    ("unclosed_driver", 1, "native_fd_open_at_end", True),
    ("missing_traceback_condition", 1, "failure_traceback_retains_client", False)]
rejected = []
for name, index, key, value in controls:
    changed = copy.deepcopy(raw)
    changed["rows"][index][key] = value
    if strict_equal(changed, raw):
        raise ValueError("ineffective control")
    try:
        audit(changed)
    except (KeyError, ValueError):
        rejected.append(name)
    else:
        raise ValueError("accepted changed-copy control: " + name)
result = {"disposition": disposition, "first_saved_cells": 12, "current_open_failure_journals": 3, "reference_open_failure_journals": 0, "native_failed_Popen_attempts": 4, "healthy_local_FIFO_cells": 2, "source_or_native_cells_replayed": 0, "rejected_controls": rejected, "raw_sha256": hashlib.sha256((ROOT / "RAW.json").read_bytes()).hexdigest(), "qualification": "Retained exception traceback and successful empty-journal close only; no general GC/leak duration, cleanup-close fault, thread-start failure, whole-call deadline, task/input/recovery efficacy or public source adoption."}
with (ROOT / "AUDIT.json").open("x", encoding="ascii") as file:
    file.write(json.dumps(result, ensure_ascii=True, indent=2) + "\n")
print(json.dumps(result, ensure_ascii=True))
