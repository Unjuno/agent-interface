def record(kind,**data):

 with event_lock:
  seq+=1;row={'seq':seq,'ns':time.monotonic_ns(),'event':kind,**data};events.write(json.dumps(row,sort_keys=True)+'\n');events.flush()

def frames(thread):
 f=sys._current_frames().get(thread.ident) if thread is not None else None;rows=[]
 while f is not None:
  loc=f.f_locals;row={'function':f.f_code.co_name,'file':f.f_code.co_filename,'line':f.f_lineno,'self_is_client':loc.get('self') is client,'self_is_shared_condition':client is not None and loc.get('self') is client._condition}
  for name in ('deadline','remaining','identifier','timeout','method'):
   if name in loc and type(loc[name]) in (str,int,float,bool,type(None)):row[name]=loc[name]
  rows.append(row);f=f.f_back
 return rows

def until(test,label,seconds=2):
 end=time.monotonic()+seconds
 while True:
  value=test()
  if value:return value
  if time.monotonic()>=end:raise RuntimeError('setup/terminal timeout: '+label)
  time.sleep(.001)

def rows(name):
 p=root/name
 if not p.exists():return []

 return [json.loads(line) for line in text.splitlines(keepends=True) if line.endswith('\n') and line.strip()]

def caller(name,fn):

  end=record(name+'_exception',type=type(exc).__name__,message=str(exc))['ns'];outcomes[name]={'kind':'exception','type':type(exc).__name__,'message':str(exc),'start_ns':begin,'end_ns':end}
