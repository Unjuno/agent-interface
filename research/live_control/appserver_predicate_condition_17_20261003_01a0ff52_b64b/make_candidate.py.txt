from pathlib import Path
p=Path(__file__).resolve().parent
b=(p/'baseline.py').read_text()
start=b.index('    def wait_notification(')
end=b.index('\n    def initialize(',start)
new='''    def wait_notification(self, predicate, timeout=120):
        deadline = time.monotonic() + timeout
        while True:
            with self._condition:
                snapshot = tuple(self._notifications)
            selected_missing = False
            for message in snapshot:
                if predicate(message):
                    with self._condition:
                        for index, current in enumerate(self._notifications):
                            if current is message:
                                del self._notifications[index]
                                return message
                    selected_missing = True
                    break
            if selected_missing:
                continue
            with self._condition:
                current = tuple(self._notifications)
                if len(current) != len(snapshot) or any(
                        left is not right for left, right in zip(current, snapshot)):
                    continue
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    raise TimeoutError("app-server notification timed out")
                if self._closed:
                    raise AppServerError("app-server closed while waiting: stderr not drained")
                self._condition.wait(remaining)
'''
c=b[:start]+new+b[end:]
(p/'candidate.py').write_text(c)

   node.body=[x for x in node.body if not(isinstance(x,ast.FunctionDef) and x.name=='wait_notification')]
assert ast.dump(ba,include_attributes=False)==ast.dump(ca,include_attributes=False)
