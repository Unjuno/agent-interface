"""Public data-only projection qualification; never execute the native producer."""
from pathlib import Path
import copy,hashlib,importlib.machinery,importlib.util,json,sys
ROOT=Path(sys.argv[1]).resolve()
def sha(data):return hashlib.sha256(data).hexdigest()
def need(ok,label):
    if not ok:raise RuntimeError(label)
manifest=json.loads((ROOT/'MANIFEST.json').read_bytes());projection=json.loads((ROOT/'PROJECTIONS.json').read_bytes())
maps={r['original_path']:r for r in projection['files']}
need(len(maps)==len(projection['files']),'unique original/public joins')
for row in manifest['files']:
    data=(ROOT/row['path']).read_bytes();need(len(data)==row['bytes'] and sha(data)==row['sha256'],'public manifest '+row['path'])
for row in projection['files']:
    data=(ROOT/row['public_path']).read_bytes();need(len(data)==row['public_bytes'] and sha(data)==row['public_sha256'],'public projection '+row['public_path'])
    if row['exact_original']:need(row['original_bytes']==row['public_bytes'] and row['original_sha256']==row['public_sha256'],'exact original declaration')
raw=json.loads((ROOT/'RAW.json').read_bytes());freeze=json.loads((ROOT/'FREEZE.json').read_bytes())
need(maps['RAW.json']['exact_original'] and maps['SOURCE.json']['exact_original'] and maps['INPUT.json']['exact_original'],'native/source/input bytes unprojected')
need(raw['freeze_sha256']==maps['FREEZE.json']['original_sha256'],'original native raw to original freeze declaration')
for name,digest in freeze['pins'].items():need(name in maps and maps[name]['original_sha256']==digest,'original freeze to original source/definition declaration '+name)
loader=importlib.machinery.SourceFileLoader('saved_original_oracle',str(ROOT/'source/auditor.py.txt'))
spec=importlib.util.spec_from_loader(loader.name,loader);oracle=importlib.util.module_from_spec(spec);loader.exec_module(oracle);oracle.ROOT=ROOT
view=copy.deepcopy(raw);view['freeze_sha256']=sha((ROOT/'FREEZE.json').read_bytes())
result=oracle.validate(view);need(result['status']==oracle.PASS,'original endpoint oracle on projection-qualified view')
mutations=json.loads((ROOT/'MUTATION_RESULT.json').read_bytes());controls=[]
for row in mutations['rows']:
    data=(ROOT/'mutants'/(row['name']+'.json')).read_bytes();need(sha(data)==row['sha256'] and len(data)==row['bytes'],'unprojected copied record')
    bad=json.loads(data);need(bad['freeze_sha256']==maps['FREEZE.json']['original_sha256'],'copied original freeze binding');bad['freeze_sha256']=view['freeze_sha256']
    try:oracle.validate(bad);reason='ACCEPTED'
    except ValueError as error:reason=str(error)
    need(reason==row['expected_reason']==row['observed_reason'],'copied intended semantic rejection');controls.append(row['name'])
need(len(controls)==16,'16 copied records')
print(json.dumps({'status':'PASS_PUBLIC_PROJECTION_AND_SAVED_ENDPOINT_RECONSTRUCTION','public_files':len(manifest['files']),'projection_joins':len(maps),'native_cells':3,'copied_rejections':len(controls),'metadata_view_only_field':'freeze_sha256 after verified declared original/public link','raw_source_input_and_native_endpoints_unchanged':True,'client_producer_peer_or_native_executed':False,'private_original_access_claimed':False}))
