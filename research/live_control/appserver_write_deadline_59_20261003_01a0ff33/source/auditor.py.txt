def need(ok,label):
    if not ok:raise ValueError(label)
def sha(data):return hashlib.sha256(data).hexdigest()
def exact_int(value):return type(value) is int

        need(row['journal_disabled'] is True and row['request_timeout_s']==0.05 and row['checkpoint_target_s']==0.25,'fixed journal/timeout/checkpoint')

        start=row['caller_state_at_checkpoint']['started_ns'];checkpoint=row['checkpoint_parent_ns'];final=row['final_parent_ns']
        need(all(exact_int(v) for v in [start,checkpoint,final,row['producer_start_ns']]) and row['producer_start_ns']<=start<checkpoint<=final,'parent-only exact timestamp order')
        need(checkpoint-start>=250_000_000,'fixed checkpoint not premature')

        need(row['child_exit_final']==0 and exact_int(row['child_exit_final']) and row['peer_final']['exit_code']==0 and row['peer_final'].get('private_release_seen') is True,'owned peer normal terminal')

        elif name=='unread_small':
            need(row['caller_alive_at_checkpoint'] is False and row['caller_done_at_checkpoint'] is True and row['write_lock_held_at_checkpoint'] is False,'small unread reaches response timeout')
            need(finished.get('error')=={'type':'TimeoutError','message':'app-server request timed out: fixture/request'} and 'result' not in finished,'small unread exact timeout control')
        else:
            need(row['caller_alive_at_checkpoint'] is True and row['caller_done_at_checkpoint'] is False and row['write_lock_held_at_checkpoint'] is True,'large unread live blocked writer')

        if name!='healthy_large':need(row['peer_final']['read_calls']==0 and row['peer_final']['request_read'] is False and row['peer_final']['response_written'] is False,'unread peer never consumed/answered request')

def audit_original():
    raw=json.loads((ROOT/'RAW.json').read_bytes());freeze=json.loads((ROOT/'FREEZE.json').read_bytes())
    need(all(sha((ROOT/name).read_bytes())==digest for name,digest in freeze['pins'].items()),'original frozen bytes')
    for name,digest in freeze['selected_environment'].items():need(sha(Path(name).read_bytes())==digest,'selected actual environment')
    for row in raw['rows']:
        case=ROOT/'cells'/row['name'];need(json.loads((case/'row.json').read_bytes())==row,'complete actual row file join')
        native=(case/'planned-request.bin').read_bytes();need(len(native)==row['planned_request_bytes'] and sha(native)==row['planned_request_sha256'],'planned native request bytes')
        request=json.loads(native);payload=b'probe' if row['name']=='unread_small' else b'X'*262144
        need(native.endswith(b'\r\n') and request=={'method':'fixture/request','id':1,'params':{'payload':payload.decode('ascii')}},'fixed Windows text-pipe bytes and JSON request')
        if row['name']=='healthy_large':
            received=(case/'peer-request.bin').read_bytes();response=(case/'peer-response.bin').read_bytes()

            need(json.loads(response)=={'id':1,'result':row['caller_final']['result']},'actual healthy wire response binding')
        else:need(not (case/'peer-request.bin').exists() and not (case/'peer-response.bin').exists(),'do not invent unread wire/response')
    return raw,validate(raw)
