 ('remove_write_frame','actual literal frozen stdin write/flush frame'),
 ('created_deadline','request deadline not created at blocked write'),
 ('advanced_request_line','request deadline not created at blocked write'),
 ('caller_ended','large unread live blocked writer'),
 ('write_lock_free','large unread live blocked writer'),
 ('premature_checkpoint','fixed checkpoint not premature'),
 ('boolean_request_id','one integer request identity consumed'),
 ('stdin_open_final','all owned parent streams closed'),
 ('reader_live_final','final caller/reader/source endpoints'),
 ('child_unreaped','owned peer normal terminal'),
 ('small_non_timeout','small unread exact timeout control'),
 ('wrong_healthy_payload','healthy exact large payload result'),
 ('unread_peer_consumed','unread peer never consumed/answered request'),
 ('wrong_source','source row binding'),
 ('wrong_peer_pid','actual native owner PID'),
 ('missing_native_count','complete three native cells')]

    elif name=='write_lock_free':large['write_lock_held_at_checkpoint']=False

    rows.append({'name':name,'bytes':len(packed),'sha256':hashlib.sha256(packed).hexdigest(),'expected_reason':reason,'observed_reason':observed,'rejected_for_intended_reason':observed==reason})
