"""One inert native JSONL request/reply/EOF peer; no provider/backend/input."""
from datetime import datetime, timezone
import json, os
from pathlib import Path
import sys, time

HERE=Path(__file__).resolve().parent
name, receipt_path=sys.argv[1:]
case=next(c for c in json.loads((HERE/'cases.json').read_bytes()) if c['name']==name)
start=datetime.now(timezone.utc).isoformat()
wire=sys.stdin.buffer.readline()
request=json.loads(wire)
if type(request) is not dict or type(request.get('id')) is not int or request['id']!=1:
    raise ValueError('fixture received unexpected request ID')
if request.get('method')!='fixture/probe' or request.get('params')!={'case':name}:
    raise ValueError('fixture received unexpected public request')
reply={'jsonrpc':'2.0','id':case['reply_id'],'result':{'kind':'fixture-payload','case':name}}
packet=(json.dumps(reply,separators=(',',':'),ensure_ascii=True)+'\n').encode('ascii')
record=dict(schema='native-rpc-id-peer-v1',case=name,peer_pid=os.getpid(),start_utc=start,
    request_line_utf8=wire.decode('ascii'),request=request,reply_line_ascii=packet.decode('ascii'),reply=reply,
    before_reply_ns=time.perf_counter_ns(),receipt_written_utc=datetime.now(timezone.utc).isoformat(),
    planned_terminal='natural process EOF after this single reply')
with Path(receipt_path).open('x',encoding='utf8',newline='\n') as f:json.dump(record,f,indent=2);f.write('\n')
sys.stdout.buffer.write(packet);sys.stdout.buffer.flush()
sys.stderr.write('fixture EOF\n');sys.stderr.flush()
