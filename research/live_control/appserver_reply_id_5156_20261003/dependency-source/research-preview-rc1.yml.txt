name: Research Preview RC1

on:
  push:
    branches:
      - release/research-preview-20260917-rc1
  workflow_dispatch:

permissions:
  contents: read

jobs:
  package-and-audit:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - name: Checkout frozen RC source closure
        uses: actions/checkout@v4
        with:
          fetch-depth: 1
          sparse-checkout-cone-mode: false
          sparse-checkout: |
            /LICENSE
            /README.md
            /SECURITY.md
            /runtime/**
            /release/**
            /research/live_control/*.py
            /research/live_control/*.json
            /research/live_control/*.txt
            /research/live_control/results/integrated-efficiency-live-01/preregistration.json
            /research/live_control/results/integrated-efficiency-live-01/report.json
            /research/live_control/results/integrated-efficiency-live-01/audit.json

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.12'

      - name: Test release tooling
        run: |
          python release/preview_bundle_v1/test_build_preview.py -v
          python release/preview_bundle_v1/test_accept_supported_host.py -v

      - name: Static first-run preflight on checkout
        run: python release/first_run_smoke_v1/preflight.py --root .

      - name: Install pinned runtime Python dependencies for retained audit only
        run: |
          python -m pip install --upgrade pip
          python -m pip install -r runtime/requirements-golden.txt

      - name: Reconstruct retained benchmark from checkout
        run: python runtime/golden_desktop_demo_v3.py audit-retained

      - name: Build deterministic Research Preview archive
        env:
          DIST: ${{ runner.temp }}/research-preview-dist
        run: |
          mkdir -p "$DIST"
          python release/preview_bundle_v1/build_preview.py --root . --out "$DIST"
          cd "$DIST"
          sha256sum -c SHA256SUMS

      - name: Verify extracted archive source closure
        env:
          DIST: ${{ runner.temp }}/research-preview-dist
          EXTRACT: ${{ runner.temp }}/research-preview-extracted
        run: |
          mkdir -p "$EXTRACT"
          tar -xzf "$DIST"/agent-interface-research-preview-*.tar.gz -C "$EXTRACT"
          ROOT="$(find "$EXTRACT" -mindepth 1 -maxdepth 1 -type d -name 'agent-interface-research-preview-*' -print -quit)"
          test -n "$ROOT"
          python "$ROOT/release/first_run_smoke_v1/preflight.py" --root "$ROOT"
          cd "$ROOT"
          python runtime/golden_desktop_demo_v3.py audit-retained

      - name: Upload Research Preview RC artifact
        uses: actions/upload-artifact@v4
        with:
          name: agent-interface-research-preview-rc1-${{ github.sha }}
          path: ${{ runner.temp }}/research-preview-dist/*
          if-no-files-found: error
          retention-days: 7
