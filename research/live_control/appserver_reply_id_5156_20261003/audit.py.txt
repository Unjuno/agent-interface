"""Independent retained-JSON audit; never imports/runs client, peer or arm."""
from copy import deepcopy
from datetime import datetime
import hashlib, json
from pathlib import Path

HERE=Path(__file__).resolve().parent
EXPECTED=(('integer_one',1,'number'),('number_float',1.0,'number'),
          ('boolean_true',True,'boolean'),('boolean_false',False,'boolean'),
          ('string_one','1','string'),('null_id',None,'null'))
sha=lambda b:hashlib.sha256(b).hexdigest()

def require(ok, label):
    if not ok:raise ValueError(label)

def same(a,b):
    if type(a) is not type(b):return False
    if isinstance(a,dict):return a.keys()==b.keys() and all(same(a[k],b[k]) for k in a)
    if isinstance(a,list):return len(a)==len(b) and all(same(x,y) for x,y in zip(a,b))
    return a==b

def load(data):
    def unique(pairs):
        out={}
        for k,v in pairs:
            require(k not in out,'duplicate JSON key');out[k]=v
        return out
    return json.loads(data,object_pairs_hook=unique)

def iso(value):
    t=datetime.fromisoformat(value)
    require(t.utcoffset() is not None and t.utcoffset().total_seconds()==0,'UTC')
    return t

def check(raws, check_disk=True):
    require(len(raws)==2,'two source arms')
    for arm,raw in zip(('baseline','candidate'),raws):
        freeze=load((HERE/(arm.upper()+'_FREEZE.json')).read_bytes())
        require(raw['schema']=='native-rpc-boolean-comparison-v1' and raw['arm']==arm,'arm identity')
        require(raw['id']=='appserver-rpc-boolean-id-5156-native-A01-20261003','execution identity')
        require(raw['source_sha256']==sha((HERE/freeze['source_path']).read_bytes()),'source identity')
        require(raw['freeze_sha256']==sha((HERE/(arm.upper()+'_FREEZE.json')).read_bytes()),'freeze identity')
        if check_disk:
            for p,h in freeze['inputs_sha256'].items():require(sha((HERE/p).read_bytes())==h,'freeze pin '+p)
        rows=raw['rows'];require(type(rows) is list and len(rows)==6,'six original native cells')
        returned=booleans=0
        for i,(row,(name,identifier,kind)) in enumerate(zip(rows,EXPECTED)):
            require(row['index']==i and row['case']==name and row['arm']==arm,'ordered census')
            require(same(row['reply_id'],identifier) and row['json_type']==kind,'literal reply ID')
            require(type(row['peer_pid']) is int and row['peer_pid']>0,'native PID')
            require(type(row['elapsed_ns']) is int and 0<=row['elapsed_ns']<=30000000000,'elapsed bound')
            require(iso(row['start_utc'])<=iso(row['end_utc']),'UTC interval')
            require(row['natural_peer_exit0'] is True and row['issues']==[],'original EOF terminal')
            cmd=row['native_command'];require(len(cmd)==5 and cmd[1]=='-B' and cmd[3]==name,'native argv')
            require(Path(cmd[2]).name=='peer.py.txt' and Path(cmd[4]).name=='peer.json','owned fixture argv')
            request={'method':'fixture/probe','id':1,'params':{'case':name}}
            payload={'kind':'fixture-payload','case':name}
            reply={'jsonrpc':'2.0','id':identifier,'result':payload}
            peer=row['peer'];require(peer['schema']=='native-rpc-id-peer-v1','peer schema')
            require(peer['case']==name and peer['peer_pid']==row['peer_pid'],'peer identity')
            require(same(peer['request'],request) and same(peer['reply'],reply),'peer frames')
            require(same(load(peer['request_line_utf8']),request),'actual sent wire')
            require(same(load(peer['reply_line_ascii']),reply),'actual reply wire')
            require(peer['reply_line_ascii']==json.dumps(reply,separators=(',',':'))+'\n','literal output frame')
            require(peer['planned_terminal']=='natural process EOF after this single reply' and type(peer['before_reply_ns']) is int,'peer terminal declaration')
            require(iso(peer['start_utc'])<=iso(peer['receipt_written_utc']),'peer UTC')
            journal=row['journal'];require(len(journal)==2,'two actual journal frames')
            for entry,direction,message in zip(journal,('sent','received'),(request,reply)):
                require(set(entry)=={'direction','observed_ns','message'},'complete journal projection')
                require(entry['direction']==direction and same(entry['message'],message),'typed journal correspondence')
                require(type(entry['observed_ns']) is int and entry['observed_ns']>0,'journal timestamp')
            require(journal[0]['observed_ns']<=journal[1]['observed_ns'],'journal sequence')
            expect_return=kind=='number' or (arm=='baseline' and name=='boolean_true')
            if expect_return:
                require(same(row['caller'],{'kind':'returned','value':payload}),'caller numeric/alias result')
                returned+=1;booleans+=kind=='boolean'
            else:
                require(same(row['caller'],{'kind':'raised','exception_type':'AppServerError',
                    'message':'app-server closed during fixture/probe: fixture EOF'}),'caller closed result')
            state=row['final_state']
            require(state['closed'] is True and type(state['next_id']) is int and state['next_id']==2,'closed client generation')
            require(state['notifications']==[] and state['reader_alive'] is False,'reader/notification endpoint')
            require(type(state['process_returncode']) is int and state['process_returncode']==0 and state['journal_closed'] is True,'process/journal endpoint')
            require(state['parent_handles_before_explicit_cleanup']==dict(stdin=False,stdout=False,stderr=False),'client handles retained')
            require(state['parent_handles_after_cleanup']==dict(stdin=True,stdout=True,stderr=True),'collector handles closed')
            expected_cache=[] if expect_return or (arm=='candidate' and kind=='boolean') else [{'id':identifier,'message':reply}]
            require(same(state['responses'],expected_cache),'complete typed residual cache')
            require(set(row['files'])=={'peer.json','client.jsonl'},'two cell artifacts')
            for filename,meta in row['files'].items():
                expected_path=f'runs/{arm}/{i:02d}-{name}/{filename}'
                require(meta['path']==expected_path and type(meta['bytes']) is int and 0<meta['bytes']<=32768,'bounded relative artifact')
                if check_disk:
                    b=(HERE/meta['path']).read_bytes()
                    require(len(b)==meta['bytes'] and sha(b)==meta['sha256'],'original file bytes')
                    parsed=load(b) if filename=='peer.json' else [load(line) for line in b.splitlines()]
                    require(same(parsed,peer if filename=='peer.json' else journal),'disk/raw full correspondence')
        require(raw['stop_error'] is None and raw['returned']==returned and raw['boolean_returns']==booleans,'summary reduction')
        require(returned==(3 if arm=='baseline' else 2) and booleans==(1 if arm=='baseline' else 0),'source comparative result')
        require(raw['decision']==('FAIL_BOOLEAN_ID_ALIAS' if arm=='baseline' else 'PASS_BOOLEAN_ID_FILTER'),'fixed source decision')
    return {'native_rows':12,'numeric_returns':[2,2],'boolean_returns':[1,0],
        'baseline_decision':'FAIL_BOOLEAN_ID_ALIAS','candidate_decision':'PASS_BOOLEAN_ID_FILTER'}

def main():
    output=HERE/'audit-result.json';require(not output.exists(),'original audit already retained')
    paths=[HERE/'baseline-raw.json',HERE/'candidate-raw.json']
    originals=[p.read_bytes() for p in paths];raws=[load(b) for b in originals]
    result=check(raws)
    def literal_id(x):x[0]['rows'][2]['reply_id']=1
    def candidate_return(x):x[1]['rows'][2]['caller']={'kind':'returned','value':{'kind':'fixture-payload','case':'boolean_true'}}
    def float_result(x):x[1]['rows'][1]['caller']={'kind':'raised','exception_type':'AppServerError','message':'wrong'}
    def sent_id(x):x[0]['rows'][0]['journal'][0]['message']['id']=True
    def wire_id(x):x[0]['rows'][0]['peer']['request_line_utf8']='{"method":"fixture/probe","id":true,"params":{"case":"integer_one"}}\n'
    def residual(x):x[1]['rows'][2]['final_state']['responses']=[{'id':True,'message':x[1]['rows'][2]['peer']['reply']}]
    def terminal(x):x[1]['rows'][5]['natural_peer_exit0']=False
    def missing(x):x[0]['rows'].pop()
    controls=[]
    for mutate in (literal_id,candidate_return,float_result,sent_id,wire_id,residual,terminal,missing):
        changed=deepcopy(raws);mutate(changed);require(not same(changed,raws),'effective corruption')
        try:check(changed,check_disk=False)
        except (ValueError,KeyError,TypeError) as e:controls.append({'name':mutate.__name__,'refused':True,'reason':str(e)})
        else:raise ValueError('control accepted '+mutate.__name__)
    require([p.read_bytes() for p in paths]==originals,'original raw custody')
    result.update(schema='native-rpc-boolean-audit-v1',integrity='PASS',controls=controls,
        original_sha256={p.name:sha(b) for p,b in zip(paths,originals)},
        scope='independent retained JSON reducer; no client/peer/source-arm execution or authenticated provider/task evidence')
    with output.open('x',encoding='utf8',newline='\n') as f:json.dump(result,f,indent=2);f.write('\n')
    print(json.dumps(result));return 0
if __name__=='__main__':raise SystemExit(main())
