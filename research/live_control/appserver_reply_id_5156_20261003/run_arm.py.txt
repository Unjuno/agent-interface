"""Ordinary native comparison. Exact snapshots; one terminal owned peer/cell."""
from datetime import datetime, timezone
import hashlib
from importlib.machinery import SourceFileLoader
from importlib.util import module_from_spec, spec_from_loader
import json
from pathlib import Path
import subprocess, sys, time

HERE=Path(__file__).resolve().parent
sha=lambda b:hashlib.sha256(b).hexdigest()

def verify(freeze):
    for path,digest in freeze['inputs_sha256'].items():
        if sha((HERE/path).read_bytes())!=digest:raise ValueError(('input pin',path))

def run_case(module, arm, case, index):
    name=case['name'];out=HERE/'runs'/arm/(str(index).zfill(2)+'-'+name)
    out.mkdir(parents=True,exist_ok=False)
    command=[sys.executable,'-B',str(HERE/'peer.py.txt'),name,str(out/'peer.json')]
    start=datetime.now(timezone.utc).isoformat();start_ns=time.perf_counter_ns()
    client=module.CodexAppServerClient(command,cwd=str(out),journal_path=str(out/'client.jsonl'))
    pid=client.process.pid;caller=None;issues=[];natural=False
    try:
        try:caller=dict(kind='returned',value=client.request('fixture/probe',{'case':name},timeout=0.5))
        except Exception as e:caller=dict(kind='raised',exception_type=type(e).__name__,message=str(e))
        try:
            waited=client.process.wait(timeout=0.5)
            natural=waited==0
            if waited!=0:issues.append('peer natural exit was not0')
        except subprocess.TimeoutExpired:
            issues.append('peer did not terminate at original EOF')
    finally:
        client.close(timeout=0.5)
        before_handles={n:getattr(client.process,n).closed for n in ('stdin','stdout','stderr')}
        for stream in (client.process.stdin,client.process.stdout,client.process.stderr):
            if not stream.closed:stream.close()
    end_ns=time.perf_counter_ns();end=datetime.now(timezone.utc).isoformat()
    if client._reader.is_alive():issues.append('source reader remains alive')
    if client.process.poll() is None:issues.append('owned process not terminal')
    peer_bytes=(out/'peer.json').read_bytes();journal_bytes=(out/'client.jsonl').read_bytes()
    if len(peer_bytes)>32768 or len(journal_bytes)>32768:raise ValueError('per-cell record cap')
    peer=json.loads(peer_bytes);journal=[json.loads(x) for x in journal_bytes.splitlines()]
    state=dict(closed=client._closed,next_id=client._next_id,
        responses=[dict(id=k,message=v) for k,v in client._responses.items()],
        notifications=list(client._notifications),reader_alive=client._reader.is_alive(),
        process_returncode=client.process.returncode,journal_closed=client._journal.closed,
        parent_handles_before_explicit_cleanup=before_handles,
        parent_handles_after_cleanup={n:getattr(client.process,n).closed for n in ('stdin','stdout','stderr')})
    return dict(index=index,case=name,arm=arm,reply_id=case['reply_id'],json_type=case['json_type'],
        start_utc=start,end_utc=end,elapsed_ns=end_ns-start_ns,peer_pid=pid,
        native_command=command,natural_peer_exit0=natural,caller=caller,peer=peer,journal=journal,
        files={'peer.json':dict(path=(out/'peer.json').relative_to(HERE).as_posix(),bytes=len(peer_bytes),sha256=sha(peer_bytes)),
               'client.jsonl':dict(path=(out/'client.jsonl').relative_to(HERE).as_posix(),bytes=len(journal_bytes),sha256=sha(journal_bytes))},
        final_state=state,issues=issues)

def main():
    arm=sys.argv[1]
    if arm not in ('baseline','candidate'):raise ValueError('source arm')
    output=HERE/(arm+'-raw.json')
    if output.exists():raise FileExistsError('first source arm already retained')
    first=json.loads((HERE/'BASELINE_FREEZE.json').read_bytes());verify(first)
    if arm=='candidate':
        freeze=json.loads((HERE/'CANDIDATE_FREEZE.json').read_bytes());verify(freeze)
    else:freeze=first
    source=HERE/freeze['source_path']
    loader=SourceFileLoader('_rpc_bool_'+arm,str(source))
    module=module_from_spec(spec_from_loader(loader.name,loader));sys.modules[loader.name]=module;loader.exec_module(module)
    cases=json.loads((HERE/'cases.json').read_bytes());rows=[];stop_error=None
    for index,case in enumerate(cases):
        try:
            row=run_case(module,arm,case,index);rows.append(row)
            if row['issues']:stop_error='owned child/source endpoint gate';break
        except Exception as e:
            stop_error=type(e).__name__+':'+str(e);break
    returned=sum(r['caller']['kind']=='returned' for r in rows)
    boolean_returns=sum(r['caller']['kind']=='returned' and r['json_type']=='boolean' for r in rows)
    expected_returned=3 if arm=='baseline' else 2
    expected_boolean=1 if arm=='baseline' else 0
    numeric_ok=all(r['caller']=={'kind':'returned','value':{'kind':'fixture-payload','case':r['case']}}
        for r in rows if r['json_type']=='number')
    other_ok=all(r['caller']['kind']=='raised' and r['caller'].get('exception_type')=='AppServerError'
        for r in rows if r['json_type'] in ('string','null') or r['case']=='boolean_false')
    if stop_error or len(rows)!=6 or returned!=expected_returned or boolean_returns!=expected_boolean or not numeric_ok or not other_ok:
        decision='STOP';exit_code=2
    elif boolean_returns:
        decision='FAIL_BOOLEAN_ID_ALIAS';exit_code=1
    else:decision='PASS_BOOLEAN_ID_FILTER';exit_code=0
    result=dict(schema='native-rpc-boolean-comparison-v1',id=first['id'],arm=arm,
        source_sha256=sha(source.read_bytes()),freeze_sha256=sha((HERE/('BASELINE_FREEZE.json' if arm=='baseline' else 'CANDIDATE_FREEZE.json')).read_bytes()),
        decision=decision,rows=rows,stop_error=stop_error,returned=returned,boolean_returns=boolean_returns,
        scope='native owned inert JSONL protocol; no genuine model/provider/backend/input/latency/task result')
    with output.open('x',encoding='utf8',newline='\n') as f:json.dump(result,f,indent=2);f.write('\n')
    print(json.dumps({k:v for k,v in result.items() if k!='rows'}))
    return exit_code
if __name__=='__main__':raise SystemExit(main())
