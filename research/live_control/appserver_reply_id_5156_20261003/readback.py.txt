"""Read public byte joins and retained JSON only; no native/source arm replay."""
import ast,hashlib,json
from importlib.machinery import SourceFileLoader
from importlib.util import module_from_spec,spec_from_loader
from pathlib import Path
HERE=Path(__file__).resolve().parent
sha=lambda b:hashlib.sha256(b).hexdigest()
def require(ok,label):
    if not ok:raise ValueError(label)
pub=json.loads((HERE/'PUBLICATION.json').read_bytes());joins=pub['files']
for n,m in joins.items():
    b=(HERE/n).read_bytes();require(len(b)==m['public_bytes'] and sha(b)==m['public_sha256'],'public byte join '+n)
for arm in ('baseline','candidate'):
    freeze=json.loads((HERE/(arm.upper()+'_FREEZE.json')).read_bytes())
    for n,h in freeze['inputs_sha256'].items():require(joins[n]['original_sha256']==h,'original pin ledger '+n)
    require(sha((HERE/freeze['source_path']).read_bytes())==freeze['source_sha256'],'exact client snapshot')
loader=SourceFileLoader('_retained_rpc_audit',str(HERE/'audit.py.txt'))
module=module_from_spec(spec_from_loader(loader.name,loader));loader.exec_module(module)
raws=[module.load((HERE/(arm+'-raw.json')).read_bytes()) for arm in ('baseline','candidate')]
result=module.check(raws,check_disk=False)
for raw in raws:
    for row in raw['rows']:
        for n,m in row['files'].items():
            b=(HERE/m['path']).read_bytes();require(len(b)==m['bytes'] and sha(b)==m['sha256'],'unprojected original cell')
            parsed=module.load(b) if n=='peer.json' else [module.load(line) for line in b.splitlines()]
            require(module.same(parsed,row['peer'] if n=='peer.json' else row['journal']),'public cell full join')
audit=json.loads((HERE/'audit-result.json').read_bytes())
require(audit['integrity']=='PASS' and len(audit['controls'])==8 and all(c['refused'] for c in audit['controls']),'retained original audit')
for n,h in audit['original_sha256'].items():require(joins[n]['original_sha256']==h,'original raw audit SHA')
for mode in ('normal','optimized'):
    receipt=json.loads((HERE/'regression'/mode/'RECEIPT.json').read_bytes())
    require(receipt['exit_code']==0,'ordinary regression receipt')
    for n,m in receipt['streams'].items():
        b=(HERE/'regression'/mode/n).read_bytes();require(len(b)==m['bytes'] and sha(b)==m['sha256'],'regression full streams')
        if n=='stderr.txt':require(b'Ran 2 tests' in b and b'\nOK' in b,'actual unittest terminal')
syntax=0
for file in HERE.rglob('*.py.txt'):
    ast.parse(file.read_bytes(),filename=file.name);syntax+=1
for file in HERE.rglob('*'):
    if file.is_file():require((b'C:'+bytes([92])+b'Users'+bytes([92])) not in file.read_bytes() and (b'C:'+bytes([47])+b'Users'+bytes([47])) not in file.read_bytes(),'private path not projected')
result.update(schema='rpc-boolean-public-readback-v1',public_joins=len(joins),
    projected_files=sum(m['projection']!='byte-identical' for m in joins.values()),syntax_images=syntax,
    original_auditor_replayed=False,source_arms_replayed=False,scope='public bytes/author private custody ledger/retained type-sensitive JSON; not authentication or a new independent worker vote')
with (HERE/'PUBLIC_READBACK.json').open('x',encoding='utf8',newline='\n') as f:json.dump(result,f,indent=2);f.write('\n')
print(json.dumps(result))
