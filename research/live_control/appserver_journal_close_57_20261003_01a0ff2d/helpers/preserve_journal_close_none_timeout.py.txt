import ast, hashlib, json
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parent.parent
private = root / 'work/repairs/appserver-journal-close-01'
out = root / 'outputs/appserver-journal-close-01'
path = private / 'codex_app_server_client_v2.py'
before = path.read_bytes()
assert hashlib.sha256(before).hexdigest() == 'da2c00920e0d833d989c93dd9d5ac6c2a845eebdb9e794d775ee6d7c238c406a'
old = b'self._journal_lock.acquire(timeout=timeout)'
new = b'self._journal_lock.acquire(timeout=-1 if timeout is None else timeout)'
assert before.count(old) == 1
after = before.replace(old, new, 1)
original = (private / 'baseline-client.py.txt').read_bytes()
def methods(source):
    return {node.name: ast.dump(node, include_attributes=False) for node in ast.walk(ast.parse(source))
            if isinstance(node, ast.FunctionDef)}
original_methods, final_methods = methods(original), methods(after)
assert original_methods.keys() == final_methods.keys()
assert [name for name in original_methods if original_methods[name] != final_methods[name]] == ['close']
record = {'created_before_final_command_utc': datetime.now(timezone.utc).isoformat(),
          'source_before_sha256': hashlib.sha256(before).hexdigest(), 'source_after_sha256': hashlib.sha256(after).hexdigest(),
          'source_bytes': len(after), 'only_changed_method_since_original': 'close',
          'compatibility': 'timeout=None keeps the explicit unbounded mode. Finite mutex waits retain the same numeric timeout. This does not promise an overall close/flush/OS-I/O deadline.',
          'first_compatibility_error': 'compatibility-red/receipt.json and exact executed source/test/log retained',
          'new_tests_or_native_child_execution_in_this_repair': False}
with (out / 'FIX_V2_BEFORE_FINAL.json').open('x', encoding='utf-8') as f:
    json.dump(record, f, indent=2)
    f.write('\n')
path.write_bytes(after)
print(json.dumps(record, indent=2))
