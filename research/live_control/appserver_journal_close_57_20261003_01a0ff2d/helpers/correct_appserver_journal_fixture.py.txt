import hashlib, json
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parent.parent
private = root / 'work/repairs/appserver-journal-close-01'
out = root / 'outputs/appserver-journal-close-01'
original = json.loads((out / 'FREEZE_BEFORE_RED.json').read_bytes())
before = (private / 'test_appserver_journal_close_01a0ff2d.py').read_bytes()
assert hashlib.sha256(before).hexdigest() == original['test_sha256']
with (private / 'first-native-test.py.txt').open('xb') as f:
    f.write(before)
tests = (root / 'work/test_appserver_journal_close_01a0ff2d.py').read_bytes()
source = (private / 'codex_app_server_client_v2.py').read_bytes()
assert hashlib.sha256(source).hexdigest() == original['sources'][0]['sha256']
(private / 'test_appserver_journal_close_01a0ff2d.py').write_bytes(tests)
freeze = dict(original, test_sha256=hashlib.sha256(tests).hexdigest(), test_bytes=len(tests),
              before_red_v2_utc=datetime.now(timezone.utc).isoformat(),
              first_red='red/receipt.json and executed source/test/cases retained unchanged:5methods/2failures; production held mutex and LF-vs-Windows-CRLF fixture error',
              correction='Snapshot actual journal bytes immediately after fixed fixture text flush; compare unchanged actual bytes, preserve all thresholds/source/cases. Add literal base64 first/final journal readback to new records.',
              first_journal_raw_qualification='Initial RED saved booleans, not original temporary journal bytes; Windows TextIO newline explanation is source/platform qualified, not authentication of inaccessible first journal bytes.')
with (out / 'FREEZE_BEFORE_RED_V2.json').open('x', encoding='utf-8') as f:
    json.dump(freeze, f, indent=2)
    f.write('\n')
print(json.dumps({'new_test_sha256': freeze['test_sha256'], 'unchanged_source_sha256': hashlib.sha256(source).hexdigest(),
                  'first_failure_preserved': True, 'before_command_utc': freeze['before_red_v2_utc']}, indent=2))
