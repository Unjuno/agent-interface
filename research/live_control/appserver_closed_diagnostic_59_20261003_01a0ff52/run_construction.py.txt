"""Bounded sequential ordinary construction; preserve all first outcomes."""
from pathlib import Path
import datetime, hashlib, json, os, signal, subprocess, sys

ROOT = Path(__file__).resolve().parent
PLAN = json.loads((ROOT / 'PLAN.json').read_text())
pins = {name: hashlib.sha256((ROOT / name).read_bytes()).hexdigest()
        for name in ['PLAN.json', 'SOURCE.json', 'codex_app_server_client_v2.py',
                     'inert_child.py', 'run_cell.py', 'run_construction.py']}
destination = ROOT / 'construction01'
destination.mkdir()
receipts = []
for cell in PLAN['cells']:
    argv = [sys.executable, '-B', str(ROOT / 'run_cell.py'), cell['id'], cell['api'],
            cell['mode'], str(destination / cell['id'])]
    receipt = {'cell': cell, 'argv': argv, 'cwd': str(ROOT), 'sources': pins,
               'started_utc': datetime.datetime.now(datetime.UTC).isoformat()}
    process = subprocess.Popen(argv, cwd=ROOT, stdout=subprocess.PIPE,
                               stderr=subprocess.PIPE, start_new_session=True)
    receipt['probe_pid'] = process.pid
    try:
        stdout, stderr = process.communicate(timeout=PLAN['limits']['outer_case_s'])
        receipt['outer_timeout'] = False
    except subprocess.TimeoutExpired:
        receipt['outer_timeout'] = True
        os.killpg(process.pid, signal.SIGKILL)  # This own case and descendants only.
        stdout, stderr = process.communicate(timeout=1)
    (destination / (cell['id'] + '.stdout')).write_bytes(stdout)
    (destination / (cell['id'] + '.stderr')).write_bytes(stderr)
    receipt.update({'exit_code': process.returncode,
                    'ended_utc': datetime.datetime.now(datetime.UTC).isoformat(),
                    'stdout_sha256': hashlib.sha256(stdout).hexdigest(),
                    'stderr_sha256': hashlib.sha256(stderr).hexdigest(),
                    'log_cap_exceeded': len(stdout) > PLAN['limits']['cell_log_cap_bytes'] or
                                        len(stderr) > PLAN['limits']['cell_log_cap_bytes']})
    (destination / (cell['id'] + '.receipt.json')).write_text(json.dumps(receipt, indent=2) + '\n')
    receipts.append(receipt)
    print(json.dumps({'cell': cell['id'], 'exit': process.returncode,
                      'outer_timeout': receipt['outer_timeout'], 'stderr': stderr.decode()[:1000]}), flush=True)
    if process.returncode or receipt['outer_timeout'] or receipt['log_cap_exceeded']:
        break
summary = {'kind': PLAN['kind'], 'first_driver_receipts': receipts,
           'completed_cells': len(receipts), 'planned_cells': len(PLAN['cells']), 'formal_replays': 0}
(destination / 'driver-summary.json').write_text(json.dumps(summary, indent=2) + '\n')
raise SystemExit(any(r['exit_code'] or r['outer_timeout'] or r['log_cap_exceeded'] for r in receipts))
