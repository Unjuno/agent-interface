def sha(data):return hashlib.sha256(data).hexdigest()

def forward_reference(s):
    returned={};bad=[];segment=-1
    for index,event in enumerate(s['events']):
        kind=event['event']
        if kind=='setup':segment+=1;returned={}
        elif kind=='read_return':returned[event['evidence_ref']]=(index,event['ns'])
        elif kind=='commit':
            pair=returned.get(event['evidence_ref'])
            if pair is None or not(type(pair[1]) is int and type(event['ns']) is int and pair[0]<index and pair[1]<=event['ns']):
                bad.append({'segment':segment,'index':index,'evidence_ref':event['evidence_ref']})
    return bad

def validate(entry,s,expected_sha):
    assert sha(encoded(s))==expected_sha,'exact saved copy bytes'
    bad=forward_reference(s)
    result=entry['result']
    if bad:
        assert any(v.endswith(':server_return_before_effect') for v in result['errors']),'missing causal refusal'
        assert result['equivalent_cohort_latency_gain_observed'] is None,'benefit on invalid evidence'
    else:
        assert result['errors']==[],'positive causal record refused'
        assert result['equivalent_cohort_latency_gain_observed'] is True,'original fixture result lost'
    return len(bad)
