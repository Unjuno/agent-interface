"""Ordinary pure saved-record tests; never invoke any producer or auditor main."""
import ast,copy,hashlib,json,sys,unittest
from pathlib import Path
ROOT=Path(__file__).resolve().parent
OUT=Path(sys.argv[1])
sys.argv=sys.argv[:1]

def load_functions(path,names):
    tree=ast.parse(path.read_bytes())
    nodes=[n for n in tree.body if isinstance(n,(ast.Import,ast.ImportFrom))
           or isinstance(n,ast.FunctionDef) and n.name in names
           or isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id in ('KINDS','ORDER') for t in n.targets)]
    ns={'__name__':'saved_functions'}
    exec(compile(ast.Module(body=nodes,type_ignores=[]),'saved-functions','exec'),ns)
    return ns

CLIENT=json.loads((ROOT/'inputs/client.json').read_bytes())
SERVER=json.loads((ROOT/'inputs/server.json').read_bytes())
AUDIT=load_functions(ROOT/'audit_v3.py.txt',('exact','scope','audit','controls'))
BASE=load_functions(ROOT/'inputs/audit_v2.py.txt',('exact','scope','audit','controls'))
OLD=load_functions(ROOT/'inputs/regressions_v2.py.txt',('specifications','apply','extra_copies'))
OLD.update(CLIENT=CLIENT,SERVER=SERVER)
records=[]
def sha(data):return hashlib.sha256(data).hexdigest()
def bytes_of(obj):return json.dumps(obj,sort_keys=True,separators=(',',':'),ensure_ascii=True).encode()+b'\n'
def row(label,c,s,result):
    record={'label':label,'client_sha256':sha(bytes_of(c)),'server_sha256':sha(bytes_of(s)),'result':result}
    records.append(record)
    return record

def negative_copies():
    copies=[];current=None
    for original_index,event in enumerate(SERVER['events']):
        if event['event']=='setup':current=(event['case'],event['policy'])
        if event['event']!='commit':continue
        s=copy.deepcopy(SERVER);events=s['events'];commit=events.pop(original_index)
        rid=next(c['result']['read_id'] for r in CLIENT['rows'] if (r['case'],r['policy'])==current
                 for c in r['consumers'] if c['request']==commit['request_id'])
        at=next(i for i,v in enumerate(events) if v['event']=='read_return' and v['read_id']==rid)
        commit['ns']=events[at-1]['ns'];events.insert(at,commit)
        for i,v in enumerate(events):v['seq']=i
        copied=bytes_of(s)
        label='commit-before-own-return/'+str(original_index)
        copies.append((label,s))
        (OUT/'copies'/('server-'+str(original_index)+'.json')).write_bytes(copied)
    if len(copies)!=25:raise ValueError('actual effect denominator changed')
    if sha(bytes_of(copies[0][1]))!='714fd93b3c26fdbfbdb7862d5ff91122308e4ac78f115b58b2e6cb563148e3b9':
        raise ValueError('first actual peer witness mismatch')
    return copies

class SavedCausalRegressions(unittest.TestCase):
    def test_original_results_unchanged(self):
        result=AUDIT['audit'](CLIENT,SERVER);before=BASE['audit'](CLIENT,SERVER)
        row('original',CLIENT,SERVER,result)
        self.assertEqual({k:v for k,v in result.items() if k!='audit_revision'},
                         {k:v for k,v in before.items() if k!='audit_revision'})
        self.assertEqual(result['errors'],[])
        self.assertEqual(result['rows'],15);self.assertEqual(result['waiters'],30)
        self.assertIs(result['equivalent_cohort_latency_gain_observed'],True)
    def test_all_25_effect_before_return_copies_reject(self):
        for label,s in negative_copies():
            result=AUDIT['audit'](CLIENT,s);row(label,CLIENT,s,result)
            with self.subTest(label=label):
                self.assertTrue(any(e.endswith(':server_return_before_effect') for e in result['errors']))
                self.assertIsNone(result['equivalent_cohort_latency_gain_observed'])
    def test_separate_clock_origin_positive_controls(self):
        c,s=copy.deepcopy(CLIENT),copy.deepcopy(SERVER)
        for e in s['events']:e['ns']+=1
        shifted=AUDIT['audit'](c,s);row('server-origin-plus-one',c,s,shifted)
        self.assertEqual(shifted,AUDIT['audit'](CLIENT,SERVER))
        c,s=copy.deepcopy(CLIENT),copy.deepcopy(SERVER)
        for r in c['rows']:
            for k in ('offered_ns','ended_ns'):r[k]=str(int(r[k])+1)
            for consumer in r['consumers']:
                for k in ('started_ns','deadline_ns','decision_ns','ended_ns'):consumer[k]=str(int(consumer[k])+1)
            for e in r['events']:
                for k in ('ns','ns_decision','deadline_ns'):
                    if k in e:e[k]=str(int(e[k])+1)
        shifted=AUDIT['audit'](c,s);row('client-origin-plus-one',c,s,shifted)
        self.assertEqual(shifted,AUDIT['audit'](CLIENT,SERVER))
    def test_equal_server_timestamp_preserves_sequence_order(self):
        s=copy.deepcopy(SERVER)
        for e in s['events']:e['ns']=1
        result=AUDIT['audit'](CLIENT,s);row('equal-server-clock-ordered-seq',CLIENT,s,result)
        self.assertEqual(result,AUDIT['audit'](CLIENT,SERVER))
    def test_all_72_old_copies_remain_rejected(self):
        for spec in OLD['specifications']():
            c,s=OLD['apply'](spec);result=AUDIT['audit'](c,s);row(spec['label'],c,s,result)
            with self.subTest(label=spec['label']):
                self.assertTrue(result['errors']);self.assertIsNone(result['equivalent_cohort_latency_gain_observed'])
    def test_12_extra_old_copies_remain_rejected(self):
        for label,c,s in OLD['extra_copies']():
            result=AUDIT['audit'](c,s);row(label,c,s,result)
            with self.subTest(label=label):
                self.assertTrue(result['errors']);self.assertIsNone(result['equivalent_cohort_latency_gain_observed'])
    def test_12_original_controls_remain_effective(self):
        controls=AUDIT['controls'](CLIENT,SERVER)
        for control in controls:records.append({'label':'existing-control/'+control['control'],'control_result':control})
        self.assertEqual(len(controls),12);self.assertTrue(all(r['rejected'] for r in controls))

if __name__=='__main__':
    OUT.mkdir(parents=True,exist_ok=False);(OUT/'copies').mkdir()
    result=unittest.TextTestRunner(verbosity=2).run(unittest.defaultTestLoader.loadTestsFromTestCase(SavedCausalRegressions))
    (OUT/'rows.json').write_bytes((json.dumps(records,indent=2)+'\n').encode())
    print(json.dumps({'methods':result.testsRun,'saved_rows':len(records),'failures':len(result.failures),'errors':len(result.errors),
                      'candidate_sha256':sha((ROOT/'audit_v3.py.txt').read_bytes()),'browser_or_formal_invocations':0}))
    sys.exit(0 if result.wasSuccessful() else 1)
