def check(rows):
    keys=[v+'-'+t+'-'+c for v,t,c in itertools.product(['current','naive','protected'],['OSError','KeyboardInterrupt'],['healthy','before','after'])]
    if len(rows)!=18 or [r['cell'] for r in rows]!=keys: raise ValueError('ordered case coverage')

        expected={'raised_type':'CleanupFault' if masked else t,'primary_identity':not masked,'cleanup_identity':masked,'context_is_primary':masked,'notes':['journal cleanup failed: CleanupFault'] if v=='protected' and c!='healthy' else [],'close_attempts':0 if v=='current' else 1,'fd_open_at_exception':not closed,'textio_closed_at_exception':closed,'fd_open_after_driver_cleanup':False,'textio_closed_after_driver_cleanup':True,'factory_calls':1,'processes_started':0,'journal_bytes':0}
        for name,value in expected.items():
            if type(r.get(name)) is not type(value) or r[name]!=value: raise ValueError(r['cell']+' '+name)
        if not r['frames'] or not r['started_utc']<=r['finished_utc']: raise ValueError('trace/time')
