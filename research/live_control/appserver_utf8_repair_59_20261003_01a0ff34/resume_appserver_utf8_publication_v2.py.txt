env=dict(os.environ,GIT_NO_LAZY_FETCH="1",GIT_INDEX_FILE=str(out/"publication.index"))
def sha(data):
    return hashlib.sha256(data).hexdigest()
def put_json(path,value):
    path.write_bytes((json.dumps(value,ensure_ascii=True,indent=2)+"\n").encode("ascii"))
def git(*args,input=None):
    result=subprocess.run(["git",*args],cwd=repo,env=env,input=input,capture_output=True)
    if result.returncode:
        raise RuntimeError(json.dumps({"command":list(args),"exit":result.returncode,"stderr":result.stderr.decode("utf-8",errors="replace")},ensure_ascii=True))
    return result.stdout
(out/"MANIFEST_prepublication_v1.json").write_bytes((archive/"MANIFEST.json").read_bytes())
(archive/"build_appserver_utf8_repair_v1.py.txt").write_bytes((root/"build_appserver_utf8_repair.py").read_bytes())
(archive/"resume_appserver_utf8_publication_v2.py.txt").write_bytes(Path(__file__).read_bytes())
failure={"actual_tool_chunk":"3e5e1e","exit_code":1,"failed_command":["git","write-tree"],
    "observed_error":"lazy fetching disabled; could not fetch b6d5dd6b3cb200c8949332391c125fa5cb4c8900 from promisor remote",
    "outer_PID_UTC_separate_raw_unavailable":True,"native_or_regression_replays":0,

manifest=[]

put_json(archive/"MANIFEST.json",{"entries":manifest,"self_excluded":True})
staged=[]
for p in sorted(stage.rglob("*")):
    if p.is_file():
        rel=p.relative_to(stage).as_posix()
        data=p.read_bytes()
        oid=git("hash-object","-w","--stdin",input=data).decode().strip()
        git("update-index","--add","--cacheinfo","100644,"+oid+","+rel)
        staged.append({"path":rel,"mode":"100644","blob":oid,"bytes":len(data),"sha256":sha(data)})
tree=git("write-tree","--missing-ok").decode().strip()
def entries(rev):
    result={}
    for row in git("ls-tree","-r","-z",rev).split(b"\0"):
        if row:
            meta,path=row.split(b"\t",1)
            result[path.decode("utf-8")]=meta.decode("ascii")
    return result
