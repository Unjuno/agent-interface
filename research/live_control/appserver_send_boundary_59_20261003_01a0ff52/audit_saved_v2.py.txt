"""Data/source-only oracle: no client/producer imports or native calls."""
from pathlib import Path
import ast,json,hashlib,sys
ROOT=Path(__file__).parent
def check(root):
 plan=json.loads((root/'PLAN.json').read_text());errors=[];rows=[]
 def need(c,e):
  if not c:errors.append(e)
 for n,pin in plan['sources'].items():
  b=(root/n).read_bytes();need(hashlib.sha256(b).hexdigest()==pin['sha256'],'source/'+n)
 tree=ast.parse((root/'source_snapshot.py').read_text());fn=next(n for n in ast.walk(tree) if isinstance(n,ast.FunctionDef) and n.name=='request')
 write_line=next(n.lineno for n in ast.walk(fn) if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute) and n.func.attr=='_write')
 deadline_line=next(n.lineno for n in ast.walk(fn) if isinstance(n,ast.Assign) and any(isinstance(t,ast.Name) and t.id=='deadline' for t in n.targets))
 need(deadline_line>write_line,'deadline/source-order')
 expected=[('current','healthy'),('existing_writer','healthy'),('current','pressure'),('existing_writer','pressure'),('current','pressure'),('existing_writer','pressure')]
 need([(x['arm'],x['mode']) for x in plan['cells']]==expected,'deck')
 for cell in plan['cells']:
  cid=cell['id'];p=root/'execution'/cid;x=json.loads((p/'result.json').read_text());receipt=json.loads((p/'driver-receipt.json').read_text());need(x['cell']==cell,cid+'/cell');need(receipt['id']==cid,cid+'/receipt')
  need(type(receipt['exit_code']) is int and receipt['exit_code']==0,cid+'/driver-exit');need(receipt['driver_reaped'] is True,cid+'/driver-reap')
  need(type(x['peer_pid']) is int and x['peer_pid']>0,cid+'/pid');need(x.get('construction_error') is None,cid+'/construction')
  for n in ['stdout','stderr']:
   b=(p/(n+'.txt')).read_bytes();need(len(b)==receipt[n]['bytes'] and hashlib.sha256(b).hexdigest()==receipt[n]['sha256'],cid+'/'+n)
  wire=(json.dumps({'method':'fixture/echo','id':1,'params':{'padding':cell['padding']}},separators=(',',':'))+'\n').encode()
  need((p/'intended-wire.txt').read_bytes()==wire,cid+'/wire');need(x['wire']['sha256']==hashlib.sha256(wire).hexdigest() and type(x['wire']['bytes']) is int and x['wire']['bytes']==len(wire),cid+'/wire-join')
  pref=(p/'prefill.txt').read_bytes();need(x['prefill']['bytes']==len(pref) and x['prefill']['sha256']==hashlib.sha256(pref).hexdigest(),cid+'/prefill-join')
  need(type(x['peer_exit']) is int and x['peer_exit']==-15,cid+'/peer-exit-native-mac');need(x['owned_peer_reaped'] is True and x['caller_after_close'] is False and x['reader_after_close'] is False,cid+'/final-owned')
  need(all(x['final_streams'].get(n) is True for n in ['stdin','stdout','stderr']),cid+'/final-pipes')
  kinds=[e['kind'] for e in x['events']];need(kinds==['checkpoint','cleanup_begin','cleanup_end'],cid+'/event-order')
  stamps=[e['driver_ns'] for e in x['events']];need(all(type(t) is int for t in stamps) and stamps==sorted(stamps),cid+'/stamps')
  cp=x['checkpoint'];out=x['caller_outcome'];need(type(out['started_ns']) is int and type(cp['driver_ns']) is int and cp['driver_ns']-out['started_ns']>=int(plan['checkpoint_seconds']*1e9),cid+'/checkpoint')
  need(type(out['ended_ns']) is int and out['ended_ns']>=out['started_ns'],cid+'/caller-end')
  peers=[json.loads(l) for l in (p/'peer-events.jsonl').read_text().splitlines()];need(peers[0]['kind']=='ready' and peers[0].get('mode')==cell['mode'] and x['ready']=={'method':'fixture/ready','params':{'mode':cell['mode']}},cid+'/peer-ready')
  if cell['mode']=='healthy':
   need(not pref and cp['caller_alive'] is False and out['kind']=='result' and out['result']=={'label':'inert','padding':cell['padding']},cid+'/healthy')
   need([e['kind'] for e in peers]==['ready','read','reply'],cid+'/peer-healthy')
   need(bytes.fromhex(peers[1]['wire_hex'])==wire and type(peers[1]['bytes']) is int and peers[1]['bytes']==len(wire),cid+'/peer-wire')
   observed='healthy_result'
  else:
   need(0<len(pref)<=plan['max_prefill_bytes'] and x['prefill_would_block'] is True,cid+'/actual-pressure')
   need(len(peers)==1,cid+'/peer-nonread')
   need(cp['peer_poll'] is None and cp['reader_alive'] is True,cid+'/live-peer-reader')
   if cell['arm']=='current':
    frames=cp['frames'];q=next((f for f in frames if f['source']=='source_snapshot.py' and f['function']=='request'),{})
    need(cp['caller_alive'] is True and q.get('line')==write_line and q.get('deadline_local_present') is False,cid+'/before-deadline')
    need(any(f['source']=='source_snapshot.py' and f['function']=='_write' for f in frames),cid+'/write-frame')
    need(out['ended_ns']>=x['events'][1]['driver_ns'],cid+'/only-after-cleanup')
    observed='current_predeadline_blocked'
   else:
    need(cp['caller_alive'] is False and out['kind']=='error' and out['error']=='WriteUncertain' and out['total']==len(wire) and type(out['sent']) is int and 0<=out['sent']<len(wire),cid+'/uncertain')
    need(out['ended_ns']<=cp['driver_ns'] and x['writer_at_checkpoint']['poisoned'] is True,cid+'/finite-poison')
    f=x.get('followup',{});need(all(type(f.get(k)) is int and f[k]>=0 for k in ['write_calls_before','write_calls_after','records_before','records_after']),cid+'/followup-count-types');need(f.get('error')=='WriteUncertain' and f.get('sent')==0 and f.get('total')==0 and f.get('write_calls_before')==f.get('write_calls_after') and f.get('records_before')==f.get('records_after'),cid+'/no-followup-write')
    observed='existing_writer_uncertain_no_append'
  rows.append({'id':cid,'observed':observed,'peer_exit':x['peer_exit']})
 return {'errors':errors,'rows':rows,'decision':'PASS_PREDEADLINE_SEND_BOUNDARY_SCOPED' if not errors else 'HOLD_OR_PROTOCOL_FAILURE','source_deadline_after_write':deadline_line>write_line,'no_runtime_imports_or_native_calls':True}
if __name__=='__main__':
 result=check(ROOT);(ROOT/'independent-result-v2.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result,indent=2));raise SystemExit(bool(result['errors']))
