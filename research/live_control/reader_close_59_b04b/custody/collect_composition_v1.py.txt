"""Finite ordinary native pipe + exact helper composition, not a game run."""
import ast, atexit, base64, ctypes, datetime, hashlib, json, os, pathlib, subprocess, sys, threading, time, traceback, types

R = pathlib.Path(__file__).resolve().parent
WIRE = b'{"method":"late-owned-frame","params":{"value":"retained"}}\n'
def utc(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def sha(b): return hashlib.sha256(b).hexdigest()
def save(p, obj):
    temp = p.with_name(p.name + '.publishing')
    with temp.open('x', encoding='utf-8') as f: json.dump(obj, f, indent=2); f.write('\n')
    os.replace(temp, p)
def wait_file(p, timeout):
    end = time.monotonic() + timeout
    while not p.exists():
        if time.monotonic() >= end: raise TimeoutError('fixture sidecar missing: ' + p.name)
        time.sleep(0.005)
def parent(d, condition):
    if condition == 'closed':
        save(d / 'parent.json', {'pid': os.getpid(), 'created_utc': utc(), 'condition': condition})
        return 0
    argv = [sys.executable, '-B', '-u', str(__file__), '--writer', str(d)]
    p = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=sys.stdout, stderr=subprocess.DEVNULL)
    save(d / 'parent.json', {'pid': os.getpid(), 'created_utc': utc(), 'condition': condition, 'writer_pid': p.pid, 'writer_argv': argv})
    wait_file(d / 'writer-ready.json', 3)
    return 0
def writer(d):
    save(d / 'writer-ready.json', {'pid': os.getpid(), 'created_utc': utc(), 'stdout_fd': sys.stdout.fileno()})
    try: wait_file(d / 'release.flag', 12)
    except TimeoutError:
        save(d / 'writer-final.json', {'pid': os.getpid(), 'exit_code': 5, 'safety_timeout': True, 'ended_utc': utc()})
        return 5
    released = time.perf_counter_ns()
    sys.stdout.write(WIRE.decode('ascii')); sys.stdout.flush()
    save(d / 'writer-final.json', {'pid': os.getpid(), 'exit_code': 0, 'safety_timeout': False, 'released_ns': released, 'ended_utc': utc(), 'wire_input_base64': base64.b64encode(WIRE).decode()})
    return 0
class ProcessHandle:
    def __init__(self, pid):
        self.k = ctypes.WinDLL('kernel32', use_last_error=True)
        for name, args, result in [('OpenProcess', [ctypes.c_ulong, ctypes.c_int, ctypes.c_ulong], ctypes.c_void_p), ('GetExitCodeProcess', [ctypes.c_void_p, ctypes.POINTER(ctypes.c_ulong)], ctypes.c_int), ('WaitForSingleObject', [ctypes.c_void_p, ctypes.c_ulong], ctypes.c_ulong), ('CloseHandle', [ctypes.c_void_p], ctypes.c_int)]:
            fn = getattr(self.k, name); fn.argtypes = args; fn.restype = result
        self.handle = self.k.OpenProcess(0x100000 | 0x1000, 0, pid)
        if not self.handle: raise ctypes.WinError(ctypes.get_last_error())
    def code(self):
        value = ctypes.c_ulong()
        if not self.k.GetExitCodeProcess(self.handle, ctypes.byref(value)): raise ctypes.WinError(ctypes.get_last_error())
        return value.value
    def wait(self):
        if self.k.WaitForSingleObject(self.handle, 3000) != 0: raise TimeoutError('owned writer not terminal')
        return self.code()
    def close(self):
        if not self.k.CloseHandle(self.handle): raise ctypes.WinError(ctypes.get_last_error())

def one(condition, out):
    d = out / condition; d.mkdir()
    ns = {'__name__': 'reader_close_candidate'}
    exec(compile((R / 'candidate-client.py.txt').read_bytes(), 'candidate-client.py.txt', 'exec'), ns)
    module_ast = ast.parse((R / 'startup-controller.py.txt').read_bytes())
    helper = next(n for n in module_ast.body if isinstance(n, ast.FunctionDef) and n.name == '_close_failed_startup')
    helper_ns = {'threading': threading, 'subprocess': subprocess, 'atexit': atexit}
    exec(compile(ast.Module(body=[helper], type_ignores=[]), 'exact-startup-helper', 'exec'), helper_ns)
    argv = [sys.executable, '-B', '-u', str(__file__), '--parent', str(d), condition]
    row = {'condition': condition, 'driver_pid': os.getpid(), 'created_utc': utc(), 'parent_argv': argv, 'status': 'STARTED', 'source_sha256': sha((R / 'candidate-client.py.txt').read_bytes()), 'helper_ast_sha256': sha(ast.dump(helper, include_attributes=False).encode())}
    save(d / 'INTENT.json', row)
    journal = d / 'journal.jsonl'
    client = ns['CodexAppServerClient'](argv, journal_path=journal)
    row['parent_pid'] = client.process.pid
    handle = None; errors = []; calls = []; exitfuncs_called = False
    previous_hook = threading.excepthook
    def hook(a):
        if a.thread is client._reader:
            errors.append({'type': a.exc_type.__name__, 'message': str(a.exc_value), 'traceback': ''.join(traceback.format_exception(a.exc_type, a.exc_value, a.exc_traceback))})
        previous_hook(a)
    threading.excepthook = hook
    actual_close = client.close
    def recorded_close(timeout=5):
        call = {'timeout': timeout, 'start_ns': time.perf_counter_ns()}; calls.append(call)
        try:
            value = actual_close(timeout=timeout); call['return_value'] = value; return value
        except BaseException as e:
            call.update(error_type=type(e).__name__, error_message=str(e)); raise
        finally: call['end_ns'] = time.perf_counter_ns()
    client.close = recorded_close
    atexit.register(client.close)
    row['real_atexit_registered'] = True
    try:
        wait_file(d / 'parent.json', 3)
        parent_row = json.loads((d / 'parent.json').read_bytes())
        if parent_row['pid'] != client.process.pid: raise ValueError('parent identity mismatch')
        if condition == 'held':
            wait_file(d / 'writer-ready.json', 3)
            ready = json.loads((d / 'writer-ready.json').read_bytes())
            if ready['pid'] != parent_row['writer_pid']: raise ValueError('writer identity mismatch')
            handle = ProcessHandle(ready['pid']); row['writer_pid'] = ready['pid']
        row['parent_exit_before_helper'] = client.process.wait(timeout=3)
        if condition == 'closed': client._reader.join(timeout=2)
        row.update(reader_alive_before_helper=client._reader.is_alive(), client_closed_before_helper=client._closed, writer_exit_before_helper=handle.code() if handle else None)
        if row['parent_exit_before_helper'] != 0 or row['reader_alive_before_helper'] is not (condition == 'held'): raise ValueError('native stdout lifetime not established')
        if handle and row['writer_exit_before_helper'] != 259: raise ValueError('writer already terminal')
        client._record('fixture', {'method': 'retained-seed'})
        row['journal_seed_base64'] = base64.b64encode(journal.read_bytes()).decode()
        row['helper_start_ns'] = time.perf_counter_ns()
        receipt = helper_ns['_close_failed_startup'](None, None, client, d)
        row.update(helper_end_ns=time.perf_counter_ns(), helper_receipt=receipt, calls_at_checkpoint=list(map(dict, calls)), reader_alive_at_checkpoint=client._reader.is_alive(), journal_closed_at_checkpoint=client._journal.closed, client_closed_at_checkpoint=client._closed, parent_exit_at_checkpoint=client.process.poll(), writer_exit_at_checkpoint=handle.code() if handle else None, journal_checkpoint_base64=base64.b64encode(journal.read_bytes()).decode())
        if condition == 'held':
            if receipt['client_closed'] is not False or receipt['errors'] != ["planner close: TimeoutError('app-server reader close timed out')"]: raise ValueError('helper incomplete receipt wrong')
            if client._journal.closed or not client._reader.is_alive(): raise ValueError('held reader/journal state wrong')
        elif receipt['client_closed'] is not True or receipt['errors'] or not client._journal.closed or client._reader.is_alive(): raise ValueError('closed helper control wrong')
        row['release_ns'] = time.perf_counter_ns()
        (d / 'release.flag').write_bytes(b'release-owned-writer\n')
        row['writer_exit_after_release'] = handle.wait() if handle else None
        client._reader.join(timeout=2)
        row.update(reader_alive_after_release=client._reader.is_alive(), client_closed_after_release=client._closed, reader_exceptions=list(errors), notifications_after_release=list(client._notifications), journal_after_release_base64=base64.b64encode(journal.read_bytes()).decode())
        if client._reader.is_alive() or errors: raise ValueError('reader not retired cleanly')
        row['atexit_run_start_ns'] = time.perf_counter_ns()
        atexit._run_exitfuncs(); exitfuncs_called = True
        row.update(atexit_run_end_ns=time.perf_counter_ns(), calls_after_real_exitfuncs=list(map(dict, calls)), final_journal_closed=client._journal.closed)
        if len(calls) != (2 if condition == 'held' else 1) or not client._journal.closed: raise ValueError('actual callback retention/removal wrong')
        if condition == 'held' and list(client._notifications) != [json.loads(WIRE)]: raise ValueError('late frame missing')
        row['status'] = 'OBSERVED'
    except BaseException as e:
        row.update(status='STOP', error_type=type(e).__name__, error_message=str(e), traceback=traceback.format_exc())
        raise
    finally:
        (d / 'release.flag').write_bytes(b'release-owned-writer\n')
        if handle:
            row['cleanup_writer_exit'] = handle.wait(); handle.close(); row['writer_handle_closed'] = True
        row['cleanup_parent_exit'] = client.process.wait(timeout=3)
        client._reader.join(timeout=2)
        row['cleanup_reader_alive'] = client._reader.is_alive()
        if not client._reader.is_alive():
            if not exitfuncs_called: atexit._run_exitfuncs()
            if not client._journal.closed: actual_close(timeout=1)
            for pipe in [client.process.stdin, client.process.stdout, client.process.stderr]: pipe.close()
        row.update(cleanup_pipe_closed=[p.closed for p in [client.process.stdin, client.process.stdout, client.process.stderr]], cleanup_journal_closed=client._journal.closed, finished_utc=utc())
        threading.excepthook = previous_hook
        save(d / 'raw.json', row)
    return row

def main():
    freeze = json.loads((R / 'FREEZE.composition.v1.json').read_bytes())
    for name, pin in freeze['sources'].items():
        b = (R / name).read_bytes()
        if len(b) != pin['bytes'] or sha(b) != pin['sha256']: raise ValueError('source changed: ' + name)
    out = R / 'composition-v1'; out.mkdir()
    save(out / 'START.json', {'pid': os.getpid(), 'started_utc': utc(), 'argv': sys.argv, 'freeze_sha256': sha((R / 'FREEZE.composition.v1.json').read_bytes())})
    rows = []
    try:
        for condition in freeze['cells']: rows.append(one(condition, out))
    finally: save(out / 'END.json', {'pid': os.getpid(), 'ended_utc': utc(), 'completed_cells': len(rows)})
    print(json.dumps({'completed_cells': len(rows), 'rows': [{'condition': r['condition'], 'helper_receipt': r['helper_receipt'], 'calls': r['calls_after_real_exitfuncs'], 'reader_alive': r['reader_alive_at_checkpoint'], 'journal_closed': r['journal_closed_at_checkpoint']} for r in rows]}, indent=2))
if __name__ == '__main__':
    if len(sys.argv) > 1 and sys.argv[1] == '--parent': raise SystemExit(parent(pathlib.Path(sys.argv[2]), sys.argv[3]))
    if len(sys.argv) > 1 and sys.argv[1] == '--writer': raise SystemExit(writer(pathlib.Path(sys.argv[2])))
    main()
