"""One versioned ordinary unit selection with prospective pins and actual receipt."""
import datetime,hashlib,json,os,pathlib,subprocess,sys
R=pathlib.Path(__file__).resolve().parent;stage=sys.argv[1]
out=R/'evidence'/stage;out.mkdir(parents=True,exist_ok=False)
cases=out/'cases';cases.mkdir();src=R/'src/research/live_control'
sha=lambda b:hashlib.sha256(b).hexdigest()
now=lambda:datetime.datetime.now(datetime.timezone.utc).isoformat()
pins={}
for name in ['codex_app_server_client_v2.py','test_appserver_reader_close_01a0ff33.py']:
    b=(src/name).read_bytes();(out/(name+'.txt')).write_bytes(b);pins[name]={'bytes':len(b),'sha256':sha(b)}
argv=[sys.executable,'-B']+(['-O'] if stage.endswith('optimized') else [])+['-m','unittest','-v','test_appserver_reader_close_01a0ff33']
intent={'stage':stage,'argv':argv,'cwd':str(src),'created_utc':now(),'source_pins':pins,'ordinary_unit':True,'native_or_formal_replay':False}
with (out/'INTENT.json').open('x',encoding='utf-8') as f:json.dump(intent,f,indent=2);f.write('\n')
env=dict(os.environ,READER_CLOSE_UNIT_EVIDENCE=str(cases),PYTHONDONTWRITEBYTECODE='1')
start=now();p=subprocess.Popen(argv,cwd=src,env=env,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
stdout,stderr=p.communicate(timeout=10)
(out/'stdout.log').write_bytes(stdout);(out/'stderr.log').write_bytes(stderr)
receipt={**intent,'pid':p.pid,'started_utc':start,'ended_utc':now(),'exit_code':p.returncode,
         'stdout_bytes':len(stdout),'stdout_sha256':sha(stdout),'stderr_bytes':len(stderr),'stderr_sha256':sha(stderr)}
with (out/'receipt.json').open('x',encoding='utf-8') as f:json.dump(receipt,f,indent=2);f.write('\n')
print(json.dumps(receipt,indent=2));print(stderr.decode('utf-8',errors='replace'));raise SystemExit(p.returncode)
