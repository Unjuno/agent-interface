"""Saved bytes and effective copied corruptions only; never import producers."""
import ast, base64, copy, datetime, hashlib, json, pathlib, sys
R = pathlib.Path(sys.argv[1]).resolve() if len(sys.argv) > 1 else pathlib.Path(__file__).resolve().parent
def sha(b): return hashlib.sha256(b).hexdigest()
def load(p): return json.loads((R / p).read_bytes())
def need(ok, text):
    if not ok: raise ValueError(text)
def num(x): return type(x) is int
def rows(b): return [json.loads(line) for line in b.splitlines()]
def audit_native(row, condition):
    held = condition == 'held'
    need(row['status'] == 'OBSERVED' and row['condition'] == condition, 'native identity/status')
    for name in ['driver_pid', 'parent_pid']:
        need(num(row[name]) and row[name] > 0, 'PID ' + name)
    for name in ['parent_exit_before_helper', 'parent_exit_at_checkpoint', 'cleanup_parent_exit']:
        need(num(row[name]) and row[name] == 0, 'parent exit ' + name)
    for name in ['reader_alive_before_helper', 'reader_alive_at_checkpoint']:
        need(row[name] is held, 'reader ' + name)
    need(row['client_closed_at_checkpoint'] is (not held), 'client close checkpoint')
    need(row['journal_closed_at_checkpoint'] is (not held), 'journal close checkpoint')
    receipt = row['helper_receipt']
    need(receipt['client_closed'] is (not held), 'helper completed')
    need(receipt['input_release_verified'] is False and receipt['session_exit_code'] is None, 'helper scope')
    need(receipt['errors'] == (["planner close: TimeoutError('app-server reader close timed out')"] if held else []), 'helper error')
    checkpoint = row['calls_at_checkpoint']; final = row['calls_after_real_exitfuncs']
    need(len(checkpoint) == 1 and len(final) == (2 if held else 1), 'real callback retention')
    need(final[0] == checkpoint[0] and type(checkpoint[0]['timeout']) is int and checkpoint[0]['timeout'] == 1, 'helper timeout/call')
    if held:
        need(checkpoint[0].get('error_type') == 'TimeoutError' and checkpoint[0].get('error_message') == 'app-server reader close timed out', 'close timeout')
        need(num(row['writer_pid']) and row['writer_pid'] > 0, 'writer PID')
        for name in ['writer_exit_before_helper', 'writer_exit_at_checkpoint']:
            need(num(row[name]) and row[name] == 259, 'live writer ' + name)
        for name in ['writer_exit_after_release', 'cleanup_writer_exit']:
            need(num(row[name]) and row[name] == 0, 'terminal writer ' + name)
        need(row['writer_handle_closed'] is True, 'handle retired')
        need(final[1].get('return_value', 'MISSING') is None and final[1]['timeout'] == 5, 'real callback returns')
    else: need(checkpoint[0].get('return_value', 'MISSING') is None, 'healthy close returns')
    for call in final:
        need(num(call['start_ns']) and num(call['end_ns']) and call['start_ns'] <= call['end_ns'], 'call clock')
    need(row['helper_start_ns'] <= final[0]['start_ns'] <= final[0]['end_ns'] <= row['helper_end_ns'] <= row['release_ns'] <= row['atexit_run_start_ns'] <= row['atexit_run_end_ns'], 'causal order')
    if held: need(row['atexit_run_start_ns'] <= final[1]['start_ns'] <= final[1]['end_ns'] <= row['atexit_run_end_ns'], 'callback causal order')
    need(row['reader_alive_after_release'] is False and row['cleanup_reader_alive'] is False and row['client_closed_after_release'] is True, 'reader retired')
    need(row['reader_exceptions'] == [], 'reader error')
    need(row['real_atexit_registered'] is True and row['final_journal_closed'] is True and row['cleanup_journal_closed'] is True, 'journal/callback retired')
    need(type(row['cleanup_pipe_closed']) is list and len(row['cleanup_pipe_closed']) == 3 and all(x is True for x in row['cleanup_pipe_closed']), 'pipe retired')
    seed = base64.b64decode(row['journal_seed_base64'], validate=True)
    checkpoint_bytes = base64.b64decode(row['journal_checkpoint_base64'], validate=True)
    after = base64.b64decode(row['journal_after_release_base64'], validate=True)
    need(seed == checkpoint_bytes and len(rows(seed)) == 1 and rows(seed)[0]['direction'] == 'fixture' and rows(seed)[0]['message'] == {'method': 'retained-seed'}, 'seed retained')
    expected = {'method': 'late-owned-frame', 'params': {'value': 'retained'}}
    need(row['notifications_after_release'] == ([expected] if held else []), 'late cache')
    need(after.startswith(seed) and len(rows(after)) == (2 if held else 1), 'journal chain')
    if held: need(rows(after)[1]['direction'] == 'received' and rows(after)[1]['message'] == expected, 'late journal')
    return True
