"""Owned inert JSONL child; no Codex, provider, game, or input API."""
import json
import os
from pathlib import Path
import sys
import time

api, scenario, root_text = sys.argv[1:]
root = Path(root_text)
record = {'pid': os.getpid(), 'api': api, 'scenario': scenario}


def save(name, data):
    temporary = root / (name + '.new')
    temporary.write_text(json.dumps(data, sort_keys=True) + '\n')
    temporary.replace(root / name)


def wait_file(name):
    deadline = time.monotonic() + 3
    while not (root / name).exists():
        if time.monotonic() >= deadline:
            raise RuntimeError('owned control file missing: ' + name)
        time.sleep(0.001)


save('ready.json', record)
if api == 'request':
    line = sys.stdin.readline()
    record['request_line'] = line
    request = json.loads(line)
else:
    wait_file('go')
    request = None
    record['request_line'] = None

if scenario == 'healthy':
    message = ({'id': request['id'], 'result': {'usable': True}} if request is not None
               else {'method': 'fixture/ready', 'params': {'usable': True}})
    wire = (json.dumps(message, separators=(',', ':')) + '\n').encode('utf-8')
    record['stdout_hex'] = wire.hex()
    record['stdout_write_count'] = os.write(1, wire)
    diagnostic = b'healthy diagnostic\n'
elif scenario == 'held':
    record['stdout_hex'] = ''
    record['stdout_write_count'] = 0
    diagnostic = b'held diagnostic remains owned\n'
elif scenario == 'invalid':
    record['stdout_hex'] = ''
    record['stdout_write_count'] = 0
    diagnostic = b'invalid diagnostic: \xff\xfe\n'
else:
    record['stdout_hex'] = ''
    record['stdout_write_count'] = 0
    diagnostic = b''

record['stderr_hex'] = diagnostic.hex()
record['stderr_write_count'] = os.write(2, diagnostic) if diagnostic else 0
record['stdout_closed_ns'] = time.monotonic_ns()
save('wire.json', record)
os.close(1)
if scenario == 'held':
    wait_file('release')
os.close(2)
save('finished.json', {'pid': os.getpid(), 'owned_release_received': scenario == 'held'})
os._exit(0)
