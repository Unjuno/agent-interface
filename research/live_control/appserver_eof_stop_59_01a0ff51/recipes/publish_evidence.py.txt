"""Exact copies or disclosed own-path projections; original files are untouched."""
import hashlib
import json
from pathlib import Path
import shutil

ROOT = Path(__file__).resolve().parent
WT = ROOT.parent / 'agent-interface-eof-59'
LEAF = WT / 'research/live_control/appserver_eof_stop_59_01a0ff51'
entries = []
prefixes = [(str(WT), 'OWN_SOURCE_WORKTREE'), (str(ROOT), 'OWN_OUTPUT_ROOT')]


def copy(source, relative):
    original = source.read_bytes()
    public = original
    applied = []
    for old, replacement in prefixes:
        if old.encode() in public:
            public = public.replace(old.encode(), replacement.encode())
            applied.append({'original_prefix_kind': 'owned source worktree' if old == str(WT) else 'owned output root',
                            'public_prefix': replacement})
    destination = LEAF / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    if destination.exists():
        if destination.read_bytes() != public:
            raise RuntimeError('refuse rewriting previously retained bytes: ' + relative)
    else:
        destination.write_bytes(public)
    entries.append({'public_file': relative, 'original_custody': 'private author owned output/' + str(source.relative_to(ROOT)),
                    'original_bytes': len(original), 'original_sha256': hashlib.sha256(original).hexdigest(),
                    'public_bytes': len(public), 'public_sha256': hashlib.sha256(public).hexdigest(),
                    'projection': applied or 'exact bytes'})


for directory in ('native-baseline', 'native-candidate', 'unit-red', 'unit-green', 'unit-optimized',
                  'capture-baseline', 'capture-candidate', 'local-ci'):
    for source in sorted((ROOT / directory).rglob('*')):
        if source.is_file():
            relative = str(source.relative_to(ROOT))
            if directory.startswith('capture-') or directory.startswith('unit-') or directory == 'local-ci':
                relative = 'captures/' + relative
            copy(source, relative)
for name in ('setup-first-error.txt', 'audit-v1-receipt.private.json', 'audit-v2-receipt.private.json'):
    copy(ROOT / name, 'captures/' + name.replace('.private', ''))
for name in ('run_native_once.py', 'run_local_ci.py', 'publish_evidence.py'):
    copy(ROOT / name, 'recipes/' + name + '.txt')
(LEAF / 'PUBLICATION.json').write_text(json.dumps({'schema': 'owned-eof-original-public-custody-v1',
        'policy': 'only literal owned host source/output path prefixes projected; native original bytes unchanged',
        'entries': entries}, indent=2, sort_keys=True) + '\n')
print(json.dumps({'copied_files': len(entries), 'projected_files': sum(e['projection'] != 'exact bytes' for e in entries),
                  'original_bytes': sum(e['original_bytes'] for e in entries), 'public_bytes': sum(e['public_bytes'] for e in entries)}))
