"""Fake-X tests for opt-in keymap-batch-edge-v1 measurement receipts.

These tests exercise the real InputOwner thread against a synthetic Xlib
display. The monotonic values are generated by the local process; they are
not latency or live-input evidence.
"""
from __future__ import annotations

import json
import sys
import threading
import types
import unittest
from pathlib import Path

LIVE = Path(__file__).resolve().parent
if str(LIVE) not in sys.path:
    sys.path.insert(0, str(LIVE))

# Load exception types before this module installs synthetic Xlib modules.
import executor_v3  # noqa: F401


class FakeDisplay:
    def __init__(self):
        self.down = set()
        self.trace = []
        self.query_count = 0
        self.fail_queries = set()
        self.query_hooks = {}
        self.drop_counts = {}
        self.persistent_drop_codes = set()
        self.focus = 41
        self.root = types.SimpleNamespace(
            query_pointer=lambda: types.SimpleNamespace(
                mask=0, root_x=0, root_y=0))

    def get_input_focus(self):
        return types.SimpleNamespace(focus=self.focus)

    def keysym_to_keycode(self, keysym):
        return {ord("W"): 38, ord("A"): 39, ord("S"): 40}.get(keysym, 0)

    def screen(self):
        return types.SimpleNamespace(root=self.root)

    def query_keymap(self):
        self.query_count += 1
        index = self.query_count
        snapshot = tuple(sorted(self.down))
        self.trace.append(("query_keymap", index, snapshot))
        if index in self.fail_queries:
            raise RuntimeError("synthetic keymap query failure")
        bitmap = bytearray(32)
        for code in self.down:
            bitmap[code // 8] |= 1 << (code % 8)
        hook = self.query_hooks.pop(index, None)
        if hook is not None:
            hook()
        return bytes(bitmap)

    def sync(self):
        self.trace.append(("sync", tuple(sorted(self.down))))

    def close(self):
        self.trace.append(("display_close",))


class FakeLease:
    def __init__(self, token="measurement-intent"):
        self.intent_token = token
        self.deadline = 2**62
        self.cancel = threading.Event()
        self.expected_focus = 41
        self.focus_invalid = False
        self.interruptions = []

    def check(self):
        if self.deadline <= 0:
            raise RuntimeError("synthetic lease expired")

    def record_interruption(self, row):
        self.interruptions.append(row)


class KeyMeasurementTests(unittest.TestCase):
    def setUp(self):
        self.fake = FakeDisplay()
        self._saved_modules = {
            name: sys.modules.get(name)
            for name in ("Xlib", "Xlib.X", "Xlib.XK", "Xlib.display",
                         "Xlib.error", "Xlib.ext", "Xlib.ext.xtest",
                         "input_owner_v12")
        }
        x = types.ModuleType("Xlib.X")
        x.KeyPress, x.KeyRelease, x.ButtonRelease = 2, 3, 5
        x.Button1Mask, x.AnyPropertyType, x.IsViewable = 256, 0, 2
        xk = types.ModuleType("Xlib.XK")
        xk.string_to_keysym = lambda key: ord(key)
        display = types.ModuleType("Xlib.display")
        display.Display = lambda _name: self.fake
        error = types.ModuleType("Xlib.error")
        error.BadWindow = type("BadWindow", (Exception,), {})
        error.BadDrawable = type("BadDrawable", (Exception,), {})
        ext = types.ModuleType("Xlib.ext")
        xtest = types.ModuleType("Xlib.ext.xtest")

        def fake_input(_display, event, code):
            self.fake.trace.append(("key_event", event, code))
            if event == x.KeyPress:
                self.fake.down.add(code)
            elif event == x.KeyRelease:
                remaining = self.fake.drop_counts.get(code, 0)
                if code in self.fake.persistent_drop_codes or remaining:
                    if remaining:
                        self.fake.drop_counts[code] = remaining - 1
                else:
                    self.fake.down.discard(code)

        xtest.fake_input = fake_input
        ext.xtest = xtest
        xlib = types.ModuleType("Xlib")
        xlib.X, xlib.XK, xlib.display, xlib.error, xlib.ext = x, xk, display, error, ext
        sys.modules.update({
            "Xlib": xlib, "Xlib.X": x, "Xlib.XK": xk,
            "Xlib.display": display, "Xlib.error": error,
            "Xlib.ext": ext, "Xlib.ext.xtest": xtest,
        })
        sys.modules.pop("input_owner_v12", None)
        from input_owner_v12 import InputOwner
        self.InputOwner = InputOwner
        self.owner = None

    def tearDown(self):
        if self.owner is not None:
            close_error = None
            try:
                self.owner.close()
            except Exception as exc:
                close_error = exc
            # Persistent-loss coverage may make release itself fail, but the
            # owner thread must still terminate and report its stop state.
            if close_error is not None:
                self.assertTrue(
                    self.fake.persistent_drop_codes,
                    "unexpected owner-close failure outside persistent-loss case: "
                    + repr(close_error))
            self.assertTrue(self.owner.stopped.wait(1), "owner thread did not stop")
            self.assertFalse(self.owner.thread.is_alive(), "owner thread leaked")
        for name, module in self._saved_modules.items():
            if module is None:
                sys.modules.pop(name, None)
            else:
                sys.modules[name] = module

    def start_owner(self, *, measure=True):
        self.owner = self.InputOwner(":fake", measure_key_edges=measure)
        return self.owner

    def _measurement(self, row):
        self.assertIsInstance(row, dict)
        result = row.get("physical_key_measurement")
        self.assertIsInstance(result, dict)
        self.assertEqual(result.get("schema"), "keymap-batch-edge-v1")
        self.assertIs(result.get("grants_input_authority"), False)
        self.assertIs(result.get("application_consumption_observed"), False)
        return result

    def _assert_confirmed(self, measurement, *, edge, basis, status, identity):
        self.assertEqual(measurement.get("edge"), edge)
        self.assertEqual(measurement.get("interval_basis"), basis)
        self.assertEqual(measurement.get("classification"), status)
        self.assertEqual(measurement.get("identity_status"), identity)
        self.assertIsInstance(measurement.get("actuation_id"), str)
        self.assertTrue(measurement["actuation_id"])
        bracket = measurement.get("bracket")
        self.assertIsInstance(bracket, dict)
        self.assertEqual(bracket.get("status"), status)
        adapter = measurement.get("adapter_edge")
        self.assertIsInstance(adapter, dict)
        self.assertEqual(adapter.get("status"), status)
        self.assertEqual(adapter.get("edge"), edge)
        self.assertEqual(adapter.get("actuation_id"), measurement["actuation_id"])
        self.assertIs(adapter.get("grants_input_authority"), False)
        self.assertIs(adapter.get("application_consumption_observed"), False)
        pre, post = measurement.get("pre_sample"), measurement.get("post_sample")
        self.assertIsInstance(pre, dict)
        self.assertIsInstance(post, dict)
        self.assertEqual(adapter.get("interval"),
                         [pre.get("finished_ns"), post.get("finished_ns")])
        interval_field = "physical_down_interval" if edge == "down" else "physical_up_interval"
        self.assertEqual(bracket.get(interval_field), adapter["interval"])
        for field in ("owner_id", "intent_token", "key", "actuation_id"):
            self.assertEqual(bracket.get(field), measurement.get(field))
        self.assertTrue(all(type(value) is int for value in adapter["interval"]))
        self.assertLessEqual(adapter["interval"][0], adapter["interval"][1])

    def _assert_unconfirmed(self, row):
        measurement = self._measurement(row)
        self.assertNotIn(measurement.get("classification"), {
            "CONFIRMED_PHYSICAL_DOWN", "CONFIRMED_PHYSICAL_UP"})
        self.assertIsNone(measurement.get("bracket"))
        self.assertIsNone(measurement.get("adapter_edge"))
        return measurement

    def _emit_raw(self, **payload):
        print(json.dumps({"test": self.id(), "trace": self.fake.trace, **payload},
                         sort_keys=True), flush=True)

    def test_two_key_down_and_batch_up_preserve_a01_pairs(self):
        self.start_owner()
        lease = FakeLease()
        downs = [self.owner.call("down", lease, key) for key in ("W", "A")]
        down_measurements = [self._measurement(row) for row in downs]
        for measurement in down_measurements:
            self._assert_confirmed(
                measurement, edge="down", basis="key_snapshot",
                status="CONFIRMED_PHYSICAL_DOWN", identity="MINTED")
        ups = self.owner.call("up_batch", lease, ["W", "A"])
        self.assertEqual([row["key"] for row in ups], ["W", "A"])
        up_measurements = [self._measurement(row) for row in ups]
        for down, up in zip(down_measurements, up_measurements):
            self._assert_confirmed(
                up, edge="up", basis="shared_batch_snapshot",
                status="CONFIRMED_PHYSICAL_UP", identity="RETIRED")
            self.assertEqual(down["actuation_id"], up["actuation_id"])
        self.assertEqual(self.fake.down, set())
        self._emit_raw(down_measurements=down_measurements, up_measurements=up_measurements)

    def test_measurement_disabled_keeps_legacy_query_count(self):
        self.start_owner(measure=False)
        lease = FakeLease()
        before = self.fake.query_count
        down = self.owner.call("down", lease, "W")
        self.assertEqual(self.fake.query_count, before)
        self.assertNotIn("physical_key_measurement", down)
        self.owner.call("up_batch", lease, ["W"])
        self.assertEqual(self.fake.query_count - before, 2)
        self._emit_raw(query_count_delta=self.fake.query_count - before,
                       measurement_present="physical_key_measurement" in down)

    def test_rehold_same_key_mints_a_distinct_identity(self):
        self.start_owner()
        lease = FakeLease()
        first_down = self._measurement(self.owner.call("down", lease, "W"))
        first_up = self._measurement(self.owner.call("up_batch", lease, ["W"])[0])
        second_down = self._measurement(self.owner.call("down", lease, "W"))
        second_up = self._measurement(self.owner.call("up_batch", lease, ["W"])[0])
        self.assertEqual(first_down["actuation_id"], first_up["actuation_id"])
        self.assertEqual(second_down["actuation_id"], second_up["actuation_id"])
        self.assertNotEqual(first_down["actuation_id"], second_down["actuation_id"])
        self._emit_raw(actuation_ids=[first_down["actuation_id"],
                                      second_down["actuation_id"]])

    def test_duplicate_down_cannot_create_or_complete_a_second_pair(self):
        self.start_owner()
        lease = FakeLease()
        first = self._measurement(self.owner.call("down", lease, "W"))
        press_count = sum(row[:2] == ("key_event", 2) for row in self.fake.trace)
        duplicate_rejected = False
        try:
            duplicate = self.owner.call("down", lease, "W")
        except (ValueError, RuntimeError):
            duplicate_rejected = True
            duplicate_measurement = None
        else:
            duplicate_measurement = self._assert_unconfirmed(duplicate)
        self.assertTrue(duplicate_rejected or duplicate_measurement is not None)
        rows = self.owner.call("up_batch", lease, ["W"])
        up = self._measurement(rows[0])
        if duplicate_rejected:
            self.assertEqual(up["actuation_id"], first["actuation_id"])
            self.assertEqual(up["classification"], "CONFIRMED_PHYSICAL_UP")
        else:
            self.assertNotEqual(up["classification"], "CONFIRMED_PHYSICAL_UP")
            self.assertIsNone(up.get("adapter_edge"))
        self.assertEqual(sum(row[:2] == ("key_event", 2) for row in self.fake.trace),
                         press_count if duplicate_rejected else press_count + 1)
        self._emit_raw(duplicate_rejected=duplicate_rejected,
                       duplicate_measurement=duplicate_measurement, up_measurement=up)

    def test_first_and_last_key_retry_keep_per_key_up_brackets(self):
        for retry_key in ("W", "A"):
            with self.subTest(retry_key=retry_key):
                self.start_owner()
                lease = FakeLease("retry-" + retry_key)
                down_rows = [self.owner.call("down", lease, key) for key in ("W", "A")]
                code = {"W": 38, "A": 39}[retry_key]
                self.fake.drop_counts[code] = 1
                ups = self.owner.call("up_batch", lease, ["W", "A"])
                self.assertEqual(len(ups), 2)
                by_key = {row["key"]: row for row in ups}
                for down_row in down_rows:
                    key = down_row["key"]
                    up_measurement = self._measurement(by_key[key])
                    self.assertEqual(up_measurement["classification"],
                                     "CONFIRMED_PHYSICAL_UP")
                    self.assertEqual(up_measurement["actuation_id"],
                                     self._measurement(down_row)["actuation_id"])
                    expected_attempts = 2 if key == retry_key else 1
                    self.assertEqual(len(by_key[key]["server_keyup_attempts"]),
                                     expected_attempts)
                self.assertEqual(self.fake.down, set())
                self._emit_raw(retry_key=retry_key, receipts=ups)
                self.owner.close()
                self.owner = None

    def test_batch_pre_or_post_sample_failure_suppresses_every_bracket(self):
        for failed_offset in (1, 2):
            with self.subTest(failed_offset=failed_offset):
                self.start_owner()
                lease = FakeLease("sample-failure-" + str(failed_offset))
                self.owner.call("down", lease, "W")
                self.owner.call("down", lease, "A")
                self.fake.fail_queries.add(self.fake.query_count + failed_offset)
                rows = self.owner.call("up_batch", lease, ["W", "A"])
                for row in rows:
                    self._assert_unconfirmed(row)
                if failed_offset == 2:
                    with self.assertRaisesRegex(RuntimeError, "release pending"):
                        self.owner.call("down", lease, "S")
                self._emit_raw(failed_query_offset=failed_offset, receipts=rows)
                self.owner.close()
                self.owner = None

    def test_one_persistently_unverified_key_suppresses_all_batch_edges_and_down(self):
        self.start_owner()
        lease = FakeLease()
        self.owner.call("down", lease, "W")
        self.owner.call("down", lease, "A")
        self.fake.persistent_drop_codes.add(38)
        rows = self.owner.call("up_batch", lease, ["W", "A"])
        self.assertTrue(any(not row["server_keyup_verified"] for row in rows))
        self.assertTrue(any(row["server_keyup_verified"] for row in rows))
        for row in rows:
            self._assert_unconfirmed(row)
        press_count = sum(row[:2] == ("key_event", 2) for row in self.fake.trace)
        with self.assertRaisesRegex(RuntimeError, "release pending"):
            self.owner.call("down", lease, "S")
        self.assertEqual(sum(row[:2] == ("key_event", 2) for row in self.fake.trace),
                         press_count)
        self._emit_raw(receipts=rows, refused_down=True)

    def test_down_measurement_query_rechecks_cancel_focus_and_expiry_before_press(self):
        for invalidator in ("cancel", "focus", "expiry"):
            with self.subTest(invalidator=invalidator):
                # Each prior case may deliberately change focus or lease state.
                self.fake.focus = 41
                self.start_owner()
                lease = FakeLease("invalid-" + invalidator)
                query_index = self.fake.query_count + 1
                if invalidator == "cancel":
                    self.fake.query_hooks[query_index] = lease.cancel.set
                elif invalidator == "focus":
                    self.fake.query_hooks[query_index] = lambda: setattr(
                        self.fake, "focus", 99)
                else:
                    self.fake.query_hooks[query_index] = lambda: setattr(lease, "deadline", 0)
                press_count = sum(row[:2] == ("key_event", 2) for row in self.fake.trace)
                try:
                    result = self.owner.call("down", lease, "W")
                except Exception as exc:
                    result = {"raised": type(exc).__name__, "message": str(exc)}
                self.assertGreaterEqual(
                    self.fake.query_count, query_index,
                    "invalidation hook must run during the measurement query")
                after_count = sum(row[:2] == ("key_event", 2) for row in self.fake.trace)
                self.assertEqual(after_count, press_count)
                self.assertNotIn(38, self.fake.down)
                self._emit_raw(invalidator=invalidator, result=result)
                self.owner.close()
                self.owner = None

    def test_cleanup_then_same_key_cannot_reuse_retired_identity(self):
        self.start_owner()
        old_lease = FakeLease("old-intent")
        old_down = self._measurement(self.owner.call("down", old_lease, "W"))
        cleanup = self.owner.call("release", old_lease)
        self.assertTrue(cleanup["verified"])
        new_lease = FakeLease("new-intent")
        new_down = self._measurement(self.owner.call("down", new_lease, "W"))
        new_up = self._measurement(self.owner.call("up_batch", new_lease, ["W"])[0])
        self.assertNotEqual(old_down["actuation_id"], new_down["actuation_id"])
        self.assertEqual(new_down["actuation_id"], new_up["actuation_id"])
        self._emit_raw(old_actuation_id=old_down["actuation_id"],
                       new_actuation_id=new_down["actuation_id"], cleanup=cleanup)
