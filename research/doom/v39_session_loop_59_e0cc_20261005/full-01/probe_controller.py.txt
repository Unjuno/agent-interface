def write_json(path, value):
    path.write_text(json.dumps(value, indent=2) + "\n")

timeline_lock = threading.Lock()
def record(event, **fields):
    with timeline_lock, (run / "harness-timeline.jsonl").open("a") as stream:
        stream.write(json.dumps({"event": event, "at_ns": time.perf_counter_ns(), **fields}) + "\n")

    tmp = signal_path.with_suffix(".tmp")
    write_json(tmp, {"health": health, "ammo": 20})
    tmp.replace(signal_path)
    record("external_pixel_signal", health=health)

class FakeClient:
    def __init__(self, *args, **kwargs):
        self.closed = False
        record("fake_model_client_created")
    def initialize(self):
        record("fake_model_client_initialized")
    def close(self, timeout=None):
        self.closed = True
        record("fake_model_client_closed")

def active_answer():
    return {
        "state": "active", "commands": [{"action": "strafe_left", "extent": "pulse"}],
        "action_validity": [{"critical_health_minimum": 35, "maximum_health_loss": 12,
                             "minimum_ammo": 0, "max_current_age_ms": 1000}],
        "contingencies": [], "next_cover": [{"action": "strafe_left", "extent": "short"}],
        "next_cover_validity": [{"signal_id": "health", "critical_health_minimum": 35,
                                 "maximum_health_loss": 12, "max_source_age_ms": 30000}],
    }

class FakePlanner:
    def __init__(self, *args, **kwargs):
        self.thread_id = "synthetic-model-thread"
        self.turns = 0
        self.interrupted = threading.Event()
    def start_session(self):
        record("fake_model_session_started")
    def begin_turn(self, prompt, output_schema, image_path):
        self.turns += 1
        handle = SimpleNamespace(thread_id=self.thread_id, turn_id=f"synthetic-turn-{self.turns}")
        record("fake_model_begin", turn=self.turns, turn_id=handle.turn_id, prompt=prompt,
               image_path=image_path, image_sha256=hashlib.sha256(Path(image_path).read_bytes()).hexdigest())
        return handle
    def await_turn(self, handle, timeout):
        turn = int(handle.turn_id.rsplit("-", 1)[-1])
        record("fake_model_await", turn=turn)
        if turn == 2 and scenario != "normal":
            time.sleep(.05)
            set_health(60)
            if not self.interrupted.wait(6):
                raise AssertionError("real controller did not interrupt the pending model")
            # Deliberately deliver an otherwise eligible stale answer after
            # cancellation. The actual controller must refuse this answer.
            time.sleep(.02)
        else:
            time.sleep(.05)
        answer = active_answer() if turn < 3 else {
            "state": "dead", "commands": [], "action_validity": [],
            "contingencies": [], "next_cover": [], "next_cover_validity": [],
        }
        record("fake_model_return", turn=turn, answer_state=answer["state"], eligible=True)
        return SimpleNamespace(answer=answer, usage={}, handle=handle, status="completed",
                               answer_eligible=True, cancellation_requested=False, error=None)
    def interrupt(self, handle):
        record("fake_model_interrupt", turn_id=handle.turn_id)
        self.interrupted.set()
        return {"requested": True, "thread_id": handle.thread_id, "turn_id": handle.turn_id}

class CommandLog:
    def __init__(self, stream):
        self.stream = stream
    def write(self, value):
        record("controller_stdin_write", text=value)
        return self.stream.write(value)
    def __getattr__(self, name):
        return getattr(self.stream, name)

children = []
def start_child(command, **kwargs):
    # Preserve every argument generated by the actual session_command.
    actual = [command[0], "-B", str(job / "child_session.py"), *command[1:]]
    write_json(run / "child-command.json", {"generated": command, "bootstrapped": actual})
    env = dict(os.environ, E0CC_TEST_EXTERNAL_DIR=str(run / "external"),
               E0CC_TEST_SIGNAL_PATH=str(signal_path), PYTHONDONTWRITEBYTECODE="1",
               OPENBLAS_NUM_THREADS="1", OMP_NUM_THREADS="1")
    child = subprocess.Popen(actual, **kwargs, env=env)
    child.stdin = CommandLog(child.stdin)
    children.append(child)
    record("actual_child_started", pid=child.pid)
    return child
