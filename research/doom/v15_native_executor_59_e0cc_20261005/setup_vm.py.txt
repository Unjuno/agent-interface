import datetime,hashlib,io,json,subprocess,tarfile
from pathlib import Path
base=Path(__file__).resolve().parent
orb=['/opt/homebrew/bin/orb','-m','v15-native-e0cc-20261005','-u','study','-w','/home/study']
def save(n,d):
 with (base/n).open('x') as f:json.dump(d,f,indent=2);f.write('\n')
def run(n,args,**kw):
 save(n+'-request.json',{'at_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'argv':args})
 p=subprocess.run(args,capture_output=True,timeout=45,**kw)
 save(n+'-result.json',{'exit_code':p.returncode,'stdout':p.stdout.decode(),'stderr':p.stderr.decode()});assert p.returncode==0
 return p
