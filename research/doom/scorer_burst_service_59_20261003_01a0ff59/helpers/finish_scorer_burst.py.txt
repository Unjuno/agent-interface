"""Post-run endpoints, preserved regression inputs and ordinary source checks."""
import hashlib
import importlib.util
import json
import subprocess
import sys
import unittest
from pathlib import Path

ROOT = Path(__file__).resolve().parent
REPO = ROOT / 'agent-interface'
OUT = REPO / 'research/doom/scorer_burst_service_59_20261003_01a0ff59'
raw = json.loads((OUT / 'raw.json').read_bytes())
audit = json.loads((OUT / 'audit.json').read_bytes())
freeze = json.loads((OUT / 'FREEZE.json').read_bytes())
assert audit['status'] == 'PASS_FINITE_SCHEDULING_REFERENCE'
assert all(hashlib.sha256((OUT / p).read_bytes()).hexdigest() == h for p, h in freeze['hashes'].items())
summary = {}
by_case = {}
for row in raw['rows']:
    case = row['case']
    key = tuple(case[k] for k in ('path', 'sample_cost_ns', 'command_cost_ns', 'length', 'layout'))
    by_case.setdefault(key, {})[case['arm']] = row
for arm in ('original', 'repair_v2', 'one_command'):
    rows = [r for r in raw['rows'] if r['case']['arm'] == arm]
    summary[arm] = {'complete': sum(r['outcome'] == 'complete' for r in rows),
                    'diagnostic_budget_stop': sum(r['outcome'] != 'complete' for r in rows),
                    'maximum_command_sample_age_periods': max(r['max_command_sample_age_ns'] or 0 for r in rows) / 100_000_000}
contrasts = []
for key, arms in by_case.items():
    prior, proposed = arms['repair_v2'], arms['one_command']
    if prior['max_command_sample_age_ns'] > proposed['max_command_sample_age_ns']:
        contrasts.append({'case': key, 'prior_age_ns': prior['max_command_sample_age_ns'],
                          'one_command_age_ns': proposed['max_command_sample_age_ns'],
                          'prior_finish_ns': prior['ended_simulated_ns'],
                          'one_command_finish_ns': proposed['ended_simulated_ns']})
censored = [r['case'] for r in raw['rows'] if r['outcome'] == 'diagnostic_budget_stop' and r['case']['sample_cost_ns'] < 100_000_000]
assert summary['one_command']['complete'] == 256 and contrasts
assert all(r['max_command_sample_age_ns'] <= 100_000_000 + r['case']['sample_cost_ns'] for r in raw['rows'] if r['case']['arm'] == 'one_command')
result = {'status': 'SUPPORTS_FINITE_BURST_SERVICE_TRADEOFF', 'summary': summary,
          'age_improved_matched_conditions': len(contrasts), 'contrasts': contrasts,
          'short_callback_diagnostic_censor_cases': censored,
          'qualification': 'Original 134 budget stops comprise 128 sustained-overrun recurrence cases and six short-callback long-stream cases censored by the declared 64-sample budget. Do not call all budget stops infinite starvation. Source/inputs/frozen raw/auditor unchanged; no extension or replay.',
          'virtual_finish_cost_can_increase': any(c['one_command_finish_ns'] > c['prior_finish_ns'] for c in contrasts)}
with (OUT / 'ENDPOINTS.json').open('x', encoding='utf-8') as handle:
    json.dump(result, handle, indent=2)
    handle.write('\n')
tests = ('test_main_thread_scorer_polling_v1.py', 'test_map01_scorer_stdio_adapter_v1.py', 'test_independent_progress_clock_v2.py')
folder = OUT / 'regression_sources'
folder.mkdir(exist_ok=False)
for name in tests:
    data = subprocess.check_output(['git', '-C', str(REPO), 'show', freeze['base_sha'] + ':research/doom/' + name])
    (folder / name).write_bytes(data)
sys.path.insert(0, str(OUT / 'sources/one_command'))
modules = []
for name in tests:
    spec = importlib.util.spec_from_file_location(name[:-3], folder / name)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    modules.append(module)
for module in modules:
    for obj in vars(module).values():
        if isinstance(obj, type) and issubclass(obj, unittest.TestCase):
            for name in ('test_real_pipe_eof_and_same_thread', 'test_pipe_wait_samples_and_wakes_on_command'):
                if hasattr(obj, name):
                    setattr(obj, name, unittest.skip('Windows select does not support the POSIX pipe fixture; source preserved')(getattr(obj, name)))
suite = unittest.TestSuite(unittest.defaultTestLoader.loadTestsFromModule(m) for m in modules)
with (OUT / 'execution/regressions.stderr.txt').open('x', encoding='utf-8') as stream:
    check = unittest.TextTestRunner(stream=stream, verbosity=2).run(suite)
receipt = {'tests_run': check.testsRun, 'skips': len(check.skipped), 'failures': len(check.failures),
           'errors': len(check.errors), 'successful': check.wasSuccessful(),
           'test_source_sha256': {p.name: hashlib.sha256(p.read_bytes()).hexdigest() for p in folder.iterdir()}}
with (OUT / 'execution/regressions.json').open('x', encoding='utf-8') as handle:
    json.dump(receipt, handle, indent=2)
print(json.dumps({'summary': summary, 'contrasts': len(contrasts), 'censored_cases': censored,
                  'max_added_finish_ns': max(c['one_command_finish_ns'] - c['prior_finish_ns'] for c in contrasts), 'regressions': receipt}))
sys.exit(0 if check.wasSuccessful() else 1)
