"""Capture two distinct ordinary reader steps; never invoke source arms."""
import datetime as dt
import hashlib
import json
import pathlib
import subprocess
import sys

ROOT = pathlib.Path(__file__).resolve().parent.parent
REPO = ROOT / 'work/agent-interface'
OUT = REPO / 'research/doom/scorer_burst_service_59_20261003_01a0ff59/terminal_scope_v2'
mode = sys.argv[1]
freeze = json.loads((OUT / 'FREEZE_V2.json').read_bytes())
source = OUT / 'reader.py.txt'
if hashlib.sha256(source.read_bytes()).hexdigest() != freeze['reader_sha256']:
    raise ValueError('fixed reader source')
if mode == 'literal':
    argv = [sys.executable, '-B', str(source), '--literal', 'literal-results.json']
elif mode == 'retained':
    if not (OUT / 'literal-results.json').exists():
        raise ValueError('literal checks first')
    argv = [sys.executable, '-B', str(source), '../raw.json', 'retained-grid.json']
else:
    raise ValueError('step')
started = dt.datetime.now(dt.timezone.utc).isoformat()
result = subprocess.run(argv, cwd=OUT, capture_output=True, timeout=30)
for suffix, data in [('stdout.txt',result.stdout),('stderr.txt',result.stderr)]:
    with (OUT / (mode + '.' + suffix)).open('xb') as f:
        f.write(data)
record = {'step':mode,'started_utc':started,'ended_utc':dt.datetime.now(dt.timezone.utc).isoformat(),
    'exit_code':result.returncode,'public_argv':['python','-B','reader.py.txt',*argv[3:]],
    'reader_sha256':freeze['reader_sha256'],'original_source_or_oracle_invocations':0,'retry':0,
    'stdout_sha256':hashlib.sha256(result.stdout).hexdigest(),'stderr_sha256':hashlib.sha256(result.stderr).hexdigest()}
with (OUT / (mode + '.receipt.json')).open('x', encoding='utf-8', newline='\n') as f:
    f.write(json.dumps(record,indent=2,sort_keys=True)+'\n')
print(json.dumps(record))
if result.stdout:
    print(result.stdout.decode())
if result.stderr:
    print(result.stderr.decode())
raise SystemExit(result.returncode)
