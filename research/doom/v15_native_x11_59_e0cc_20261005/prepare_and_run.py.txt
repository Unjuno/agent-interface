sha = lambda p: hashlib.sha256(p.read_bytes()).hexdigest()
def save(name, value):
    with (base/name).open('x') as f: json.dump(value,f,indent=2); f.write('\n')
lock=json.loads((base/'source-lock.json').read_bytes())
for r in lock['files']:
    assert sha(base/'source'/r['path']) == r['sha256'], r['path']

    compile((base/n).read_text(),n,'exec')

save('freeze.json',freeze)
buffer=io.BytesIO()

        tar.add(base/n,arcname=n)
