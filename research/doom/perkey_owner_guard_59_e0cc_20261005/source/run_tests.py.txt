import hashlib
import json
from pathlib import Path
import subprocess
import sys
import time

root = Path(__file__).resolve().parent
destination = root / sys.argv[1]
destination.mkdir(exist_ok=False)
source = root / 'source'
command = [sys.executable, '-B', str(source / 'research/doom/test_perkey_owner_source_guard.py'), '-v']
paths = sorted(source.rglob('*.py'))
manifest = {str(path.relative_to(source)): hashlib.sha256(path.read_bytes()).hexdigest() for path in paths}
(destination / 'sources.json').write_text(json.dumps(manifest, indent=2) + '\n')
started = time.time_ns()
completed = subprocess.run(command, capture_output=True, timeout=100)
ended = time.time_ns()
(destination / 'stdout.txt').write_bytes(completed.stdout)
(destination / 'stderr.txt').write_bytes(completed.stderr)
receipt = {'command': command, 'started_unix_ns': started, 'ended_unix_ns': ended,
           'exit_code': completed.returncode, 'stdout_sha256': hashlib.sha256(completed.stdout).hexdigest(),
           'stderr_sha256': hashlib.sha256(completed.stderr).hexdigest()}
(destination / 'receipt.json').write_text(json.dumps(receipt, indent=2) + '\n')
print(json.dumps(receipt))
print(completed.stderr.decode())
sys.exit(completed.returncode)
