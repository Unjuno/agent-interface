from pathlib import Path
import importlib.util, json, os, sys, unittest
ROOT=Path(__file__).parent
spec=importlib.util.spec_from_file_location("prep",ROOT/"prepare_scorer_service.py")
prep=importlib.util.module_from_spec(spec);spec.loader.exec_module(prep)
DOOM=ROOT/"agent-interface/research/doom"
if __name__=="__main__":
    if len(sys.argv)>1 and sys.argv[1]=="suite":
        sys.path.insert(0,str(DOOM))
        names=("test_main_thread_scorer_polling_v1","test_map01_scorer_stdio_adapter_v1",
               "test_independent_progress_clock_v2","test_map01_scorer_scheduler_replay_3270",
               "test_scorer_command_service_01a0ff58")
        if os.name=="nt":
            import test_main_thread_scorer_polling_v1 as polling
            import test_map01_scorer_stdio_adapter_v1 as stdio
            reason="Windows select.select supports sockets, not os.pipe; no POSIX pipe result claimed"
            polling.PollingTests.test_real_pipe_eof_and_same_thread=unittest.skip(reason)(polling.PollingTests.test_real_pipe_eof_and_same_thread)
            stdio.Tests.test_pipe_wait_samples_and_wakes_on_command=unittest.skip(reason)(stdio.Tests.test_pipe_wait_samples_and_wakes_on_command)
        suite=unittest.defaultTestLoader.loadTestsFromNames(names)
        result=unittest.TextTestRunner(verbosity=2).run(suite)
        sys.exit(0 if result.wasSuccessful() else 1)
    sources=[]
    for name in ("main_thread_scorer_polling_v1.py","map01_scorer_stdio_adapter_v1.py"):
        original=prep.git("cat-file","blob",prep.PRIOR+":"+prep.PACKAGE+"/sources/fair_v2/"+name)
        actual=(DOOM/name).read_bytes()
        normalized=actual.replace(b"\r\n",b"\n")
        if normalized!=original:raise ValueError("v2 adoption mismatch: "+name)
        sources.append(dict(path="research/doom/"+name,working_sha256=prep.sha(actual),git_normalized_sha256=prep.sha(normalized),
                            upstream_v2_sha256=prep.sha(original),upstream_v2_git_blob=prep.git("rev-parse",prep.PRIOR+":"+prep.PACKAGE+"/sources/fair_v2/"+name).decode().strip()))
    prep.dump(prep.OUT/"ADOPTION.json",dict(upstream_pr=6896,upstream_head=prep.PRIOR,exact_normalized_v2=True,sources=sources))
    (prep.OUT/"run.py").write_bytes(Path(__file__).read_bytes())
    for label,opt in (("focused-green",False),("focused-optimized",True)):
        argv=[sys.executable,"-B",*(["-O"] if opt else []),"-m","unittest","-v","test_scorer_command_service_01a0ff58"]
        prep.run(label,DOOM,argv,0)
    for label,opt in (("portable-suite",False),("portable-suite-optimized",True)):
        prep.run(label,DOOM,[sys.executable,"-B",*(["-O"] if opt else []),str(Path(__file__).resolve()),"suite"],0)
