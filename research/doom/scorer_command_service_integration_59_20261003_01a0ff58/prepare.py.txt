def run(label,cwd,argv,expected):
    started=datetime.now(timezone.utc).isoformat()
    result=subprocess.run(argv,cwd=cwd,capture_output=True,timeout=30)
    ended=datetime.now(timezone.utc).isoformat()
    stdout=result.stdout;stderr=result.stderr
    (OUT/(label+".stdout.log")).write_bytes(stdout);(OUT/(label+".stderr.log")).write_bytes(stderr)
    receipt=dict(label=label,argv=argv,cwd=str(cwd),started_utc=started,ended_utc=ended,
        exit_code=result.returncode,expected_exit=expected,stdout_sha256=sha(stdout),stderr_sha256=sha(stderr))
    dump(OUT/(label+".receipt.json"),receipt)
    print(json.dumps(receipt),flush=True)
    if result.returncode!=expected: raise ValueError("unexpected run status")
