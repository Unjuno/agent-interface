"""Raw-only literal endpoint and complete native byte accounting."""
import hashlib
import json
from pathlib import Path
import sys

def audit(rows, cases):
    errors = []
    if type(rows) is not list or len(rows) != len(cases):
        return ['case count'], {}
    summary = {'rows': len(rows), 'pre_drain_commands': 0,
               'blocked_direct_rows': 0, 'capacity_refusals': 0}
    for row, case in zip(rows, cases):
        prefix = case['id'] + ':'
        def require(condition, message):
            if not condition:
                errors.append(prefix + message)
        require(json.dumps(row['case'], sort_keys=True) == json.dumps(case, sort_keys=True), 'case identity')
        for flag, desired in [('timed_out', False), ('sink_seen', True), ('readers_terminal', True)]:
            require(type(row[flag]) is bool and row[flag] is desired, flag)
        require(type(row['exit_code']) is int and row['exit_code'] == 0, 'exit')
        require(row['reader_errors'] == [], 'reader errors')
        events = row['events']
        kinds = [e['event'] for e in events]
        require(all(type(e['index']) is int for e in events) and
                [e['index'] for e in events] == list(range(1, len(events) + 1)), 'event order')
        require(all(type(e['native_ns']) is int and type(e['thread_id']) is int for e in events), 'event integer types')
        require(len({e['thread_id'] for e in events}) == 1, 'callback owner thread')
        require(kinds.count('pipe_ready') == 1 and kinds.count('sink_enter') == 1 and kinds.count('sample') == 1, 'one sample/pipe')
        require('source_exception' not in kinds, 'unexpected source exception')
        pre = row['pre_drain_events']
        require(json.dumps(pre, sort_keys=True) == json.dumps(events[:len(pre)], sort_keys=True), 'causal pre-drain prefix')
        pre_commands = [e['line'] for e in pre if e['event'] == 'command']
        overflow = case['capacity_bytes'] == 1
        blocked = case['pipe'] == 'full' and case['sink'] == 'direct'
        expected_pre = [] if blocked or overflow else ['FINISH']
        require(pre_commands == expected_pre, 'pre-drain command endpoint')
        summary['pre_drain_commands'] += len(pre_commands)
        summary['blocked_direct_rows'] += int(blocked)
        summary['capacity_refusals'] += int(overflow)
        commands = [e['line'] for e in events if e['event'] == 'command']
        require(commands == ([] if overflow else ['FINISH']), 'eventual command endpoint')
        require(kinds.count('capacity_refused') == int(overflow), 'capacity refusal')
        require(kinds.count('source_terminal') == int(not overflow), 'source terminal')
        require(kinds.count('sink_return') == int(not overflow), 'sink return')
        ready = next(e for e in events if e['event'] == 'pipe_ready')
        require(ready['nonblocking'] is True, 'native nonblocking initialization')
        require(type(ready['prefill_bytes']) is int and ready['prefill_bytes'] >= 0, 'prefill type')
        require((ready['prefill_bytes'] > 0) is (case['pipe'] == 'full'), 'native prefill')
        enter = next(e for e in events if e['event'] == 'sink_enter')
        require(enter['row']['controller_visible'] is False, 'controller isolation')
        literal = {'controller_visible': False, 'missed_periods_before': 0,
                   'sample_finished_ns': 0, 'sample_started_ns': 0,
                   'scheduled_ns': 0, 'start_lateness_ns': 0,
                   'payload': {'schema': 'independent-progress-sample-v2',
                               'sample_ns': 0, 'kill_count': 0, 'death_count': 0,
                               'episode_finished': False, 'player_dead': False, 'map_exit': False}}
        require(json.dumps(enter['row'], sort_keys=True) == json.dumps(literal, sort_keys=True), 'literal typed sample receipt')
        frame = (json.dumps(enter['row'], sort_keys=True, separators=(',', ':')) + '\n').encode()
        require(hashlib.sha256(frame).hexdigest() == enter['sha256'] and len(frame) == enter['bytes'], 'frame identity')
        actual = bytes.fromhex(row['stdout_hex'])
        expected = b'P' * ready['prefill_bytes'] + (b'' if overflow else frame)
        require(actual == expected, 'complete native bytes/order')
        require(type(row['stdout_bytes']) is int and hashlib.sha256(actual).hexdigest() == row['stdout_sha256'] and len(actual) == row['stdout_bytes'], 'native byte hash/length')
        reconciled = next(e for e in events if e['event'] == 'sink_reconciled')
        require(type(reconciled['pending_bytes']) is int and reconciled['pending_bytes'] == 0, 'no pending output at child exit')
        if not overflow:
            terminal = next(e for e in events if e['event'] == 'source_terminal')
            stats = terminal['stats']
            require(type(stats['samples']) is int and stats['samples'] == 1, 'source samples')
            require(type(stats['commands']) is int and stats['commands'] == 1, 'source commands')
            require(type(stats['owner_thread_id']) is int and all(e['thread_id'] == stats['owner_thread_id'] for e in events), 'source owner identity')
            literal_stats = {'owner_thread_id': events[0]['thread_id'], 'samples': 1,
                             'commands': 1, 'missed_sample_periods': 0, 'eof': False}
            if case['adapter'] == 'polling':
                literal_stats.update(stopped_by_command=True, started_ns=0, ended_ns=0)
            else:
                literal_stats['sample_hz'] = 1.0
            require(json.dumps(stats, sort_keys=True) == json.dumps(literal_stats, sort_keys=True), 'complete typed source stats')
            require(type(terminal['pending_bytes']) is int and terminal['pending_bytes'] == (len(frame) if case['pipe'] == 'full' and case['sink'] == 'spool' else 0), 'terminal/persistence separation')
    return errors, summary

if __name__ == '__main__':
    root = Path(__file__).resolve().parent
    cases = json.loads((root / 'FREEZE.json').read_text())['cases']
    raw = json.loads(Path(sys.argv[1]).read_text())
    errors, summary = audit(raw, cases)
    expected = json.loads((root / 'FREEZE.json').read_text())['expected']
    if json.dumps(summary, sort_keys=True) != json.dumps(expected, sort_keys=True):
        errors.append('frozen aggregate endpoint mismatch')
    print(json.dumps({'disposition': 'PASS_PIPE_CONSTRUCTION_SCOPED' if not errors else 'FAIL_PIPE_CONSTRUCTION',
                      'errors': errors, 'summary': summary}, indent=2, sort_keys=True))
    raise SystemExit(bool(errors))
