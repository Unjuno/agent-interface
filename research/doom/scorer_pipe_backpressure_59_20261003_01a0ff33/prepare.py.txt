from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import platform
import subprocess
import sys

root = Path(__file__).resolve().parent
repo = root.parent / 'agent-interface'
sources = {'main403': '403a21a6951662401925835fbc70670bab84b457',
           'repair6913': '8f5872874760b671b558a7febe4e14a23261d753'}
names = ['main_thread_scorer_polling_v1.py', 'map01_scorer_stdio_adapter_v1.py',
         'independent_progress_clock_v2.py']
pins = []
for arm, commit in sources.items():
    for name in names:
        path = 'research/doom/' + name
        data = subprocess.check_output(['git', 'show', commit + ':' + path], cwd=repo)
        tree = subprocess.check_output(['git', 'ls-tree', commit, '--', path], cwd=repo).decode().strip()
        target = root / 'sources' / arm / name
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes(data)
        mode, kind, oid_path = tree.split(' ', 2)
        oid = oid_path.split('\t')[0]
        pins.append({'arm': arm, 'commit': commit, 'repo_path': path,
                     'copy_path': target.relative_to(root).as_posix(), 'mode': mode,
                     'blob_oid': oid, 'bytes': len(data), 'sha256': hashlib.sha256(data).hexdigest()})
(root / 'SOURCE.json').write_text(json.dumps(pins, indent=2, sort_keys=True) + '\n', encoding='utf-8')
environment = {'python': sys.version, 'python_executable_sha256': hashlib.sha256(Path(sys.executable).read_bytes()).hexdigest(),
               'platform': platform.platform(), 'machine': platform.machine(),
               'pid': __import__('os').getpid(), 'scope': 'native Windows / stdlib / no external source dependency'}
(root / 'ENVIRONMENT.json').write_text(json.dumps(environment, indent=2, sort_keys=True) + '\n', encoding='utf-8')
cases = []
for source in sources:
    for adapter in ('polling', 'stdin'):
        for pipe in ('empty', 'full'):
            for sink in ('direct', 'spool'):
                cases.append({'id': '-'.join((source, adapter, pipe, sink)),
                              'source': source, 'adapter': adapter, 'pipe': pipe,
                              'sink': sink, 'capacity_bytes': 4096})
        cases.append({'id': '-'.join((source, adapter, 'full', 'overflow')),
                      'source': source, 'adapter': adapter, 'pipe': 'full',
                      'sink': 'spool', 'capacity_bytes': 1})
files = []
for path in sorted(root.rglob('*')):
    if (path.is_file() and path.name != 'FREEZE.json' and
            'run-01' not in path.parts and '__pycache__' not in path.parts):
        data = path.read_bytes()
        files.append({'path': path.relative_to(root).as_posix(), 'bytes': len(data),
                      'sha256': hashlib.sha256(data).hexdigest()})
freeze = {'worker': '01a0ff33-df63-7d60-871f-a7ecd2649d07', 'policy_version': 'FINAL-v5',
          'claim_comment': 5965233761, 'frozen_utc': datetime.now(timezone.utc).isoformat(),
          'producer_invocations_before_freeze': 0, 'cases': cases, 'files': files,
          'expected': {'rows': 20, 'pre_drain_commands': 12, 'blocked_direct_rows': 4, 'capacity_refusals': 4},
          'caps': {'sink_entry_seconds': 2, 'pre_drain_seconds': .2, 'child_after_drain_seconds': 5,
                   'producer_outer_seconds': 45, 'output_stream_bytes': 1048576},
          'scope': 'ordinary finite native-I/O construction; not a live/formal allocation'}
target = root / 'FREEZE.json'
if target.exists():
    raise RuntimeError('freeze already exists')
target.write_text(json.dumps(freeze, indent=2, sort_keys=True) + '\n', encoding='utf-8')
print(json.dumps({'freeze_sha256': hashlib.sha256(target.read_bytes()).hexdigest(),
                  'cases': len(cases), 'pins': len(files), 'frozen_utc': freeze['frozen_utc']}))
