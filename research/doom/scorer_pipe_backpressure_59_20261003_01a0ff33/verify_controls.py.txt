"""Directed copied-raw witnesses, no producer or source import."""
import copy, hashlib, importlib.util, json
from pathlib import Path
import sys
root=Path(__file__).resolve().parent
spec=importlib.util.spec_from_file_location('raw_auditor',root/'audit.py')
module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module)
raw_path=root/'run-01'/'raw.json'
original=raw_path.read_bytes();raw=json.loads(original)
cases=json.loads((root/'FREEZE.json').read_text())['cases']
def digest(value):
    return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':')).encode()).hexdigest()
if '--verify-only' in sys.argv:
    outcomes=[]
    for path in sorted((root/'controls-01').glob('control-*.json')):
        witness=json.loads(path.read_text())
        mutant=copy.deepcopy(raw)
        if witness['operation']=='drop_last':mutant.pop()
        else:mutant[witness['row_index']]=witness['changed_row']
        errors,_=module.audit(mutant,cases)
        outcomes.append({'control':witness['control'],'rejected':bool(errors),'errors':errors})
    assert len(outcomes)==8 and all(x['rejected'] for x in outcomes)
    assert raw_path.read_bytes()==original
    print(json.dumps({'controls':len(outcomes),'all_rejected':True,'raw_unchanged':True},sort_keys=True))
    raise SystemExit(0)
out=root/'controls-01';out.mkdir(exist_ok=False)
mutations=[]
def edited(label, edit):
    data=copy.deepcopy(raw);edit(data)
    mutations.append((label,data))
def terminal(row):return next(e for e in row['events'] if e['event']=='source_terminal')
edited('sample_count_boolean_alias',lambda data:terminal(data[0])['stats'].__setitem__('samples',True))
edited('command_count_float_alias',lambda data:terminal(data[0])['stats'].__setitem__('commands',1.0))
def index_alias(data):
    data[0]['events'][0]['index']=True;data[0]['pre_drain_events'][0]['index']=True
edited('event_index_boolean_alias',index_alias)
edited('pending_bytes_boolean_alias',lambda data:terminal(data[0]).__setitem__('pending_bytes',False))
def consistent_payload_alias(data):
    row=data[0]
    for events in (row['events'],row['pre_drain_events']):
        e=next(e for e in events if e['event']=='sink_enter')
        e['row']['payload']['kill_count']=False
        frame=(json.dumps(e['row'],sort_keys=True,separators=(',',':'))+'\n').encode()
        e['bytes']=len(frame);e['sha256']=hashlib.sha256(frame).hexdigest()
    row['stdout_hex']=frame.hex();row['stdout_bytes']=len(frame);row['stdout_sha256']=hashlib.sha256(frame).hexdigest()
    next(e for e in row['events'] if e['event']=='sink_reconciled')['produced_sha256']=row['stdout_sha256']
edited('consistent_payload_boolean_alias',consistent_payload_alias)
def output_loss(data):
    row=data[0];native=bytes.fromhex(row['stdout_hex'])[:-1]
    row.update(stdout_hex=native.hex(),stdout_bytes=len(native),stdout_sha256=hashlib.sha256(native).hexdigest())
edited('one_native_byte_lost_with_rehashed_output',output_loss)
edited('pre_drain_command_removed',lambda data:data[0].__setitem__('pre_drain_events',[e for e in data[0]['pre_drain_events'] if e['event']!='command']))
mutations.append(('inventory_row_removed',copy.deepcopy(raw[:-1])))
results=[]
for index,(label,mutant) in enumerate(mutations):
    errors,_=module.audit(mutant,cases)
    changed=[i for i,(before,after) in enumerate(zip(raw,mutant)) if digest(before)!=digest(after)]
    witness={'control':label,'raw_sha256':hashlib.sha256(original).hexdigest(),
             'operation':'drop_last' if len(mutant)!=len(raw) else 'replace_row',
             'row_index':changed[0] if changed else None,
             'changed_row':mutant[changed[0]] if changed else None,
             'mutant_canonical_sha256':digest(mutant),'errors':errors,
             'unchanged_other_rows_sha256':digest([r for i,r in enumerate(raw) if i not in changed and i<len(mutant)]),
             'rejected':bool(errors)}
    (out/('control-%02d.json'%index)).write_text(json.dumps(witness,indent=2,sort_keys=True)+'\n',encoding='utf-8')
    results.append({'control':label,'rejected':bool(errors),'errors':errors})
report={'controls':results,'all_rejected':all(r['rejected'] for r in results),'raw_unchanged':raw_path.read_bytes()==original}
(out/'RESULT.json').write_text(json.dumps(report,indent=2,sort_keys=True)+'\n',encoding='utf-8')
print(json.dumps(report,sort_keys=True))
raise SystemExit(not report['all_rejected'] or not report['raw_unchanged'])
