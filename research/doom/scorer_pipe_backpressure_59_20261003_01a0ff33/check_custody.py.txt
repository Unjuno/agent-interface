"""Bind each actual native command stream back to the first raw record."""
import hashlib,json
from pathlib import Path
root=Path(__file__).resolve().parent
rows=json.loads((root/'run-01'/'raw.json').read_text())
for row in rows:
    name=row['case']['id'];out=root/'run-01'
    native=(out/(name+'.stdout.bin')).read_bytes()
    stderr=(out/(name+'.stderr.jsonl')).read_bytes()
    assert native==bytes.fromhex(row['stdout_hex'])
    assert hashlib.sha256(native).hexdigest()==row['stdout_sha256']
    assert hashlib.sha256(stderr).hexdigest()==row['stderr_sha256']
    assert json.dumps([json.loads(line) for line in stderr.splitlines()],sort_keys=True)==json.dumps(row['events'],sort_keys=True)
freeze=json.loads((root/'FREEZE.json').read_text())
assert all(hashlib.sha256((root/p['path']).read_bytes()).hexdigest()==p['sha256'] for p in freeze['files'])
print(json.dumps({'native_stdout_stderr_pairs':len(rows),'frozen_files_unchanged':len(freeze['files'])},sort_keys=True))
