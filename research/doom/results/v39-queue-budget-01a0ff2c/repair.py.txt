import hashlib,json
from pathlib import Path
root=Path(__file__).parent
p=root/'map01_overlap_controller_v39.py'
b=p.read_bytes()
if hashlib.sha256(b).hexdigest()!='a0bcfa076970b7cf6d048155478952958280b7958e0bbe486c0f1f12a55e4f0e': raise RuntimeError('baseline pin')
old=b'incoming.get(timeout=min(.25,max(.1,end-time.monotonic())))'
new=b'incoming.get(timeout=min(.25,max(0.,end-time.monotonic())))'
if b.count(old)!=1: raise RuntimeError('repair site')
(root/'baseline.py.txt').write_bytes(b)
p.write_bytes(b.replace(old,new))
print(json.dumps({'source_bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest(),'changed_sites':1,'all_other_bytes_exact':p.read_bytes().replace(new,old)==b}))
