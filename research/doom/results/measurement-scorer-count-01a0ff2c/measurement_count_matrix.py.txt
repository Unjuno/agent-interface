"""Ordinary constructed-fixture characterization; no formal allocation."""
from pathlib import Path
import json, sys, tempfile
from audit_map01_measurement_integration_v1 import audit
from test_audit_map01_measurement_integration_v1 import Tests

plan = json.loads(Path(sys.argv[1]).read_text(encoding='utf-8'))
destination = Path(sys.argv[2])
assert not destination.exists()
with destination.open('x', encoding='utf-8', newline='\n') as stream:
    for cell in plan['cells']:
        with tempfile.TemporaryDirectory() as tmp:
            run, research = Tests().make(tmp)
            (run/'scorer-summary.json').write_text(json.dumps(cell['summary']), encoding='utf-8')
            row = {'cell_id':cell['cell_id'], 'summary':cell['summary']}
            try:
                row['result'] = audit(run, research)
            except AttributeError as exc:
                row['exception'] = {'type':type(exc).__name__, 'message':str(exc)}
            stream.write(json.dumps(row, sort_keys=True)+'\n')
print(json.dumps({'rows':len(plan['cells'])}))
