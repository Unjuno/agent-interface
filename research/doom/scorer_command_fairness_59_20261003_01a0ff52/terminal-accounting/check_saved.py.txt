"""Independent grid enumeration on saved terminal records, with type checks."""
from pathlib import Path
import json

root = Path(__file__).resolve().parent
data = json.loads((root/'raw.json').read_text())
failures, comparisons = [], []
expected_keys = {(k,s,c) for k in ('polling_finish','polling_eof','polling_cap','stdin_finish','stdin_eof')
                 for s in ('sample','sink') for c in (350_000_000,10_000_000)}
actual_keys = {(r['kind'],r['cost_stage'],r['cost_ns']) for r in data['rows']}
if len(data['rows']) != 20 or actual_keys != expected_keys: failures.append('case coverage')
for r in data['rows']:
    identity = [r['kind'],r['cost_stage'],r['cost_ns']]
    if any(type(r[k]) is not int for k in ('period_ns','initial_ns','ended_ns','cost_ns')):
        failures.append([identity,'clock type']); continue
    if type(r['stats']['missed_sample_periods']) is not int:
        failures.append([identity,'counter type']); continue
    # Grid enumeration is deliberately independent of scheduler elapsed arithmetic.
    unserved = []
    tick = r['initial_ns'] + r['period_ns']
    while tick < r['ended_ns']:
        unserved.append(tick); tick += r['period_ns']
    expected_endpoint = 'eof' if r['kind'].endswith('eof') else ('cap' if r['kind'].endswith('cap') else 'FINISH')
    expected_commands = ['FINISH'] if expected_endpoint == 'FINISH' else []
    if (r['endpoint'] != expected_endpoint or r['commands'] != expected_commands or
        r['ended_ns'] != r['cost_ns'] or r['stats']['samples'] != 1 or
        r['same_owner'] is not True or len(r['receipts']) != 1):
        failures.append([identity,'construction endpoint'])
    expected_ticks = [100_000_000,200_000_000,300_000_000] if r['cost_ns'] == 350_000_000 else []
    if unserved != expected_ticks or r['stats']['missed_sample_periods'] != 0:
        failures.append([identity,'claimed defect/control differs'])
    comparisons.append({'case':identity,'strict_interior_unserved_deadlines_ns':unserved,
        'reported_missed':r['stats']['missed_sample_periods'],
        'underreports':len(unserved)>r['stats']['missed_sample_periods']})
result = {'failures':failures,'comparisons':comparisons,
          'overrun_witnesses':sum(r['underreports'] for r in comparisons),
          'fast_controls':sum(not r['underreports'] for r in comparisons),
          'diagnostic_verified':not failures}
(root/'independent-result.json').write_text(json.dumps(result,indent=2,sort_keys=True)+'\n')
print(json.dumps({k:v for k,v in result.items() if k!='comparisons'}))
raise SystemExit(bool(failures))
