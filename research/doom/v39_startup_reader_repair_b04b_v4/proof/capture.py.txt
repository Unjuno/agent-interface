import datetime, hashlib, json, pathlib, subprocess, sys
R = pathlib.Path(__file__).resolve().parent

def utc(): return datetime.datetime.now(datetime.timezone.utc).isoformat()
def pin(p):
 b = p.read_bytes(); return {'bytes': len(b), 'sha256': hashlib.sha256(b).hexdigest()}
argv = [sys.executable, '-X', 'utf8', str(R / 'regression.py'), str(source), str(dest / 'RAW.json')]
before = {'utc': utc(), 'argv': argv, 'source': pin(source), 'runner': pin(R / 'regression.py')}; (dest / 'BEFORE.json').write_text(json.dumps(before, indent=2), encoding='utf-8')

 p.stdout.close(); p.stderr.close()
(dest / 'stdout.bin').write_bytes(out); (dest / 'stderr.bin').write_bytes(err)

(dest / 'RECEIPT.json').write_text(json.dumps(receipt, indent=2), encoding='utf-8'); print(json.dumps(receipt)); print(err.decode('utf-8')); sys.exit(p.returncode)
