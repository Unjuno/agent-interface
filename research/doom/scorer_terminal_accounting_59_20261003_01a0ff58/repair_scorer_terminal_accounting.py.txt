def sha(data):
    return hashlib.sha256(data).hexdigest()
def dump(path, data):
    path.write_text(json.dumps(data, indent=2) + '\n', encoding='utf-8', newline='\n')

def run(label, argv, expected_exit):
    target = OUT / (label + '.receipt.json')
    if target.exists():
        raise RuntimeError('Retained receipt already exists: ' + label)
    before = sources()
    start = datetime.now(timezone.utc).isoformat()
    result = subprocess.run(argv, cwd=DOOM, capture_output=True, timeout=45)
    end = datetime.now(timezone.utc).isoformat()
    after = sources()
    if before != after:
        raise RuntimeError('Source changed during ordinary check')
    (OUT / (label + '.stdout.log')).write_bytes(result.stdout)
    (OUT / (label + '.stderr.log')).write_bytes(result.stderr)

              'label': label, 'argv': argv, 'cwd': str(DOOM), 'started_utc': start, 'ended_utc': end,

              'source_sha256': before, 'source_unchanged_after': True,
              'stdout_sha256': sha(result.stdout), 'stderr_sha256': sha(result.stderr),
              'stdout_bytes': len(result.stdout), 'stderr_bytes': len(result.stderr)}
    dump(target, record)
    print(json.dumps(record))
    if result.returncode != expected_exit:
        raise RuntimeError('Unexpected ordinary-check exit; inspect retained logs')
