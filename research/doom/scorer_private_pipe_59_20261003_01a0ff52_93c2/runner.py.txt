"""Single-use formal producer/auditor and bounded engineering copied-raw controls."""
import datetime
import hashlib
import json
from pathlib import Path
import resource
import subprocess
import sys
import time

ROOT=Path(__file__).resolve().parent

def sha(data):
    return hashlib.sha256(data).hexdigest()

def main():
    out=Path(sys.argv[1]);out.mkdir(exist_ok=True)
    protocol=json.loads((ROOT/'PROTOCOL.json').read_text())
    freeze=json.loads((ROOT/'FREEZE.json').read_text())
    for item in freeze['inputs']:
        if sha((ROOT/item['path']).read_bytes())!=item['sha256']:
            raise ValueError('pre-run frozen input mismatch')
    with (out/'RUN_INTENT.json').open('x') as f:
        f.write(json.dumps({'allocation_id':protocol['allocation_id'],'utc':datetime.datetime.now(datetime.UTC).isoformat(),'freeze_sha256':sha((ROOT/'FREEZE.json').read_bytes()),'producer_limit':1,'auditor_limit':1,'retry_limit':0},indent=2)+'\n')
    dispositions=[]
    run_start=time.perf_counter_ns()
    for label,argv in [('candidate',[sys.executable,'-B',str(ROOT/'candidate.py.txt'),'formal',str(out/'raw.jsonl')]),
                       ('audit',[sys.executable,'-B',str(ROOT/'audit.py.txt'),str(out/'raw.jsonl'),str(out/'audit.json')]),
                       ('controls',[sys.executable,'-B',str(ROOT/'controls.py.txt'),str(out/'raw.jsonl'),str(out/'controls.json')])]:
        start=datetime.datetime.now(datetime.UTC).isoformat()
        cpu_before=resource.getrusage(resource.RUSAGE_CHILDREN)
        try:
            run=subprocess.run(argv,capture_output=True,timeout=protocol['outer_timeout_s'])
            stdout,stderr,code=run.stdout,run.stderr,run.returncode
            disposition='COMPLETED' if code==0 else 'FAIL'
        except subprocess.TimeoutExpired as exc:
            stdout,stderr,code=exc.stdout or b'',exc.stderr or b'',None
            disposition='STOP_TIMEOUT'
        cpu_after=resource.getrusage(resource.RUSAGE_CHILDREN)
        for name,data in [('stdout',stdout),('stderr',stderr)]:(out/f'{label}.{name}.log').write_bytes(data)
        receipt={'label':label,'argv':argv,'start':start,'end':datetime.datetime.now(datetime.UTC).isoformat(),'exit_code':code,'disposition':disposition,
                 'stdout_sha256':sha(stdout),'stderr_sha256':sha(stderr),'child_user_cpu_s':cpu_after.ru_utime-cpu_before.ru_utime,
                 'child_system_cpu_s':cpu_after.ru_stime-cpu_before.ru_stime}
        with (out/f'{label}.receipt.json').open('x') as f:f.write(json.dumps(receipt,indent=2)+'\n')
        dispositions.append({'label':label,'disposition':disposition,'exit_code':code})
        if disposition!='COMPLETED':
            break
    unchanged=all(sha((ROOT/item['path']).read_bytes())==item['sha256'] for item in freeze['inputs'])
    status='PASS_SCOPED' if len(dispositions)==3 and all(x['exit_code']==0 for x in dispositions) and unchanged else 'STOP' if any(x['disposition'].startswith('STOP') for x in dispositions) else 'FAIL'
    result={'allocation_id':protocol['allocation_id'],'status':status,'runs':dispositions,'source_unchanged_after':unchanged,'elapsed_s':(time.perf_counter_ns()-run_start)/1e9,
            'formal_producer_count':sum(x['label']=='candidate' for x in dispositions),'formal_auditor_count':sum(x['label']=='audit' for x in dispositions),
            'formal_retries':0,'copied_controls_are_not_trial_replay':True,'scope':protocol['U']}
    with (out/'EXECUTION.json').open('x') as f:f.write(json.dumps(result,indent=2)+'\n')
    print(json.dumps(result))
    raise SystemExit(0 if status=='PASS_SCOPED' else 1)

if __name__=='__main__':main()
