"""Posthoc same-raw deadline interpretation; no candidate/backend/trial execution."""
import hashlib
import json
from pathlib import Path
import sys

raw=Path(sys.argv[1]).read_bytes()
rows=[json.loads(line) for line in raw.decode().splitlines()]
results=[]
for row in rows:
    if row['variant']!='v2' or row['condition']=='fast':
        continue
    receipt=row['receipts'][0]
    command=next(event for event in row['events'] if event['kind']=='command_enter')['ns']
    results.append({'case_id':row['case_id'],'receipt_scheduled_ns':receipt['scheduled_ns'],'command_ns':command,
                    'period_ns':row['period_ns'],'complete_period_boundaries_after_scheduled_by_command':(command-receipt['scheduled_ns'])//row['period_ns'],
                    'source_reported_missed_sample_periods':row['stats']['missed_sample_periods'],'sample_count':row['stats']['samples']})
print(json.dumps({'scope':'Posthoc counter interpretation, not prospective formal service gate or new trial',
                  'raw_sha256':hashlib.sha256(raw).hexdigest(),'results':results},indent=2))
