"""Independent raw reducer. No candidate or retained scheduler import/execution."""
import hashlib
import json
from pathlib import Path
import sys

ROOT = Path(__file__).resolve().parent
P = 20_000_000
MESSAGE = b'{"op":"finish"}\n'
LINE = MESSAGE[:-1].decode()
LIMIT = 4

def require(ok, message):
    if not ok:
        raise ValueError(message)

def equal(actual, expected):
    if type(actual) is not type(expected):
        return False
    if isinstance(expected, dict):
        return actual.keys() == expected.keys() and all(equal(actual[k], v) for k, v in expected.items())
    if isinstance(expected, list):
        return len(actual) == len(expected) and all(equal(a, b) for a, b in zip(actual, expected))
    return actual == expected

class Trace:
    def __init__(self, events):
        require(type(events) is list and bool(events), 'events list')
        self.events = events
        self.index = 0
        previous = -1
        for event in events:
            require(type(event) is dict and type(event.get('ns')) is int and event['ns'] >= previous, 'monotone exact integer event time')
            require(event.get('owner') is True, 'owner')
            previous = event['ns']
    def take(self, kind, **fields):
        require(self.index < len(self.events), 'trace ended early')
        event = self.events[self.index]
        self.index += 1
        expected = {'kind': kind, 'owner': True, 'ns': event['ns'], **fields}
        require(equal(event, expected), f'event {self.index}: expected {kind} exact fields/types')
        return event['ns']
    def clock(self):
        return self.take('clock')
    def wait_and_read(self, timeout):
        self.take('wait_enter', timeout=timeout)
        self.take('wait_exit', ready=True)
        self.take('read_enter', size=65536)
        self.take('read_exit', hex=MESSAGE.hex())
    def command(self):
        ns = self.take('command_enter', line=LINE)
        self.take('command_exit')
        return ns

def audit_row(row, case, allocation):
    expected_keys = set(case).intersection({'index','case_id','repetition','condition','variant','adapter'})
    require(expected_keys == {'index','case_id','repetition','condition','variant','adapter'}, 'case fields')
    allowed = expected_keys | {'allocation_id','period_ns','load_ns','sample_limit','events','commands','receipts','stats','outcome','unread_hex'}
    require(type(row) is dict and row.keys() == allowed, 'row fields')
    for key in expected_keys:
        require(equal(row[key], case[key]), f'case field {key}')
    require(equal([row['allocation_id'], row['period_ns'], row['load_ns'], row['sample_limit']], [allocation,P,40_000_000,LIMIT]), 'identity/constants')
    t = Trace(row['events'])
    t.take('write_enter', hex=MESSAGE.hex())
    written_ns = t.take('write_exit', bytes=len(MESSAGE))
    t.take('precheck_enter')
    ready_ns = t.take('precheck_exit', ready=True)
    run_ns = t.take('run_enter')
    receipts = []
    samples = 0
    missed = 0
    next_ns = None
    command_ns = None
    load_durations = []

    def sample(scheduled, skipped):
        nonlocal samples
        started = t.clock()
        entered = t.take('sample_enter', ordinal=samples+1)
        if samples == LIMIT:
            t.take('sample_budget', completed=LIMIT)
            return False
        payload = {'private_scorer':17, 'ordinal':samples+1}
        exited = t.take('sample_exit', payload=payload)
        if row['condition'] == 'sample_sleep':
            require(exited - entered >= 40_000_000, 'actual sample load at least requested 40ms')
            load_durations.append(exited - entered)
        finished = t.clock()
        receipt = {'scheduled_ns':scheduled,'sample_started_ns':started,'sample_finished_ns':finished,
                   'start_lateness_ns':max(0,started-scheduled),'missed_periods_before':skipped,'payload':payload}
        sink_entered = t.take('sink_enter', receipt=receipt)
        sink_exited = t.take('sink_exit')
        if row['condition'] == 'sink_sleep':
            require(sink_exited - sink_entered >= 40_000_000, 'actual sink load at least requested 40ms')
            load_durations.append(sink_exited - sink_entered)
        receipts.append(receipt)
        samples += 1
        return True

    def due(now):
        nonlocal next_ns, missed
        if next_ns is None:
            next_ns = now
        if now < next_ns:
            return None
        elapsed = (now-next_ns)//P + 1
        skipped = elapsed - 1
        scheduled = next_ns
        next_ns += elapsed*P
        missed += skipped
        return sample(scheduled, skipped)

    if row['variant'] == 'read_first':
        require(row['adapter']=='reference', 'reference adapter')
        t.wait_and_read(0)
        command_ns = t.command()
        expected_stats = {'samples':0,'commands':1,'owner':True}
        outcome = 'SERVED'
    elif row['adapter'] == 'polling':
        started = t.clock()
        next_ns = started
        while True:
            now = t.clock()
            sampled = due(now)
            if sampled is True:
                if samples == LIMIT:
                    break
                if row['variant'] == 'retained':
                    continue
            if row['variant'] == 'v2':
                now = t.clock()
                timeout = max(0, (next_ns-now)/1_000_000_000)
            else:
                timeout = (next_ns-now)/1_000_000_000
            t.wait_and_read(timeout)
            command_ns = t.command()
            break
        ended = t.clock()
        outcome = 'SERVED' if command_ns is not None else 'CENSORED_CAP'
        expected_stats = {'samples':samples,'commands':int(command_ns is not None),'missed_sample_periods':missed,
                          'eof':False,'stopped_by_command':command_ns is not None,'started_ns':started,'ended_ns':ended,'owner':True}
    else:
        require(row['adapter']=='stdin', 'stdin adapter')
        buffered = False
        while True:
            if row['variant']=='v2' and buffered:
                command_ns = t.command()
                break
            now = t.clock()
            sampled = due(now)
            if sampled is False:
                break
            if row['variant']=='retained' and sampled is True:
                continue
            if buffered:
                command_ns = t.command()
                break
            now = t.clock()
            t.wait_and_read(max(0,(next_ns-now)/1e9))
            buffered = True
        outcome = 'SERVED' if command_ns is not None else 'CENSORED_DIAGNOSTIC'
        expected_stats = {'samples':samples,'commands':int(command_ns is not None),'missed_sample_periods':missed,
                          'eof':False,'sample_hz':50.0,'owner':True}
    t.take('run_exit', outcome=outcome)
    t.take('postcheck_enter')
    left = command_ns is None
    t.take('postcheck_exit', ready=left)
    if left:
        t.take('read_enter', size=65536)
        t.take('read_exit', hex=MESSAGE.hex())
    t.take('private_pipe_closed')
    require(t.index == len(t.events), 'trailing trace')
    require(equal(row['stats'], expected_stats), 'source stats reconstructed')
    require(equal(row['receipts'], receipts), 'complete source receipts reconstructed')
    require(equal(row['commands'], [] if left else [LINE]), 'command payload isolation and completion')
    require(equal(row['unread_hex'], MESSAGE.hex() if left else '') and row['outcome']==outcome, 'unread/termination')
    overloaded = row['condition']!='fast'
    expected_served = row['variant']!='retained' or not overloaded
    require((command_ns is not None) == expected_served, 'prospective command service hypothesis')
    expected_samples = 0 if row['variant']=='read_first' else 1 if expected_served else LIMIT
    require(samples == expected_samples, 'prospective completed sample count')
    return {'case_id':row['case_id'],'variant':row['variant'],'adapter':row['adapter'],'condition':row['condition'],
            'outcome':outcome,'samples':samples,'readiness_to_command_ns':None if left else command_ns-ready_ns,
            'run_to_command_ns':None if left else command_ns-run_ns,'write_to_command_ns':None if left else command_ns-written_ns,
            'measured_load_ns':load_durations,'censored':left}

def pairs(items):
    result = {}
    for k,v in items:
        require(k not in result, 'duplicate JSON key')
        result[k]=v
    return result

def strict_json(text):
    return json.loads(text, object_pairs_hook=pairs, parse_constant=lambda x: (_ for _ in ()).throw(ValueError('nonfinite JSON')))

def main():
    raw_path, output = map(Path, sys.argv[1:])
    protocol = strict_json((ROOT/'PROTOCOL.json').read_text())
    freeze = strict_json((ROOT/'FREEZE.json').read_text())
    for item in freeze['inputs']:
        require(hashlib.sha256((ROOT/item['path']).read_bytes()).hexdigest()==item['sha256'], 'frozen input')
    raw = raw_path.read_bytes()
    rows = [strict_json(line) for line in raw.decode().splitlines()]
    require(len(rows)==len(protocol['cases'])==60, 'exact raw coverage')
    results = [audit_row(row, case, protocol['allocation_id']) for row,case in zip(rows,protocol['cases'])]
    result = {'disposition':'PASS_SCOPED','allocation_id':protocol['allocation_id'],'rows':len(rows),'raw_sha256':hashlib.sha256(raw).hexdigest(),
              'reconstruction_imports_candidate':False,'results':results,'scope':protocol['U']}
    with output.open('x') as f:
        f.write(json.dumps(result,indent=2)+'\n')
    print(json.dumps({k:v for k,v in result.items() if k!='results'}))

if __name__=='__main__':
    main()
