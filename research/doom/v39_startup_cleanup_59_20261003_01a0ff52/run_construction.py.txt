"""Finite supervisor of the three ordinary caller cases."""
from datetime import datetime,timezone
import argparse
import hashlib
import json
import os
from pathlib import Path
import signal
import subprocess
import sys

root=Path(__file__).resolve().parent
parser=argparse.ArgumentParser()
parser.add_argument('--run-id',default='construction01')
parser.add_argument('--case',action='append')
parser.add_argument('--session-child',default='session_child.py')
args=parser.parse_args()
names=args.case or ['normal','missing_fixture','outer_cleanup']
run=root/args.run_id;run.mkdir()
receipts=[]
for name in names:
    case=run/name
    argv=[sys.executable,'-B',str(root/'probe_case.py'),name,'--output',str(case),
          '--session-child',args.session_child]
    r={'case':name,'argv':argv,'started_utc':datetime.now(timezone.utc).isoformat(),
       'plan_sha256':hashlib.sha256((root/'PLAN.json').read_bytes()).hexdigest(),
       'runner_sha256':hashlib.sha256((root/'probe_case.py').read_bytes()).hexdigest(),
       'child_sha256':hashlib.sha256((root/args.session_child).read_bytes()).hexdigest()}
    (run/(name+'.receipt.json')).write_text(json.dumps(r,indent=2)+'\n')
    p=subprocess.Popen(argv,cwd=root,stdout=subprocess.PIPE,stderr=subprocess.PIPE,
                       start_new_session=True)
    try:stdout,stderr=p.communicate(timeout=5);r['timeout']=False
    except subprocess.TimeoutExpired:
        os.killpg(p.pid,signal.SIGKILL);stdout,stderr=p.communicate(timeout=1);r['timeout']=True
    for suffix,data in [('stdout',stdout),('stderr',stderr)]:
        (run/(name+'.'+suffix)).write_bytes(data)
        r[suffix+'_sha256']=hashlib.sha256(data).hexdigest();r[suffix+'_bytes']=len(data)
    r.update(ended_utc=datetime.now(timezone.utc).isoformat(),exit_code=p.returncode,
             output_bound_ok=len(stdout)+len(stderr)<=65536)
    (run/(name+'.receipt.json')).write_text(json.dumps(r,indent=2)+'\n');receipts.append(r)
    if p.returncode or r['timeout'] or not r['output_bound_ok']:break
(run/'receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
print(json.dumps([{'case':r['case'],'exit':r['exit_code'],'timeout':r['timeout']} for r in receipts]))
raise SystemExit(int(len(receipts)!=len(names) or any(r['exit_code'] or r['timeout'] or not r['output_bound_ok'] for r in receipts)))
