"""Exclusive ordinary capture; never overwrite first outcomes."""
import argparse
import hashlib
import json
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

parser = argparse.ArgumentParser()
parser.add_argument('--phase', required=True)
parser.add_argument('--source-root', type=Path, required=True)
parser.add_argument('--cases', nargs='+', default=['normal', 'owner-file-refusal'])
args = parser.parse_args()
root = Path(__file__).resolve().parent
experiment = root / 'v12-owner-record-cleanup'
phase = experiment / args.phase
phase.mkdir(exist_ok=False)
helper = root / 'v12_cleanup_case.py'
source = args.source_root / 'research/doom/session_map01_v12.py'
def hash_file(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()
plan = {'kind': 'ordinary construction/regression, no formal allocation', 'utc': datetime.now(timezone.utc).isoformat(),
        'phase': args.phase, 'cases': args.cases, 'python': sys.version,
        'helper_sha256': hash_file(helper), 'session_source_sha256': hash_file(source),
        'predicates': {'normal': 'no main error; game/session/backend close1; child0 at main return; tmp removed',
                       'original_fault': 'OSError at owner file; backend close1; game/session close0; child live at main return; outer EOF child0',
                       'repaired_fault': 'same OSError preserved; backend/game/session close1; child0 at main return; tmp removed'},
        'limits': 'one private inert child at a time; child self cap12s; subprocess capture deadline18s; no GUI/input'}
(phase / 'BEFORE.json').write_text(json.dumps(plan, indent=2) + '\n', encoding='utf-8')
for i, case in enumerate(args.cases):
    case_dir = phase / f'{i:02d}-{case}'
    command = [sys.executable, '-I', '-u', str(helper), '--source-root', str(args.source_root),
               '--case', case, '--out', str(case_dir)]
    started = datetime.now(timezone.utc).isoformat()
    result = subprocess.run(command, capture_output=True, timeout=18)
    ended = datetime.now(timezone.utc).isoformat()
    prefix = phase / f'{i:02d}-{case}'
    stdout_path = prefix.with_suffix('.stdout.bin')
    stderr_path = prefix.with_suffix('.stderr.bin')
    stdout_path.write_bytes(result.stdout)
    stderr_path.write_bytes(result.stderr)
    receipt = {'command': command, 'started_utc': started, 'ended_utc': ended, 'exit_code': result.returncode,
               'stdout_sha256': hash_file(stdout_path), 'stdout_bytes': len(result.stdout),
               'stderr_sha256': hash_file(stderr_path), 'stderr_bytes': len(result.stderr)}
    prefix.with_suffix('.invocation.json').write_text(json.dumps(receipt, indent=2) + '\n', encoding='utf-8')
    print(json.dumps({'phase': args.phase, 'case': case, 'exit_code': result.returncode,
                      'stdout_bytes': len(result.stdout), 'stderr_bytes': len(result.stderr)}))
    if result.returncode:
        raise SystemExit(result.returncode)
    data = json.loads((case_dir / 'FINAL-RECEIPT.json').read_text(encoding='utf-8'))
    if case == 'normal' and (data['main_error'] is not None or data['session_close_at_main_return'] != 1
                            or data['game_close_at_main_return'] != 1 or data['owned_child_final_exit'] != 0):
        raise SystemExit('normal control failed; stop before subsequent case')
