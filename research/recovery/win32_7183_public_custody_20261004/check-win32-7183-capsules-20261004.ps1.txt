function Hash([byte[]]$Bytes) {
    $h = [Security.Cryptography.SHA256]::Create()
    try { ([BitConverter]::ToString($h.ComputeHash($Bytes))).Replace('-','').ToLowerInvariant() } finally { $h.Dispose() }
}
function Decode([string]$Text) {
    $compact = $Text -replace '\s',''
    $b = [Convert]::FromBase64String($compact)
    if ([Convert]::ToBase64String($b) -cne $compact) { throw 'Noncanonical base64' }
    return ,$b
}
