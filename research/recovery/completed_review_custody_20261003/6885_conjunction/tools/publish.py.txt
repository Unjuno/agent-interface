def add(source, relative):
    data = source.read_bytes()
    value = data.replace(str(workspace).encode(), b"<REVIEWER_WORKSPACE>")
    value = value.replace(str(workspace).replace("\\", "\\\\").encode(), b"<REVIEWER_WORKSPACE>")

    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(value)
    originals.append({"artifact": relative, "original_sha256": hashlib.sha256(data).hexdigest(),
                      "public_sha256": hashlib.sha256(value).hexdigest(), "path_redacted": data != value,
                      "original_retained": source.read_bytes() == data})

def git(*args, data=None):
    return subprocess.check_output(["git", *args], cwd=repo, env=env, input=data)


assert subprocess.run(["git", "show-ref", "--verify", "--quiet", ref], cwd=repo).returncode == 1
git("read-tree", head)
paths = []
for source in sorted(public.rglob("*")):
    if not source.is_file():
        continue
    path = prefix + "/" + source.relative_to(public).as_posix()
    data = source.read_bytes()
    oid = git("hash-object", "-w", "--stdin", data=data).decode().strip()
    assert git("cat-file", "blob", oid) == data
    git("update-index", "--add", "--cacheinfo", "100644", oid, path)
    paths.append(path)
assert git("diff", "--cached", "--name-only", head).decode().splitlines() == paths
git("diff", "--cached", "--check", "--", prefix + "/README.md")
tree = git("write-tree", "--missing-ok").decode().strip()
commit = git("commit-tree", tree, "-p", head,
             data=b"review(#6885): retain independent raw reconstruction [skip ci]\n").decode().strip()
git("update-ref", ref, commit, "0" * 40)
assert (repo / ".git/index").read_bytes() == primary
