import assert from 'node:assert/strict';
import {mkdir,readFile,writeFile,readdir} from 'node:fs/promises';
import {join,dirname} from 'node:path';
import {fileURLToPath,pathToFileURL} from 'node:url';
import {createHash} from 'node:crypto';
const root = dirname(fileURLToPath(import.meta.url));
const [sourceLabel, outputName] = process.argv.slice(2);
if (!['before','fixed'].includes(sourceLabel)) throw new Error('explicit source label');
const {createRelayClient} = await import(pathToFileURL(join(root,sourceLabel,'relay_client.mjs')));
const out = join(root,outputName);
await mkdir(out);
const digest = data => createHash('sha256').update(data).digest('hex');
function deadline(promise,ms) {
  let timer;
  const expiry = new Promise((_,reject) => {timer=setTimeout(()=>reject(new Error('owned review guard')),ms);timer.unref();});
  return Promise.race([promise,expiry]).finally(()=>clearTimeout(timer));
}
async function waitBlocked(client) {
  const began=Date.now();
  while(!client.state().blocked) {
    if(Date.now()-began>2500) throw new Error('blocked-state review guard');
    await new Promise(r=>setTimeout(r,2));
  }
}
const results=[];
for(const mode of ['line','fragment']) {
  const startedUTC=new Date().toISOString();
  const caseDir=join(out,mode);
  await mkdir(caseDir);
  const evidenceDirectory=join(caseDir,'evidence'),childRecord=join(caseDir,'child-exit.json');
  const client=await createRelayClient({command:process.execPath,
    args:[join(root,'child_fixture.cjs'),mode,childRecord],evidenceDirectory});
  let outcome,setupError=null,first,waitSame=false;
  try {
    const accepted=client.send('owned-review-only',{literal:'入力の同一性'});
    waitSame=client.wait()===accepted;
    first=await deadline(accepted,3500);
    if(mode==='line') await waitBlocked(client);
    outcome=await deadline(client.close(),3500);
    const repeated=await deadline(client.close(),1000);
    let sendRejected=false;
    try {client.send('must-not-send',{});} catch {sendRejected=true;}
    const state=client.state();
    const files={};
    for(const name of (await readdir(evidenceDirectory)).sort()) {
      const bytes=await readFile(join(evidenceDirectory,name));
      files[name]={bytes:bytes.length,sha256:digest(bytes),utf8:bytes.toString('utf8')};
    }
    const child=JSON.parse(await readFile(childRecord,'utf8'));
    let pidAbsent=false,pidProbe=null;
    try {process.kill(child.pid,0);pidProbe='still-present';}
    catch(error) {pidProbe=error.code;pidAbsent=error.code==='ESRCH';}
    let terminal=null;
    if(files['exit.json'])terminal=JSON.parse(files['exit.json'].utf8);
    const checks={
      firstReplyExact:first?.id===1&&first?.tool==='owned-review-only'&&first?.status==='returned'
        &&first?.result?.content?.[0]?.text==='保全された正常返答',
      sameAcceptedPromise:waitSame&&client.wait()===accepted,
      oneAttemptOneRequest:state.attempts===1&&child.requests===1
        &&Object.keys(files).filter(n=>n.startsWith('request-')).length===1,
      noSecondReply:Object.keys(files).filter(n=>n.startsWith('reply-')).length===1,
      requestExact:JSON.parse(files['request-1.json'].utf8).arguments.literal==='入力の同一性',
      originalReplyUnchanged:JSON.stringify(JSON.parse(files['reply-1.json'].utf8))===JSON.stringify(first),
      firstFaultRetained:typeof outcome.journal_error==='string'
        &&outcome.journal_error.includes('unsolicited relay response'),
      blockedNoReplay:sendRejected&&state.pending===false&&state.blocked.includes('unsolicited relay response'),
      actualNonzeroExit:outcome.code===(mode==='line'?29:31)&&outcome.signal===null
        &&child.code===outcome.code&&child.reason==='stdin-eof',
      independentTerminalReceipt:terminal!==null&&JSON.stringify(terminal)===JSON.stringify(outcome),
      noSecondTerminalError:!Object.hasOwn(outcome,'exit_journal_error'),
      repeatCloseSameOutcome:JSON.stringify(repeated)===JSON.stringify(outcome),
      ownedChildGone:pidAbsent,
    };
    const row={sourceLabel,mode,startedUTC,endedUTC:new Date().toISOString(),firstReply:first,
      state,close:outcome,repeatedClose:repeated,child,pidProbe,checks,files,
      status:Object.values(checks).every(Boolean)?'PASS':'FAIL'};
    await writeFile(join(caseDir,'RESULT.json'),JSON.stringify(row,null,2)+'\n',{flag:'wx'});
    results.push(row);
  } catch(error) {
    setupError=String(error);
    try {
      if(client.state().pending)await deadline(client.wait().catch(()=>{}),4000);
      outcome=await deadline(client.close(),4000);
    } catch(cleanupError) {setupError+='; cleanup '+String(cleanupError);}
    const row={sourceLabel,mode,startedUTC,endedUTC:new Date().toISOString(),setupError,
      outcome,status:'ERROR',state:client.state()};
    await writeFile(join(caseDir,'RESULT.json'),JSON.stringify(row,null,2)+'\n',{flag:'wx'});
    results.push(row);
  }
}
console.log(JSON.stringify({sourceLabel,results:results.map(r=>({mode:r.mode,status:r.status,
  failedChecks:Object.entries(r.checks||{}).filter(([,v])=>!v).map(([k])=>k),
  setupError:r.setupError,childCode:r.child?.code,childRequests:r.child?.requests,
  pidProbe:r.pidProbe}))}));
process.exitCode=results.every(r=>r.status==='PASS')?0:1;
