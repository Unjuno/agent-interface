"""Capture actual command byte streams before deriving public path redactions."""
from pathlib import Path
import hashlib
import json
import os
import subprocess
import sys
from datetime import datetime, timezone

root = Path(__file__).resolve().parent
package = root / 'thread-exit-repo/research/concurrency/singleflight_thread_exit_6501_01a0ff35'
private = root / 'thread-exit-private'
private.mkdir(exist_ok=True)
label = sys.argv[1]
command = sys.argv[2:]
start = datetime.now(timezone.utc).isoformat()
process = subprocess.run(command, cwd=package, capture_output=True, timeout=60,
    env=dict(os.environ, PYTHONDONTWRITEBYTECODE='1'))
end = datetime.now(timezone.utc).isoformat()
record = dict(label=label, argv=command, start_utc=start, end_utc=end, exit_code=process.returncode)
public = package / 'evidence'
public.mkdir(exist_ok=True)
for name, data in [('stdout', process.stdout), ('stderr', process.stderr)]:
    target = private / f'{label}.{name}'
    with target.open('xb') as f:
        f.write(data)
    clean = data.replace(str(root.parent).encode(), b'<PRIVATE_WORKSPACE>')
    clean = clean.replace(str(Path.home()).encode(), b'<PRIVATE_HOME>')
    with (public / f'{label}.{name}.txt').open('xb') as f:
        f.write(clean)
    record[name] = {'original_sha256': hashlib.sha256(data).hexdigest(),
        'public_sha256': hashlib.sha256(clean).hexdigest(), 'redaction': 'local path prefixes only'}
with (public / f'{label}.receipt.json').open('x', encoding='utf-8', newline='\n') as f:
    json.dump(record, f, indent=2)
    f.write('\n')
print(json.dumps(record))
print(process.stdout.decode(errors='replace'))
print(process.stderr.decode(errors='replace').replace(str(root.parent), '<PRIVATE_WORKSPACE>').replace(str(Path.home()), '<PRIVATE_HOME>'))
sys.exit(process.returncode)
