def git(*args, **kw):
    return subprocess.check_output([G, '-C', str(repo), *args], env=env, **kw)


def sha(b):
    return hashlib.sha256(b).hexdigest()


def lossless(name, b):
    if b'\0' in b or any(line.rstrip(b'\r\n').endswith((b' ', b'\t')) for line in b.splitlines(keepends=True)):
        return name + '.base64.json', json.dumps({'encoding': 'base64', 'original_path': name,
                                                 'original_bytes': len(b), 'original_sha256': sha(b),
                                                 'data': base64.b64encode(b).decode()}, sort_keys=True).encode() + b'\n'
    try:
        b.decode('utf8')
    except UnicodeDecodeError:
        return name + '.base64.json', json.dumps({'encoding': 'base64', 'original_path': name,
                                                 'original_bytes': len(b), 'original_sha256': sha(b),
                                                 'data': base64.b64encode(b).decode()}, sort_keys=True).encode() + b'\n'
    return name, b
