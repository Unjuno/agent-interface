import io,json,threading,unittest
from collections import deque
from unittest.mock import patch
import codex_app_server_client_v2 as module
class Fresh(unittest.TestCase):
    def test_reader_appended_match_is_rechecked_before_eof(self):
        client=module.CodexAppServerClient.__new__(module.CodexAppServerClient)
        old={'method':'turn/completed','params':{'intent':'保持した意図','row':'old'}}
        fresh={'method':'turn/completed','params':{'intent':'保持した意図','row':'fresh'}}
        client._condition=threading.Condition();client._notifications=deque([old]);client._responses={}
        client._notification_consumptions=0;client._closed=False;client._record=lambda _:None
        client.process=type('Inert',(),{'stdout':io.StringIO(json.dumps(fresh,ensure_ascii=False)+'\n')})()
        nested=[];first=True
        def predicate(row):
            nonlocal first
            if first:
                first=False;nested.append(client.wait_notification(lambda x:x is old,timeout=0))
                client._read()  # actual reader method, inert one-line input then EOF
            return True
        with patch.object(module.time,'monotonic',return_value=0):
            result=client.wait_notification(predicate,timeout=1)
        self.assertEqual(result,fresh);self.assertEqual(nested,[old])
        self.assertEqual(list(client._notifications),[]);self.assertEqual(client._notification_consumptions,2)
        self.assertTrue(client._closed)
