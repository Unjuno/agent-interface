"""Hand-authored saved oracle only; never launches task or X server."""
import copy,unittest
from auditor import verify
def fixture():
    specs=[dict(id=a+'-'+('paused' if p else 'healthy'),arm=a,paused=p) for a in ('keepalive','terminate_before_accept') for p in (False,True)]
    plan=dict(allocation='hand-oracle',source_hashes={},checkpoint_ns=150,resume_ns=400,program=dict(ops=['hand-ops']))
    plan['cells']=specs;raw=dict(schema='requested-process-fence/raw-v1',allocation='hand-oracle',source_hashes={},cells=[])
    for spec in specs:
        killed=spec['paused'] and spec['arm']=='terminate_before_accept'
        def q(t,keys=[]):return dict(start_ns=t,end_ns=t+1,keys=list(keys),buttons=0)
        identity=dict(cell=spec['id']);tr=[dict(kind='ready',ns=10,pid=123,identity=identity),dict(kind='dispatch_start',ns=20,pid=123,program=dict(ops=['hand-ops'],authority=dict(expires_at_ns=10000)))]
        for key,t in [('F8',30),('F9',60)]:
            tr.append(dict(kind='positive_check',key=key,permitted=True,ns=t,pid=123))
            if key=='F9' and spec['paused']:tr.append(dict(kind='pause_entered',key=key,ns=61,pid=123))
            if key=='F9' and killed:break
            start=510 if key=='F9' and spec['paused'] else t+1
            tr.extend(dict(kind=k,key=key,ns=start+n,pid=123) for n,k in enumerate(['native_start','native_end']))
        if not killed:tr.extend([dict(kind='dispatch_return',receipt=dict(status='completed',admission='accepted'),ns=600,pid=123),dict(kind='owner_done',ns=610,pid=123)])
        events=[('press',74),('release',74)] if killed else [('press',74),('release',74),('press',75),('release',75)] if spec['paused'] else [('press',74),('press',75),('release',74),('release',75)]
        row=dict(spec,errors=[],xvfb_exit=0,initial=q(1),final_query=q(720),before_emergency=q(710),emergency=[],keycodes=dict(F8=74,F9=75),owner_rows=tr,owner_pid=123,prearmed=dict(identity=identity,at_ns=5),start_sent_ns=15,owner_exit=-9 if killed else 0,completed_ns=730,pipe_eof_ns=104 if killed else 650,cleanup=[])
        row['events']=[dict(kind=k,keycode=c,received_ns=660) for k,c in events]
        if spec['paused']:
            row.update(exposure=q(70,[74]),request_ns=100,accepted_ns=105,release_query=q(110),checkpoint=dict(query=q(251),owner_alive=not killed),cleanup=[dict(key='F8',kind='release',start_ns=106,end_ns=109)])
            if killed:row.update(kill_sent_ns=101,reaped_ns=103)
            else:row.update(resume_sent_ns=501)
        raw['cells'].append(row)
    return plan,raw
class Audit(unittest.TestCase):
    def test_hand_oracle(self):
        p,r=fixture();self.assertEqual(verify(p,r)['errors'],[])
    def test_independent_corruptions(self):
        p,r=fixture()
        mutations=[lambda d:d['cells'].pop(),lambda d:d['cells'][3].update(owner_exit=0),lambda d:d['cells'][3].update(accepted_ns=102),lambda d:d['cells'][3]['checkpoint']['query'].update(keys=[74]),lambda d:d['cells'][1]['checkpoint'].update(owner_alive=False),lambda d:d['cells'][1]['owner_rows'][5].update(permitted=False),lambda d:d['cells'][0].update(emergency=[{}]),lambda d:d.update(source_hashes={'bad':'hash'}),lambda d:d['cells'][3]['events'].append(dict(kind='press',keycode=75,received_ns=660)),lambda d:d['cells'][1].update(resume_sent_ns=1000),lambda d:d['cells'][2]['owner_rows'][0].update(pid=999)]
        for i,change in enumerate(mutations):
            with self.subTest(mutation=i):
                d=copy.deepcopy(r);change(d);self.assertTrue(verify(p,d)['errors'])
    def test_missing_evidence(self):
        p,r=fixture();del r['cells'][0]['owner_rows'];self.assertTrue(verify(p,r)['errors'])
if __name__=='__main__':unittest.main()
