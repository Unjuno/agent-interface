"""Saved-only, separately implemented evidence reconstruction. Same author."""
import hashlib,json,pathlib,sys
def verify(plan,raw):
    errors=[];metrics=[]
    def require(v,m):
        if not v:errors.append(m)
    try:
        require(raw['schema']=='requested-process-fence/raw-v1','schema')
        require(raw['allocation']==plan['allocation'],'allocation')
        require(raw['source_hashes']==plan['source_hashes'],'source identity')
        require(len(raw['cells'])==4,'cell count')
        for spec,r in zip(plan['cells'],raw['cells']):
            cid=spec['id'];ok=lambda v,m:require(v,cid+': '+m)
            for k,v in spec.items():ok(r[k]==v,'cell identity '+k)
            ok(not r['errors'],'producer errors')
            ok(r['xvfb_exit']==0,'Xvfb exit')
            ok(r['initial']['keys']==[] and r['initial']['buttons']==0,'initial neutral')
            ok(r['final_query']['keys']==[] and r['final_query']['buttons']==0,'terminal neutral')
            ok(r['final_query']['end_ns']<=r['completed_ns'],'terminal chronology')
            ok(r['before_emergency']['keys']==[] and not r['emergency'],'no emergency-success substitution')
            codes=r['keycodes'];f8,f9=codes['F8'],codes['F9']
            ok(type(f8)is int and type(f9)is int and f8!=f9 and f8>0 and f9>0,'distinct keycodes')
            traces=r['owner_rows'];pid=r['owner_pid']
            ok(all(t['pid']==pid for t in traces),'same owner PID')
            ok(all(a['ns']<=b['ns'] for a,b in zip(traces,traces[1:])),'owner clock order')
            ok(traces[0]['kind']=='ready' and traces[0]['identity']==r['prearmed']['identity'],'ready identity')
            ok(r['prearmed']['at_ns']<=traces[0]['ns']<=r['start_sent_ns'],'prearmed before dispatch')
            dispatch=[t for t in traces if t['kind']=='dispatch_start']
            ok(len(dispatch)==1,'dispatch exactly once')
            program=dispatch[0]['program']
            ok(program['ops']==plan['program']['ops'],'public program')
            ok(program['authority']['expires_at_ns']>r['completed_ns'],'unexpired scientific trace')
            checks=[t for t in traces if t['kind']=='positive_check']
            ok([t['key'] for t in checks]==['F8','F9'] and all(t['permitted'] is True for t in checks),'two positive checks')
            native=[t for t in traces if t['kind'].startswith('native_')]
            killed=spec['paused'] and spec['arm']=='terminate_before_accept'
            expected=['F8'] if killed else ['F8','F9']
            ok([(t['kind'],t['key']) for t in native]==[(k,key) for key in expected for k in ('native_start','native_end')],'native prefix/pairs')
            for key in expected:
                check=next(t for t in checks if t['key']==key)
                start=next(t for t in native if t['key']==key and t['kind']=='native_start')
                ok(check['ns']<=start['ns'],'positive check before native')
            expected_events=[(k,codes[key]) for key in expected for k in ('press','release')]
            actual=[(e['kind'],e['keycode']) for e in r['events']]
            # Healthy release_all releases F8 and F9 after both presses.
            if not spec['paused']:expected_events=[('press',f8),('press',f9),('release',f8),('release',f9)]
            ok(actual==expected_events,'exact app events')
            ok(all(e['received_ns']<=r['completed_ns'] for e in r['events']),'observer retirement')
            returns=[t for t in traces if t['kind']=='dispatch_return']
            if killed:
                ok(r['owner_exit']==-9 and not returns,'killed owner has no public completion')
                ok(r['request_ns']<=r['kill_sent_ns']<=r['reaped_ns']<=r['pipe_eof_ns']<=r['accepted_ns'],'death+EOF before acceptance')
                ok(all(t['ns']<r['accepted_ns'] for t in native),'no native after acceptance')
                ok(r['checkpoint']['owner_alive'] is False,'retired at checkpoint')
                ok(traces[-1]['kind']=='pause_entered','killed at frozen pause')
            else:
                ok(r['owner_exit']==0 and len(returns)==1,'normal owner completion')
                ok(returns[0]['receipt']['status']=='completed' and returns[0]['receipt']['admission']=='accepted','genuine public receipt')
                ok(traces[-1]['kind']=='owner_done','normal close')
            ok(r['pipe_eof_ns']<=r['completed_ns'],'EOF observed')
            if spec['paused']:
                pause=[t for t in traces if t['kind']=='pause_entered']
                ok(len(pause)==1 and pause[0]['key']=='F9','F9 pause')
                ok(checks[-1]['ns']<=pause[0]['ns']<=r['exposure']['start_ns']<=r['exposure']['end_ns']<=r['request_ns'],'pause/exposure/request')
                ok(r['exposure']['keys']==[f8] and r['exposure']['buttons']==0,'held-F8 exposure')
                ok(len(r['cleanup'])==1 and r['cleanup'][0]['key']=='F8' and r['cleanup'][0]['kind']=='release','prearmed cleanup only')
                ok(r['accepted_ns']<=r['cleanup'][0]['start_ns']<=r['cleanup'][0]['end_ns']<=r['release_query']['start_ns'],'release chronology')
                ok(r['release_query']['keys']==[] and r['release_query']['buttons']==0,'independent release query')
                cp=r['checkpoint']['query']
                ok(cp['start_ns']>=r['request_ns']+plan['checkpoint_ns'] and cp['end_ns']<r['request_ns']+plan['resume_ns'],'fresh pre-resume checkpoint')
                ok(cp['keys']==[] and cp['buttons']==0,'checkpoint neutral')
                if not killed:
                    ok(r['checkpoint']['owner_alive'] is True,'live acknowledged owner at checkpoint')
                    f9start=next(t['ns'] for t in native if t['kind']=='native_start' and t['key']=='F9')
                    ok(r['request_ns']+plan['resume_ns']<=r['resume_sent_ns']<=f9start and r['accepted_ns']<f9start,'late positive native after accepted')
                metrics.append(dict(cell=cid,request_to_accepted_ms=(r['accepted_ns']-r['request_ns'])/1e6,request_to_release_readback_ms=(r['release_query']['end_ns']-r['request_ns'])/1e6,owner_exit=r['owner_exit']))
            else:ok(not r['cleanup'] and not any(t['kind']=='pause_entered' for t in traces),'healthy parity/no supervisor emission')
    except (KeyError,TypeError,ValueError,IndexError,StopIteration) as e:errors.append('missing/malformed evidence: '+repr(e))
    return dict(disposition='SCOPED_REQUESTED_PROCESS_FENCE_SUPPORTED' if not errors else 'STOP_OR_FAILED_GATES',errors=errors,metrics=metrics,scope='Private healthy Xvfb; intentional pre-native pause; same-author saved-only independent implementation; no product/hard-deadline/process-tree or hardware claim')
if __name__=='__main__':
    src=pathlib.Path(sys.argv[1]);data=pathlib.Path(sys.argv[2]);out=pathlib.Path(sys.argv[3])
    plan=json.loads((src/'PLAN.json').read_text());raw=json.loads((data/'raw.json').read_text());result=verify(plan,raw)
    for path,digest in plan['source_hashes'].items():
        if hashlib.sha256((src/path).read_bytes()).hexdigest()!=digest:result['errors'].append('saved source altered: '+path)
    for r in raw['cells']:
        b=(data/(r['id']+'.owner.stdout')).read_bytes()
        if hashlib.sha256(b).hexdigest()!=r['stdout_sha256'] or [json.loads(s) for s in b.splitlines()]!=r['owner_rows']:result['errors'].append('original pipe trace mismatch: '+r['id'])
    if result['errors']:result['disposition']='STOP_OR_FAILED_GATES'
    result['raw_sha256']=hashlib.sha256((data/'raw.json').read_bytes()).hexdigest()
    out.write_text(json.dumps(result,sort_keys=True,indent=2)+'\n');print(json.dumps(result));sys.exit(1 if result['errors'] else 0)
