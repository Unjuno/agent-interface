"""Separate retained-raw reducer. Never imports/executes producer or runtime."""
import errno
import hashlib
import json
import pathlib
import sys

ROOT = pathlib.Path(__file__).resolve().parent


def require(ok, message):
    if not ok:
        raise ValueError(message)


def equal(a, b):
    return json.dumps(a, sort_keys=True, separators=(",", ":")) == json.dumps(b, sort_keys=True, separators=(",", ":"))


def validate(raw_bytes):
    records = [json.loads(line) for line in raw_bytes.splitlines()]
    deck = json.loads((ROOT / "INPUT.json").read_text())
    freeze = json.loads((ROOT / "FREEZE.json").read_text())
    require(len(records) == 22, "complete twenty-row denominator")
    header, footer = records[0], records[-1]
    require(set(header) == {"kind", "allocation", "utc_ns", "producer_pid", "input_sha256", "selector_class"}, "header fields")
    require(type(header["producer_pid"]) is int and header["producer_pid"] > 0, "actual producer PID type")
    require(header["kind"] == "header" and header["allocation"] == freeze["allocation"], "allocation")
    require(header["input_sha256"] == hashlib.sha256((ROOT / "INPUT.json").read_bytes()).hexdigest(), "input identity")
    require(header["selector_class"] == "KqueueSelector", "actual selected backend")
    require(equal({k: v for k, v in footer.items() if k != "utc_ns"},
                  {"kind": "footer", "rows": 20, "all_rows_emitted": True}), "footer types/count")
    require(type(header["utc_ns"]) is int and type(footer["utc_ns"]) is int, "clock type")
    previous_utc = header["utc_ns"]
    previous_mono = 0
    summaries = []
    truth = {"both": "control", "data_only": "data", "control_only": "control", "empty": None}
    for rec, case in zip(records[1:-1], deck):
        require(set(rec) == {"kind", "case", "events", "failure"}, "trial fields")
        require(rec["kind"] == "trial" and rec["failure"] is None, "producer first exception")
        require(equal(rec["case"], case), "prospective row identity/order/types")
        events = rec["events"]
        payloads = []
        for n, event in enumerate(events):
            require(type(event["seq"]) is int and event["seq"] == n, "event sequence type/order")
            for k in ("utc_ns", "mono_ns"):
                require(type(event[k]) is int and event[k] > 0, "clock type")
            require(event["utc_ns"] >= previous_utc and event["mono_ns"] >= previous_mono, "observed clock/order")
            previous_utc, previous_mono = event["utc_ns"], event["mono_ns"]
            for key in ("fd", "count", "events", "errno"):
                if key in event:
                    require(type(event[key]) is int, "exact integer " + key)
            payloads.append({k: v for k, v in event.items() if k not in ("seq", "utc_ns", "mono_ns")})
        cursor = 0

        def take(kind, expected=None):
            nonlocal cursor
            require(cursor < len(payloads), "missing event " + kind)
            event = payloads[cursor]; cursor += 1
            require(event.get("kind") == kind, "causal stage " + kind)
            if expected is not None:
                require(equal(event, dict(kind=kind, **expected)), "event binding " + kind)
            return event

        names = ("data_r", "data_w", "control_r", "control_w", "selector")
        fds = {}
        for name in names:
            event = take("owned")
            require(type(event.get("fd")) is int and event["fd"] >= 0, "owned FD type")
            fds[name] = event["fd"]
            take_expected = {"resource": name, "fd": event["fd"]}
            take_expected.update(selector_class="KqueueSelector") if name == "selector" else take_expected.update(nonblocking=True)
            require(equal(event, dict(kind="owned", **take_expected)), "owned FD fields")
        require(len(set(fds.values())) == 5, "row-local FD uniqueness")
        for name in case["register_order"]:
            take("registered", {"resource": name, "fd": fds[name + "_r"], "events": 1})
        for name in case["write_order"]:
            byte = {"data": "44", "control": "43"}[name]
            take("write_requested", {"resource": name, "fd": fds[name + "_w"], "hex": byte})
            take("write_return", {"resource": name, "fd": fds[name + "_w"], "count": 1})
        take("poll_requested", {"timeout": 0})
        polled = take("poll_return")
        require(set(polled) == {"kind", "ready"} and type(polled["ready"]) is list, "poll fields")
        ready = []
        for key in polled["ready"]:
            require(type(key) is dict and set(key) == {"resource", "fd", "events"}, "native key fields")
            name = key["resource"]
            require(name in ("data", "control"), "registered key identity")
            require(type(key["fd"]) is int and type(key["events"]) is int, "native key exact types")
            require(key["fd"] == fds[name + "_r"] and key["events"] == 1, "native key FD/mask")
            ready.append(name)
        require(len(ready) == len(set(ready)) and set(ready) == set(case["write_order"]), "ready set equals prewritten bytes")
        choice_event = take("selected")
        require(set(choice_event) == {"kind", "resource"}, "decision fields")
        selected = choice_event["resource"]
        require(selected in ready if ready else selected is None, "decision uses returned readiness")
        if case["policy"] == "first_ready":
            require(selected == (ready[0] if ready else None), "documented first-ready baseline")
        if selected is not None:
            expected = {"purpose": "primary", "resource": selected, "fd": fds[selected + "_r"]}
            take("read_requested", expected)
            take("read_return", dict(expected, hex={"data": "44", "control": "43"}[selected]))
        take("primary_checkpoint", {})
        leftovers = set(case["write_order"]) - {selected}
        for name in ("data", "control"):
            expected = {"purpose": "cleanup", "resource": name, "fd": fds[name + "_r"]}
            take("read_requested", expected)
            if name in leftovers:
                take("read_return", dict(expected, hex={"data": "44", "control": "43"}[name]))
            else:
                take("read_return", dict(expected, errno=errno.EAGAIN))
        for name in case["register_order"]:
            take("unregistered", {"resource": name, "fd": fds[name + "_r"]})
        for name in ("selector", "data_r", "data_w", "control_r", "control_w"):
            take("closed", {"resource": name, "fd": fds[name]})
        for name in names:
            take("close_checked", {"resource": name, "fd": fds[name], "open": False, "errno": errno.EBADF})
        require(cursor == len(payloads), "unexpected trailing event")
        summaries.append(dict(trial_id=case["trial_id"], policy=case["policy"], condition=case["condition"],
                              register_order=case["register_order"], write_order=case["write_order"],
                              ready_order=ready, primary=selected, expected=truth[case["condition"]],
                              contract_violation=selected != truth[case["condition"]], all5FDs_closed=True))
    require(footer["utc_ns"] >= previous_utc, "footer chronology")
    candidate_bad = sum(x["contract_violation"] for x in summaries if x["policy"] == "control_priority")
    baseline_bad = sum(x["contract_violation"] for x in summaries if x["policy"] == "first_ready")
    decision = "H_FAIL" if candidate_bad else "H_PASS_CO_READY_SCOPED" if baseline_bad else "H_NOT_EXPOSED"
    return dict(method="METHOD_PASS_SCOPED", hypothesis=decision, rows=20,
                control_priority_violations=candidate_bad, first_ready_violations=baseline_bad,
                rows_detail=summaries, raw_bytes=len(raw_bytes), raw_sha256=hashlib.sha256(raw_bytes).hexdigest(),
                producer_imports_or_replays=0, scope="one serial authored native macOS decision point per condition")


if __name__ == "__main__":
    result = validate(pathlib.Path(sys.argv[1]).read_bytes())
    with pathlib.Path(sys.argv[2]).open("x") as out:
        json.dump(result, out, indent=2); out.write("\n")
    print(json.dumps({k: v for k, v in result.items() if k != "rows_detail"}))
