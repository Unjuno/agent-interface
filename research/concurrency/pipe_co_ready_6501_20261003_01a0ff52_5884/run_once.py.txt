"""Exclusive attempt record and one process per frozen allocation."""
import hashlib
import json
import os
import pathlib
import platform
import resource
import selectors
import subprocess
import sys
import time

ROOT = pathlib.Path(__file__).resolve().parent
GIT = "/run/current-system/sw/bin/git"


def save(path, obj, exclusive=False):
    with path.open("x" if exclusive else "w") as f:
        json.dump(obj, f, indent=2); f.write("\n"); f.flush(); os.fsync(f.fileno())


def main():
    source_commit = sys.argv[1]
    freeze = json.loads((ROOT / "FREEZE.json").read_text())
    for name, descriptor in freeze["source_pins"].items():
        content = (ROOT / name).read_bytes()
        if hashlib.sha256(content).hexdigest() != descriptor["sha256"]:
            raise RuntimeError("source mismatch: " + name)
    for name, pin in freeze["environment"]["selected_runtime_pins"].items():
        if hashlib.sha256(pathlib.Path(pin["path"]).read_bytes()).hexdigest() != pin["sha256"]:
            raise RuntimeError("runtime mismatch: " + name)
    if platform.platform() != freeze["environment"]["platform"] or sys.version != freeze["environment"]["python"]:
        raise RuntimeError("environment mismatch")
    if selectors.DefaultSelector.__name__ != "KqueueSelector":
        raise RuntimeError("expected native macOS selector")
    repo = subprocess.check_output([GIT, "rev-parse", "--show-toplevel"], cwd=ROOT, text=True).strip()
    relative = ROOT.relative_to(repo).as_posix()
    for name in list(freeze["source_pins"]) + ["FREEZE.json"]:
        committed = subprocess.check_output([GIT, "show", source_commit + ":" + relative + "/" + name], cwd=repo)
        if committed != (ROOT / name).read_bytes():
            raise RuntimeError("not exact frozen Git source: " + name)
    out = ROOT / "run-01"
    out.mkdir(exist_ok=False)
    record = dict(allocation=freeze["allocation"], source_commit=source_commit,
                  source_freeze_sha256=hashlib.sha256((ROOT / "FREEZE.json").read_bytes()).hexdigest(),
                  producer_invocations=0, primary_audit_invocations=0, formal_retries=0,
                  state="PREPARED_BEFORE_SEND", prepared_utc_ns=time.time_ns(), commands=[],
                  child_maxrss_scope="all launcher children including Git identity reads; not an isolated producer peak")
    save(out / "attempt.json", record, True)
    for name, command in [("producer", [sys.executable, "-B", str(ROOT / "candidate.py.txt"), str(out / "raw.jsonl")]),
                          ("primary_audit", [sys.executable, "-B", str(ROOT / "audit.py.txt"), str(out / "raw.jsonl"), str(out / "audit.json")])]:
        record[name + "_invocations"] += 1
        receipt = dict(name=name, command=command, cwd=str(ROOT), started_utc_ns=time.time_ns(), exit_code=None)
        record["commands"].append(receipt); record["state"] = "SENT_ONCE_" + name
        save(out / "attempt.json", record)
        try:
            with (out / (name + ".stdout.txt")).open("xb") as stdout, (out / (name + ".stderr.txt")).open("xb") as stderr:
                result = subprocess.run(command, cwd=ROOT, stdout=stdout, stderr=stderr, timeout=10, check=False)
            receipt["exit_code"] = result.returncode
        except subprocess.TimeoutExpired:
            receipt["timeout"] = True; receipt["exit_code"] = None
        receipt["ended_utc_ns"] = time.time_ns()
        receipt["stdout_sha256"] = hashlib.sha256((out / (name + ".stdout.txt")).read_bytes()).hexdigest()
        receipt["stderr_sha256"] = hashlib.sha256((out / (name + ".stderr.txt")).read_bytes()).hexdigest()
        record["child_maxrss_bytes_observed"] = resource.getrusage(resource.RUSAGE_CHILDREN).ru_maxrss
        save(out / "attempt.json", record)
        if receipt["exit_code"] != 0:
            record["state"] = "STOP_FIRST_RESULT_RETAINED"; save(out / "attempt.json", record)
            return 2
    record["state"] = "FIRST_PRODUCER_AND_PRIMARY_AUDIT_TERMINAL"; record["ended_utc_ns"] = time.time_ns()
    save(out / "attempt.json", record)
    print(json.dumps(dict(state=record["state"], counts=[record["producer_invocations"], record["primary_audit_invocations"]])))
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
