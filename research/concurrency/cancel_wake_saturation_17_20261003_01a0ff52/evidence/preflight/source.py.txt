import array,datetime,errno,fcntl,hashlib,json,os,pathlib,platform,stat,sys,termios,threading,time
mods=[array,datetime,fcntl,hashlib,json,pathlib,platform,stat,termios,threading]
paths={m.__name__:m.__file__ for m in mods if getattr(m,"__file__",None)}
paths["python_executable"]=os.path.realpath(sys.executable)
pins={n:hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest() for n,p in paths.items()}
c={n:pathlib.Path("/sys/fs/cgroup/"+n).read_text().strip() for n in ["cpu.max","memory.max","memory.swap.max","pids.max"]}
r,w=os.pipe();fcntl.fcntl(w,fcntl.F_SETPIPE_SZ,4096);capacity=fcntl.fcntl(w,fcntl.F_GETPIPE_SZ);os.close(r);os.close(w)
print(json.dumps(dict(pid=os.getpid(),native_tid=threading.get_native_id(),utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),python=sys.version,kernel=platform.release(),machine=platform.machine(),cgroup=c,pipe_capacity=capacity,dependency_paths=paths,dependency_sha256=pins)))
if c!={"cpu.max":"25000 100000","memory.max":"134217728","memory.swap.max":"0","pids.max":"32"} or capacity!=4096:sys.exit(2)
