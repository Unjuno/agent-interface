"""Fresh owned pipe actor; all releases after primary remain explicitly separate."""
import asyncio
import concurrent.futures
import errno
import json
import os
import queue
import sys
import threading
import time

case = sys.argv[1]
if case not in {'reader_close', 'normal_D', 'writer_EOF', 'wrapper_cancel'}:
    raise ValueError('fixed case')
events = []
lock = threading.Lock()
identity = queue.Queue()
read_fd, write_fd = os.pipe()
read_owned = write_owned = True
close_actor = None
close_result = None
loop = asyncio.new_event_loop()


def event(name, **extra):
    with lock:
        events.append({'ordinal': len(events), 'event': name, 'pid': os.getpid(),
                       'native_id': threading.get_native_id(), 'mono_ns': time.monotonic_ns(), **extra})


def probe(fd):
    try:
        os.fstat(fd)
        return {'state': 'OPEN', 'errno': None}
    except OSError as exc:
        return {'state': 'EBADF' if exc.errno == errno.EBADF else 'ERROR', 'errno': exc.errno}


def read():
    name = threading.current_thread().name
    event('read_about_to_call', fd=read_fd, lifetime='data-reader-0', thread_name=name)
    identity.put({'native_id': threading.get_native_id(), 'name': name})
    try:
        data = os.read(read_fd, 1)
        value = {'kind': 'READ_EOF' if data == b'' else 'READ_DATA', 'hex': data.hex(), 'errno': None}
    except OSError as exc:
        value = {'kind': 'READ_ERROR', 'hex': None, 'errno': exc.errno}
    event('read_return', value=value)
    return value


executor = concurrent.futures.ThreadPoolExecutor(max_workers=1, thread_name_prefix='owned-pipe-read')
native = executor.submit(read)
reader = identity.get(timeout=1)


async def wait_native():
    return await asyncio.wrap_future(native, loop=loop)


task = loop.create_task(wait_native())
loop.run_until_complete(asyncio.sleep(0))


def snapshot(stage):
    value = native.result() if native.done() and not native.cancelled() else None
    return {'stage': stage, 'mono_ns': time.monotonic_ns(), 'future_done': native.done(),
            'future_running': native.running(), 'future_cancelled': native.cancelled(), 'value': value,
            'task_done': task.done(), 'task_cancelled': task.cancelled(),
            'reader_fd': read_fd, 'writer_fd': write_fd, 'reader_lifetime': 'data-reader-0',
            'writer_lifetime': 'data-writer-0', 'reader_probe': probe(read_fd), 'writer_probe': probe(write_fd),
            'executor_threads_alive': [thread.is_alive() for thread in executor._threads],
            'close_actor_alive': close_actor.is_alive() if close_actor is not None else False,
            'close_result': close_result}


def emit(name, **extra):
    print(json.dumps({'message': name, 'pid': os.getpid(), 'ppid': os.getppid(), 'case': case, **extra}), flush=True)


emit('READY', reader=reader, main_native_id=threading.get_native_id(), before=snapshot('ready'))
if sys.stdin.readline().strip() != 'CHECK':
    raise RuntimeError('exact CHECK before actor')
emit('PRE_ACT', before=snapshot('pre_actor'))
if sys.stdin.readline().strip() != 'ACT':
    raise RuntimeError('exact ACT once')
event('actor_requested', case=case)
actor_start_ns = time.monotonic_ns()
if case == 'reader_close':
    read_owned = False

    def close_reader():
        global close_result
        event('reader_close_call_start', fd=read_fd, lifetime='data-reader-0')
        try:
            os.close(read_fd)
            close_result = {'returned': True, 'errno': None}
        except OSError as exc:
            close_result = {'returned': True, 'errno': exc.errno}
        event('reader_close_return', result=close_result)

    close_actor = threading.Thread(target=close_reader, name='owned-reader-close')
    close_actor.start()
elif case == 'normal_D':
    count = os.write(write_fd, b'D')
    event('actor_write_D', count=count)
elif case == 'writer_EOF':
    write_owned = False
    os.close(write_fd)
    event('actor_writer_closed', fd=write_fd, lifetime='data-writer-0')
else:
    task.cancel()
    loop.run_until_complete(asyncio.sleep(0))
    event('actor_wrapper_cancelled', task_done=task.done(), task_cancelled=task.cancelled())
remaining = 0.25 - (time.monotonic_ns() - actor_start_ns) / 1_000_000_000
if remaining > 0:
    time.sleep(remaining)
loop.run_until_complete(asyncio.sleep(0))
event('primary_snapshot')
emit('PRIMARY', actor_start_ns=actor_start_ns, primary=snapshot('primary'), events=list(events))
if sys.stdin.readline().strip() != 'CLEANUP':
    raise RuntimeError('exact cleanup after primary')
event('harness_cleanup_start')
if not native.done() and write_owned:
    count = os.write(write_fd, b'D')
    event('harness_release_D', count=count)
value = native.result(timeout=1)
loop.run_until_complete(asyncio.sleep(0))
if close_actor is not None:
    close_actor.join(timeout=1)
    if close_actor.is_alive():
        raise RuntimeError('owned close actor still pending after harness')
if read_owned:
    read_owned = False
    os.close(read_fd)
    event('harness_reader_closed', fd=read_fd, lifetime='data-reader-0')
if write_owned:
    write_owned = False
    os.close(write_fd)
    event('harness_writer_closed', fd=write_fd, lifetime='data-writer-0')
executor.shutdown(wait=True)
if task.done() and not task.cancelled():
    task.result()
loop.close()
event('executor_joined_and_loop_closed')
emit('FINAL', final=snapshot('final'), events=list(events), loop_closed=loop.is_closed(), native_value=value)
