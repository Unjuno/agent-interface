"""One prospective native-stack-qualified comparison; each owned peer once."""
import argparse
import datetime
import hashlib
import importlib.machinery
import importlib.util
import json
import os
from pathlib import Path
import selectors
import subprocess
import sys
import time

ROOT = Path(__file__).resolve().parent
loader = importlib.machinery.SourceFileLoader('native_stack_reader', str(ROOT / 'stack_reader.py.txt'))
spec = importlib.util.spec_from_loader('native_stack_reader', loader)
reader = importlib.util.module_from_spec(spec)
loader.exec_module(reader)
CASES = ['reader_close', 'normal_D', 'writer_EOF', 'wrapper_cancel']


def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


class Lines:
    def __init__(self, pipe):
        self.pipe, self.pending, self.raw = pipe, b'', b''
        os.set_blocking(pipe.fileno(), False)

    def read(self, timeout):
        deadline = time.monotonic() + timeout
        with selectors.DefaultSelector() as selector:
            selector.register(self.pipe, selectors.EVENT_READ)
            while b'\n' not in self.pending:
                remaining = deadline - time.monotonic()
                if remaining <= 0 or not selector.select(remaining):
                    raise TimeoutError('owned bounded stdout frame')
                chunk = os.read(self.pipe.fileno(), 8192)
                if not chunk:
                    raise RuntimeError('owned peer EOF before frame')
                self.pending += chunk
                self.raw += chunk
                if len(self.raw) > 131072:
                    raise RuntimeError('peer stdout ceiling')
        line, self.pending = self.pending.split(b'\n', 1)
        return json.loads(line)


def cell(case, output):
    row = {'case': case, 'started_utc': utc(), 'driver_pid': os.getpid(), 'physical_input': False,
           'sample': None, 'ready': None, 'pre_actor': None, 'primary': None, 'final': None,
           'actor_sent': False, 'error': None}
    peer_argv = [sys.executable, '-B', '-u', str(ROOT / 'peer.py.txt'), case]
    peer = subprocess.Popen(peer_argv, stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    row.update(peer_pid=peer.pid, peer_argv=peer_argv)
    lines = Lines(peer.stdout)
    sampler = None
    peer_stderr = b''
    remainder = b''
    try:
        ready = lines.read(2)
        row['ready'] = ready
        if ready['message'] != 'READY' or ready['pid'] != peer.pid or ready['ppid'] != os.getpid():
            raise ValueError('owned peer readiness identity')
        before = ready['before']
        if (before['future_done'] is not False or before['future_running'] is not True or
                before['writer_probe']['state'] != 'OPEN' or before['reader_probe']['state'] != 'OPEN'):
            raise ValueError('pending owned silent read before sample')
        report = output / (case + '.sample.txt')
        argv = ['/usr/bin/sample', str(peer.pid), '1', '10', '-file', str(report)]
        started_utc, started_ns = utc(), time.monotonic_ns()
        sampler = subprocess.Popen(argv, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
        sample_timeout = False
        try:
            sample_stdout, sample_stderr = sampler.communicate(timeout=5)
        except subprocess.TimeoutExpired:
            sampler.kill()
            sample_stdout, sample_stderr = sampler.communicate(timeout=2)
            sample_timeout = True
        ended_utc, ended_ns = utc(), time.monotonic_ns()
        (output / (case + '.sample.stdout.log')).write_bytes(sample_stdout)
        (output / (case + '.sample.stderr.log')).write_bytes(sample_stderr)
        report_raw = report.read_bytes() if report.exists() else b''
        row['sample'] = {'argv': argv, 'pid': sampler.pid, 'exit_code': sampler.returncode,
                         'target_pid': peer.pid, 'started_utc': started_utc, 'ended_utc': ended_utc,
                         'started_ns': started_ns, 'ended_ns': ended_ns, 'bytes': len(report_raw),
                         'sha256': sha(report_raw), 'stdout_sha256': sha(sample_stdout),
                         'stderr_sha256': sha(sample_stderr), 'watchdog_fired': sample_timeout}
        if sample_timeout:
            raise RuntimeError('owned sampler watchdog')
        if sampler.returncode != 0 or not report_raw or len(report_raw) > 1_048_576:
            raise ValueError('native sampler exit/output gate')
        gate = reader.inspect(report_raw.decode(), peer.pid, os.getpid(),
                              ready['reader']['native_id'], ready['reader']['name'])
        row['sample']['gate'] = gate
        if gate['qualified'] is not True or ready['reader']['native_id'] == ready['main_native_id']:
            raise ValueError('exact read-worker native-stack gate')
        peer.stdin.write(b'CHECK\n')
        peer.stdin.flush()
        pre = lines.read(2)
        row['pre_actor'] = pre
        if (pre['message'] != 'PRE_ACT' or pre['before']['future_done'] is not False or
                pre['before']['future_running'] is not True or pre['before']['writer_probe']['state'] != 'OPEN'):
            raise ValueError('same native Future still pending before actor')
        row['actor_send_ns'] = time.monotonic_ns()
        row['actor_send_utc'] = utc()
        peer.stdin.write(b'ACT\n')
        peer.stdin.flush()
        row['actor_sent'] = True
        row['primary'] = lines.read(3)
        if row['primary']['message'] != 'PRIMARY':
            raise ValueError('literal primary frame')
        peer.stdin.write(b'CLEANUP\n')
        peer.stdin.flush()
        row['final'] = lines.read(3)
        if row['final']['message'] != 'FINAL':
            raise ValueError('literal final frame')
        os.set_blocking(peer.stdout.fileno(), True)
        remainder, peer_stderr = peer.communicate(timeout=2)
        row['status'] = 'COMPLETE_FIRST_CELL'
    except Exception as exc:
        row['error'] = {'type': type(exc).__name__, 'message': str(exc)}
        row['status'] = 'STOP_FIRST_CELL'
    finally:
        if sampler is not None and sampler.poll() is None:
            sampler.kill()
            sampler.communicate(timeout=2)
        if peer.poll() is None:
            peer.terminate()
            try:
                remainder, peer_stderr = peer.communicate(timeout=2)
            except subprocess.TimeoutExpired:
                peer.kill()
                remainder, peer_stderr = peer.communicate(timeout=2)
                row['cleanup_escalated_owned_kill'] = True
            row['owned_peer_terminated_on_stop'] = True
        else:
            row['owned_peer_terminated_on_stop'] = False
        for pipe in (peer.stdin, peer.stdout, peer.stderr):
            if pipe is not None:
                pipe.close()
        stdout_raw = lines.raw + remainder
        (output / (case + '.peer.stdout.log')).write_bytes(stdout_raw)
        (output / (case + '.peer.stderr.log')).write_bytes(peer_stderr)
        row.update(peer_exit=peer.returncode, ended_utc=utc(), peer_stdout_sha256=sha(stdout_raw),
                   peer_stderr_sha256=sha(peer_stderr), peer_stdout_bytes=len(stdout_raw),
                   pipes_closed=all(pipe.closed for pipe in (peer.stdin, peer.stdout, peer.stderr)),
                   sampler_terminal=sampler.poll() is not None if sampler is not None else None)
    (output / (case + '.receipt.json')).write_text(json.dumps(row, indent=2, sort_keys=True) + '\n')
    return row


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--output', required=True)
    args = parser.parse_args()
    output = Path(args.output).resolve()
    output.mkdir(parents=False, exist_ok=False)
    freeze_raw = (ROOT / 'FREEZE.json').read_bytes()
    freeze = json.loads(freeze_raw)
    for pin in freeze['inputs']:
        raw = (ROOT / pin['path']).read_bytes()
        if len(raw) != pin['bytes'] or sha(raw) != pin['sha256']:
            raise ValueError('pre-execution frozen input mismatch: ' + pin['path'])
    environment = json.loads((ROOT / 'environment.json').read_bytes())
    if sys.version != environment['python_version'] or sys.platform != 'darwin':
        raise ValueError('frozen native macOS interpreter')
    for pin in environment['external_inputs']:
        raw = Path(pin['path']).read_bytes()
        if len(raw) != pin['bytes'] or sha(raw) != pin['sha256']:
            raise ValueError('pre-execution external dependency drift: ' + pin['label'])
    start = {'allocation': freeze['allocation'], 'argv': sys.argv, 'pid': os.getpid(), 'started_utc': utc(),
             'freeze_sha256': sha(freeze_raw), 'formal_producer_invocations': 1, 'formal_retries': 0,
             'source_root': str(ROOT), 'inputs_verified': True}
    (output / 'START.json').write_text(json.dumps(start, indent=2) + '\n')
    rows = []
    stopped = False
    for case in CASES:
        if stopped:
            rows.append({'case': case, 'status': 'NOT_RUN_AFTER_FIRST_STOP'})
            continue
        row = cell(case, output)
        rows.append(row)
        stopped = row['status'] != 'COMPLETE_FIRST_CELL'
    result = {'allocation': freeze['allocation'], 'started_utc': start['started_utc'], 'ended_utc': utc(),
              'freeze_sha256': sha(freeze_raw), 'cases': rows, 'physical_input': False,
              'kernel_trace': False, 'formal_producer_invocations': 1, 'formal_retries': 0,
              'status': 'STOP_RETAINED_FIRST' if stopped else 'COMPLETE_FIRST_COMPARISON'}
    raw = (json.dumps(result, indent=2, sort_keys=True) + '\n').encode()
    (output / 'result.json').write_bytes(raw)
    print(json.dumps({'status': result['status'], 'rows': len(rows), 'sha256': sha(raw), 'bytes': len(raw)}))
    raise SystemExit(20 if stopped else 0)


if __name__ == '__main__':
    main()
