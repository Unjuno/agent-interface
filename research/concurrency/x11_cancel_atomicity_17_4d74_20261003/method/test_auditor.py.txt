"""Hand-derived saved oracle and coherent mutations; no producer/gate/X11 imports."""
import copy
import hashlib
import json
from pathlib import Path
import tempfile
import unittest

from auditor import AuditError, audit, read_json


MS = 1_000_000
ORDER = [("precheck-healthy", "precheck", False), ("atomic-paused", "atomic", True),
         ("precheck-paused", "precheck", True), ("atomic-healthy", "atomic", False)]
OPS = [{"op": "focus", "target": "owned"}, {"op": "key_state", "key": "F8", "down": True},
       {"op": "key_state", "key": "F9", "down": True}, {"op": "wait_update", "timeout_ms": 40},
       {"op": "release_all"}]


def physical(base, start, end, keys):
    return {"started_ns": base + start * MS, "ended_ns": base + end * MS,
            "keys_down": keys, "buttons_mask": 0}


def neutral(at):
    return {"keys_down": [], "buttons_down": [], "verified": True, "monotonic_ns": at}


def hand_cell(index, root):
    """Explicit schedule written before formal output; durations are hand-authored."""
    identity, arm, paused = ORDER[index]
    base = 1_000_000_000 + index * 2_000_000_000
    native9, wait_start, wait_end, release_at, end = (411, 413, 453, 454, 456) if paused else (6, 8, 48, 49, 51)
    log = b"hand-authored saved Xvfb log\n"
    (root / (identity + ".xvfb.log")).write_bytes(log)
    checks = [{"key": "F8", "checked_ns": base + 2 * MS, "allowed": True, "lock_held": arm == "atomic"},
              {"key": "F9", "checked_ns": base + 5 * MS, "allowed": True, "lock_held": arm == "atomic"}]
    calls = [{"kind": "key_state", "key": "F8", "down": True, "started_ns": base + 3 * MS,
              "ended_ns": base + 4 * MS, "thread_id": 123},
             {"kind": "key_state", "key": "F9", "down": True, "started_ns": base + native9 * MS,
              "ended_ns": base + (native9 + 1) * MS, "thread_id": 123}]
    samples = []
    if paused:
        samples.append({"started_ns": base + 4_500_000, "ended_ns": base + 4_600_000,
                        "keys_down": [74], "buttons_mask": 0})
        query = physical(base, 160, 161, [] if arm == "precheck" else [74])
        samples.append(query)
        if arm == "precheck":
            samples.append(physical(base, 425, 426, [75]))
        lock, accepted, ended = (11, 12, 14) if arm == "precheck" else (413, 414, 416)
        cancel = {"request_ns": base + 10 * MS, "lock_acquired_ns": base + lock * MS,
                  "accepted_ns": base + accepted * MS, "ended_ns": base + ended * MS,
                  "receipt": neutral(base + (accepted + 1) * MS)}
        checkpoint = {"at_ns": base + 162 * MS, "program_alive": True,
                      "canceller_alive": arm == "atomic", "cancel_accepted": arm == "precheck",
                      "query": copy.deepcopy(query)}
        pause = {"entered_ns": base + 6 * MS, "resumed_ns": base + 410 * MS}
        event_order = [("press", 74), ("release", 74), ("press", 75), ("release", 75)] if arm == "precheck" else [
            ("press", 74), ("press", 75), ("release", 74), ("release", 75)]
        arrival = [8, 16, 420, 460] if arm == "precheck" else [8, 420, 421, 422]
    else:
        samples.append(physical(base, 20, 21, [74, 75]))
        pause = cancel = checkpoint = None
        event_order = [("press", 74), ("press", 75), ("release", 74), ("release", 75)]
        arrival = [10, 11, 55, 56]
    samples.append(physical(base, end + 9, end + 10, []))
    events = [{"type": kind, "keycode": code, "received_ns": base + t * MS, "server_time_ms": index * 1000 + i + 20}
              for i, ((kind, code), t) in enumerate(zip(event_order, arrival))]
    execution = {"started_ns": base + MS, "ended_ns": base + (end - 1) * MS,
                 "emissions": 4, "program_emissions": 4, "observations": [], "activations": [],
                 "completed_ops": [0, 1, 2, 3, 4], "releases": [neutral(base + release_at * MS)],
                 "waits": [{"operation_index": 3, "requested_ms": 40, "started_ns": base + wait_start * MS,
                            "ended_ns": base + wait_end * MS, "completed": True, "kind": "fixed_delay",
                            "update_observed": None}]}
    return {"id": identity, "arm": arm, "paused": paused, "keycodes": {"F8": 74, "F9": 75},
            "program": {"schema": "agent-interface/program-v1", "program_id": "program-" + identity,
                        "source": {"observation_seq": 1, "binding_revision": 1},
                        "authority": {"lease_id": "owned-" + identity, "expires_at_ns": base + 4_000_000_000},
                        "terminal": {"release_all_required": True}, "ops": copy.deepcopy(OPS)},
            "dispatch_started_ns": base + 500_000, "dispatch_ended_ns": base + end * MS,
            "checks": checks, "native_calls": calls, "pause": pause, "cancel": cancel,
            "checkpoint": checkpoint, "samples": samples, "events": events,
            "program_receipt": {"status": "completed", "admission": "accepted",
                                "required_capabilities": ["clock.monotonic", "event.feedback", "input.keyboard",
                                                          "input.release_all", "window.focus"],
                                "execution": execution, "recovery_required": False},
            "terminal": {"keys_down": [], "buttons_mask": 0, "threads_alive": [], "Xvfb_exit": -15,
                         "query": physical(base, end + 14, end + 15, [])},
            "errors": [], "thread_errors": [], "Xvfb_pid": 100 + index, "window_id": 200 + index,
            "Xvfb_argv": ["Xvfb", ":" + str(91 + index), "-screen", "0", "320x240x24", "-nolisten", "tcp", "-ac"],
            "cleanup_receipt": neutral(base + (end + 20) * MS), "cell_elapsed_ns": 600 * MS,
            "xvfb_log_sha256": hashlib.sha256(log).hexdigest()}


class SavedOracleTests(unittest.TestCase):
    def setUp(self):
        self.directory = tempfile.TemporaryDirectory()
        self.addCleanup(self.directory.cleanup)
        self.root = Path(self.directory.name)
        source = self.root / "sources" / "inert.txt"
        source.parent.mkdir()
        source.write_bytes(b"inert hand-authored custody fixture; never executed\n")
        pins = {"sources/inert.txt": hashlib.sha256(source.read_bytes()).hexdigest()}
        self.plan = {"schema": "x11-cancel-atomicity-plan-v1", "raw_schema": "x11-cancel-atomicity-raw-v1",
                     "allocation": "X11-CANCEL-ATOMICITY-7012-4D74-A01-20261003",
                     "source_hashes": pins, "cells": [{"id": i, "arm": a, "paused": p} for i, a, p in ORDER],
                     "constants": {"checkpoint_delay_ns": 150_000_000, "resume_delay_ns": 400_000_000, "healthy_wait_ms": 40},
                     "program_ops": copy.deepcopy(OPS)}
        encoded = (json.dumps(self.plan, sort_keys=True) + "\n").encode()
        self.plan_digest = hashlib.sha256(encoded).hexdigest()
        self.raw = {"schema": self.plan["raw_schema"], "allocation": self.plan["allocation"],
                    "source_hashes_before": pins.copy(), "source_hashes_after": pins.copy(),
                    "plan_sha256": self.plan_digest, "environment": {"python": "hand oracle", "platform": "saved-only",
                                                                    "clock": "monotonic", "pid": 99},
                    "cells": [hand_cell(i, self.root) for i in range(4)], "first_disposition": "COLLECTED_NOT_SCORED"}

    def check(self, raw=None, plan=None):
        return audit(self.raw if raw is None else raw, self.plan if plan is None else plan,
                     self.root, self.plan_digest, self.root)

    def test_hand_oracle_separates_both_results(self):
        result = self.check()
        self.assertEqual(result["status"], "SCOPED_RESULTS_RECONSTRUCTED")
        self.assertEqual(result["dispositions"], ["POST_ACCEPTED_INPUT_COUNTEREXAMPLE",
                         "ATOMIC_SERIALIZATION_WITH_CANCEL_REACHABILITY_DELAY"])
        self.assertNotIn("PASS", result["status"])
        self.assertEqual([c["post_accepted_positive"] for c in result["cells"]], [False, False, True, False])

    def test_atomic_receiver_lag_is_not_late_injection(self):
        row = self.raw["cells"][1]
        self.assertGreater(row["events"][1]["received_ns"], row["cancel"]["accepted_ns"])
        self.assertFalse(self.check()["cells"][1]["post_accepted_positive"])
        self.assertFalse(any(75 in s["keys_down"] for s in row["samples"]))

    def test_normal_or_sigterm_endpoint_is_explicit_not_coerced(self):
        copied = copy.deepcopy(self.raw)
        copied["cells"][0]["terminal"]["Xvfb_exit"] = 0
        self.assertEqual(self.check(copied)["status"], "SCOPED_RESULTS_RECONSTRUCTED")
        for invalid in (False, None, -9):
            copied["cells"][0]["terminal"]["Xvfb_exit"] = invalid
            with self.assertRaises(AuditError):
                self.check(copied)

    def test_coherent_copied_corruptions_rejected(self):
        # Each copy remains parseable; no generic outer digest blocks these semantic checks.
        def change(raw, key):
            pre, atomic = raw["cells"][2], raw["cells"][1]
            if key == "boolean_counter": pre["program_receipt"]["execution"]["completed_ops"][1] = True
            elif key == "false_native": pre["native_calls"][1]["down"] = 1
            elif key == "receiver_only_late":
                pre["cancel"].update(lock_acquired_ns=pre["native_calls"][1]["ended_ns"] + MS,
                                     accepted_ns=pre["native_calls"][1]["ended_ns"] + 2 * MS)
                pre["cancel"]["ended_ns"] = pre["cancel"]["accepted_ns"] + 2 * MS
                pre["cancel"]["receipt"] = neutral(pre["cancel"]["accepted_ns"] + MS)
            elif key == "early_atomic_acceptance":
                atomic["cancel"].update(lock_acquired_ns=atomic["pause"]["entered_ns"],
                                        accepted_ns=atomic["pause"]["entered_ns"] + MS)
                atomic["cancel"]["receipt"] = neutral(atomic["cancel"]["accepted_ns"] + MS)
            elif key == "stale_checkpoint":
                q = pre["checkpoint"]["query"]; q["started_ns"] -= 50 * MS; q["ended_ns"] -= 50 * MS
                pre["samples"][1] = copy.deepcopy(q)
            elif key == "duplicate_cell": raw["cells"][3] = copy.deepcopy(raw["cells"][0])
            elif key == "app_census": pre["events"].pop()
            elif key == "physical_f9": pre["samples"][2]["keys_down"] = []
            elif key == "cleanup": pre["terminal"]["threads_alive"] = ["dispatch"]
            elif key == "lease": pre["program"]["authority"]["expires_at_ns"] = pre["dispatch_ended_ns"]
            elif key == "source": raw["source_hashes_after"]["sources/inert.txt"] = "0" * 64
            elif key == "plan": raw["plan_sha256"] = "0" * 64
            elif key == "capabilities": pre["program_receipt"]["required_capabilities"] = ["input.keyboard"]
        controls = ["boolean_counter", "false_native", "receiver_only_late", "early_atomic_acceptance",
                    "stale_checkpoint", "duplicate_cell", "app_census", "physical_f9", "cleanup", "lease", "source", "plan",
                    "capabilities"]
        for key in controls:
            with self.subTest(control=key):
                changed = copy.deepcopy(self.raw)
                change(changed, key)
                with self.assertRaises(AuditError):
                    self.check(changed)

    def test_real_producer_disposition_and_observer_retirement(self):
        for kind in ("missing_disposition", "stopped_disposition", "late_sample", "late_event"):
            with self.subTest(kind=kind):
                changed = copy.deepcopy(self.raw)
                if kind == "missing_disposition": del changed["first_disposition"]
                elif kind == "stopped_disposition": changed["first_disposition"] = "STOP_NATIVE_CELL"
                elif kind == "late_sample":
                    row = changed["cells"][0]
                    row["samples"][-1] = physical(row["terminal"]["query"]["ended_ns"], 1, 2, [])
                else:
                    row = changed["cells"][0]
                    row["events"][-1]["received_ns"] = row["terminal"]["query"]["ended_ns"] + MS
                with self.assertRaises((AuditError, KeyError)):
                    self.check(changed)

    def test_local_source_and_log_bytes_are_verified(self):
        (self.root / "precheck-healthy.xvfb.log").write_bytes(b"coherent replacement outside raw")
        with self.assertRaises(AuditError):
            self.check()
        source = self.root / "sources" / "inert.txt"
        source.write_bytes(b"changed source")
        with self.assertRaises(AuditError):
            self.check()

    def test_json_duplicate_keys_and_nonfinite_refused(self):
        path = self.root / "bad.json"
        for text in ('{"x": 1, "x": 2}', '{"x": NaN}'):
            path.write_text(text)
            with self.assertRaises(AuditError):
                read_json(path)


if __name__ == "__main__":
    unittest.main()
