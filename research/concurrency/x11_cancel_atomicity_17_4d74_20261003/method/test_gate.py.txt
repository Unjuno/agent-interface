"""Construction controls for the real gate, with no display or native input.

These tests do not start a producer, consume a formal cell, or exercise the
paused timing discriminator. Only the effect boundaries are replaced by stubs.
"""
from __future__ import annotations

from pathlib import Path
import sys
import threading
import unittest
from unittest.mock import Mock


ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "sources"))
sys.path.insert(0, str(ROOT / "method"))

from cancel_gate import CancelRaceBackend, X11BackendError


class GateConstructionTests(unittest.TestCase):
    def make_backend(self, arm):
        # Bypass X11Backend.__init__: no display connection or input is created.
        backend = CancelRaceBackend.__new__(CancelRaceBackend)
        backend.arm = arm
        backend.paused = False
        backend.owner_lock = threading.RLock()
        backend.cancelled = threading.Event()
        backend.checks = []
        backend.cancel_row = None
        backend.program_id = "construction-owned-program"
        backend._native = Mock(return_value="native-stub-result")

        receipt = {"verified": True, "keys_down": [], "buttons_down": []}
        cleanup_marker_states = []

        def neutral_cleanup():
            cleanup_marker_states.append(backend.cancelled.is_set())
            return receipt

        backend.release_all = Mock(side_effect=neutral_cleanup)
        return backend, receipt, cleanup_marker_states

    def test_healthy_positive_input_reaches_native_boundary(self):
        for arm in ("precheck", "atomic"):
            with self.subTest(arm=arm):
                backend, _, cleanup_states = self.make_backend(arm)
                result = backend.key_state("F9", True)
                self.assertEqual(result, "native-stub-result")
                backend._native.assert_called_once_with("F9", True)
                self.assertEqual(len(backend.checks), 1)
                self.assertTrue(backend.checks[0]["allowed"])
                self.assertFalse(backend.cancelled.is_set())
                backend.release_all.assert_not_called()
                self.assertEqual(cleanup_states, [])

    def test_already_accepted_cancellation_blocks_positive_input(self):
        for arm in ("precheck", "atomic"):
            with self.subTest(arm=arm):
                backend, _, cleanup_states = self.make_backend(arm)
                backend.cancel_program(backend.program_id)
                with self.assertRaises(X11BackendError):
                    backend.key_state("F9", True)
                backend._native.assert_not_called()
                self.assertEqual(len(backend.checks), 1)
                self.assertFalse(backend.checks[0]["allowed"])
                self.assertTrue(backend.cancelled.is_set())
                backend.release_all.assert_called_once_with()
                self.assertEqual(cleanup_states, [True])

    def test_cleanup_key_up_remains_allowed_after_cancellation(self):
        for arm in ("precheck", "atomic"):
            with self.subTest(arm=arm):
                backend, _, cleanup_states = self.make_backend(arm)
                backend.cancel_program(backend.program_id)
                result = backend.key_state("F8", False)
                self.assertEqual(result, "native-stub-result")
                backend._native.assert_called_once_with("F8", False)
                self.assertEqual(backend.checks, [])
                self.assertTrue(backend.cancelled.is_set())
                backend.release_all.assert_called_once_with()
                self.assertEqual(cleanup_states, [True])

    def test_mismatched_program_cannot_mark_or_clean_owned_input(self):
        for arm in ("precheck", "atomic"):
            with self.subTest(arm=arm):
                backend, _, cleanup_states = self.make_backend(arm)
                with self.assertRaises(X11BackendError):
                    backend.cancel_program("different-program")
                self.assertFalse(backend.cancelled.is_set())
                self.assertIsNone(backend.cancel_row)
                backend.release_all.assert_not_called()
                backend._native.assert_not_called()
                self.assertEqual(cleanup_states, [])
                self.assertEqual(backend.checks, [])

    def test_matching_program_marks_before_one_cleanup_and_retains_receipt(self):
        for arm in ("precheck", "atomic"):
            with self.subTest(arm=arm):
                backend, receipt, cleanup_states = self.make_backend(arm)
                row = backend.cancel_program(backend.program_id)
                self.assertTrue(backend.cancelled.is_set())
                backend.release_all.assert_called_once_with()
                self.assertEqual(cleanup_states, [True])
                self.assertIs(row, backend.cancel_row)
                self.assertIs(row["receipt"], receipt)
                self.assertTrue(row["receipt"]["verified"])
                self.assertEqual(row["receipt"]["keys_down"], [])
                self.assertEqual(row["receipt"]["buttons_down"], [])
                backend._native.assert_not_called()
                self.assertEqual(backend.checks, [])


if __name__ == "__main__":
    unittest.main()
