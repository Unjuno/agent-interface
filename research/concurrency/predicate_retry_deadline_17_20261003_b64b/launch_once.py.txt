"""One frozen model, then one saved-only audit; exact owned guest only."""
import pathlib,subprocess,json,hashlib,tarfile,io,datetime,sys
root=pathlib.Path(__file__).resolve().parent
prefix=['/opt/homebrew/bin/orb','run','-m','research-6695-docker-01a0ff52','-u','root']
freeze=json.loads((root/'FREEZE.json').read_text());files=freeze['frozen_inputs'];image=freeze['image']
def stamp():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def run_receipt(name,args,data=None,timeout=15):
    start=stamp();r=subprocess.run(args,input=data,capture_output=True,timeout=timeout)
    (root/(name+'.stdout')).write_bytes(r.stdout);(root/(name+'.stderr')).write_bytes(r.stderr)
    receipt={'argv':args,'utc_start':start,'utc_end':stamp(),'exit_code':r.returncode,'stdout':{'bytes':len(r.stdout),'sha256':hashlib.sha256(r.stdout).hexdigest()},'stderr':{'bytes':len(r.stderr),'sha256':hashlib.sha256(r.stderr).hexdigest()}}
    (root/(name+'.receipt.json')).write_text(json.dumps(receipt,indent=2)+'\n');return r
for name,pin in files.items():
    b=(root/name).read_bytes();assert len(b)==pin['bytes'] and hashlib.sha256(b).hexdigest()==pin['sha256'],name
with (root/'ANALYTICAL_SEND_ONCE.json').open('x') as f:json.dump({'allocation':freeze['allocation'],'utc':stamp(),'freeze_sha256':hashlib.sha256((root/'FREEZE.json').read_bytes()).hexdigest()},f)
gsrc='/var/tmp/predicate-retry-a01-b64b-src';gout='/var/tmp/predicate-retry-a01-b64b-out';buffer=io.BytesIO()
with tarfile.open(fileobj=buffer,mode='w') as t:
    for name in [*files,'FREEZE.json']:
        b=(root/name).read_bytes();i=tarfile.TarInfo(name);i.size=len(b);i.mode=0o444;t.addfile(i,io.BytesIO(b))
setup="import pathlib,tarfile,sys;src=pathlib.Path(sys.argv[1]);out=pathlib.Path(sys.argv[2]);src.mkdir();out.mkdir();out.chmod(0o777);tarfile.open(fileobj=sys.stdin.buffer,mode='r|').extractall(src,filter='data')"
r=run_receipt('source-transfer',prefix+['python3','-c',setup,gsrc,gout],buffer.getvalue());assert r.returncode==0
code="import pathlib,hashlib,json,sys;r=pathlib.Path(sys.argv[1]);print(json.dumps({str(p.relative_to(r)):{'bytes':len(p.read_bytes()),'sha256':hashlib.sha256(p.read_bytes()).hexdigest()} for p in r.rglob('*') if p.is_file()},sort_keys=True))"
r=run_receipt('source-readback',prefix+['python3','-c',code,gsrc]);assert r.returncode==0;readback=json.loads(r.stdout)
for name,pin in files.items():assert readback[name]==pin,name
assert readback['FREEZE.json']['sha256']==hashlib.sha256((root/'FREEZE.json').read_bytes()).hexdigest()
terminal=[];model_success=False;audit_success=False
for kind in ('model','audit'):
    if kind=='audit' and not model_success:break
    name='predicate-retry-'+kind+'-a01-b64b'
    with (root/(kind+'-SEND-ONCE.json')).open('x') as f:json.dump({'utc':stamp(),'allocation':freeze['allocation'],'kind':kind},f)
    args=prefix+['docker','run','--name',name,'--label','worker=01a0ff52-b64b','--label','allocation='+freeze['allocation'],'--pull','never','--network','none','--read-only','--cap-drop','ALL','--security-opt','no-new-privileges','--cpus','0.25','--memory','128m','--memory-swap','128m','--pids-limit','32','--tmpfs','/tmp:rw,nosuid,nodev,noexec,size=8m','--user','65534:65534','-v',gsrc+':/src:ro','-v',gout+':/out:rw',image,'python','-B','/src/container_entry.py',kind]
    try:r=run_receipt(kind,args,timeout=30);success=r.returncode==0
    except subprocess.TimeoutExpired as e:
        (root/(kind+'.timeout.stdout')).write_bytes(e.stdout or b'');(root/(kind+'.timeout.stderr')).write_bytes(e.stderr or b'');run_receipt(kind+'-timeout-stop',prefix+['docker','stop','--time','1',name],timeout=5);success=False
        (root/(kind+'.receipt.json')).write_text(json.dumps({'argv':args,'utc_end':stamp(),'exit_code':None,'decision':'STOP_FIRST_WALL_CONTAINMENT','timeout_seconds':30})+'\n')
    ins=run_receipt(kind+'-inspect',prefix+['docker','inspect',name]);assert ins.returncode==0
    doc=json.loads(ins.stdout)[0];assert not doc['State']['Running'];terminal.append({'name':name,'id':doc['Id'],'State':doc['State'],'Image':doc['Image'],'HostConfig':{k:doc['HostConfig'][k] for k in ['NetworkMode','ReadonlyRootfs','Memory','MemorySwap','NanoCpus','PidsLimit','SecurityOpt','CapDrop']},'mounts':doc['Mounts']})
    if kind=='model':model_success=success
    else:audit_success=success
    if not success:break
r=run_receipt('output-export',prefix+['tar','-cf','-','-C',gout,'.']);assert r.returncode==0 and len(r.stdout)<2097152
(root/'guest-output.tar').write_bytes(r.stdout);raw=root/'raw';raw.mkdir(exist_ok=False)
with tarfile.open(fileobj=io.BytesIO(r.stdout)) as t:t.extractall(raw,filter='data')
assert sum(p.stat().st_size for p in raw.rglob('*') if p.is_file())<2097152
with (root/'terminal-containers.json').open('x') as f:f.write(json.dumps(terminal,indent=2)+'\n')
print(json.dumps({'model_exit0':model_success,'saved_audit_exit0':audit_success,'terminal_containers':len(terminal),'raw_files':[p.name for p in raw.iterdir() if p.is_file()]}))
raise SystemExit(0 if model_success and audit_success else 2)
