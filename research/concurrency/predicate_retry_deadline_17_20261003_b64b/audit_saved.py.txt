CLI: audit_saved.py DATA_ROOT --source-root SOURCE_ROOT [--output AUDIT.json]
Callable: audit_document(decoded_model, decoded_construction, source_root).
The callable returns an audit document or raises AuditError. Source files are
read as bytes and parsed into AST only. Complete transitions are derived here,
and cyclic actor progress is decided by closed-walk product-state BFS, without
SCC/Tarjan or importing any subject, construction helper, or model generator.

class AuditError(ValueError):
    pass


def require(value, message):
    if not value:
        raise AuditError(message)


def typed_equal(a, b):
    if type(a) is not type(b):
        return False
    if type(a) is dict:
        return a.keys() == b.keys() and all(typed_equal(a[k], b[k]) for k in a)
    if type(a) is list:
        return len(a) == len(b) and all(typed_equal(x, y) for x, y in zip(a, b))
    return a == b


def same(a, b, message):
    require(typed_equal(a, b), message)


def integer(value, message, minimum=0):
    require(type(value) is int and value >= minimum, message)
    return value


def pairs(values):
    result = {}
    for key, value in values:
        require(key not in result, 'duplicate JSON key: ' + key)
        result[key] = value
    return result


def reject_constant(value):
    raise AuditError('nonfinite JSON constant: ' + value)


def finite(value):
    if type(value) is float:
        require(math.isfinite(value), 'nonfinite JSON float')
    elif type(value) is dict:
        for item in value.values(): finite(item)
    elif type(value) is list:
        for item in value: finite(item)


def parse(raw):
    try:
        value = json.loads(raw, object_pairs_hook=pairs, parse_constant=reject_constant)
        finite(value)
        return value
    except (UnicodeError, json.JSONDecodeError, RecursionError) as exc:
        raise AuditError('malformed saved JSON: ' + str(exc)) from exc


def raw_file(path):
    require(path.is_file(), 'missing saved file: ' + str(path))
    require(path.stat().st_size <= CAP, 'saved file byte cap: ' + str(path))
    return path.read_bytes()


def document(path):
    value = parse(raw_file(path))
    require(type(value) is dict, 'saved document must be object: ' + str(path))
    return value


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def dump(node):
    return ast.dump(node, include_attributes=False)


def method(tree):
    classes = [n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == 'CodexAppServerClient']
    require(len(classes) == 1, 'source client class count')
    rows = [n for n in classes[0].body if isinstance(n, ast.FunctionDef) and n.name == 'wait_notification']
    require(len(rows) == 1, 'source notification method count')
    return rows[0]


def without_method(tree):
    tree = copy.deepcopy(tree)
    for node in tree.body:
        if isinstance(node, ast.ClassDef) and node.name == 'CodexAppServerClient':
            node.body = [n for n in node.body
                         if not isinstance(n, ast.FunctionDef) or n.name != 'wait_notification']
    return dump(tree)


def timeout_if(node):
    if not isinstance(node, ast.If): return False
    return any(isinstance(child, ast.Raise) and isinstance(child.exc, ast.Call)
               and isinstance(child.exc.func, ast.Name) and child.exc.func.id == 'TimeoutError'
               for child in node.body)

        trees[name] = ast.parse(raw, filename=name)
        methods[name] = method(trees[name])
        source_report[name] = {'bytes': size, 'sha256': sha}
    reference = without_method(trees['baseline.py'])
    for name in EXPECTED_SOURCES:
        same(without_method(trees[name]), reference, 'unchanged other-method/import AST: ' + name)
        same(dump(methods[name].args), dump(methods['baseline.py'].args), 'same notification signature')
    for name in ('outside.py', 'deadline_first.py'):

        require(len(body) == 2 and isinstance(body[1], ast.While), 'outside loop structure')
        loop = body[1].body
        require(isinstance(loop[0], ast.With) and isinstance(loop[-1], ast.With), 'snapshot/recheck sections')
        same(dump(loop[0].body[0]), dump(ast.parse('snapshot = tuple(self._notifications)').body[0]), 'source snapshot expression')

        same(dump(checks[retry[0]].test),
             dump(ast.parse('len(current) != len(snapshot) or any(left is not right for left, right in zip(current, snapshot))', mode='eval').body),
             'ordered identity queue-change guard')
        require(retry[0] < timeouts[0] if name == 'outside.py' else timeouts[0] < retry[0], 'source policy guard order')
        for i in timeouts:
            same(dump(checks[i].test), dump(ast.parse('remaining <= 0', mode='eval').body), 'source expired guard')
    baseline = methods['baseline.py'].body
    require(len(baseline) == 2 and isinstance(baseline[1], ast.With), 'baseline shared critical section')
    require(len(baseline[1].body) == 1 and isinstance(baseline[1].body[0], ast.While), 'baseline critical loop')
    scan = baseline[1].body[0].body
    require(isinstance(scan[0], ast.For) and timeout_if(scan[2]), 'baseline false scan then deadline')

    same(spec.get('queue_capacity'), 1, 'finite queue capacity')

def canonical(pc, q, snap):
    names = {}; next_name = 0
    for value in (snap, q):
        if value is not None and value not in names:
            names[value] = next_name; next_name += 1
    return (pc, None if q is None else names[q], None if snap is None else names[snap])


def successors(policy, state):
    pc, q, snap = state
    if pc == 'DONE': return []
    choices = []
    if policy == 'baseline':
        choices.append(('A', 'A_ATOMIC_TIMEOUT', canonical('DONE', q, snap)))
    elif pc == 'SNAP':
        choices.append(('A', 'A_SNAPSHOT', canonical('SCAN', q, q)))
    elif pc == 'SCAN':
        choices.append(('A', 'A_FALSE_SCAN', canonical('CHECK', q, snap)))
    elif pc == 'CHECK':
        if policy == 'outside' and q != snap:
            choices.append(('A', 'A_RETRY', canonical('SNAP', q, snap)))
        else:
            choices.append(('A', 'A_TIMEOUT', canonical('DONE', q, snap)))
    else:
        raise AuditError('unexpected live pc')
    if q is not None:
        choices.append(('B', 'B_CONSUME', canonical(pc, None, snap)))
    else:
        fresh = 0 if snap is None else snap + 1
        choices.append(('R', 'R_FRESH_APPEND', canonical(pc, fresh, snap)))
    return choices


def reachable_relation(policy):
    initial = ('SNAP', 0, None)
    seen = {initial}; pending = deque([initial]); relation = set()
    while pending:
        source = pending.popleft()
        for actor, event, target in successors(policy, source):
            relation.add((source, target, actor, event))
            if target not in seen:
                seen.add(target); pending.append(target)
        require(len(seen) <= 64 and len(relation) <= 192, 'independent finite-domain cap')
    return seen, relation


def state_record(value):
    require(type(value) is dict and set(value) == {'pc', 'q', 'snap'}, 'exact state schema')
    require(type(value['pc']) is str and value['pc'] in ('SNAP', 'SCAN', 'CHECK', 'DONE'), 'pc type/value')
    for key in ('q', 'snap'):
        require(value[key] is None or type(value[key]) is int and value[key] in (0, 1), 'identity handle type/value')
    state = (value['pc'], value['q'], value['snap'])
    same(canonical(*state), state, 'canonical snapshot-first identity naming')
    return state


def edge_record(edge, count):
    require(type(edge) is dict and set(edge) == {'from', 'to', 'actor', 'event'}, 'exact edge schema')
    a = integer(edge['from'], 'edge from integer'); b = integer(edge['to'], 'edge to integer')
    require(a < count and b < count, 'edge state index bounds')
    require(type(edge['actor']) is str and edge['actor'] in ('A', 'B', 'R'), 'actor type/value')
    require(type(edge['event']) is str, 'edge event type')
    return (a, b, edge['actor'], edge['event'])


def progress_closed_walk(states, edges, required_mask):
    """Independent product BFS: return nonempty closed walk with actor coverage."""
    adjacency = {i: [] for i in range(len(states))}
    for edge in edges:
        if states[edge[0]][0] != 'DONE' and states[edge[1]][0] != 'DONE':
            adjacency[edge[0]].append(edge)
    bits = {'A': 1, 'B': 2, 'R': 4}
    for origin in range(len(states)):
        if states[origin][0] == 'DONE': continue
        start = (origin, 0); pending = deque([start]); parents = {start: None}
        while pending:
            position = pending.popleft()
            for edge in adjacency[position[0]]:
                target = (edge[1], position[1] | bits[edge[2]])
                if edge[1] == origin and target[1] & required_mask == required_mask:
                    path = [edge]; cursor = position
                    while parents[cursor] is not None:
                        previous, traversed = parents[cursor]
                        path.append(traversed); cursor = previous
                    return list(reversed(path))
                if target not in parents:
                    parents[target] = (position, edge); pending.append(target)
    return None


def check_graph(graph):
    require(type(graph) is dict, 'policy graph object')
    policy = graph.get('policy'); require(type(policy) is str and policy in POLICIES, 'policy name')
    require(type(graph.get('states')) is list and 0 < len(graph['states']) <= 64, 'state list cap')
    states = [state_record(s) for s in graph['states']]
    require(len(set(states)) == len(states), 'duplicate states')
    same(graph.get('initial'), 0, 'initial index0')
    same(states[0], ('SNAP', 0, None), 'initial one irrelevant row')
    require(type(graph.get('edges')) is list and len(graph['edges']) <= 192, 'edge list cap')
    edges = [edge_record(e, len(states)) for e in graph['edges']]
    require(len(set(edges)) == len(edges), 'duplicate edges')
    wanted_states, wanted_edges = reachable_relation(policy)
    same(set(states), wanted_states, 'complete independently reachable state set: ' + policy)
    actual_edges = {(states[a], states[b], actor, event) for a, b, actor, event in edges}
    same(actual_edges, wanted_edges, 'complete independent transition relation: ' + policy)
    a_walk = progress_closed_walk(states, edges, 1)
    fair_walk = progress_closed_walk(states, edges, 7)
    same(graph.get('fair_lasso'), fair_walk is not None, 'producer fairness decision: ' + policy)
    if fair_walk is None:
        same(graph.get('lasso'), None, 'no producer lasso for negative policy')
    else:
        lasso = graph.get('lasso')
        require(type(lasso) is dict and set(lasso) == {'start', 'edges'}, 'lasso schema')
        cursor = integer(lasso['start'], 'lasso start integer')
        require(cursor < len(states) and states[cursor][0] != 'DONE', 'reachable nonterminal lasso start')
        require(type(lasso['edges']) is list and 0 < len(lasso['edges']) <= 1024, 'nonempty bounded producer lasso')
        actor_set = set(); first = cursor
        for raw in lasso['edges']:
            edge = edge_record(raw, len(states))
            require(edge in set(edges), 'lasso edge not in graph')
            require(edge[0] == cursor and states[edge[1]][0] != 'DONE', 'lasso continuity/nonterminal')
            cursor = edge[1]; actor_set.add(edge[2])
        same(cursor, first, 'lasso closed walk')
        same(actor_set, {'A', 'B', 'R'}, 'all three actors progress on producer cycle')
    same(a_walk is not None, policy == 'outside', 'restricted A-progress cycle expectation: ' + policy)
    same(fair_walk is not None, policy == 'outside', 'restricted all3-progress cycle expectation: ' + policy)
    return {'policy': policy, 'states': len(states), 'edges': len(edges),
            'A_progress_closed_walk': a_walk is not None, 'all3_progress_closed_walk': fair_walk is not None,
            'independent_witness': None if fair_walk is None else [
                {'from': a, 'to': b, 'actor': actor, 'event': event} for a, b, actor, event in fair_walk]}


def expected_control_trace(policy, scenario):
    """Source-derived ledger; no extracted method or helper is executed."""
    trace = []

    def enter_exit(actor, entry, logical_seconds=2.0):
        trace.extend(({'actor': actor, 'event': 'CONDITION_ENTER', 'entry': entry,
                       'logical_seconds': logical_seconds},
                      {'actor': actor, 'event': 'CONDITION_EXIT'}))

    if policy == 'baseline':
        enter_exit('A', 1)
        return trace
    inject = scenario in ('churn', 'fresh_match')
    cycles = 8 if policy == 'outside' and scenario == 'churn' else 1
    for cycle in range(cycles):
        enter_exit('A', 2 * cycle + 1, 0.0 if scenario == 'fresh_match' else 2.0)
        if inject:
            enter_exit('B', 2 * cycle + 1, .25 if scenario == 'fresh_match' else 2.0)
            enter_exit('B', 2 * cycle + 2, .25 if scenario == 'fresh_match' else 2.0)
            trace.extend(({'actor': 'B', 'event': 'B_CONSUME', 'removed_method': 'noise'},
                          {'actor': 'R', 'event': 'R_FRESH_APPEND',
                           'method': 'target' if scenario == 'fresh_match' else 'noise',
                           'fresh_identity': True,
                           'logical_arrival_seconds': .5 if scenario == 'fresh_match' else 2.0}))
        enter_exit('A', 2 * cycle + 2)
    if policy == 'outside' and scenario == 'fresh_match':
        enter_exit('A', 3); enter_exit('A', 4)
    return trace


def check_construction(construction, pins):
    same(construction.get('schema'), 'predicate-retry-source-controls-v2', 'source-control coherent-clock schema')
    same(construction.get('source_pins'), pins, 'source-control exact pins')
    same(construction.get('all_frozen_expectations'), True, 'source-control frozen expectations')
    same(construction.get('actual_A_source_method_calls'), 9, 'nine actual A calls')
    same(construction.get('actual_B_source_method_calls'), 11, 'eleven actual B calls')
    require(type(construction.get('python')) is str and construction['python'].startswith('3.14.5 '), 'ordinary construction Python provenance')
    same(construction.get('kind'), 'new ordinary source construction, custom deterministic coherent Condition/clock; no prior native replay', 'construction category')
    for key in ('subject_constructors', 'real_threads', 'native_peers'):
        same(construction.get(key), 0, 'constructor-free saved control: ' + key)
    rows = construction.get('rows')
    require(type(rows) is list and len(rows) == 9, 'nine fixed source-control rows')
    expected = {(p, s): 'TimeoutError' for p in POLICIES for s in ('quiescent', 'churn')}
    expected[('outside', 'churn')] = 'DiagnosticCutoff'
    expected.update({('outside', 'fresh_match'): 'MATCH', ('deadline_first', 'fresh_match'): 'TimeoutError',
                     ('baseline', 'match_oracle'): 'MATCH'})
    seen = set(); reports = []
    for row in rows:
        require(type(row) is dict, 'source-control row object')
        key = (row.get('policy'), row.get('scenario'))
        require(key in expected and key not in seen, 'source-control unique fixed roster')
        seen.add(key); same(row.get('outcome'), expected[key], 'source-control exact outcome: ' + str(key))
        require(type(row.get('queue_methods')) is list and all(type(s) is str for s in row['queue_methods']), 'queue method strings')
        expected_queue = [] if expected[key] == 'MATCH' else ['target'] if key == ('deadline_first', 'fresh_match') else ['noise']
        same(row['queue_methods'], expected_queue, 'source-control complete queue result')
        for field in ('A_predicate_calls', 'B_predicate_calls', 'A_condition_entries'):
            integer(row.get(field), 'source-control typed counter: ' + field)
        policy, scenario = key
        expected_trace = expected_control_trace(policy, scenario)
        same(row.get('trace'), expected_trace, 'independently reconstructed exact control trace: ' + str(key))
        emitted_A = sum(t['event'] == 'CONDITION_ENTER' and t['actor'] == 'A' for t in expected_trace)
        consumes = sum(t['event'] == 'B_CONSUME' for t in expected_trace)
        same(row['A_condition_entries'], emitted_A + (1 if expected[key] == 'DiagnosticCutoff' else 0), 'attempted/emitted A entry counter join')
        same(row['B_predicate_calls'], consumes, 'B predicate / actual consumes join')
        expected_A_calls = 8 if expected[key] == 'DiagnosticCutoff' else 2 if key == ('outside', 'fresh_match') else 1
        same(row['A_predicate_calls'], expected_A_calls, 'A false/matching scan count')
        A_clock_reads = 1 if expected[key] in ('MATCH', 'DiagnosticCutoff') else 2
        same(row.get('clock_reads'), A_clock_reads + consumes, 'shared A/B mocked clock reads')
        readings = [0.0]
        if scenario == 'fresh_match':
            readings.append(.25)
            if policy == 'deadline_first': readings.append(2.0)
        elif expected[key] == 'DiagnosticCutoff':
            readings.extend([2.0] * 8)
        elif expected[key] != 'MATCH':
            readings.extend([2.0] * (1 + consumes))
        same(row.get('clock_readings'), readings, 'actual source-derived monotonic return ledger: ' + str(key))
        same(len(readings), row['clock_reads'], 'clock ledger/read count join')
        require(all(left <= right for left, right in zip(readings, readings[1:])), 'coherent monotonic returned clock')
        timeline = [t['logical_seconds'] if t['event'] == 'CONDITION_ENTER' else t['logical_arrival_seconds']
                    for t in expected_trace if t['event'] in ('CONDITION_ENTER', 'R_FRESH_APPEND')]
        require(all(left <= right for left, right in zip(timeline, timeline[1:])), 'coherent Condition/arrival chronology')
        if scenario == 'fresh_match':
            same(timeline[:5], [0.0, .25, .25, .5, 2.0], 'fresh A snapshot/B consume/R arrival/A recheck causal chronology')
            require(readings[1] < .5 < 1 < timeline[4], 'fresh arrival before A deadline and expired A reacquisition')
        for field, value in [('synthetic_timeout_seconds', 1), ('synthetic_initial_seconds', 0), ('synthetic_reacquisition_seconds', 2)]:
            same(row.get(field), value, 'synthetic clock constant: ' + field)
        expected_detail = {'method': 'target'} if expected[key] == 'MATCH' else '8 completed false snapshot/recheck cycles; not an infinite execution claim' if expected[key] == 'DiagnosticCutoff' else 'app-server notification timed out'
        same(row.get('detail'), expected_detail, 'exact first source-control detail')
        reports.append({'policy': policy, 'scenario': scenario, 'outcome': row['outcome'],
                        'A_condition_entries': row['A_condition_entries'], 'successful_A_entries': emitted_A,
                        'B_consumes': consumes, 'clock_reads': row['clock_reads'],
                        'clock_readings': readings, 'coherent_synthetic_chronology': True})
    same(seen, set(expected), 'complete source-control roster')
    same([(r['policy'], r['scenario']) for r in rows],
         [(p, s) for s in ('quiescent', 'churn') for p in POLICIES]
         + [('outside', 'fresh_match'), ('deadline_first', 'fresh_match'), ('baseline', 'match_oracle')],
         'source construction exact first-row order')
    return reports


def audit_document(model, construction, root):
    root = Path(root)
    require(type(model) is dict and type(construction) is dict, 'saved document root objects')
    finite(model); finite(construction)
    spec = document(root / 'SPEC.json')
    pins, sources = verify_sources(root, spec)
    control_raw = raw_file(root / 'source_controls.py')
    same((len(control_raw), digest(control_raw)), EXPECTED_CONTROL_SOURCE, 'coherent V2 source-control helper bytes')
    same(model.get('schema'), 'predicate-retry-model-v1', 'model schema')
    same(model.get('allocation'), spec.get('allocation'), 'model allocation')
    same(model.get('source_pins'), pins, 'model full source pins')
    same(model.get('decision'), DECISION, 'producer restricted decision')
    require(type(model.get('limits')) is list and all(type(s) is str for s in model['limits']), 'model limits strings')
    graphs = model.get('graphs')
    require(type(graphs) is list and len(graphs) == 3, 'three policy graphs')
    require(all(type(g) is dict for g in graphs), 'graph objects')
    same([g.get('policy') for g in graphs], list(POLICIES), 'ordered model policy roster')
    graph_reports = [check_graph(g) for g in graphs]
    control_reports = check_construction(construction, pins)
    return {'schema': 'predicate-retry-independent-saved-audit-v1', 'passed': True, 'decision': DECISION,
            'source_pins': pins, 'source_sha256': sources, 'other_method_AST_unchanged': True,
            'independent_algorithm': 'expanded actor-bit closed-walk BFS; no Tarjan/SCC or generator import',
            'graphs': graph_reports, 'construction': control_reports,
            'spec_sha256': digest(raw_file(root / 'SPEC.json')),
            'source_pins_sha256': digest(raw_file(root / 'source-pins.json')),
            'source_control_helper_sha256': digest(control_raw),
            'checker_sha256': digest(Path(__file__).read_bytes()),
            'limits': ['Exact finite expired all-false equality quotient only; model traces are not native scheduling observations.',
                       'Two comparison-relevant handles, not a total live Python-reference bound for every schedule.',
                       'Source controls are saved records; no subject/generator/constructor/thread/native execution by auditor.',
                       'Saved consistency/source identity, not evidence authenticity or universal adversarial coverage.',
                       'Fresh-match loss qualifies simple alternative; no production adoption or whole-call/task-effect claim.']}
