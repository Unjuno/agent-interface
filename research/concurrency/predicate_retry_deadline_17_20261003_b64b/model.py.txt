import json,pathlib,collections,hashlib,sys
ROOT=pathlib.Path(__file__).resolve().parent
def canon(pc,q,snap):
    names={}
    def name(x):
        if x is None:return None
        if x not in names:names[x]=len(names)
        return names[x]
    s=name(snap);q=name(q)
    return {'pc':pc,'q':q,'snap':s}
def key(s):return (s['pc'],s['q'],s['snap'])
def successors(s,policy):
    pc,q,sn=s['pc'],s['q'],s['snap'];out=[]
    if pc=='DONE':return out
    if policy=='baseline':out.append(('A','A_ATOMIC_TIMEOUT',canon('DONE',q,sn)))
    elif pc=='SNAP':out.append(('A','A_SNAPSHOT',canon('SCAN',q,q)))
    elif pc=='SCAN':out.append(('A','A_FALSE_SCAN',canon('CHECK',q,sn)))
    elif pc=='CHECK':
        if policy=='outside' and q!=sn:out.append(('A','A_RETRY',canon('SNAP',q,sn)))
        else:out.append(('A','A_TIMEOUT',canon('DONE',q,sn)))
    if q is not None:out.append(('B','B_CONSUME',canon(pc,None,sn)))
    else:out.append(('R','R_FRESH_APPEND',canon(pc,0 if sn is None else 1,sn)))
    return out
def graph(policy):
    states=[canon('SNAP',0,None)];indexes={key(states[0]):0};edges=[];todo=collections.deque([0])
    while todo:
        i=todo.popleft()
        for actor,event,state in successors(states[i],policy):
            k=key(state)
            if k not in indexes:indexes[k]=len(states);states.append(state);todo.append(indexes[k])
            edges.append({'from':i,'to':indexes[k],'actor':actor,'event':event})
    return {'policy':policy,'states':states,'edges':edges,'initial':0}
def components(g):
    adj=collections.defaultdict(list)
    for e in g['edges']:
        if g['states'][e['to']]['pc']!='DONE':adj[e['from']].append(e['to'])
    index={};low={};stack=[];on=set();result=[]
    def visit(v):
        index[v]=low[v]=len(index);stack.append(v);on.add(v)
        for w in adj[v]:
            if w not in index:visit(w);low[v]=min(low[v],low[w])
            elif w in on:low[v]=min(low[v],index[w])
        if low[v]==index[v]:
            c=[]
            while True:
                w=stack.pop();on.remove(w);c.append(w)
                if w==v:break
            result.append(set(c))
    for v,s in enumerate(g['states']):
        if s['pc']!='DONE' and v not in index:visit(v)
    return result
def witness(g):
    for c in components(g):
        es=[e for e in g['edges'] if e['from'] in c and e['to'] in c]
        if {e['actor'] for e in es}!={'A','B','R'}:continue
        for start in sorted(c):
            todo=collections.deque([(start,0,[])]);seen={(start,0)}
            while todo:
                v,mask,path=todo.popleft()
                for e in es:
                    if e['from']!=v:continue
                    m=mask|{'A':1,'B':2,'R':4}[e['actor']];p=path+[e]
                    if e['to']==start and m==7:return {'start':start,'edges':p}
                    k=(e['to'],m)
                    if k not in seen:seen.add(k);todo.append((e['to'],m,p))
    return None
def main():
    spec=json.loads((ROOT/'SPEC.json').read_text());pins=json.loads((ROOT/'source-pins.json').read_text())
    for name,p in pins['paths'].items():
        b=(ROOT/name).read_bytes();assert len(b)==p['bytes'] and hashlib.sha256(b).hexdigest()==p['sha256']
    graphs=[]
    for policy in spec['policies']:
        g=graph(policy);g['lasso']=witness(g);g['fair_lasso']=g['lasso'] is not None;graphs.append(g)
    assert [g['fair_lasso'] for g in graphs]==[False,True,False]
    result={'schema':'predicate-retry-model-v1','allocation':spec['allocation'],'source_pins':pins,'graphs':graphs,'decision':'PASS_SCOPED_EXISTENTIAL_RETRY_COUNTEREXAMPLE','limits':spec['limits']}
    out=pathlib.Path(sys.argv[1]);out.write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps({'decision':result['decision'],'graphs':[{'policy':g['policy'],'states':len(g['states']),'edges':len(g['edges']),'fair_lasso':g['fair_lasso']} for g in graphs]},sort_keys=True))
if __name__=='__main__':main()
