"""Exclusive native repair receipts, with original source/raw preserved."""
from pathlib import Path
import datetime, hashlib, json, subprocess, sys

here=Path(__file__).resolve().parent
stage,*argv=sys.argv[1:]
for suffix in ('.stdout','.stderr','.receipt.json'):assert not (here/(stage+suffix)).exists()
start=datetime.datetime.now(datetime.timezone.utc).isoformat()
child=subprocess.Popen(argv,cwd=here,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
stdout,stderr=child.communicate(timeout=120)
end=datetime.datetime.now(datetime.timezone.utc).isoformat()
for suffix,raw in [('.stdout',stdout),('.stderr',stderr)]:
    with (here/(stage+suffix)).open('xb') as f:f.write(raw)
record=dict(argv=argv,cwd=str(here),started_utc=start,ended_utc=end,pid=child.pid,exit_code=child.returncode,
    check_sha256=hashlib.sha256((here/'check_repair.py').read_bytes()).hexdigest(),
    verifier_sha256=hashlib.sha256((here/'repair_v2/audit_v2.py').read_bytes()).hexdigest() if (here/'repair_v2/audit_v2.py').exists() else None,
    interpreter_sha256=hashlib.sha256(Path(argv[0]).read_bytes()).hexdigest(),
    stdout_sha256=hashlib.sha256(stdout).hexdigest(),stderr_sha256=hashlib.sha256(stderr).hexdigest(),
    scope='ordinary retained-JSON repair; no producer/original audit main/formal replay')
with (here/(stage+'.receipt.json')).open('x',encoding='utf-8') as f:json.dump(record,f,sort_keys=True,indent=2);f.write('\n')
print(json.dumps(record));print(stderr.decode('utf-8','replace')[-4000:])
raise SystemExit(child.returncode)
