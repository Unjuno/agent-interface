"""Ordinary pure saved-record ownership checks; no original main invocation."""
import ast,copy,hashlib,json,pathlib,sys,unittest
ROOT=pathlib.Path(__file__).resolve().parent
OUT=pathlib.Path(sys.argv[1]);SOURCE=ROOT/'audit_v2.py.txt';sys.argv=sys.argv[:1]
def sha(raw):return hashlib.sha256(raw).hexdigest()
def encoded(value):return json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=True,allow_nan=False).encode()
def selected(path):
    nodes=[n for n in ast.parse(path.read_bytes()).body if isinstance(n,ast.FunctionDef) and n.name in ('check','integer','inspect','inspect_all')]
    space={'__name__':'selected_saved_ownership'};exec(compile(ast.Module(body=nodes,type_ignores=[]),'saved-definitions','exec'),space);return space
AUDIT=selected(SOURCE);OLD=selected(ROOT/'inputs/audit.py.txt')
RAW_BYTES=(ROOT/'inputs/raw.jsonl').read_bytes();RAW=[json.loads(line) for line in RAW_BYTES.splitlines()];SPECS=json.loads((ROOT/'inputs/input.json').read_bytes());EXPECTED=OLD['inspect_all'](RAW,SPECS);records=[]
def relabel_fd(row,role,value):
    row=copy.deepcopy(row)
    if role=='SELECTOR':row['events'][0]['selector_fd']=value
    else:row['events'][0]['fds'][role]['fd']=value
    for event in row['events']:
        if event['tag']=='poll' and role in ('DR','CR'):
            for key in event['ready']:
                if key['role']+'R'==role:key['fd']=value
        if event['tag']=='closed' and event['role']==role:event['fd']=value
    return row
def second_open(row,placement):
    row=copy.deepcopy(row);opened=copy.deepcopy(row['events'][0]);at=len(row['events']) if placement=='after-close' else 1
    highest=max([v['fd'] for v in opened['fds'].values()]+[opened['selector_fd']]);base=highest+1
    for i,role in enumerate(('DR','DW','CR','CW')):opened['fds'][role]['fd']=base+i;opened['fds'][role]['inode']+=1000000
    opened['selector_fd']=base+4;opened['t']=row['events'][at-1]['t'];row['events'].insert(at,opened)
    for i,e in enumerate(row['events']):e['seq']=i
    return row
def evaluate(row,label,expected_refusal):
    before=OLD['inspect'](row,SPECS[row['spec']['id']]);reason=None
    try:after=AUDIT['inspect'](row,SPECS[row['spec']['id']])
    except ValueError as exc:after=None;reason=str(exc)
    record=dict(label=label,input_sha256=sha(encoded(row)+b'\n'),old_result=before,new_result=after,new_refusal=reason,expected_refusal=expected_refusal);records.append(record)
    (OUT/'copies'/(label.replace('/','__')+'.json')).write_bytes(encoded(row)+b'\n')
    return after,reason
class OwnershipRepair(unittest.TestCase):
    def test_original15_typed_results_preserved(self):
        self.assertEqual(sha(RAW_BYTES),'242439e8d7a96f2c938668955c5469e40e1ccaeb97a754d1f1295e07f156efa0')
        actual=AUDIT['inspect_all'](RAW,SPECS);self.assertEqual(encoded(actual),encoded(EXPECTED));records.append(dict(label='original15',input_sha256=sha(RAW_BYTES),result=actual))
    def test_exact_three_discovered_witnesses_refused(self):
        for name in ('negative_control_read_fd','negative_selector_fd','unclosed_second_open'):
            data=(ROOT/'internal-analysis'/(name+'.json')).read_bytes();row=json.loads(data)
            with self.subTest(name=name):
                after,reason=evaluate(row,'retained/'+name,True);self.assertIsNone(after);self.assertIsNotNone(reason)
        for name in ('negative_control_read_fd_full15','unclosed_second_open_full15'):
            data=(ROOT/'internal-analysis'/(name+'.jsonl')).read_bytes();rows=[json.loads(line) for line in data.splitlines()]
            with self.subTest(full15=name):
                self.assertEqual(encoded(OLD['inspect_all'](rows,SPECS)),encoded(EXPECTED))
                with self.assertRaises(ValueError):AUDIT['inspect_all'](rows,SPECS)
    def test_all75_negative_fd_copies_refused(self):
        for row in RAW:
            for role in ('DR','DW','CR','CW','SELECTOR'):
                with self.subTest(row=row['spec']['id'],role=role):
                    after,reason=evaluate(relabel_fd(row,role,-1),f"negative-fd/{row['spec']['id']}/{role}",True);self.assertIsNone(after);self.assertIsNotNone(reason)
    def test_all30_extra_open_copies_refused(self):
        for row in RAW:
            for placement in ('after-close','after-first-open'):
                with self.subTest(row=row['spec']['id'],placement=placement):
                    after,reason=evaluate(second_open(row,placement),f"extra-open/{row['spec']['id']}/{placement}",True);self.assertIsNone(after);self.assertIsNotNone(reason)
    def test_all75_zero_fd_domain_positives_preserved(self):
        for row in RAW:
            for role in ('DR','DW','CR','CW','SELECTOR'):
                with self.subTest(row=row['spec']['id'],role=role):
                    after,reason=evaluate(relabel_fd(row,role,0),f"valid-zero-fd/{row['spec']['id']}/{role}",False);self.assertIsNone(reason);self.assertEqual(encoded(after),encoded(EXPECTED[row['spec']['id']]))
    def test_existing11_corruptions_remain_rejected(self):
        summary=json.loads((ROOT/'internal-analysis/RESULT.json').read_bytes())
        controls=[x for x in summary['controls'] if x['accepted'] is False];self.assertEqual(len(controls),11)
        for item in controls:
            row=json.loads((ROOT/'internal-analysis'/(item['name']+'.json')).read_bytes())
            with self.subTest(name=item['name']):
                with self.assertRaises(ValueError):AUDIT['inspect'](row,SPECS[row['spec']['id']])
                records.append(dict(label='existing-rejection/'+item['name'],input_sha256=sha(encoded(row)+b'\n'),rejected=True))
if __name__=='__main__':
    OUT.mkdir(parents=True,exist_ok=False);(OUT/'copies').mkdir()
    result=unittest.TextTestRunner(verbosity=2).run(unittest.defaultTestLoader.loadTestsFromTestCase(OwnershipRepair))
    (OUT/'rows.json').write_text(json.dumps(records,indent=2)+'\n')
    print(json.dumps(dict(methods=result.testsRun,failures=len(result.failures),errors=len(result.errors),records=len(records),source_sha256=sha(SOURCE.read_bytes()),test_sha256=sha(pathlib.Path(__file__).read_bytes()),original_raw_sha256=sha(RAW_BYTES),formal_native_or_producer_invocations=0)))
    sys.exit(0 if result.wasSuccessful() else 1)
