"""Separate byte-ledger and causal raw reconstruction; no candidate import."""
import hashlib
import json
import pathlib
import sys


def check(condition, reason):
    if not condition:
        raise ValueError(reason)


def integer(value):
    return type(value) is int


def inspect(record, spec):
    check(record.keys() == {'spec','events'} and record['spec'] == spec, 'row/spec')
    # Equality alone aliases bool/float to integer. Check the recursively typed spec.
    for key in ['id','budget','cancel_after']:
        check(integer(record['spec'][key]), 'typed spec')
    check(type(record['spec']['profile']) is str, 'profile type')
    events = record['events']
    check(type(events) is list and len(events) > 5, 'events')
    schemas = {
        'open':{'fds','selector_fd'}, 'write':{'role','data','wrote','cause_read'},
        'poll':{'cycle','ready'}, 'read':{'role','data','cycle','count'},
        'batch_end':{'cycle','reads','reason'}, 'terminal':{'kind','read_count','cycles'},
        'cleanup_read':{'role','data'}, 'cleanup_empty':{'role'}, 'closed':{'role','fd','err'}
    }
    for i, event in enumerate(events):
        tag = event.get('tag')
        check(tag in schemas and event.keys() == schemas[tag] | {'tag','seq','t'}, 'event schema')
        check(tag != 'open' or i == 0, 'single opening event')
        check(integer(event['seq']) and event['seq'] == i and integer(event['t']) and event['t'] >= 0, 'sequence/time type')
        check(i == 0 or event['t'] >= events[i-1]['t'], 'clock reversal')
        for field in ['cycle','count','reads','read_count','cycles','fd','err','wrote','cause_read','selector_fd']:
            if field in event:
                check(integer(event[field]), 'integer:'+field)
    first = events[0]
    check(first['tag'] == 'open' and first['fds'].keys() == {'DR','DW','CR','CW'}, 'open roles')
    check(first['selector_fd'] >= 0, 'nonnegative selector fd')
    fdmap = first['fds']
    for value in fdmap.values():
        check(value.keys() == {'fd','inode'} and integer(value['fd']) and value['fd'] >= 0 and integer(value['inode']) and value['inode'] > 0, 'fd identity')
    check(len({v['fd'] for v in fdmap.values()} | {first['selector_fd']}) == 5, 'unique owned fds')
    check(fdmap['DR']['inode'] == fdmap['DW']['inode'] and fdmap['CR']['inode'] == fdmap['CW']['inode'] and fdmap['DR']['inode'] != fdmap['CR']['inode'], 'pipe inode pairs')
    pending = {'D':0, 'C':0}
    writes = {'D':0, 'C':0}
    read_count = 0
    last_poll = None
    cycle = 0
    batch_reads = 0
    need_batch_end = False
    terminal = None
    cleanup_empty = set()
    closed = set()
    previous = first
    for event in events[1:]:
        tag = event['tag']
        if tag in ('write','poll','read','batch_end'):
            check(terminal is None, 'primary after terminal')
        if tag == 'write':
            role = event['role']
            check(role in pending, 'write role')
            data = bytes.fromhex(event['data'])
            check(data and data == (b'D' if role == 'D' else b'C') * len(data) and event['wrote'] == len(data), 'write byte/size')
            check(event['cause_read'] == read_count, 'write cause')
            if role == 'D':
                if writes['D'] == 0:
                    check(read_count == 0 and data == (b'DDDD' if spec['profile'] in ('cold_late','data_only') else b'D'), 'initial data')
                else:
                    check(spec['profile'].startswith('hot') and data == b'D', 'hot refill only')
                    check(previous['tag'] == 'read' and previous['role'] == 'D' or previous['tag'] == 'write' and previous['role'] == 'C', 'refill ordering')
                    check(writes['D'] == read_count, 'exactly one refill per read')
            else:
                check(writes['C'] == 0 and data == b'C', 'single cancel')
                check(read_count == (0 if spec['profile'] == 'co_ready' else spec['cancel_after']), 'cancel injection read')
                check(read_count == 0 or previous['tag'] == 'read' and previous['role'] == 'D' and previous['count'] == read_count, 'cancel after read')
            writes[role] += 1
            pending[role] += len(data)
        elif tag == 'poll':
            check(not need_batch_end and event['cycle'] == cycle + 1 and cycle < 32, 'poll order/batch')
            cycle = event['cycle']
            ready = event['ready']
            check(type(ready) is list, 'ready list')
            seen = set()
            for key in ready:
                check(key.keys() == {'role','fd','mask','inode'} and key['role'] in pending, 'ready schema')
                role = key['role']
                check(role not in seen, 'duplicate readiness')
                seen.add(role)
                check(all(integer(key[f]) for f in ['fd','mask','inode']) and key['mask'] == 1, 'ready typed mask')
                check(key['fd'] == fdmap[role+'R']['fd'] and key['inode'] == fdmap[role+'R']['inode'], 'ready identity')
            check(seen == {r for r,n in pending.items() if n}, 'actual readiness byte ledger')
            last_poll = seen
            batch_reads = 0
        elif tag == 'read':
            role = event['role']
            check(last_poll is not None and role in last_poll and event['cycle'] == cycle, 'poll/read join')
            check(pending[role] > 0 and event['data'] == ('44' if role == 'D' else '43'), 'read conservation')
            pending[role] -= 1
            if role == 'D':
                check('C' not in last_poll, 'control priority')
                read_count += 1
                batch_reads += 1
                need_batch_end = True
                check(spec['budget'] == 0 or batch_reads <= spec['budget'], 'per-poll budget')
            else:
                check('C' in last_poll, 'control ready')
            check(event['count'] == read_count, 'read count')
        elif tag == 'batch_end':
            check(event['cycle'] == cycle and event['reads'] == batch_reads and need_batch_end, 'batch join')
            reason = event['reason']
            check(reason in ('eagain','budget','cutoff'), 'batch reason')
            if reason == 'eagain':
                check(pending['D'] == 0, 'EAGAIN with data pending')
            elif reason == 'budget':
                check(spec['budget'] > 0 and batch_reads == spec['budget'] and read_count < 16, 'budget return')
            else:
                check(read_count == 16, 'fixed diagnostic cutoff')
            need_batch_end = False
        elif tag == 'terminal':
            check(terminal is None and not need_batch_end and event['read_count'] == read_count and event['cycles'] == cycle, 'terminal join')
            terminal = event
            if event['kind'] == 'CANCEL_ACK':
                check(previous['tag'] == 'read' and previous['role'] == 'C' and pending['C'] == 0, 'ACK custody')
            elif event['kind'] == 'EMPTY':
                check(last_poll == set() and previous['tag'] == 'poll', 'empty terminal')
            else:
                check(event['kind'] == 'CUTOFF' and previous['tag'] == 'batch_end' and previous['reason'] == 'cutoff', 'cutoff custody')
        elif tag == 'cleanup_read':
            role = event['role']
            check(terminal is not None and role in pending and role not in cleanup_empty and not closed, 'cleanup phase')
            data = bytes.fromhex(event['data'])
            check(data and data == (b'D' if role == 'D' else b'C') * len(data) and pending[role] >= len(data), 'cleanup byte conservation')
            pending[role] -= len(data)
        elif tag == 'cleanup_empty':
            role = event['role']
            check(terminal is not None and role in pending and pending[role] == 0 and role not in cleanup_empty and not closed, 'cleanup empty')
            cleanup_empty.add(role)
        elif tag == 'closed':
            role = event['role']
            check(cleanup_empty == {'D','C'} and role in {'DR','DW','CR','CW','SELECTOR'} and role not in closed, 'close phase/role')
            check(event['fd'] == (first['selector_fd'] if role == 'SELECTOR' else fdmap[role]['fd']) and event['err'] == 9, 'close witness')
            closed.add(role)
        previous = event
    check(closed == {'DR','DW','CR','CW','SELECTOR'} and terminal is not None, 'complete cleanup')
    profile = spec['profile']
    budget = spec['budget']
    if profile.startswith('hot'):
        expected_reads = 16 if budget == 0 else ((spec['cancel_after'] - 1)//budget + 1)*budget
        expected_kind = 'CUTOFF' if budget == 0 else 'CANCEL_ACK'
        check(writes == {'D':1+read_count,'C':1}, 'hot exact writes')
    elif profile == 'co_ready':
        expected_reads, expected_kind = 0, 'CANCEL_ACK'
        check(writes == {'D':1,'C':1}, 'co-ready writes')
    elif profile == 'cold_late':
        expected_reads, expected_kind = (1 if budget == 1 else 4), 'CANCEL_ACK'
        check(writes == {'D':1,'C':1}, 'cold cancel writes')
    else:
        expected_reads, expected_kind = 4, 'EMPTY'
        check(profile == 'data_only' and writes == {'D':1,'C':0}, 'data-only writes')
    check(read_count == expected_reads and terminal['kind'] == expected_kind, 'prospective decision')
    return {'id':spec['id'],'profile':profile,'budget':budget,'reads':read_count,'terminal':terminal['kind']}


def inspect_all(raw, specs):
    check(len(raw) == len(specs) == 15, 'complete15 denominator')
    return [inspect(record, spec) for record, spec in zip(raw,specs)]


if __name__ == '__main__':
    raw_path = pathlib.Path('/out/raw.jsonl')
    specs = json.loads(pathlib.Path('/source/input.json').read_text())
    raw = [json.loads(line) for line in raw_path.read_text().splitlines()]
    result = {'scope':'directed native byte/read-budget boundary only','rows':inspect_all(raw,specs),'raw_sha256':hashlib.sha256(raw_path.read_bytes()).hexdigest()}
    pathlib.Path('/out/audit.json').write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps({'audited_rows':len(result['rows']),'raw_sha256':result['raw_sha256']}))
