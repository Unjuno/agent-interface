"""Finite copied-JSON controls; AST-load pure audit defs only, never any main."""
import ast
import copy
import hashlib
import json
from pathlib import Path

ARTIFACT = Path('<OWNED_WORKSPACE>/outputs/hot-drain-17/artifact')
OUT = Path(__file__).resolve().parent


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(name, value):
    path = OUT / name
    path.write_text(json.dumps(value, indent=2, sort_keys=True) + '\n')
    return dict(path=str(path), sha256=sha(path))


def pure_audit():
    tree = ast.parse((ARTIFACT / 'audit.py.txt').read_text())
    expected = {'check', 'integer', 'inspect', 'inspect_all'}
    defs = [node for node in tree.body if isinstance(node, ast.FunctionDef)]
    if {node.name for node in defs} != expected:
        raise RuntimeError('unexpected function definitions')
    if any(node.decorator_list for node in defs):
        raise RuntimeError('unexpected decorators')
    namespace = {}
    exec(compile(ast.Module(body=defs, type_ignores=[]), str(ARTIFACT / 'audit.py.txt') + ':pure_defs', 'exec'), namespace)
    return namespace


def renumber(record):
    for seq, event in enumerate(record['events']):
        event['seq'] = seq
        event['t'] = seq


def fd_rewrite(record, role, value):
    first = record['events'][0]
    if role == 'SELECTOR':
        first['selector_fd'] = value
    else:
        first['fds'][role]['fd'] = value
    for event in record['events']:
        if event['tag'] == 'closed' and event['role'] == role:
            event['fd'] = value
        if event['tag'] == 'poll' and role.endswith('R'):
            for ready in event['ready']:
                if ready['role'] == role[0]:
                    ready['fd'] = value


def first(record, tag, role=None):
    return next(event for event in record['events'] if event['tag'] == tag and (role is None or event.get('role') == role))


def run():
    audit = pure_audit()
    specs = json.loads((ARTIFACT / 'input.json').read_text())
    raw = [json.loads(line) for line in (ARTIFACT / 'results/raw.jsonl').read_text().splitlines()]
    baseline = audit['inspect_all'](copy.deepcopy(raw), copy.deepcopy(specs))
    report = dict(scope='bounded internal copied-JSON controls; no formal run or eligible approval', baseline_rows=baseline,
                  source_hashes={name: sha(ARTIFACT / name) for name in ['audit.py.txt', 'input.json', 'PLAN.md', 'FREEZE.json', 'results/raw.jsonl']},
                  controls=[])

    def case(name, row_id, mutate, expected):
        changed = copy.deepcopy(raw[row_id])
        mutate(changed)
        stored = save(name + '.json', changed)
        try:
            result = audit['inspect'](changed, copy.deepcopy(specs[row_id]))
        except (ValueError, KeyError, TypeError) as exc:
            outcome = dict(accepted=False, exception=type(exc).__name__, reason=str(exc))
        else:
            outcome = dict(accepted=True, result=result)
        report['controls'].append(dict(name=name, row_id=row_id, expected=expected, **stored, **outcome))

    case('negative_control_read_fd', 6, lambda r: fd_rewrite(r, 'CR', -1), 'reject impossible Linux FD identity')
    case('negative_selector_fd', 6, lambda r: fd_rewrite(r, 'SELECTOR', -1), 'reject impossible Linux FD identity')
    case('float_control_read_fd', 6, lambda r: fd_rewrite(r, 'CR', 6.0), 'reject typed identity alias')
    case('boolean_selector_fd', 6, lambda r: fd_rewrite(r, 'SELECTOR', False), 'reject typed identity alias')
    case('wrong_control_write_byte', 6, lambda r: first(r, 'write', 'C').update(data='44'), 'reject wrong command byte')
    case('wrong_control_read_byte', 6, lambda r: first(r, 'read', 'C').update(data='44'), 'reject wrong command byte')
    case('false_ack_at_cutoff', 0, lambda r: first(r, 'terminal').update(kind='CANCEL_ACK'), 'reject cleanup mistaken for ACK')

    def premature_cleanup(r):
        events = r['events']
        a = next(i for i,e in enumerate(events) if e['tag'] == 'terminal')
        b = next(i for i,e in enumerate(events) if e['tag'] == 'cleanup_read' and e['role'] == 'C')
        events[a], events[b] = events[b], events[a]
        renumber(r)
    case('cleanup_before_terminal', 0, premature_cleanup, 'reject primary/cleanup phase crossing')

    def omit_pending_control(r):
        r['events'] = [e for e in r['events'] if not (e['tag'] == 'cleanup_read' and e['role'] == 'C')]
        renumber(r)
    case('omit_pending_control_cleanup', 0, omit_pending_control, 'reject missing pending command byte')

    def hidden_control(r):
        event = next(e for e in r['events'] if e['tag'] == 'poll' and e['cycle'] == 2)
        event['ready'] = [key for key in event['ready'] if key['role'] != 'C']
    case('pending_control_hidden_from_poll', 1, hidden_control, 'reject pending-control accounting mismatch')

    def early_cutoff(r):
        r['events'] = [e for e in r['events'] if not (e['tag'] == 'read' and e.get('role') == 'D' and e['count'] == 16)
                       and not (e['tag'] == 'write' and e.get('role') == 'D' and e['cause_read'] == 16)]
        first(r, 'batch_end').update(reads=15)
        first(r, 'terminal').update(read_count=15)
        renumber(r)
    case('cutoff_at15', 0, early_cutoff, 'reject wrong primary diagnostic cutoff')

    def service8(r):
        r['events'] = [e for e in r['events'] if not (e['tag'] == 'batch_end' and e['cycle'] == 1)
                       and not (e['tag'] == 'poll' and e['cycle'] == 2)]
        for event in r['events']:
            if 'cycle' in event and event['cycle'] >= 2:
                event['cycle'] -= 1
            if event['tag'] == 'batch_end':
                event['reads'] = 8
            if event['tag'] == 'terminal':
                event['cycles'] -= 1
        renumber(r)
    case('budget4_service8_same_poll', 5, service8, 'reject per-poll service above fixed k')

    def primary_after(r):
        first(r, 'cleanup_read', 'C').update(tag='read', count=16, cycle=1)
    case('primary_control_read_after_cutoff', 0, primary_after, 'reject primary after terminal')

    # Recoverable denominator-preserving example: replace one row only.
    changed_raw = copy.deepcopy(raw)
    fd_rewrite(changed_raw[6], 'CR', -1)
    lines = ''.join(json.dumps(record, sort_keys=True) + '\n' for record in changed_raw)
    path = OUT / 'negative_control_read_fd_full15.jsonl'
    path.write_text(lines)
    report['denominator_preserving_counterexample'] = dict(path=str(path), sha256=sha(path),
        rows=audit['inspect_all'](copy.deepcopy(changed_raw), copy.deepcopy(specs)),
        changed_fields=['row6.events[0].fds.CR.fd: 6 -> -1', 'row6.events[3].ready[1].fd: 6 -> -1', 'row6.events[12].fd: 6 -> -1'])
    report['helper_source_sha256'] = sha(Path(__file__))
    print(json.dumps(report, indent=2, sort_keys=True))
    save('RESULT.json', report)


if __name__ == '__main__':
    try:
        run()
    except BaseException as exc:
        failure = OUT / 'FIRST_HELPER_FAILURE.json'
        if not failure.exists():
            save(failure.name, dict(exception=type(exc).__name__, reason=str(exc), helper_source_sha256=sha(Path(__file__))))
        raise
