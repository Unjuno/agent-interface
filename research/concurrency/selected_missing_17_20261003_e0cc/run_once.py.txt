"""First model then first saved oracle; exclusive intent files, owned PID/terminal receipts."""
import datetime,hashlib,json,os,pathlib,resource,subprocess,sys
ROOT=pathlib.Path(__file__).resolve().parent
def limits():
    resource.setrlimit(resource.RLIMIT_CPU,(10,10));resource.setrlimit(resource.RLIMIT_FSIZE,(1048576,1048576))
def run(phase):
    path=ROOT/(phase+'.intent.json');intent={'phase':phase,'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'argv':[sys.executable,'-B',str(ROOT/(phase+'.py.txt'))]}
    with path.open('x') as f:json.dump(intent,f)
    env=os.environ.copy();env.update(OMP_NUM_THREADS='1',OPENBLAS_NUM_THREADS='1',PYTHONDONTWRITEBYTECODE='1')
    child=subprocess.Popen(intent['argv'],cwd=ROOT,stdout=subprocess.PIPE,stderr=subprocess.PIPE,env=env,preexec_fn=limits)
    receipt=dict(intent,pid=child.pid,exit_code=None);(ROOT/(phase+'.receipt.json')).write_text(json.dumps(receipt,indent=2)+'\n')
    try:out,err=child.communicate(timeout=20)
    except subprocess.TimeoutExpired:
        child.kill();out,err=child.communicate();receipt['containment_timeout']=True
    for stream,data in [('stdout',out),('stderr',err)]:
        (ROOT/(phase+'.'+stream)).write_bytes(data);receipt[stream]={'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest()}
    receipt.update(exit_code=child.returncode,utc_end=datetime.datetime.now(datetime.timezone.utc).isoformat())
    (ROOT/(phase+'.receipt.json')).write_text(json.dumps(receipt,indent=2)+'\n');print(json.dumps(receipt))
    if len(out)>65536 or len(err)>65536 or child.returncode!=0:raise SystemExit(1)
    return receipt
if __name__=='__main__':
    run('model');run('audit_saved')
