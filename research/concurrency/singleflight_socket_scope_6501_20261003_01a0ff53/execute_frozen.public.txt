"""One bounded invocation pair against the already committed prospective freeze."""
import concurrent.futures.thread
import datetime
import hashlib
import json
import socket
import subprocess
import sys
import threading
from pathlib import Path

ROOT = Path('<PRIVATE_REPO>')
REL = 'research/concurrency/singleflight_socket_scope_6501_20261003_01a0ff53'
PKG = ROOT / REL
SOURCE = '36dcd20bf0b84bab9ce9b5d82398f9b351342efe'

def digest(blob):
    return hashlib.sha256(blob).hexdigest()

def utc():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()

def write(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2, sort_keys=True)
        stream.write('\n')

freeze = json.loads((PKG / 'FREEZE.json').read_text())
start = json.loads((PKG / 'START_GATE.json').read_text())
for name, expected in freeze['sha256'].items():
    blob = subprocess.check_output(['git', 'show', f'{SOURCE}:{REL}/{name}'], cwd=ROOT)
    if digest(blob) != expected or digest((PKG/name).read_bytes()) != expected:
        raise SystemExit('STOP source mismatch: ' + name)
if digest(Path(sys.executable).read_bytes()) != freeze['executable_sha256']:
    raise SystemExit('STOP executable mismatch')
for name, module in [('socket', socket), ('threading', threading), ('concurrent.futures.thread', concurrent.futures.thread)]:
    if digest(Path(module.__file__).read_bytes()) != start['stdlib_sha256'][name]:
        raise SystemExit('STOP stdlib mismatch: ' + name)
out = PKG / 'run-a01'
out.mkdir(exist_ok=False)
write(out/'PRE_RUN.json', dict(source_commit=SOURCE, observed_utc=utc(),
      launcher_sha256=digest(Path(__file__).read_bytes()),
      freeze_sha256=digest((PKG/'FREEZE.json').read_bytes()),
      start_gate_sha256=digest((PKG/'START_GATE.json').read_bytes()),
      source_and_dependencies_verified=True, allocation=freeze['run_id'], retries=0))

def invoke(label, argv):
    began = utc()
    with (out/f'{label}.stdout.txt').open('xb') as stdout, (out/f'{label}.stderr.txt').open('xb') as stderr:
        try:
            result = subprocess.run(argv, cwd=PKG, stdout=stdout, stderr=stderr,
                                    timeout=freeze['timeout_seconds'], check=False)
            exit_code, state = result.returncode, 'completed'
        except subprocess.TimeoutExpired:
            exit_code, state = None, 'timeout'
    record = dict(argv=[Path(argv[0]).name, *argv[1:]], cwd=REL, start_utc=began,
                  end_utc=utc(), actual_exit=exit_code, state=state,
                  executable_sha256=freeze['executable_sha256'], timeout_seconds=freeze['timeout_seconds'])
    write(out/f'{label}.execution.json', record)
    print(label, json.dumps(record, sort_keys=True), flush=True)
    return exit_code == 0

if not invoke('candidate', [sys.executable, '-B', 'study.py', 'fixtures.json', 'run-a01/raw.json']):
    raise SystemExit('STOP first candidate outcome retained')
if (out/'raw.json').stat().st_size > freeze['output_max_bytes']:
    raise SystemExit('STOP candidate output bound')
if not invoke('auditor', [sys.executable, '-B', 'audit.py', 'fixtures.json', 'run-a01/raw.json', 'run-a01/audit.json']):
    raise SystemExit('STOP first auditor outcome retained')
print('Primary pair completed; mutation gates remain pending.', flush=True)
