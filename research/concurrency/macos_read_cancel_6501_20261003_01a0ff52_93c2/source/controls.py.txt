"""Ordinary copy-only negative controls, never rerun the OS producer."""
from pathlib import Path
import argparse, copy, hashlib, json, shutil
import audit

def event(c,name):
    return next(e for e in c['events'] if e['name']==name)

def changes(raw):
    def primary(k,v,index):
        raw['cases'][index]['primary'][k]=v
        event(raw['cases'][index],'primary_checkpoint')['data']['snapshot'][k]=v
    return [
        ('boolean_not_integer',lambda:primary('task_done',1,6)),
        ('wrapper_native_future_cancelled',lambda:primary('future_cancelled',True,0)),
        ('control_reader_still_open',lambda:primary('reader_state','OPEN',4)),
        ('caller_close_not_started',lambda:primary('caller_close_started',False,2)),
        ('worker_actor_changed',lambda:event(raw['cases'][6],'read_return').update(tid=raw['cases'][6]['events'][0]['tid'],native_tid=raw['cases'][6]['events'][0]['native_tid'])),
        ('read_result_mismatch',lambda:raw['cases'][6]['final']['result'].update(byte_hex='45')),
        ('pipe_close_wrong_owner',lambda:next(e for e in raw['cases'][6]['events'] if e['name']=='fd_closed')['data'].update(role='worker_data_reader')),
        ('final_fd_still_open',lambda:raw['cases'][6]['final']['fds'].update({next(iter(raw['cases'][6]['final']['fds'])):'OPEN'})),
        ('event_sequence_gap',lambda:event(raw['cases'][6],'worker_return').update(seq=999)),
        ('allocation_changed',lambda:raw.update(allocation_id='unallocated-copy')),
    ]

def synchronize(directory,raw):
    for c in raw['cases']:
        # Explicit independent snapshots: do not rely on aliasing within Python dicts.
        event(c,'case_final')['data']['snapshot']=copy.deepcopy(c['final'])
        name=c['mode']+'-'+str(c['rep'])
        stdout=''.join(json.dumps({'event':e},sort_keys=True)+'\n' for e in c['events'])+json.dumps({'case':c},sort_keys=True)+'\n'
        (directory/(name+'.stdout.log')).write_text(stdout)
        receipt_path=directory/(name+'.receipt.json');receipt=json.loads(receipt_path.read_text())
        receipt['stdout_sha256']=hashlib.sha256(stdout.encode()).hexdigest();receipt_path.write_text(json.dumps(receipt,indent=2)+'\n')
    (directory/'raw.json').write_text(json.dumps(raw,indent=2)+'\n')

def run(directory,protocol,sha,out):
    baseline=audit.audit(directory,protocol,sha);out.mkdir(exist_ok=False,parents=True)
    original_hash=hashlib.sha256((directory/'raw.json').read_bytes()).hexdigest();results=[]
    for index in range(10):
        raw=json.loads((directory/'raw.json').read_text());name,mutate=changes(raw)[index]
        before=json.dumps(raw,sort_keys=True);mutate();audit.require(before!=json.dumps(raw,sort_keys=True),'effective mutation')
        target=out/name;shutil.copytree(directory,target);synchronize(target,raw)
        try:audit.audit(target,protocol,sha)
        except (ValueError,KeyError,TypeError,json.JSONDecodeError) as exc:results.append({'name':name,'rejected':True,'reason':str(exc)})
        else:results.append({'name':name,'rejected':False});raise RuntimeError('accepted semantic corruption '+name)
    audit.require(hashlib.sha256((directory/'raw.json').read_bytes()).hexdigest()==original_hash,'original unchanged')
    result={'classification':'ordinary retained-raw copy controls; zero OS replay','baseline':baseline['status'],'all_rejected':True,'count':len(results),'results':results,'raw_sha256':original_hash}
    (out/'RESULT.json').write_text(json.dumps(result,indent=2)+'\n');return result

if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('directory',type=Path);p.add_argument('--protocol',type=Path,required=True);p.add_argument('--producer-sha',required=True);p.add_argument('--out',type=Path,required=True);a=p.parse_args()
    print(json.dumps(run(a.directory,a.protocol,a.producer_sha,a.out),sort_keys=True))
