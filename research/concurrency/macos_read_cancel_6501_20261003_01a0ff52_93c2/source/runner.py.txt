"""Single formal invocation with process-group watchdog and literal receipts."""
from pathlib import Path
import argparse, hashlib, json, os, signal, subprocess, sys, time
ROOT=Path(__file__).resolve().parent
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def run_one(argv,out,label,timeout):
    start=time.time_ns();p=subprocess.Popen(argv,stdout=subprocess.PIPE,stderr=subprocess.PIPE,start_new_session=True)
    timed_out=False
    try:stdout,stderr=p.communicate(timeout=timeout)
    except subprocess.TimeoutExpired:
        timed_out=True;os.killpg(p.pid,signal.SIGKILL);stdout,stderr=p.communicate()
    end=time.time_ns();(out/(label+'.stdout.log')).write_bytes(stdout);(out/(label+'.stderr.log')).write_bytes(stderr)
    receipt={'argv':argv,'cwd':str(Path.cwd()),'start_utc_ns':start,'end_utc_ns':end,'exit_code':p.returncode,'watchdog_timeout':timed_out,'stdout_sha256':hashlib.sha256(stdout).hexdigest(),'stderr_sha256':hashlib.sha256(stderr).hexdigest()}
    (out/(label+'.receipt.json')).write_text(json.dumps(receipt,indent=2)+'\n');return receipt
if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('out',type=Path);a=p.parse_args();a.out.mkdir(parents=True,exist_ok=False)
    freeze=json.loads((ROOT/'FREEZE.json').read_text())
    for name,identity in freeze['files'].items():
        if sha(ROOT/name)!=identity['sha256']:raise SystemExit('source changed: '+name)
    formal_start=time.monotonic();producer=run_one([sys.executable,'-B',str(ROOT/'producer.py'),'--produce',str(a.out/'raw')],a.out,'producer',30)
    remaining=max(.1,30-(time.monotonic()-formal_start))
    reader=run_one([sys.executable,'-B',str(ROOT/'audit.py'),str(a.out/'raw'),'--protocol',str(ROOT/'PROTOCOL.json'),'--producer-sha',sha(ROOT/'producer.py')],a.out,'primary-audit',remaining)
    result={'allocation_id':json.loads((ROOT/'PROTOCOL.json').read_text())['allocation_id'],'producer_count':1,'primary_auditor_count':1,'formal_retries':0,'producer':producer,'primary_audit':reader,'freeze_sha256':sha(ROOT/'FREEZE.json'),'status':'COMPLETE_AUDITED' if producer['exit_code']==0 and reader['exit_code']==0 and not producer['watchdog_timeout'] and not reader['watchdog_timeout'] else 'FIRST_RESULT_STOP_HOLD_RETAIN'}
    (a.out/'FORMAL_RUN.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps(result,sort_keys=True));raise SystemExit(0 if result['status']=='COMPLETE_AUDITED' else 2)
