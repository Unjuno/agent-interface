"""Independent raw-only order/resource reader; never imports producer code."""
from pathlib import Path
import argparse, hashlib, json

MODES = ['wrapper_cancel','caller_close','owned_control','normal_data']
PRIMARY_KEYS = {'task_done','task_cancelled','future_done','future_cancelled','future_running','worker_active','worker_thread_alive','caller_close_started','caller_close_returned','reader_state','control_reader_state'}
def require(ok, message):
    if not ok: raise ValueError(message)
def integer(x): require(type(x) is int,'integer type')
def same(a,b): return json.dumps(a,sort_keys=True,separators=(',',':')) == json.dumps(b,sort_keys=True,separators=(',',':'))

def case(c, mode, rep):
    require(set(c)=={'mode','rep','pid','primary','final','events'},'case schema')
    require(c['mode']==mode,'mode');integer(c['rep']);require(c['rep']==rep,'repetition');integer(c['pid']);require(c['pid']>0,'PID')
    events=c['events'];require(type(events) is list and len(events)>8,'events')
    prior=-1
    for i,e in enumerate(events):
        require(set(e)=={'seq','ns','pid','tid','native_tid','name','data'},'event schema')
        for k in ['seq','ns','pid','tid','native_tid']:integer(e[k])
        require(e['seq']==i and e['ns']>=prior and e['pid']==c['pid'] and e['tid']>0 and e['native_tid']>0,'ordered identity')
        prior=e['ns'];require(type(e['name']) is str and type(e['data']) is dict,'event types')
    def one(name):
        found=[x for x in events if x['name']==name];require(len(found)==1,'exact event '+name);return found[0]
    def before(a,b):require(one(a)['seq']<one(b)['seq'],a+' before '+b)
    begin=one('case_begin'); require(begin['seq']==0 and begin['data']['mode']==mode and type(begin['data']['rep']) is int and begin['data']['rep']==rep,'begin identity')
    owner=(begin['tid'],begin['native_tid']); worker_event=one('worker_enter'); worker=(worker_event['tid'],worker_event['native_tid'])
    require(owner!=worker,'separate owner and worker thread')
    owner_events={'case_begin','entry_checkpoint','wrapper_cancel_request','wrapper_cancel_observed','normal_data_request','control_request','primary_checkpoint','harness_release','executor_join','case_final'}
    worker_events={'worker_enter','read_return','worker_return','selector_created','selector_closed','control_ack'}
    allowed=owner_events|worker_events|{'fd_closed','future_callback','caller_close_enter','caller_close_return'}
    for e in events:
        require(e['name'] in allowed,'known event name')
        if e['name'] in owner_events:require((e['tid'],e['native_tid'])==owner,'owner actor')
        if e['name'] in worker_events:require((e['tid'],e['native_tid'])==worker,'worker actor')
    fds=[begin['data'][k] for k in ['r','w','cr','cw'] if begin['data'][k] is not None]
    for fd in fds:integer(fd);require(fd>2,'owned fd')
    require(len(set(fds))==len(fds),'distinct descriptor integers')
    require((begin['data']['cr'] is not None)==(mode=='owned_control'),'control descriptors')
    before('worker_enter','entry_checkpoint'); entry=one('entry_checkpoint')['data']
    require(entry['future_running'] is True and entry['future_done'] is False and entry['active'] is True,'pending entered operation')
    primary=c['primary'];require(set(primary)==PRIMARY_KEYS,'primary schema')
    for k in PRIMARY_KEYS-{'reader_state','control_reader_state'}:require(type(primary[k]) is bool,'boolean '+k)
    require(primary['reader_state'] in {'OPEN','EBADF','CLOSE_PENDING'},'reader observation')
    require(primary['control_reader_state'] in {None,'EBADF'},'control reader observation')
    require(same(one('primary_checkpoint')['data']['snapshot'],primary),'primary literal join')
    require(primary['task_done'] is True and primary['future_cancelled'] is False,'wrapper/future terminal separation')
    before('entry_checkpoint','primary_checkpoint');before('primary_checkpoint','executor_join');before('executor_join','case_final')
    require(one('executor_join')['data']['thread_alive'] is False,'actual thread join')
    releases=[x for x in events if x['name']=='harness_release']
    require(len(releases)<=1,'single cleanup release')
    if releases:require(releases[0]['seq']>one('primary_checkpoint')['seq'] and releases[0]['data']['byte_hex']=='44','cleanup attribution')
    final=c['final'];require(set(final)=={'future_done','future_cancelled','worker_active','worker_thread_alive','result','fds'},'final schema')
    require(final['future_done'] is True and final['future_cancelled'] is False and final['worker_active'] is False and final['worker_thread_alive'] is False,'final completion')
    require(same(one('case_final')['data']['snapshot'],final),'final literal join')
    result=final['result'];require(set(result)=={'outcome','byte_hex','errno'},'result schema')
    if mode=='owned_control':require(same(result,one('control_ack')['data']['result']),'control result literal join')
    else:
        require(same(result,one('read_return')['data']['result']),'read result literal join');before('read_return','worker_return')
    require(result['outcome'] in {'READ_DATA','READ_EOF','READ_ERROR','CONTROL_ACK'},'complete result')
    if result['outcome']=='READ_ERROR':integer(result['errno']);require(result['errno']>0 and result['byte_hex'] is None,'typed read error')
    else:require(result['errno'] is None,'no errno');require(result['byte_hex']==({'READ_DATA':'44','READ_EOF':'','CONTROL_ACK':'43'}[result['outcome']]),'actual byte')
    closed=[x for x in events if x['name']=='fd_closed']
    for e in closed: integer(e['data']['fd'])
    require(len(closed)==len(fds) and {x['data']['fd'] for x in closed}==set(fds),'one close per owned pipe fd')
    require(len({x['data']['fd'] for x in closed})==len(closed),'no duplicate close')
    expected_roles={begin['data']['w']:'owner_writer',begin['data']['r']:'worker_data_reader' if mode=='owned_control' else 'caller_reader' if mode=='caller_close' else 'owner_reader'}
    if mode=='owned_control':expected_roles.update({begin['data']['cr']:'worker_control_reader',begin['data']['cw']:'owner_control_writer'})
    closer_actor=None
    if mode=='caller_close':
        closer_event=one('caller_close_enter');closer_actor=(closer_event['tid'],closer_event['native_tid'])
        require(closer_actor not in [owner,worker],'separate caller close thread')
        require((one('caller_close_return')['tid'],one('caller_close_return')['native_tid'])==closer_actor,'closer actor')
    for e in closed:
        role=expected_roles[e['data']['fd']];require(e['data']['role']==role,'endpoint close role')
        expected_actor=worker if role.startswith('worker_') else closer_actor if role=='caller_reader' else owner
        require((e['tid'],e['native_tid'])==expected_actor,'endpoint close actor')
    if mode=='owned_control':
        created=one('selector_created')['data'];integer(created['fd']);require(created['fd'] not in fds and created['backend']=='KqueueSelector','native selector')
        fds.append(created['fd']);require(one('selector_closed')['data']=={'fd':created['fd'],'state':'EBADF'},'selector close observation')
        before('control_request','control_ack');before('selector_closed','control_ack');before('control_ack','worker_return');before('worker_return','primary_checkpoint')
        require(primary['future_done'] is True and primary['future_running'] is False and primary['worker_active'] is False and primary['task_cancelled'] is False,'control operation done')
        require(primary['reader_state']=='EBADF' and primary['control_reader_state']=='EBADF' and result['outcome']=='CONTROL_ACK','owned endpoints before ack')
        for e in closed:
            if e['data']['fd'] in [begin['data']['r'],begin['data']['cr']]:require(e['seq']<one('control_ack')['seq'],'close before ack')
        require(not releases,'control succeeds without cleanup D')
    elif mode=='normal_data':
        before('normal_data_request','read_return');before('read_return','worker_return');before('worker_return','primary_checkpoint')
        require(primary['future_done'] is True and primary['future_running'] is False and primary['worker_active'] is False and primary['task_cancelled'] is False and result['outcome']=='READ_DATA','normal data positive')
        require(primary['reader_state']=='OPEN' and not releases,'normal data ownership until cleanup')
    else:
        before('wrapper_cancel_request','wrapper_cancel_observed');before('wrapper_cancel_observed','primary_checkpoint')
        require(one('wrapper_cancel_request')['data']['accepted'] is True and primary['task_cancelled'] is True,'actual wrapper cancellation')
        observed=one('wrapper_cancel_observed')['data'];require(observed['done'] is True and observed['cancelled'] is True and observed['future_cancelled'] is False,'cancelled wrapper not native Future')
        if mode=='wrapper_cancel':
            if primary['future_done'] is False:
                require(primary['future_running'] is True and primary['worker_active'] is True and primary['reader_state']=='OPEN' and primary['worker_thread_alive'] is True,'coherent pending operation')
                require(len(releases)==1,'required separate cleanup');before('harness_release','read_return')
        else:
            before('wrapper_cancel_observed','caller_close_enter');before('caller_close_enter','primary_checkpoint')
            require(primary['caller_close_started'] is True,'close attempted')
            returns=[e for e in events if e['name']=='caller_close_return'];require(len(returns)==1,'final close return')
            require(primary['caller_close_returned']==(returns[0]['seq']<one('primary_checkpoint')['seq']),'close checkpoint chronology')
            if not primary['caller_close_returned']:require(primary['reader_state']=='CLOSE_PENDING','pending close is unknown descriptor state')
    require(set(final['fds'])=={str(fd) for fd in fds} and all(v=='EBADF' for v in final['fds'].values()),'final native descriptor observations')
    return {'mode':mode,'rep':rep,'primary_future_done':primary['future_done'],'reader':primary['reader_state'],'active_operation':primary['worker_active'],'primary_executor_thread_alive':primary['worker_thread_alive'],'final_outcome':result['outcome'],'harness_release':bool(releases)}

def audit(directory, protocol, producer_sha):
    raw=json.loads((directory/'raw.json').read_text());require(set(raw)=={'schema','allocation_id','environment','producer_sha256','protocol_sha256','cases','status'},'root schema')
    p=json.loads(protocol.read_text());require(raw['schema']=='macos-owned-read-completion-v1' and raw['allocation_id']==p['allocation_id'] and raw['status']=='COMPLETE','allocation completion')
    require(raw['producer_sha256']==producer_sha and raw['protocol_sha256']==hashlib.sha256(protocol.read_bytes()).hexdigest(),'source protocol pins')
    require(raw['environment']['platform']=='macOS-27.0.1-arm64-arm-64bit-Mach-O' and raw['environment']['python'].startswith('3.14.5 '),'selected environment')
    require(type(raw['cases']) is list and len(raw['cases'])==8,'eight rows')
    summaries=[];pids=[];prior_end=0
    for c,(mode,rep) in zip(raw['cases'],[(m,r) for m in MODES for r in [0,1]]):
        summaries.append(case(c,mode,rep));pids.append(c['pid']);name=mode+'-'+str(rep)
        receipt=json.loads((directory/(name+'.receipt.json')).read_text());integer(receipt['exit_code']);require(receipt['exit_code']==0,'child exit')
        integer(receipt['start_utc_ns']);integer(receipt['end_utc_ns']);require(prior_end<=receipt['start_utc_ns']<=receipt['end_utc_ns'],'serial actual receipts');prior_end=receipt['end_utc_ns']
        for stream in ['stdout','stderr']:require(hashlib.sha256((directory/(name+'.'+stream+'.log')).read_bytes()).hexdigest()==receipt[stream+'_sha256'],'actual stream hash')
        records=[json.loads(x) for x in (directory/(name+'.stdout.log')).read_text().splitlines()];footers=[x['case'] for x in records if 'case' in x];events=[x['event'] for x in records if 'event' in x]
        require(len(footers)==1 and same(footers[0],c) and same(events,c['events']),'retained stdout/raw literal join')
    require(len(set(pids))==8,'eight fresh private child PIDs')
    supported=all(not x['primary_future_done'] and x['active_operation'] for x in summaries if x['mode']=='wrapper_cancel')
    return {'status':'METHOD_PASS_H_SUPPORTED_SCOPED' if supported else 'METHOD_PASS_H_CONTRADICTED_RETAIN','rows':8,'summaries':summaries,'raw_sha256':hashlib.sha256((directory/'raw.json').read_bytes()).hexdigest(),'kernel_read_proof':False,'physical_release_or_performance':False}

if __name__=='__main__':
    a=argparse.ArgumentParser();a.add_argument('directory',type=Path);a.add_argument('--protocol',type=Path,required=True);a.add_argument('--producer-sha',required=True);args=a.parse_args()
    try:result=audit(args.directory,args.protocol,args.producer_sha)
    except (ValueError,KeyError,TypeError,json.JSONDecodeError) as exc:print(json.dumps({'status':'HOLD_AUDIT','error':str(exc)}));raise SystemExit(2)
    print(json.dumps(result,sort_keys=True))
