"""New reviewer regressions for actual accepted-begin/U composition, not old decks."""
import os
from pathlib import Path
import sys
import unittest

ROOT = Path(os.environ['REVIEW6894_SOURCE_ROOT']).resolve()
sys.path.insert(0, str(ROOT))
from runtime.kernel import (Action, ActionKind, AuthorityLease, ContractError,
                            EffectOccurrence, ExecutionReceipt, ExecutionRequest,
                            Observation, ReleaseReceipt, RequestLifecycle, Stage,
                            TargetBinding)
import runtime.kernel as kernel
from runtime.kernel import backend, contracts, lifecycle

for module in (kernel, backend, contracts, lifecycle):
    expected = ROOT/'runtime/kernel'/('__init__.py' if module is kernel else module.__name__.split('.')[-1]+'.py')
    if Path(module.__file__).resolve() != expected:
        raise RuntimeError('unexpected actual imported source origin')

ORIGINS = (0, 2**53+17, 10**24+17)

def authorized(origin):
    flow = RequestLifecycle()
    observation = Observation(19, origin+1, 'joint-surface', 'a'*64, 3, 3, 'rgb24')
    binding = TargetBinding('joint-target', 19, 'joint-surface', 'b'*64)
    lease = AuthorityLease('joint-lease', 19, 'joint-surface', origin+100, frozenset({ActionKind.KEY}))
    request = ExecutionRequest('joint-command', 'c'*64, binding, lease,
                               (Action('joint-action', ActionKind.KEY, 'press-release'),))
    flow.record_observation(observation)
    flow.bind(binding)
    flow.authorize(lease, now_ns=origin+2)
    return flow, request

def receipt(origin, start_offset, occurrence):
    return ExecutionReceipt('joint-command', 'joint-backend', 'c'*64, 'joint-lease',
                            19, 'joint-surface', origin+start_offset, origin+60, 1,
                            occurrence, ReleaseReceipt(origin+70, True))

class JointBoundaryTests(unittest.TestCase):
    def unchanged(self, flow, before):
        self.assertEqual(set(vars(flow)), set(before))
        for key, value in before.items():
            self.assertIs(vars(flow)[key], value, key)

    def test_refused_stale_none_then_lost_receipt_cancel_preserves_possibility(self):
        for origin in ORIGINS:
            with self.subTest(origin=origin):
                flow, request = authorized(origin)
                flow.begin_execution(request, now_ns=origin+40)
                before = dict(vars(flow))
                with self.assertRaisesRegex(ContractError, 'starts before accepted begin'):
                    flow.record_execution(receipt(origin, 39, EffectOccurrence.NONE))
                self.unchanged(flow, before)
                for release in (ReleaseReceipt(origin+39, True), ReleaseReceipt(origin+40, False)):
                    with self.assertRaises(ContractError):
                        flow.stop('cleanup-not-current-or-unverified', release=release)
                    self.unchanged(flow, before)
                flow.stop('execution-return-lost-after-stale-none-refusal', release=ReleaseReceipt(origin+40, True))
                self.assertIs(flow.request, request)
                self.assertIsNone(flow.execution)
                self.assertIsNone(flow.effect)
                result = flow.outcome()
                self.assertIs(result.stage, Stage.STOPPED)
                self.assertEqual(result.command_id, 'joint-command')
                self.assertTrue(result.release_verified)
                self.assertFalse(result.effect_verified)
                self.assertTrue(result.effect_occurred)

    def test_current_explicit_occurrence_after_stale_none_refusal_controls_outcome(self):
        for origin in ORIGINS:
            for occurrence in EffectOccurrence:
                with self.subTest(origin=origin, occurrence=occurrence):
                    flow, request = authorized(origin)
                    flow.begin_execution(request, now_ns=origin+40)
                    before = dict(vars(flow))
                    with self.assertRaises(ContractError):
                        flow.record_execution(receipt(origin, 39, EffectOccurrence.NONE))
                    self.unchanged(flow, before)
                    current = receipt(origin, 40, occurrence)
                    flow.record_execution(current)
                    self.assertIs(flow.execution, current)
                    flow.stop('current-explicit-receipt-controls-occurrence')
                    result = flow.outcome()
                    self.assertIs(result.effect_occurred, occurrence in (EffectOccurrence.POSSIBLE, EffectOccurrence.OBSERVED))
                    self.assertFalse(result.effect_verified)
                    self.assertTrue(result.release_verified)
                    stopped = dict(vars(flow))
                    with self.assertRaises(ContractError):
                        flow.stop('second-terminal-cleanup')
                    self.unchanged(flow, stopped)

    def test_expired_begin_does_not_create_uncertain_execution(self):
        for origin in ORIGINS:
            with self.subTest(origin=origin):
                flow, request = authorized(origin)
                before = dict(vars(flow))
                with self.assertRaisesRegex(ContractError, 'authority expired'):
                    flow.begin_execution(request, now_ns=origin+100)
                self.unchanged(flow, before)
                self.assertIsNone(flow.request)
                self.assertIsNone(flow.execution_started_ns)
                flow.stop('no-admitted-begin', release=ReleaseReceipt(origin+100, True))
                result = flow.outcome()
                self.assertIsNone(result.command_id)
                self.assertFalse(result.effect_occurred)
                self.assertFalse(result.effect_verified)
                self.assertTrue(result.release_verified)

if __name__ == '__main__':
    unittest.main()
