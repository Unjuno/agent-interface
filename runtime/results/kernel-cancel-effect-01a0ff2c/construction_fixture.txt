"""Sequential contract regressions; no native backend or physical input."""
import unittest
from runtime.kernel import (
    ContractError, EffectOccurrence, EffectReceipt, EffectStatus,
    RequestLifecycle,
)
from runtime.kernel.test_kernel import (
    E, M, binding, execution, lease, observation, released, request,
)


def authorized():
    flow = RequestLifecycle()
    flow.record_observation(observation())
    flow.bind(binding())
    flow.authorize(lease(), now_ns=200)
    return flow


class InertInterruptedBackend:
    """Two possible application histories with the same missing receipt."""
    def __init__(self, apply_before_failure):
        self.apply_before_failure = apply_before_failure
        self.saved_count = 0

    def execute(self, req):
        if self.apply_before_failure:
            self.saved_count += 1
        raise OSError('receipt unavailable')


class CancellationEffectTests(unittest.TestCase):
    def test_missing_receipt_after_begin_preserves_possible_effect(self):
        for applied in [False, True]:
            with self.subTest(inert_effect_applied=applied):
                flow = authorized()
                req = request()
                flow.begin_execution(req, now_ns=300)
                backend = InertInterruptedBackend(applied)
                with self.assertRaises(OSError):
                    backend.execute(req)
                self.assertEqual(backend.saved_count, int(applied))
                flow.stop('cancelled after unavailable receipt', release=released(800))
                result = flow.outcome()
                # Existing API maps POSSIBLE and OBSERVED to True. This is
                # possible occurrence, not proof of the in-memory effect.
                self.assertTrue(result.effect_occurred)
                self.assertFalse(result.effect_verified)
                self.assertTrue(result.release_verified)
                self.assertEqual(result.command_id, 'cmd-1')

    def test_stop_before_begin_keeps_no_effect(self):
        flow = authorized()
        flow.stop('cancelled before input', release=released(800))
        self.assertFalse(flow.outcome().effect_occurred)
        self.assertIsNone(flow.outcome().command_id)

    def test_rejected_begin_does_not_create_possible_effect(self):
        flow = authorized()
        with self.assertRaises(ContractError):
            flow.begin_execution(request(), now_ns=1000)
        flow.stop('expired request refused', release=released(1000))
        self.assertFalse(flow.outcome().effect_occurred)

    def test_completed_none_keeps_explicit_no_effect(self):
        flow = authorized()
        flow.begin_execution(request(), now_ns=300)
        flow.record_execution(execution(EffectOccurrence.NONE))
        flow.stop('completed without effect')
        self.assertFalse(flow.outcome().effect_occurred)

    def test_completed_possible_is_still_possible(self):
        flow = authorized()
        flow.begin_execution(request(), now_ns=300)
        flow.record_execution(execution(EffectOccurrence.POSSIBLE))
        flow.stop('effect unavailable')
        self.assertTrue(flow.outcome().effect_occurred)

    def test_completed_observed_is_still_observed(self):
        flow = authorized()
        flow.begin_execution(request(), now_ns=300)
        flow.record_execution(execution(EffectOccurrence.OBSERVED))
        flow.stop('cancelled after effect')
        self.assertTrue(flow.outcome().effect_occurred)

    def test_contradicted_observed_history_is_preserved(self):
        flow = authorized()
        flow.begin_execution(request(), now_ns=300)
        flow.record_execution(execution(EffectOccurrence.OBSERVED))
        flow.record_effect(EffectReceipt('cmd-1', M, 900, EffectStatus.CONTRADICTED, E))
        self.assertTrue(flow.outcome().effect_occurred)
        self.assertFalse(flow.outcome().effect_verified)

    def test_unreleased_stop_stays_nonterminal(self):
        flow = authorized()
        flow.begin_execution(request(), now_ns=300)
        with self.assertRaises(ContractError):
            flow.stop('cancelled')
        self.assertIsNone(flow.execution)
        with self.assertRaises(ContractError):
            flow.outcome()


if __name__ == '__main__':
    unittest.main()
