"""Emit six distinct contract histories; inert effects are oracle-only labels."""
import json
from dataclasses import asdict
from runtime.kernel import EffectOccurrence
from runtime.kernel.test_kernel import execution, released, request
from test_cancel_effect import InertInterruptedBackend, authorized

for case in ['unbegun', 'begun_failure_before_effect', 'begun_failure_after_effect',
             'completed_none', 'completed_possible', 'completed_observed']:
    flow = authorized()
    events = ['authorized']
    receipt_occurrence = None
    inert_effect = None
    if case != 'unbegun':
        req = request()
        flow.begin_execution(req, now_ns=300)
        events.append('begin_accepted')
        if case.startswith('begun_failure'):
            backend = InertInterruptedBackend(case.endswith('after_effect'))
            try:
                backend.execute(req)
            except OSError:
                events.append('execute_receipt_unavailable')
            inert_effect = backend.saved_count
        else:
            receipt_occurrence = case.removeprefix('completed_')
            flow.record_execution(execution(EffectOccurrence(receipt_occurrence)))
            events.append('receipt_accepted')
    flow.stop('cancelled', release=released(800))
    events.append('stop_with_verified_release')
    print(json.dumps({'case':case, 'events':events,
                      'receipt_occurrence':receipt_occurrence,
                      'inert_application_effect_count':inert_effect,
                      'outcome':asdict(flow.outcome())},sort_keys=True))
