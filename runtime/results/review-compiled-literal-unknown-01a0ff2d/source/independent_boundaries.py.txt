"""Independent finite literal/multi-condition boundary checks; no I/O backend."""
import atexit, copy, importlib.util, json, os, pathlib, sys, unittest
ROOT = pathlib.Path(__file__).resolve().parent
VARIANT = sys.argv.pop(1)
paths = {
    'baseline': ROOT/'projection/runtime/results/compiled-literal-unknown-01a0ff35/source/baseline-compiled_gui.py.txt',
    'candidate': ROOT/'projection/runtime/core_v1/compiled_gui.py',
    'composed': ROOT/'composed-compiled_gui.py',
}
module = importlib.util.module_from_spec(importlib.util.spec_from_file_location('reviewed_'+VARIANT, paths['candidate']))
exec(compile(paths[VARIANT].read_bytes(), str(paths[VARIANT]), 'exec'), module.__dict__)
ROWS = []
def save_rows():
    p = pathlib.Path(os.environ['REVIEW_RAW'])
    if p.exists(): raise RuntimeError('raw target already exists')
    p.write_text(json.dumps({'variant':VARIANT,'rows':ROWS},indent=2)+'\n',encoding='utf-8')
atexit.register(save_rows)

class Fixture:
    def __init__(self, label, middle=None, final=None, verdict='succeeded',
                 stale=False, late=False, reference='independent-witness'):
        self.label=label
        self.middle={'phase':'unknown','ready':True} if middle is None else middle
        self.final={'phase':'done','ready':True} if final is None else final
        self.verdict=verdict; self.stale=stale; self.late=late; self.reference=reference
        self.now=0; self.calls={k:[] for k in ('observe','admit','execute','verify')}
        self.plan={
          'format':'compiled-gui-interface-v1','interface_id':'review-mixed-predicates',
          'session_scope':'private-review-fixture','surface':'form',
          'predicates':['phase','ready'],
          'symbols':{'field':{'kind':'target_reference','target_reference':'field',
                     'identity_predicate':'phase','dependencies':['phase','ready']}},
          'actions':{
             'enter':{'target_symbol':'field','operation':'enter',
                      'expected_effect':{'phase':'unknown','ready':True}},
             'save':{'target_symbol':'field','operation':'save',
                     'expected_effect':{'phase':'done','ready':True}}},
          'method':{'name':'bounded-two-step','version':'1','initial_state':'initial',
                    'max_transitions':2,'max_runtime_ms':10,'states':{
             'initial':{'branches':[{'when':{'phase':'initial','ready':True},'outcome':'action',
                         'action':'enter','next_state':'middle','reason':None}]},
             'middle':{'branches':[{'when':{'phase':'unknown','ready':True},'outcome':'action',
                        'action':'save','next_state':'final','reason':None}]},
             'final':{'branches':[{'when':{'phase':'done','ready':True},'outcome':'complete',
                        'action':None,'next_state':None,'reason':None}]}}}}
    def observe(self,payload):
        self.calls['observe'].append(copy.deepcopy(payload))
        i=len(self.calls['observe'])
        if i>3: raise AssertionError('unbounded observation')
        predicates=[{'phase':'initial','ready':True},self.middle,self.final][i-1]
        return {'sequence':i,'captured_ns':self.now,'surface':'form',
                'predicates':copy.deepcopy(predicates),'evidence_ref':'frame-'+str(i),
                'evidence_digest':'frame-1' if self.stale and i==2 else 'frame-'+str(i)}
    def admit(self,payload):
        self.calls['admit'].append(copy.deepcopy(payload))
        return {'eligible':True,'status':'revalidated','authorization':'one-use',
                'expected_sequence':payload['observation']['sequence'],'valid_until_ns':100_000_000}
    def execute(self,payload):
        self.calls['execute'].append(copy.deepcopy(payload))
        if len(self.calls['execute'])>2: raise AssertionError('extra action')
        return {'status':'completed','action_id':'action-'+str(len(self.calls['execute'])),
                'effect_ref':'stub-effect','release':{'verified':True,'keys_down':[],'buttons_down':[]}}
    def verify(self,payload):
        self.calls['verify'].append(copy.deepcopy(payload))
        if self.late and len(self.calls['verify'])==1: self.now=10_000_000
        status=self.verdict if len(self.calls['verify'])==1 else 'succeeded'
        return {'status':status,'evidence_ref':self.reference if status=='succeeded' else None}
    def run(self):
        before=copy.deepcopy(self.plan)
        row={'case':self.label,'middle':copy.deepcopy(self.middle),'final':copy.deepcopy(self.final),
             'verdict':self.verdict,'reference':copy.deepcopy(self.reference),'stale':self.stale,'late':self.late}
        try:
            result=module.run(self.plan,{'observe':self.observe,'admit':self.admit,
              'execute':self.execute,'verify_effect':self.verify,'cancelled':lambda:False},
              clock=lambda:self.now)
            row['result']=copy.deepcopy(result)
            return result
        except Exception as e:
            row['raised']=type(e).__name__
            raise
        finally:
            row['calls']=copy.deepcopy(self.calls)
            row['caller_plan_unchanged']=self.plan==before
            ROWS.append(row)
            if self.plan!=before: raise AssertionError('mutated caller plan')

class LiteralBoundaryTests(unittest.TestCase):
    def stop(self, fixture, reason, actions=1, verifiers=0):
        result=fixture.run()
        self.assertEqual((result['outcome'],result['reason']),('SAFE_YIELD',reason))
        self.assertEqual(result['completed_transitions'],actions)
        self.assertEqual(len(fixture.calls['execute']),actions)
        self.assertEqual(len(fixture.calls['verify']),verifiers)
        self.assertEqual([x['operation'] for x in fixture.calls['execute']],['enter','save'][:actions])
        return result
    def test_matching_multi_predicate_literal(self):
        fixture=Fixture('matching-mixed-literal')
        result=fixture.run()
        self.assertEqual((result['outcome'],result['reason']),('TASK_SUCCEEDED','method_complete'))
        self.assertEqual(result['completed_transitions'],2)
        self.assertEqual(len(fixture.calls['verify']),2)
        self.assertEqual(fixture.calls['verify'][0]['expected_effect'],{'phase':'unknown','ready':True})
        self.assertEqual([x['operation'] for x in fixture.calls['execute']],['enter','save'])
    def test_missing_each_predicate_stops_without_callback(self):
        for middle in ({'phase':'unknown'},{'ready':True},{}):
            with self.subTest(middle=middle):
                self.stop(Fixture('missing-component',middle=middle),'effect_unavailable')
    def test_other_predicate_cross_type_refuses(self):
        for value in (0,1,'true'):
            with self.subTest(value=value):
                self.stop(Fixture('other-type-mismatch',middle={'phase':'unknown','ready':value}),'effect_failed')
    def test_other_unknown_component_stays_unavailable(self):
        self.stop(Fixture('other-unknown',middle={'phase':'unknown','ready':'unknown'}),'effect_unavailable')
    def test_negative_callback_preserves_pending_command(self):
        for status in ('failed','unavailable'):
            with self.subTest(status=status):
                result=self.stop(Fixture('callback-'+status,verdict=status),'effect_'+status,verifiers=1)
                self.assertEqual(result['pending_effect']['action'],'enter')
    def test_unchanged_digest_precedes_matching_literal_callback(self):
        self.stop(Fixture('same-digest',stale=True),'no_progress')
    def test_callback_exhausting_exact_deadline_blocks_next_action(self):
        result=self.stop(Fixture('late-callback',late=True),'budget_exhausted',verifiers=1)
        self.assertEqual(result['ended_ns'],10_000_000)
        self.assertIsNone(result['pending_effect'])
    def test_second_missing_effect_preserves_both_completed_actions(self):
        result=self.stop(Fixture('second-missing',final={'phase':'done'}),'effect_unavailable',actions=2,verifiers=1)
        self.assertEqual(result['pending_effect']['action'],'save')
        self.assertEqual([r['action'] for r in result['transitions']],['enter','save'])

class ComposedReferenceTests(unittest.TestCase):
    def test_successful_callback_requires_named_reference(self):
        self.assertEqual(VARIANT,'composed')
        for reference in (None,'',0,False,[]):
            with self.subTest(reference=reference):
                fixture=Fixture('bad-reference',reference=reference)
                with self.assertRaisesRegex(ValueError,'effect evidence reference'):
                    fixture.run()
                self.assertEqual(len(fixture.calls['execute']),1)
                self.assertEqual(len(fixture.calls['verify']),1)

if __name__=='__main__': unittest.main()
