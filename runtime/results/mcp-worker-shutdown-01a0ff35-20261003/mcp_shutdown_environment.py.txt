from datetime import datetime, timezone
import hashlib
from importlib import metadata
import json
from pathlib import Path
import platform
import sys
import asyncio.futures
import asyncio.tasks
import concurrent.futures.thread
import mcp.server.fastmcp.server
import anyio
import pydantic

output = Path(sys.argv[1])
if output.exists():
    raise FileExistsError(output)
sources = {}
for module in (asyncio.futures, asyncio.tasks, concurrent.futures.thread,
               mcp.server.fastmcp.server, anyio, pydantic):
    path = Path(module.__file__)
    sources[module.__name__] = dict(path=str(path), sha256=hashlib.sha256(path.read_bytes()).hexdigest())
records = sorted((dict(name=d.metadata['Name'], version=d.version) for d in metadata.distributions()),
                 key=lambda d: (d['name'].lower(), d['version']))
output.write_text(json.dumps(dict(classification='ordinary environment observation, not prospective formal freeze',
    observed_utc=datetime.now(timezone.utc).isoformat(), python=sys.version,
    executable=sys.executable, executable_sha256=hashlib.sha256(Path(sys.executable).read_bytes()).hexdigest(),
    platform=platform.platform(), packages=records, relevant_module_sources=sources), indent=2)+'\n', encoding='utf-8')
print(json.dumps(dict(python=platform.python_version(), mcp=metadata.version('mcp'),
                     anyio=metadata.version('anyio'), pydantic=metadata.version('pydantic'),
                     packages=len(records))))
