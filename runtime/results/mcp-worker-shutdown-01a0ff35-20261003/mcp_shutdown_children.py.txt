from datetime import datetime, timezone
import json
from pathlib import Path
import re

root = Path(__file__).resolve().parent/'mcp-shutdown-evidence'
text = (root/'green-final-mcp312.stderr.txt').read_text(encoding='utf-8')
started = sorted(set(map(int, re.findall(r'execute program .*?pid=(\d+) running', text))))
ended = [(int(pid), int(code)) for pid, code in re.findall(r'pid=(\d+)[^\n]*exited with return code (\d+)', text)]
if len(started) != 3 or set(started) != {pid for pid, code in ended} or any(code for pid, code in ended):
    raise ValueError('actual inert stdio child terminal receipt mismatch')
path = root/'stdio-child-receipts.json'
if path.exists():
    raise FileExistsError(path)
path.write_text(json.dumps(dict(observed_utc=datetime.now(timezone.utc).isoformat(),
    source='green-final-mcp312.stderr.txt', classification='actual SDK inert-fixture child exit receipts, not native backend effect',
    started_pids=started, terminal_exits=ended, all_recorded_terminal=True), indent=2)+'\n', encoding='utf-8')
print(json.dumps(dict(started_pids=started, terminal_exits=ended)))
