from datetime import datetime, timezone
import json
from pathlib import Path
import sys
import time

output = Path(sys.argv[1])
if output.exists():
    raise FileExistsError(output)
clocks = {}
for name in ('monotonic', 'perf_counter'):
    info = time.get_clock_info(name)
    clocks[name] = dict(implementation=info.implementation, monotonic=info.monotonic,
                        adjustable=info.adjustable, resolution_seconds=info.resolution)
record = dict(observed_utc=datetime.now(timezone.utc).isoformat(), python=sys.version, clocks=clocks,
              qualification='read-only clock metadata; no timing trial or physical sleep-duration measurement')
output.write_text(json.dumps(record, indent=2)+'\n', encoding='utf-8')
print(json.dumps(record))
