from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import re
import time

ROOT = Path(__file__).resolve().parent
OUT = ROOT/'mcp-shutdown-evidence'
REPO = ROOT/'mcp-shutdown-repo'
path = OUT/'native-diagnosis.json'
if path.exists():
    raise FileExistsError(path)
def clock(name):
    value = time.get_clock_info(name)
    return dict(implementation=value.implementation, monotonic=value.monotonic,
                adjustable=value.adjustable, resolution_seconds=value.resolution)
def summary(path):
    text = path.read_text(encoding='utf-8')
    counts = re.search(r'Ran (\d+) tests in ', text)
    verdict = next(line for line in reversed(text.splitlines()) if line.startswith(('FAILED (', 'OK')))
    return dict(methods=int(counts.group(1)), verdict=verdict,
                errors=re.findall(r'^ERROR: (.+)$', text, re.M), failures=re.findall(r'^FAIL: (.+)$', text, re.M))
checks = {name:summary(OUT/'native-shared-entry'/(name+'.stderr.log')) for name in ('protocol','harness')}
baseline = summary(OUT/'native-baseline-three.stderr.txt')
result = dict(classification='retained ordinary shared-entry FAIL and bounded source/clock diagnosis; not formal result or retry',
    observed_utc=datetime.now(timezone.utc).isoformat(), checks=checks, baseline_three=baseline,
    clock_observation={'monotonic':clock('monotonic'), 'perf_counter':clock('perf_counter')},
    clock_observation_interpreter='orchestrator3.11.9; supported3.12 clock implementation separately available through captured native environment/runtime source; no physical sleep duration measured here',
    source_facts=['Legacy native exchange/allocation/finish/cleanup require POSIX and /proc/self/ns/pid or os.O_DIRECTORY; these sources are unchanged and were not ported.',
                  'Archive fixture identifies itself as Linux/X11 and asserts zip module filenames begin archive path plus literal slash; unchanged original-server check reproduces that assertion failure.',
                  'One no-inline-summary20ms assertion reproduces as15ms on original-server source; explicit-wait case passes in that one baseline check. Both candidate failures remain; no favorable-row selection or causal frequency inference.',
                  'The runtime wait calls time.sleep and timestamps it with time.monotonic_ns, both unchanged. Coarse Windows monotonic resolution is a plausible explanation for unstable lower-bound measurement, not proof of actual physical wait duration.',
                  'No full shared-native PASS or Linux-supported backend result is claimed. Appropriate compatible native checks/explicit review remain necessary before runtime adoption.'],
    original_server_blob='05f952d26c90d66ccc2af3b24dd4704e0ffd596e',
    original_server_sha256=hashlib.sha256((OUT/'source-red/mcp_server.py.txt').read_bytes()).hexdigest(),
    restored_final_server_sha256=hashlib.sha256((REPO/'runtime/cli_v1/mcp_server.py').read_bytes()).hexdigest(),
    formal_allocations=0)
path.write_text(json.dumps(result, indent=2)+'\n', encoding='utf-8')
print(json.dumps(dict(shared_entry='FAIL', protocol=checks['protocol']['verdict'], harness=checks['harness']['verdict'],
                     baseline_three=baseline['verdict'], monotonic=clock('monotonic'), adoption='HOLD compatible-native confirmation')))
