import sys,site,json,os,time
from pathlib import Path
from datetime import datetime,timezone
R=Path(__file__).resolve().parent
D=R.parent/'review6879-2d0b'/'deps313';sys.path.insert(0,str(D));site.addsitedir(str(D))
from mcp.server.fastmcp import FastMCP
OUT=Path(sys.argv[1]);server=FastMCP('owned-inert-unicode-boundary')
def emit(event,**data):
    with (OUT/'server.jsonl').open('a',encoding='utf-8') as f:f.write(json.dumps({'event':event,'pid':os.getpid(),'utc':datetime.now(timezone.utc).isoformat(),'monotonic_ns':time.monotonic_ns(),**data},ensure_ascii=True)+'\n')
@server.tool()
def interface_validate(program:dict)->dict:
    emit('actual_tool_entry',program=program)
    return {'received':program}
if __name__=='__main__':
    emit('server_start')
    try:server.run(transport='stdio')
    finally:emit('server_run_finally')
