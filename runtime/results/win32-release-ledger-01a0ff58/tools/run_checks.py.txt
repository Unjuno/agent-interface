"""Own finite regression runner; preserves each source/command/stream/UTC result."""
import datetime
import hashlib
import json
import pathlib
import shutil
import subprocess
import sys

root = pathlib.Path(__file__).resolve().parent
which, mode, label = sys.argv[1:]
source = root / which
test_path = source / "runtime/backends/win32_v1/test_release_ledger.py"
shutil.copyfile(root / "test_release_ledger.py", test_path)
out = root / "runs" / label
out.mkdir(parents=True, exist_ok=False)
argv = [sys.executable, "-X", "utf8"] + (["-O"] if mode == "optimized" else [])
argv += ["-m", "unittest", "-v", "runtime.backends.win32_v1.test_release_ledger",
         "runtime.backends.win32_v1.test_integration.PureWin32HelperTests"]
pins = {str(p.relative_to(source)).replace("\\", "/"):
        hashlib.sha256(p.read_bytes()).hexdigest()
        for p in source.rglob("*.py")}
receipt = {"kind": "ordinary inert-API regression", "argv": argv, "cwd": str(source),
           "source_sha256": pins, "started_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(),
           "no_live_native_integration_class": True}
(out / "START.json").write_text(json.dumps(receipt, indent=2) + "\n", encoding="utf-8")
with (out / "stdout.txt").open("wb") as stdout, (out / "stderr.txt").open("wb") as stderr:
    proc = subprocess.Popen(argv, cwd=source, stdout=stdout, stderr=stderr)
    receipt["pid"] = proc.pid
    try:
        receipt["exit_code"] = proc.wait(timeout=20)
    except subprocess.TimeoutExpired:
        proc.kill()
        receipt["exit_code"] = proc.wait()
        receipt["timeout_killed_only_own_child"] = True
receipt["ended_utc"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
receipt["streams"] = {n: {"bytes": (out / n).stat().st_size,
                         "sha256": hashlib.sha256((out / n).read_bytes()).hexdigest()}
                      for n in ["stdout.txt", "stderr.txt"]}
(out / "RESULT.json").write_text(json.dumps(receipt, indent=2) + "\n", encoding="utf-8")
print(json.dumps({"run": label, "exit_code": receipt["exit_code"], "pid": proc.pid,
                  "started_utc": receipt["started_utc"], "ended_utc": receipt["ended_utc"]}))
print((out / "stderr.txt").read_text(encoding="utf-8"))
sys.exit(receipt["exit_code"])
