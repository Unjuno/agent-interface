"""Record bounded command receipts without confusing harness and child exits."""
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess
import sys

root = Path(__file__).resolve().parent
cwd = root / 'receipt-adoption-v1'
label, *args = sys.argv[1:]
private = root / 'receipt-adoption-private'
private.mkdir(exist_ok=True)
logs = cwd / 'logs'
logs.mkdir(exist_ok=True)
receipt_path = logs / f'{label}.receipt.json'
if receipt_path.exists():
    raise SystemExit('label already used; first result must be retained')
command = [sys.executable, '-B', *args]
started = datetime.now(timezone.utc).isoformat()
child = subprocess.run(command, cwd=cwd, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=60)
ended = datetime.now(timezone.utc).isoformat()
record = {'label': label, 'command': ['python', '-B', *args], 'python': sys.version, 'started_utc': started, 'ended_utc': ended, 'exit_code': child.returncode, 'streams': {}}
for name, data in (('stdout', child.stdout), ('stderr', child.stderr)):
    rawpath = private / f'{label}.{name}.log'
    with rawpath.open('xb') as f:
        f.write(data)
    public = data.replace(str(root).encode(), b'PRIVATE_WORK')
    publicpath = logs / f'{label}.{name}.log'
    with publicpath.open('xb') as f:
        f.write(public)
    record['streams'][name] = {'original_sha256': hashlib.sha256(data).hexdigest(), 'published_sha256': hashlib.sha256(public).hexdigest(), 'published_path': publicpath.relative_to(cwd).as_posix(), 'bytes': len(public), 'private_path_prefix_redacted': data != public}
with receipt_path.open('x', encoding='utf-8', newline='\n') as f:
    json.dump(record, f, indent=2)
    f.write('\n')
print(json.dumps(record))
print(child.stderr.decode(errors='replace')[-1800:].replace(str(root), 'PRIVATE_WORK'))
raise SystemExit(child.returncode)
