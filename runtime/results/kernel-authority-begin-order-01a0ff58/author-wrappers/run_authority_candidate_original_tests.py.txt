"""Check the isolated candidate's changed shared API against the nine pinned sources."""
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path
import subprocess
import sys

root = Path(__file__).resolve().parent
package = root / 'agent-interface/runtime/results/kernel-authority-begin-order-01a0ff58'
private = root / 'authority-begin-order-private'
freeze = json.loads((package / 'FINAL_FREEZE.json').read_text())
out = private / 'candidate-original-suite'
out.mkdir()
source = {}
for name, identity in freeze['source'].items():
    raw = (package / 'source' / (Path(name).name + '.txt')).read_bytes()
    assert hashlib.sha256(raw).hexdigest() == identity['sha256']
    if name.endswith('/lifecycle.py'):
        raw = (package / 'authorization_floor_lifecycle.py.txt').read_bytes()
    destination = out / name
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_bytes(raw)
    source[name] = hashlib.sha256(raw).hexdigest()
(out / 'SUITE_FREEZE.json').write_text(json.dumps({'utc': datetime.now(timezone.utc).isoformat(), 'source_sha256': source, 'candidate_source_from_final_freeze': True}, indent=2) + '\n')
for label, flags in (('candidate-kernel-normal', []), ('candidate-kernel-optimized', ['-O'])):
    args = [sys.executable, '-B', *flags, '-m', 'unittest', 'discover', '-s', 'runtime/kernel', '-p', 'test_*.py', '-v']
    started = datetime.now(timezone.utc).isoformat()
    (out / (label + '.SEND_STARTED.json')).write_text(json.dumps({'utc': started, 'command': args, 'cwd': str(out)}, indent=2) + '\n')
    child = subprocess.Popen(args, cwd=out, stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    stdout, stderr = child.communicate(timeout=30)
    ended = datetime.now(timezone.utc).isoformat()
    assert len(stdout) + len(stderr) < 100000
    (out / (label + '.stdout.original.txt')).write_bytes(stdout)
    (out / (label + '.stderr.original.txt')).write_bytes(stderr)
    published = stderr.decode().replace(str(out), '<ISOLATED_SUITE>').replace(str(Path.home()), '<HOME>').encode()
    sha = lambda b: hashlib.sha256(b).hexdigest()
    record = {'label': label, 'command': ['python', '-B', *flags, '-m', 'unittest', 'discover', '-s', 'runtime/kernel', '-p', 'test_*.py', '-v'], 'cwd': '<ISOLATED_SUITE>', 'started_utc': started, 'ended_utc': ended, 'child_pid': child.pid, 'exit_code': child.returncode, 'original_stream_sha256': {'stdout': sha(stdout), 'stderr': sha(stderr)}, 'published_stream_sha256': {'stdout': sha(stdout), 'stderr': sha(published)}, 'source_sha256': source, 'scope': 'ordinary isolated authorization-memory candidate against original43 pinned kernel tests; not original baseline replay or real backend evidence'}
    (package / 'logs' / (label + '.stdout.txt')).write_bytes(stdout)
    (package / 'logs' / (label + '.stderr.txt')).write_bytes(published)
    (package / 'logs' / (label + '.receipt.json')).write_text(json.dumps(record, indent=2) + '\n')
    (out / (label + '.RECEIPT.json')).write_text(json.dumps(record, indent=2) + '\n')
    print(json.dumps({'label':label, 'exit_code':child.returncode, 'started_utc':started, 'ended_utc':ended}))
    print(published.decode()[-150:])
    assert child.returncode == 0 and 'Ran 43 tests' in published.decode() and 'OK' in published.decode()
    for name, digest in source.items():
        assert hashlib.sha256((out / name).read_bytes()).hexdigest() == digest
