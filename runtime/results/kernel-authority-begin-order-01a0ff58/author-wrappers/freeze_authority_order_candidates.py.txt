import difflib
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

root = Path(__file__).resolve().parent
package = root / 'agent-interface/runtime/results/kernel-authority-begin-order-01a0ff58'
baseline = json.loads((package / 'BEFORE_BASELINE.json').read_text())
receipt = json.loads((package / 'logs/baseline-boundary.receipt.json').read_text())
stderr = (package / 'logs/baseline-boundary.stderr.txt').read_text()
assert receipt['exit_code'] == 1 and 'Ran 8 tests' in stderr and 'FAILED (failures=5)' in stderr
assert 'FAILED (errors=' not in stderr
for name, identity in baseline['source'].items():
    assert hashlib.sha256((package / 'source' / (Path(name).name + '.txt')).read_bytes()).hexdigest() == identity['sha256']
for name, digest in baseline['new_helper_sha256'].items():
    assert hashlib.sha256((package / name).read_bytes()).hexdigest() == digest
original = (package / 'source/lifecycle.py.txt').read_text()
anchor = '        if now_ns >= self.lease.valid_until_ns:\n'
assert original.count(anchor) == 1
observation = original.replace(anchor, '        if self.observation is None or now_ns < self.observation.captured_ns:\n            raise ContractError("execution precedes observation capture")\n' + anchor)
authorization = original.replace('        self.execution_started_ns: int | None = None\n', '        self.execution_started_ns: int | None = None\n        self.authorized_ns: int | None = None\n')
assert authorization.count('        self.lease = lease\n') == 1
authorization = authorization.replace('        self.lease = lease\n', '        self.lease = lease\n        self.authorized_ns = now_ns\n')
authorization = authorization.replace(anchor, '        if self.authorized_ns is None or now_ns < self.authorized_ns:\n            raise ContractError("execution precedes accepted authorization")\n' + anchor)
for arm, source in (('observation_floor', observation), ('authorization_floor', authorization)):
    destination = package / (arm + '_lifecycle.py.txt')
    assert not destination.exists()
    destination.write_text(source, encoding='utf-8', newline='\n')
    patch = ''.join(difflib.unified_diff(original.splitlines(True), source.splitlines(True), fromfile='a/runtime/kernel/lifecycle.py', tofile='b/isolated/' + arm + '/lifecycle.py'))
    (package / (arm + '.patch.txt')).write_text(patch, encoding='utf-8', newline='\n')
helpers = {p.name: hashlib.sha256(p.read_bytes()).hexdigest() for p in package.iterdir() if p.is_file() and p.suffix in {'.py', '.md', '.txt'}}
final = {**baseline, 'recorded_utc': datetime.now(timezone.utc).isoformat(), 'phase': 'final prospective freeze before new108-row matrix/candidate checks', 'run_id': 'authority-begin-order-01a0ff58-ordinary-01', 'final_helpers_sha256': helpers, 'matrix': {'arms': ['baseline', 'observation_floor', 'authorization_floor'], 'grants_ns': [200,400], 'begin_ns': [0,99,100,199,200,300,399,400,1000], 'request_matches': [True,False], 'rows':108}, 'source_result_reuse': {'nine_bytes_match': True, 'prior_original_kernel_methods':43, 'prior_normal_and_optimized': 'existing receipt-main-context-v3 results; original43 baseline not executed anew', 'public_reference':'https://github.com/Unjuno/agent-interface/pull/6901#issuecomment-5964986578'}, 'resource': {'CPU': 'one small sequential stdlib child', 'formal_allocations':0, 'input_backend_gui_model_container_gpu':False}, 'decision':'PASS_ORDER_MEMORY_SCOPED only on complete108-row/source/ordinary test/data-control gates; no physical or production claim'}
(package / 'FINAL_FREEZE.json').write_text(json.dumps(final, indent=2) + '\n', encoding='utf-8')
(root / 'authority-begin-order-private/FINAL_FREEZE.json').write_bytes((package / 'FINAL_FREEZE.json').read_bytes())
print(json.dumps({'freeze_utc': final['recorded_utc'], 'helpers': len(helpers), 'source_files': len(final['source']), 'row_budget':108, 'matrix_executed':False}))
