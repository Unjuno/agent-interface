import argparse
import hashlib
import json
import subprocess
from datetime import datetime, timezone
from pathlib import Path

ROOT=Path(__file__).resolve().parent
REPO=ROOT/'agent-interface'
E=ROOT/'relay-utf8-repair'
PYTHON=ROOT/'relay312-venv/Scripts/python.exe'
parser=argparse.ArgumentParser()
parser.add_argument('--phase',required=True)
parser.add_argument('--full',action='store_true')
args=parser.parse_args()
E.mkdir(exist_ok=True)
OUT=E/args.phase
OUT.mkdir(exist_ok=False)
source=REPO/'runtime/cli_v1/mcp_relay.py'
test=REPO/'runtime/cli_v1/test_mcp_relay.py'
def sha(data):return hashlib.sha256(data).hexdigest()
names=['test_serve_preserves_utf8_wire_despite_text_stream_encoding',
       'test_bytes_require_utf8_before_dispatch_and_id_consumption',
       'test_unicode_bytes_keep_sdk_uncertainty_and_never_replay',
       'test_serve_keeps_already_decoded_text_stream_compatibility']
targets=['runtime.cli_v1.test_mcp_relay'] if args.full else ['runtime.cli_v1.test_mcp_relay.PublicRelayTests.'+n for n in names]
command=[str(PYTHON),'-m','unittest','-v',*targets]
before={'kind':'ordinary author regression; no formal/reviewer-producer replay',
        'started_utc':datetime.now(timezone.utc).isoformat(), 'command':command,
        'source_sha256':sha(source.read_bytes()),'test_sha256':sha(test.read_bytes()),
        'git_head':subprocess.check_output(['git','-C',str(REPO),'rev-parse','HEAD']).decode().strip(),
        'predicates':'UTF8 literal/escaped key duplicate refuses ID1; valid Unicode forwarded exactly; invalid bytes/UTF16/32 refuse before SDK; SDK accepted error consumes ID and never replays; already decoded text stream compatible',
        'limits':'mocked SDK contexts only in new4 methods; no actual GUI/model/backend/input/server subprocess'}
(OUT/'BEFORE.json').write_text(json.dumps(before,indent=2)+'\n',encoding='utf-8')
(OUT/'runtime-source.py.txt').write_bytes(source.read_bytes())
(OUT/'test-source.py.txt').write_bytes(test.read_bytes())
result=subprocess.run(command,cwd=REPO,capture_output=True,timeout=50)
(OUT/'stdout.bin').write_bytes(result.stdout);(OUT/'stderr.bin').write_bytes(result.stderr)
after={**before,'ended_utc':datetime.now(timezone.utc).isoformat(),'exit_code':result.returncode,
       'stdout_sha256':sha(result.stdout),'stderr_sha256':sha(result.stderr)}
(OUT/'RECEIPT.json').write_text(json.dumps(after,indent=2)+'\n',encoding='utf-8')
print(json.dumps({'phase':args.phase,'exit_code':result.returncode,'test_sha256':before['test_sha256']}))
print(result.stderr.decode('utf-8',errors='backslashreplace'))
