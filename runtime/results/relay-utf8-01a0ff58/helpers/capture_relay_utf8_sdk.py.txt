"""Distinct ordinary current-archive UTF8 pipe/SDK construction, no input backend."""
import argparse
import hashlib
import importlib.metadata
import io
import json
import os
import subprocess
import sys
import zipfile
from datetime import datetime, timezone
from pathlib import Path

ROOT=Path(__file__).resolve().parent
REPO=ROOT/'agent-interface'
E=ROOT/'relay-utf8-repair'
parser=argparse.ArgumentParser()
parser.add_argument('--phase',required=True)
parser.add_argument('--expected',choices=['original','repaired'],required=True)
args=parser.parse_args()
OUT=E/args.phase
OUT.mkdir(exist_ok=False)
sys.path.insert(0,str(REPO))
from runtime.distribution_v2.build import build,GENERATED,SOURCE_FILES
def sha(raw):return hashlib.sha256(raw).hexdigest()
def check(condition,message):
    if not condition:raise ValueError(message)
head=subprocess.check_output(['git','-C',str(REPO),'rev-parse','HEAD']).decode().strip()
archive=OUT/'runtime.pyz'
manifest=build(REPO,archive,OUT/'BUILD-MANIFEST.json',OUT/'SHA256SUMS')
check(manifest['source_revision']==head,'build pinned wrong revision')
pins={}
query=''.join(f'{head}:{name}\n' for name in SOURCE_FILES).encode()
stream=io.BytesIO(subprocess.check_output(['git','-C',str(REPO),'cat-file','--batch'],input=query))
with zipfile.ZipFile(archive) as package:
    for name in SOURCE_FILES:
        oid,kind,size=stream.readline().decode().split()
        raw=stream.read(int(size))
        check(stream.read(1)==b'\n' and kind=='blob','Git batch boundary')
        check(package.read(name)==raw,'archive source differs from exact Git')
        pins[name]={'git_blob':oid,'sha256':sha(raw),'bytes':len(raw)}
    for name,raw in GENERATED.items():
        check(package.read(name)==raw,'archive generated entry mismatch')
(OUT/'SOURCE-PINS.json').write_text(json.dumps(pins,indent=2)+'\n',encoding='utf-8')
# Static validation never opens a backend, evaluates live authority or dispatches.
program={'schema':'agent-interface/program-v1','program_id':'ordinary-utf8-static',
         'source':{'observation_seq':0,'binding_revision':0},
         'authority':{'lease_id':'ordinary-not-an-input-permit','expires_at_ns':1},
         'terminal':{'release_all_required':True},
         'ops':[{'op':'text','text':'\u65e5\u672c\u8a9e\U0001f600'},{'op':'release_all'}]}
duplicate=(r'{"id":1,"tool":"interface_validate","arguments":{"program":{"\ud83d\ude00":1,"'
           +'\U0001f600'+'":2}}}\n').encode('utf-8')
clock1=b'{"id":1,"tool":"interface_clock","arguments":{}}\n'
valid=(json.dumps({'id':2,'tool':'interface_validate','arguments':{'program':program}},
                  ensure_ascii=False,separators=(',',':'))+'\n').encode('utf-8')
invalid=b'{"id":3,"tool":"interface_clock","arguments":{"text":"\x80"}}\n'
clock3=b'{"id":3,"tool":"interface_clock","arguments":{}}\n'
close4=b'{"id":4,"tool":"interface_close","arguments":{}}\n'
wire=duplicate+clock1+valid+invalid+clock3+close4
(OUT/'stdin.bin').write_bytes(wire)
(OUT/'targets.json').write_text('{"app":123}\n',encoding='utf-8')
probe_code='import sys,json,importlib.metadata;print(json.dumps({"python":sys.version,"utf8_mode":sys.flags.utf8_mode,"stdin_encoding":sys.stdin.encoding,"stdin_errors":sys.stdin.errors,"mcp":importlib.metadata.version("mcp")},ensure_ascii=True))'
probe=subprocess.run([sys.executable,'-I','-c',probe_code],input=b'',capture_output=True,timeout=8)
(OUT/'native-stdio-probe.stdout.bin').write_bytes(probe.stdout)
(OUT/'native-stdio-probe.stderr.bin').write_bytes(probe.stderr)
check(probe.returncode==0,'native encoding probe failed')
environment=json.loads(probe.stdout)
check(environment['mcp']=='1.30.0','SDK version not pinned')
command=[sys.executable,'-I',str(archive),'relay','--','--targets',str(OUT/'targets.json'),
         '--output-directory',str(OUT/'calls'),'--session-mode','persistent-x11']
before={'kind':'new ordinary author repair construction, no historical/formal/reviewer replay',
        'head':head,'started_utc':datetime.now(timezone.utc).isoformat(),'command':command,
        'environment':environment,'archive_sha256':sha(archive.read_bytes()),'archive_bytes':archive.stat().st_size,
        'source_files':len(pins),'stdin_sha256':sha(wire),'stdin_bytes':len(wire),
        'input_row_sha256':[sha(line) for line in wire.splitlines(keepends=True)],
        'expected':args.expected,
        'predicates':{'original':'native default first duplicate crosses SDK/consumesID; Unicode static call records failure, without asserting server execution; neutral close terminates',
                      'repaired':'six responses refused/returned/returned/refused/returned/returned; nextIDs1/2/3/3/4/5; duplicate/invalid preSDK no ID; static valid true/no backend; close no connection open'},
        'limits':'one relay/one own SDK server, no observe/dispatch, max20s; each fresh exclusive output'}
(OUT/'BEFORE.json').write_text(json.dumps(before,indent=2)+'\n',encoding='utf-8')
started=datetime.now(timezone.utc).isoformat()
process=subprocess.Popen(command,cwd=OUT,stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
stdout,stderr=process.communicate(wire,timeout=20)
ended=datetime.now(timezone.utc).isoformat()
(OUT/'stdout.bin').write_bytes(stdout);(OUT/'stderr.bin').write_bytes(stderr)
receipt={'relay_pid':process.pid,'started_utc':started,'ended_utc':ended,'exit_code':process.returncode,
         'stdout_sha256':sha(stdout),'stdout_bytes':len(stdout),'stderr_sha256':sha(stderr),'stderr_bytes':len(stderr)}
(OUT/'RECEIPT.json').write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8')
check(process.returncode==0,'actual relay did not exit0')
rows=[json.loads(line) for line in stdout.splitlines()]
check(len(rows)==6,'response count')
if args.expected=='original':
    check(rows[0]['status']=='unknown_requires_reconciliation' and rows[0]['next_id']==2,
          'original Unicode duplicate did not reproduce native SDK uncertainty')
    check(rows[1]['status']=='refused' and rows[1]['next_id']==2,'original sameID not refused')
    check(rows[2]['status']=='unknown_requires_reconciliation' and rows[2]['next_id']==3,
          'original Unicode static validation did not reproduce native encoding failure')
else:
    check([row['status'] for row in rows]==['refused','returned','returned','refused','returned','returned'],
          'repaired response statuses')
    check([row['next_id'] for row in rows]==[1,2,3,3,4,5],'repaired ID boundaries')
    check(rows[0]['dispatched'] is False and rows[3]['dispatched'] is False,'decode did not refuse before SDK')
    static=json.loads(rows[2]['result']['content'][0]['text'])
    check(static['static_valid'] is True and static['backend_checked'] is False
          and static['task_success'] is None,'Unicode static validation scope')
close=json.loads(rows[-1]['result']['content'][0]['text'])
check(close['status']=='closed' and close['connection_close_attempted'] is False,'backend was opened or close mismatch')
report={'statuses':[row['status'] for row in rows],'next_ids':[row['next_id'] for row in rows],
        'expected':args.expected,'exit_code':process.returncode,'archive_sha256':before['archive_sha256'],
        'stdout_sha256':receipt['stdout_sha256'],'backend_connection_close_attempted':False,
        'limits':'static SDK transfer/serialization only; no actual native input/application/physical release/task/performance'}
(OUT/'RESULT.json').write_text(json.dumps(report,indent=2)+'\n',encoding='utf-8')
print(json.dumps(report))
