import datetime, hashlib, json, pathlib, subprocess, sys
root = pathlib.Path(__file__).resolve().parent
which, mode, label = sys.argv[1:]
source = root / which
out = root / 'runs' / label
out.mkdir(parents=True, exist_ok=False)
argv = [sys.executable, '-X', 'utf8', '-B'] + (['-O'] if mode == 'optimized' else [])
argv += ['-m', 'unittest', '-v', 'runtime.backends.win32_v1.test_explicit_up_ledger', 'runtime.backends.win32_v1.test_release_ledger', 'runtime.backends.win32_v1.test_integration.PureWin32HelperTests']
def pins():
    return {p.relative_to(source).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest() for p in source.rglob('*.py')}
receipt = {'unit': 'win32-explicit-up-adoption-01', 'kind': 'ordinary inert source adoption regression', 'argv': argv, 'cwd': str(source), 'source_sha256': pins(), 'plan_sha256': hashlib.sha256((root / 'PLAN.json').read_bytes()).hexdigest(), 'runner_sha256': hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(), 'started_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'live_native_integration_class': False, 'native_backend_constructors_API_endpoints': 'forbidden or inert'}
(out / 'START.json').write_text(json.dumps(receipt, indent=2) + '\n', encoding='utf-8')
with (out / 'stdout.txt').open('wb') as stdout, (out / 'stderr.txt').open('wb') as stderr:
    p = subprocess.Popen(argv, cwd=source, stdout=stdout, stderr=stderr)
    receipt['pid'] = p.pid
    try:
        receipt['exit_code'] = p.wait(timeout=20)
    except subprocess.TimeoutExpired:
        p.kill()
        receipt['exit_code'] = p.wait()
        receipt['timeout_killed_only_own_child'] = True
receipt.update({'ended_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'source_unchanged': pins() == receipt['source_sha256'], 'streams': {n: {'bytes': (out / n).stat().st_size, 'sha256': hashlib.sha256((out / n).read_bytes()).hexdigest()} for n in ['stdout.txt', 'stderr.txt']}})
receipt['output_size_gate'] = all(x['bytes'] <= 524288 for x in receipt['streams'].values())
(out / 'RESULT.json').write_text(json.dumps(receipt, indent=2) + '\n', encoding='utf-8')
print(json.dumps({'run': label, 'pid': p.pid, 'exit_code': receipt['exit_code'], 'source_unchanged': receipt['source_unchanged'], 'output_size_gate': receipt['output_size_gate'], 'started_utc': receipt['started_utc'], 'ended_utc': receipt['ended_utc']}))
print((out / 'stderr.txt').read_text(encoding='utf-8'))
sys.exit(receipt['exit_code'])
