"""One ordinary retained-data reconciliation; no producer/test invocation."""
from datetime import datetime, timezone
from pathlib import Path
import hashlib,json,os,subprocess
ROOT=Path(__file__).resolve().parent
REPO=ROOT/'paired-clock-repo'
REL='runtime/results/paired-clock-01a0ff35-20261003'
PKG=REPO/REL
def sha(data):return hashlib.sha256(data).hexdigest()
receipt_path=PKG/'public-audit.receipt.json'
if receipt_path.exists(): raise ValueError('retain first public reconciliation')
source=PKG/'public_read_rows.py.txt'
command=['../mcp-shutdown-venv312/Scripts/python.exe','-B','-O',REL+'/public_read_rows.py.txt',
         '--package',REL,'--report',REL+'/public-audit.json']
receipt={'kind':'ordinary retained-data audit; no A01 producer replay','argv':command,
    'cwd':'.','cwd_reference':'own paired-clock checkout at4ada7f511ce451caa4050c78b7fa4b3afe6b6f72 with the disclosed test-only change',
    'started_utc':datetime.now(timezone.utc).isoformat(),'source_sha256':sha(source.read_bytes()),
    'publication_sha256':sha((PKG/'PUBLICATION.json').read_bytes()),
    'wrapper_sha256':sha(Path(__file__).read_bytes()),'requested_env':{'PYTHONDONTWRITEBYTECODE':'1','PYTHONIOENCODING':'utf-8'}}
receipt_path.write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8')
result=subprocess.run(command,cwd=REPO,capture_output=True,env=dict(os.environ,**receipt['requested_env']))
for name,data in [('stdout',result.stdout),('stderr',result.stderr)]:
    (PKG/('public-audit.'+name+'.txt')).write_bytes(data); receipt[name+'_sha256']=sha(data)
receipt.update(ended_utc=datetime.now(timezone.utc).isoformat(),exit_code=result.returncode,
               source_unchanged=sha(source.read_bytes())==receipt['source_sha256'],
               publication_unchanged=sha((PKG/'PUBLICATION.json').read_bytes())==receipt['publication_sha256'])
receipt_path.write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8')
(PKG/'public-audit-wrapper.py.txt').write_bytes(Path(__file__).read_bytes())
print(json.dumps({'exit_code':result.returncode,'source_unchanged':receipt['source_unchanged'],
    'publication_unchanged':receipt['publication_unchanged'],'stdout':result.stdout.decode('utf-8',errors='replace'),
    'stderr':result.stderr.decode('utf-8',errors='replace')}))
raise SystemExit(result.returncode)
