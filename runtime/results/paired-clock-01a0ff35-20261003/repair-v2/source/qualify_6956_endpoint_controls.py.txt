"""Ordinary copied-function controls; never invoke the formal clock producer."""
import ast, hashlib, inspect, io, json, textwrap, unittest
from pathlib import Path
from datetime import datetime, timezone
from unittest.mock import patch
from runtime.cli_v1 import mcp_session
from runtime.cli_v1.test_post_dispatch_capture import PostDispatchCaptureTests

OUT=Path(__file__).resolve().parent/'6956-v2-intake'
name='test_wait_receipt_records_actual_clock_reads_on_success_and_error'
start=datetime.now(timezone.utc).isoformat()
original=mcp_session.MCPSessionOwner.inspect_after_dispatch
text=textwrap.dedent(inspect.getsource(original))
needle="capture_wait['ended_ns'] = time.monotonic_ns()"
if text.count(needle)!=1:raise ValueError('one exact endpoint in pinned function required')
controls=[('healthy',text),('reuse-start',text.replace(needle,"capture_wait['ended_ns'] = capture_wait['started_ns']")),
    ('outer-start',text.replace(needle,"capture_wait['ended_ns'] = started")),
    ('wrong-reading',text.replace(needle,"capture_wait['ended_ns'] = time.monotonic_ns() + 1"))]
rows=[]
for label,source in controls:
    target=OUT/(label+'.function.py.txt')
    with target.open('x',encoding='utf-8',newline='\n') as f:f.write(source)
    module=ast.parse(source)
    if len(module.body)!=1 or not isinstance(module.body[0],ast.FunctionDef):raise ValueError('function-only private control')
    namespace=dict(mcp_session.__dict__)
    exec(compile(module,str(target),'exec'),namespace)
    private=namespace['inspect_after_dispatch']
    methods=['test_explicit_wait_is_after_release_and_retained_lookup_never_waits_again',
        'test_wait_duration_survives_a_coarse_receipt_clock',name]
    for method in methods:
        stream=io.StringIO()
        with patch.object(mcp_session.MCPSessionOwner,'inspect_after_dispatch',private):
            result=unittest.TextTestRunner(stream=stream,verbosity=2).run(unittest.TestSuite([PostDispatchCaptureTests(method)]))
        output=stream.getvalue()
        with (OUT/(label+'-'+method+'.log.txt')).open('x',encoding='utf-8',newline='\n') as f:f.write(output)
        should_pass=label=='healthy' or method!=name
        rows.append({'variant':label,'method':method,'tests_run':result.testsRun,
            'failures':len(result.failures),'errors':len(result.errors),'skipped':len(result.skipped),
            'was_successful':result.wasSuccessful(),'expected_success':should_pass,
            'matches_expected':result.wasSuccessful()==should_pass,
            'source_sha256':hashlib.sha256(source.encode()).hexdigest(),
            'native_unittest_log_sha256':hashlib.sha256(output.encode()).hexdigest()})
if mcp_session.MCPSessionOwner.inspect_after_dispatch is not original:raise ValueError('private function patch not retired')
packet={'started_utc':start,'ended_utc':datetime.now(timezone.utc).isoformat(),
    'rows':rows,'native_or_formal_producer_replays':0,'production_source_mutations':0,
    'disposition':'ALL_PROSPECTIVE_CONTROLS_MATCH' if all(r['matches_expected'] for r in rows) else 'CONTROL_MISMATCH_RETAIN',
    'limit':'Existing two regression methods only for old-test survival; exact endpoint method uses trusted synchronous callback fixtures, no physical-time/backend/input claim.'}
with (OUT/'endpoint-controls.json').open('x',encoding='utf-8') as f:json.dump(packet,f,indent=2)
print(json.dumps(packet,indent=2))
raise SystemExit(0 if all(r['matches_expected'] for r in rows) else 1)
