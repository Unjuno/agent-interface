"""Post-observation public data reader. Never imports or runs the frozen probe."""
from datetime import datetime
from pathlib import Path
import argparse, copy, hashlib, json

def digest(data): return hashlib.sha256(data).hexdigest()
def need(ok, why):
    if not ok: raise ValueError(why)
def utc(value):
    need(type(value) is str, 'UTC must be a string')
    result=datetime.fromisoformat(value)
    need(result.utcoffset() is not None and result.utcoffset().total_seconds()==0, 'UTC offset')
    return result

def analyse(plan, start, end, raw):
    need(plan['allocation_id']==start['allocation_id']==end['allocation_id'], 'allocation identity')
    need(type(start['rows_declared']) is int and start['rows_declared']==12, 'declared denominator')
    need(type(end['rows_completed']) is int and end['rows_completed']==12, 'completed denominator')
    need(end['execution_completed'] is True, 'literal terminal completion')
    first,last=utc(start['started_utc']),utc(end['completed_utc'])
    need(utc(plan['frozen_utc'])<=first<=last, 'prospective original UTC sequence')
    need(raw.endswith(b'\n') and digest(raw)==end['rows_sha256'], 'complete raw identity')
    rows=[json.loads(line) for line in raw.splitlines()]
    need(len(rows)==len(plan['rows'])==12, 'full fixed12 rows')
    output=[]; previous=first
    for index,row in enumerate(rows):
        for key in ('row_id','requested_ns','monotonic_started_ns','monotonic_ended_ns','perf_started_ns','perf_ended_ns'):
            need(type(row[key]) is int and row[key]>=0, 'strict nonnegative integer '+key)
        need(row['row_id']==index and row['requested_ns']==20_000_000, 'fixed row/order/request')
        need(plan['rows'][index]=={'row_id':index,'requested_ns':20_000_000}, 'fixed declared condition')
        before,after=utc(row['started_utc']),utc(row['ended_utc'])
        need(previous<=before<=after<=last, 'ordered row UTC'); previous=after
        # Each subtraction remains within one API's clock domain.
        m=row['monotonic_ended_ns']-row['monotonic_started_ns']
        p=row['perf_ended_ns']-row['perf_started_ns']
        need(m>=0 and p>=0, 'clock backwards')
        output.append({'row_id':index,'requested_ns':20_000_000,'monotonic_elapsed_ns':m,
                       'perf_elapsed_ns':p,'paired_witness':m<20_000_000<=p})
    witness=[r['row_id'] for r in output if r['paired_witness']]
    short=[r['row_id'] for r in output if r['perf_elapsed_ns']<20_000_000]
    return {'allocation_id':plan['allocation_id'],'rows':12,'observations':output,
        'decision':'COUNTEREVIDENCE_SHORT_PERF' if short else 'SUPPORTED_IN_FIXED_RUN' if witness else 'HOLD_NO_WITNESS',
        'witness_rows':witness,'short_perf_rows':short,
        'monotonic_range_ns':[min(r['monotonic_elapsed_ns'] for r in output),max(r['monotonic_elapsed_ns'] for r in output)],
        'perf_range_ns':[min(r['perf_elapsed_ns'] for r in output),max(r['perf_elapsed_ns'] for r in output)]}

def controls(plan,start,end,raw):
    results=[]; original=[json.loads(line) for line in raw.splitlines()]
    for case in ('boolean_row_id','duplicate_id','missing_row','backwards_perf','future_freeze','short_perf'):
        p,s,e,rows=copy.deepcopy((plan,start,end,original))
        if case=='boolean_row_id': rows[0]['row_id']=False
        elif case=='duplicate_id': rows[1]['row_id']=0
        elif case=='missing_row': rows.pop()
        elif case=='backwards_perf': rows[0]['perf_ended_ns']=rows[0]['perf_started_ns']-1
        elif case=='future_freeze': p['frozen_utc']='2099-01-01T00:00:00+00:00'
        elif case=='short_perf': rows[0]['perf_ended_ns']=rows[0]['perf_started_ns']+19_000_000
        altered=b''.join((json.dumps(row)+'\n').encode() for row in rows)
        e['rows_sha256']=digest(altered)
        # Byte/projection checks are deliberately lifted: these exercise semantics.
        try:
            value=analyse(p,s,e,altered)
        except ValueError as error:
            need(case!='short_perf', 'valid counterevidence must retain a decision')
            results.append({'case':case,'outcome':'REJECTED','reason':str(error)})
        else:
            need(case=='short_perf' and value['decision']=='COUNTEREVIDENCE_SHORT_PERF' and value['short_perf_rows']==[0], 'counterevidence suppressed')
            results.append({'case':case,'outcome':value['decision'],'short_perf_rows':[0]})
    return results

def main():
    ap=argparse.ArgumentParser(description=__doc__)
    ap.add_argument('--package',required=True);ap.add_argument('--report',required=True)
    args=ap.parse_args(); root=Path(args.package)
    publication=json.loads((root/'PUBLICATION.json').read_bytes())
    for name,item in publication['files'].items():
        path=Path(name);need(not path.is_absolute() and '..' not in path.parts, 'relative projection member')
        data=(root/path).read_bytes()
        need(len(data)==item['public_bytes'] and digest(data)==item['public_sha256'], 'projected file identity '+name)
        if not item['path_only_projection']:
            need(item['original_sha256']==item['public_sha256'] and item['original_bytes']==item['public_bytes'], 'unprojected identity '+name)
    plan=json.loads((root/'PLAN.json').read_bytes())
    start=json.loads((root/'execution-A01/START.json').read_bytes())
    end=json.loads((root/'execution-A01/END.json').read_bytes())
    need(start['plan_sha256']==publication['original_frozen_plan_sha256']==publication['files']['PLAN.json']['original_sha256'], 'original frozen plan join')
    for name,value in plan['source_sha256'].items():
        need(digest((root/name).read_bytes())==value, 'unchanged frozen source '+name)
    raw=(root/'execution-A01/rows.jsonl').read_bytes()
    value=analyse(plan,start,end,raw)
    frozen=json.loads((root/'AUDIT.json').read_bytes())
    need(all(frozen[key]==data for key,data in value.items()), 'independent arithmetic differs from original audit')
    report={'kind':'post-observation data-only public reconciliation under -O; not prospective allocation or producer execution',
        'projection_files_checked':len(publication['files']),'unchanged_raw_sha256':digest(raw),
        'original_frozen_plan_sha256':publication['original_frozen_plan_sha256'],
        'public_plan_sha256':digest((root/'PLAN.json').read_bytes()),'result':value,
        'semantic_controls':controls(plan,start,end,raw),
        'qualification':'Local separate arithmetic implementation; not a nonauthor vote, execution authentication, physical calibration or general failure-frequency estimate. Original reader remains byte-identical and its original -O run used the private unprojected PLAN.'}
    with Path(args.report).open('x',encoding='utf-8',newline='\n') as stream:
        stream.write(json.dumps(report,indent=2)+'\n')
    print(json.dumps({'decision':value['decision'],'rows':value['rows'],'witness_rows':value['witness_rows'],
        'projection_files_checked':len(publication['files']),'semantic_controls':len(report['semantic_controls'])}))
if __name__=='__main__': main()
