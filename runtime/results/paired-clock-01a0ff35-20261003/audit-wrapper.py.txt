from datetime import datetime,timezone
from pathlib import Path
import hashlib,json,subprocess

ROOT=Path(__file__).resolve().parent
PACKAGE=ROOT/'paired-clock-repo/runtime/results/paired-clock-01a0ff35-20261003'
PYTHON=ROOT/'mcp-shutdown-venv312/Scripts/python.exe'
def sha(data):return hashlib.sha256(data).hexdigest()
plan=PACKAGE/'PLAN.json';spec=json.loads(plan.read_bytes())
source=PACKAGE/'read_rows.py.txt'
if sha(source.read_bytes())!=spec['source_sha256']['read_rows.py.txt']:raise ValueError('frozen independent reader')
receipt_path=PACKAGE/'audit.receipt.json'
if receipt_path.exists()or(PACKAGE/'AUDIT.json').exists():raise ValueError('data audit already attempted; preserve first output')
argv=[str(PYTHON),'-B','-O',str(source),'--plan',str(plan),'--output',str(PACKAGE/'execution-A01'),'--report',str(PACKAGE/'AUDIT.json')]
receipt={'argv':argv,'cwd':str(PACKAGE),'started_utc':datetime.now(timezone.utc).isoformat(),
         'source_sha256':sha(source.read_bytes()),'kind':'ordinary retained-data arithmetic; no sleep/backend/producer import or replay'}
value=subprocess.run(argv,cwd=PACKAGE,capture_output=True)
for name,data in [('stdout',value.stdout),('stderr',value.stderr)]:
    (PACKAGE/('audit.'+name+'.txt')).write_bytes(data);receipt[name+'_sha256']=sha(data)
receipt.update(ended_utc=datetime.now(timezone.utc).isoformat(),exit_code=value.returncode,
               source_unchanged=sha(source.read_bytes())==receipt['source_sha256'])
receipt_path.write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8')
print(json.dumps({'exit_code':value.returncode,'stdout':value.stdout.decode(errors='replace'),'stderr':value.stderr.decode(errors='replace'),
                  'source_unchanged':receipt['source_unchanged']}))
raise SystemExit(value.returncode)
