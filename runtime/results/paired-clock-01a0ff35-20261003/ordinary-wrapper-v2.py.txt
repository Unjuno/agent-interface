from datetime import datetime,timezone
from pathlib import Path
import hashlib,json,os,subprocess,sys

ROOT=Path(__file__).resolve().parent
REPO=ROOT/'paired-clock-repo'
PACKAGE=REPO/'runtime/results/paired-clock-01a0ff35-20261003'
PYTHON=ROOT/'mcp-shutdown-venv312/Scripts/python.exe'
def sha(data):return hashlib.sha256(data).hexdigest()
label=sys.argv[1];args=sys.argv[2:]
receipt_path=PACKAGE/(label+'.receipt.json')
if receipt_path.exists():raise ValueError('unique ordinary stage required; preserve first results')
sources={str(p.relative_to(REPO)).replace('\\','/'):sha(p.read_bytes()) for parent in ('runtime','research/live_control')
         for p in (REPO/parent).rglob('*.py') if '__pycache__' not in p.parts}
source=REPO/'runtime/cli_v1/test_post_dispatch_capture.py'
snapshot=PACKAGE/(label+'.test-source.py.txt');snapshot.write_bytes(source.read_bytes())
receipt={'argv':[str(PYTHON),'-B',*args],'cwd':str(REPO),'started_utc':datetime.now(timezone.utc).isoformat(),
         'source_sha256':sources,'snapshot_sha256':sha(snapshot.read_bytes()),
         'wrapper_sha256':sha(Path(__file__).read_bytes()),
         'requested_env':{'PYTHONDONTWRITEBYTECODE':'1','PYTHONIOENCODING':'utf-8',
                          'OMP_NUM_THREADS':'1','OPENBLAS_NUM_THREADS':'1','MKL_NUM_THREADS':'1'},
         'kind':'ordinary engineering regression/current-source integration, not paired-clock A01 or any original formal producer replay'}
receipt_path.write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8')
value=subprocess.run(receipt['argv'],cwd=REPO,capture_output=True,
                     env=dict(os.environ,**receipt['requested_env']))
for name,data in [('stdout',value.stdout),('stderr',value.stderr)]:
    (PACKAGE/(label+'.'+name+'.txt')).write_bytes(data);receipt[name+'_sha256']=sha(data)
after={p:sha((REPO/p).read_bytes()) for p in sources}
receipt.update(ended_utc=datetime.now(timezone.utc).isoformat(),exit_code=value.returncode,source_unchanged=after==sources,
               thread_settings='requested OMP/BLAS1, not claimed enforced/global resources')
receipt_path.write_text(json.dumps(receipt,indent=2)+'\n',encoding='utf-8')
print(json.dumps({'label':label,'exit_code':value.returncode,'source_files':len(sources),'source_unchanged':after==sources,
                  'stdout_tail':value.stdout.decode(errors='replace')[-700:],'stderr_tail':value.stderr.decode(errors='replace')[-1700:]}))
raise SystemExit(value.returncode)
