"""Capture ordinary test commands/exits; process-local scratch placement only."""
import hashlib,json,os,pathlib,shutil,subprocess,sys
from datetime import datetime,timezone
root=pathlib.Path(__file__).resolve().parent
label=sys.argv[1]
output=root/'execution'/label;output.mkdir(parents=True)
scratch=root/'scratch'/label;scratch.mkdir(parents=True)
env=os.environ.copy();env['TEMP']=str(scratch);env['TMP']=str(scratch)
node=shutil.which('node')
argv=[node,'--test',str(root/'source/test_relay_client.mjs')]
start=datetime.now(timezone.utc).isoformat()
p=subprocess.run(argv,capture_output=True,env=env,timeout=30)
logs={}
for name,data in [('stdout',p.stdout),('stderr',p.stderr)]:
    with (output/(name+'.txt')).open('xb') as f:f.write(data)
    logs[name]=dict(bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
record=dict(argv=argv,start_utc=start,end_utc=datetime.now(timezone.utc).isoformat(),exit=p.returncode,logs=logs,node_sha256=hashlib.sha256(pathlib.Path(node).read_bytes()).hexdigest(),
  source_sha256=hashlib.sha256((root/'source/relay_client.mjs').read_bytes()).hexdigest(),test_sha256=hashlib.sha256((root/'source/test_relay_client.mjs').read_bytes()).hexdigest(),scope='ordinary real owned synthetic child regression; no native effects')
with (output/'receipt.json').open('x',encoding='utf-8',newline='\n') as f:json.dump(record,f,indent=2);f.write('\n')
print(json.dumps(record))
sys.exit(p.returncode)
