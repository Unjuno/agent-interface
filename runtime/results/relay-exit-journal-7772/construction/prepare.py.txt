import hashlib,json,pathlib,subprocess
from datetime import datetime,timezone
root=pathlib.Path(__file__).resolve().parent
repo=root.parent/'agent-interface'
base='9c26e204c0475bee30919e3d0e6681f393078ef1'
files=['relay_client.mjs','relay_host.mjs','test_relay_client.mjs','README.md']
pins={}
for name in files:
    path='runtime/host_v1/'+name
    raw=subprocess.check_output(['git','show',base+':'+path],cwd=repo)
    target=root/'source'/name;target.parent.mkdir(parents=True,exist_ok=True)
    with target.open('xb') as f:f.write(raw)
    pins[path]=dict(blob=subprocess.check_output(['git','rev-parse',base+':'+path],cwd=repo).decode().strip(),sha256=hashlib.sha256(raw).hexdigest())
with (root/'SOURCE.json').open('x',encoding='utf-8',newline='\n') as f:json.dump(dict(base=base,utc=datetime.now(timezone.utc).isoformat(),pins=pins),f,indent=2);f.write('\n')
print(json.dumps(dict(base=base,files=len(files))))
