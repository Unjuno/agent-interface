"""Exact existing Node workflow selection and its static local module closure."""
import hashlib,json,os,pathlib,posixpath,re,shutil,subprocess,sys
from datetime import datetime,timezone
root=pathlib.Path(__file__).resolve().parent;repo=root.parent/'agent-interface'
base=json.loads((root/'SOURCE.json').read_text())['base']
def git(*args):return subprocess.check_output(['git',*args],cwd=repo)
workflow=git('show',base+':.github/workflows/native-mcp-v1.yml')
selection=re.search(rb'run: node --test ([^\r\n]+)',workflow).group(1).decode().split()
assert len(selection)==13
dest=root/'combined';dest.mkdir()
queue=list(selection);exported={}
overrides={'runtime/host_v1/'+n:root/'source'/n for n in ['relay_client.mjs','test_relay_client.mjs','README.md']}
while queue:
    path=queue.pop(0)
    if path in exported:continue
    original=git('show',base+':'+path)
    raw=overrides[path].read_bytes() if path in overrides else original
    target=dest/path;target.parent.mkdir(parents=True,exist_ok=True)
    with target.open('xb') as f:f.write(raw)
    exported[path]=dict(original_blob=git('rev-parse',base+':'+path).decode().strip(),original_sha256=hashlib.sha256(original).hexdigest(),tested_sha256=hashlib.sha256(raw).hexdigest(),changed=raw!=original)
    for dependency in re.findall(rb"(?:from\s*|import\s*\()\s*['\"](\.[^'\"]+\.mjs)['\"]",raw):
        resolved=posixpath.normpath(posixpath.join(posixpath.dirname(path),dependency.decode()))
        if resolved.startswith('../'):raise ValueError('dependency outside export')
        queue.append(resolved)
with (root/'AFFECTED_SOURCE.json').open('x',encoding='utf-8',newline='\n') as f:json.dump(dict(base=base,workflow_sha256=hashlib.sha256(workflow).hexdigest(),selection=selection,closure=exported),f,indent=2);f.write('\n')
out=root/'execution/affected';out.mkdir();scratch=root/'scratch/affected';scratch.mkdir()
env=os.environ.copy();env['TEMP']=str(scratch);env['TMP']=str(scratch)
node=shutil.which('node');argv=[node,'--test',*[str(dest/p) for p in selection]]
start=datetime.now(timezone.utc).isoformat();p=subprocess.run(argv,cwd=dest,env=env,capture_output=True,timeout=35)
logs={}
for name,data in [('stdout',p.stdout),('stderr',p.stderr)]:
    with (out/(name+'.txt')).open('xb') as f:f.write(data)
    logs[name]=dict(bytes=len(data),sha256=hashlib.sha256(data).hexdigest())
with (out/'receipt.json').open('x',encoding='utf-8',newline='\n') as f:json.dump(dict(argv=argv,start_utc=start,end_utc=datetime.now(timezone.utc).isoformat(),exit=p.returncode,logs=logs,scope='existing explicit Node workflow selection; hosted/Python/native job not invoked'),f,indent=2);f.write('\n')
print(json.dumps(dict(exit=p.returncode,selection=len(selection),closure=len(exported),stdout_bytes=len(p.stdout),stderr_bytes=len(p.stderr))))
sys.exit(p.returncode)
