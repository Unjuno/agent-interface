from pathlib import Path
from datetime import datetime,timezone
import hashlib,json,platform,re,subprocess,sys

root=Path.cwd();out=root/'outputs/kernel-nominal-uncertainty-context-v3'
freeze=json.loads((out/'FREEZE.json').read_bytes());started=datetime.now(timezone.utc).isoformat()
def sha(path):return hashlib.sha256(path.read_bytes()).hexdigest()
def check_source(label,cwd):
    key=label.removesuffix('_optimized');actual={str(p.relative_to(cwd)).replace('\\','/'):sha(p) for p in cwd.rglob('*') if p.is_file()}
    assert actual==freeze['source_sha256'][key],(label,'source changed')
receipts=[]
for case in freeze['commands']:
    cwd=Path(case['cwd']);check_source(case['label'],cwd)
    begin=datetime.now(timezone.utc).isoformat();result=subprocess.run(case['argv'],cwd=cwd,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=30)
    end=datetime.now(timezone.utc).isoformat();dest=out/'checks'/case['label'];dest.mkdir(parents=True)
    (dest/'stdout.txt').write_bytes(result.stdout);(dest/'stderr.txt').write_bytes(result.stderr)
    text=(result.stdout+result.stderr).decode('utf8');match=re.search(r'Ran (\d+) tests? in',text);assert match
    count=int(match.group(1));errors=int(re.search(r'errors=(\d+)',text).group(1)) if 'errors=' in text else 0
    failures=int(re.search(r'failures=(\d+)',text).group(1)) if 'failures=' in text else 0
    marker=True
    if case['label']=='nominal_only':marker='self.assertTrue(result.effect_occurred)' in text and 'AssertionError: False is not true' in text
    if case['label']=='uncertainty_only':marker='flow.record_execution(shaped(execution()))' in text and 'AssertionError: ContractError not raised' in text
    record={**case,'started_utc':begin,'ended_utc':end,'exit_code':result.returncode,'actual_methods':count,'actual_failures':failures,'actual_errors':errors,'semantic_failure_marker_matches':marker,'stdout_sha256':sha(dest/'stdout.txt'),'stderr_sha256':sha(dest/'stderr.txt'),'formal_replayed':False}
    (dest/'receipt.json').write_text(json.dumps(record,indent=2,sort_keys=True)+'\n',encoding='utf8',newline='\n');receipts.append(record)
    check_source(case['label'],cwd)
    assert result.returncode==case['expected_exit'] and count==case['expected_methods'] and failures==case['expected_failures'] and errors==case['expected_errors'] and marker,(case['label'],'decision gate')
    if case['expected_exit']==0:assert '\nOK\n' in text and 'skipped=' not in text
verification={'worker':freeze['worker'],'kind':freeze['kind'],'started_utc':started,'ended_utc':datetime.now(timezone.utc).isoformat(),'python':platform.python_version(),'platform':platform.platform(),'freeze_sha256':sha(out/'FREEZE.json'),'plan_sha256':sha(out/'PLAN.md'),'runner_sha256':sha(Path(__file__)),'status':'PASS_CURRENT_CONTEXT_N_U_SCOPED','commands':receipts,'all_source_bytes_unchanged':True,'original_candidate_or_formal_replay':False,'author_branch_or_main_written':False,'physical_task_effect_verified':False}
(out/'VERIFICATION.json').write_text(json.dumps(verification,indent=2,sort_keys=True)+'\n',encoding='utf8',newline='\n')
print(json.dumps({k:v for k,v in verification.items() if k!='commands'}));print(json.dumps([{'label':r['label'],'methods':r['actual_methods'],'exit':r['exit_code'],'failures':r['actual_failures'],'errors':r['actual_errors'],'semantic_marker':r['semantic_failure_marker_matches']} for r in receipts]))
