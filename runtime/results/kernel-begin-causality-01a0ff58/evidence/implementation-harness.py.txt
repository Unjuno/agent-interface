"""Run directed implementation mutations only in private isolated copies."""
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess

work = Path(__file__).parent
source = work / 'agent-interface/runtime/results/kernel-begin-causality-01a0ff58'
original = (source / 'candidate.py').read_text(encoding='utf-8')
mutations = {
    'missing-guard': original.replace('receipt.started_ns < self.accepted_begin_ns', 'False'),
    'reversed-guard': original.replace('receipt.started_ns < self.accepted_begin_ns', 'receipt.started_ns > self.accepted_begin_ns'),
    'reject-equality': original.replace('receipt.started_ns < self.accepted_begin_ns', 'receipt.started_ns <= self.accepted_begin_ns'),
    'remember-before-admission': original.replace('        super().begin_execution(request, now_ns=now_ns)\n        self.accepted_begin_ns = now_ns', '        self.accepted_begin_ns = now_ns\n        super().begin_execution(request, now_ns=now_ns)'),
}
records = []
for label, text in mutations.items():
    target = work / 'private-begin-causality' / label
    target.mkdir(exist_ok=False)
    for name in ('fixtures.py', 'test_boundary.py'):
        shutil.copyfile(source / name, target / name)
    shutil.copytree(source / 'frozen_kernel', target / 'frozen_kernel', ignore=shutil.ignore_patterns('__pycache__'))
    (target / 'candidate.py').write_text(text, encoding='utf-8', newline='\n')
    run = subprocess.run(['python', '-B', '-m', 'unittest', '-v', 'test_boundary'], cwd=target,
        env=dict(os.environ, BEGIN_CAUSALITY_ARM='candidate'), stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=60)
    (target / 'original.log').write_bytes(run.stdout)
    records.append({'mutation': label, 'source_sha256': hashlib.sha256(text.encode()).hexdigest(),
        'exit_code': run.returncode, 'log_sha256': hashlib.sha256(run.stdout).hexdigest(),
        'summary': run.stdout.decode().splitlines()[-1], 'argv': ['python','-B','-m','unittest','-v','test_boundary']})
out = source / 'evidence/implementation-mutations.json'
with out.open('x', encoding='utf-8', newline='\n') as f:
    json.dump(records, f, indent=2)
    f.write('\n')
print(json.dumps(records))
raise SystemExit(0 if all(r['exit_code'] == 1 and r['summary'].startswith('FAILED (failures=') for r in records) else 1)
